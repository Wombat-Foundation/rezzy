1use rezzy::reconcile::{
2    build_bucket_sketches, ElementHash, EventIdFormat, ReconciliationClient, RemoteDigest,
3    ResidentKernel,
4};
5use std::fs::File;
6use std::io::{BufRead, BufReader};
7use std::time::Instant;
8
9fn load_uuids(filename: &str) -> (ResidentKernel, Vec<ElementHash>) {
10    let mut resident = ResidentKernel::new();
11    let mut elements = Vec::new();
12
13    let file = File::open(filename).unwrap();
14    for line in BufReader::new(file).lines() {
15        let uuid = format!("${}", line.unwrap());
16        if uuid == "$" {
17            continue;
18        }
19        let hash = ElementHash::from_matrix_event_id(&uuid, EventIdFormat::Legacy).unwrap();
20        resident.insert(hash).unwrap();
21        elements.push(hash);
22    }
23    (resident, elements)
24}

26fn main() {
27    println!("Loading A.txt...");
28    let start = Instant::now();
29    let (local_resident, local_elements) = load_uuids("A.txt");
30    println!("Loaded A.txt in {:?}", start.elapsed());
31
32    println!("Loading B.txt...");
33    let start = Instant::now();
34    let (remote_resident, remote_elements) = load_uuids("B.txt");
35    println!("Loaded B.txt in {:?}", start.elapsed());
36
37    let client = ReconciliationClient::default();
38
39    // Step 1: Client determines the difference and generates bucket requests
40    let start_triage = Instant::now();
41    let remote_digest = RemoteDigest {
42        digest: remote_resident.accumulator().digest(),
43        known_event_count: remote_resident.accumulator().known_event_count(),
44        strata: *remote_resident.strata(),
45        frame_matches: true,
46        has_unknown_extremity: false,
47    };
48
49    let action = client.select_action(&local_resident, remote_digest, 0);
50    let triage_duration = start_triage.elapsed();
51    println!("Client triage completed in {triage_duration:?}");
52
53    match action {
54        rezzy::reconcile::ClientAction::BucketSketches { requests, .. } => {
55            println!("Requested {} buckets.", requests.len());
56
57            // Step 2: Server builds the requested sketches
58            let start_server = Instant::now();
59            let mut sorted_remote_h64: Vec<u64> = remote_elements.iter().map(|h| h.h64).collect();
60            sorted_remote_h64.sort_unstable();
61            let sketches = build_bucket_sketches(&sorted_remote_h64, &requests).unwrap();
62            let server_duration = start_server.elapsed();
63            println!("Server extraction completed in {server_duration:?}");
64
65            // Step 3: Client decodes the sketches
66            let start_decode = Instant::now();
67            let mut sorted_local_h64: Vec<u64> = local_elements.iter().map(|h| h.h64).collect();
68            sorted_local_h64.sort_unstable();
69            let local_sketches = build_bucket_sketches(&sorted_local_h64, &requests).unwrap();
70
71            let mut total_roots: usize = 0;
72            let mut failed_buckets: usize = 0;
73
74            for (mut remote_sketch, local_sketch) in sketches.into_iter().zip(local_sketches) {
75                remote_sketch.xor(&local_sketch).unwrap();
76
77                if let Ok(roots) = remote_sketch.decode_elements(remote_sketch.capacity()) {
78                    total_roots = total_roots.saturating_add(roots.len());
79                } else {
80                    failed_buckets = failed_buckets.saturating_add(1);
81                }
82            }
83
84            let decode_duration = start_decode.elapsed();
85            println!("Client decoding completed in {decode_duration:?}");
86            println!(
87                "Decode summary: {total_roots} roots recovered, {failed_buckets} buckets failed"
88            );
89        }
90        _ => {
91            println!("Client action: {action:?}");
92        }
93    }
94}
