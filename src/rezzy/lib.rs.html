3//! # Rezzy — Matrix State Resolution Engine
4//!
5//! A high-performance, spec-compliant implementation of [Matrix](https://spec.matrix.org/)
6//! state resolution versions **V1**, **V2**, **V2.1** ([MSC4297]), **V2.1.1**
7//! (experimental), and **V2.2** (experimental [MSC4242]) support.
8//!
9//! Rezzy is designed for correctness-first operation inside homeservers, bridges,
10//! and formal-verification toolchains. It runs in `#![no_std]` environments (with
11//! `alloc`) and optionally leverages SIMD-width bitmask sweeps for CDO filtering.
12//!
13//! ## Quick Start Example (Room V11 / State Res V2)
14//!
15//! ```rust,no_run
16//! use rezzy::{resolve_iterative_sort, LeanEvent, SharedState, StateResVersion, HashMap};
17//!
18//! // Build the unconflicted state (agreed upon by all forks).
19//! let unconflicted_state = SharedState::new();
20//!
21//! // Populate conflicted events and full auth context.
22//! let conflicted_subgraph: HashMap<String, LeanEvent> = HashMap::new();
23//! let auth_context: HashMap<String, LeanEvent> = HashMap::new();
24//!
25//! // Resolve the winning state.
26//! let resolved = resolve_iterative_sort(
27//!     unconflicted_state,
28//!     conflicted_subgraph,
29//!     &auth_context,
30//!     StateResVersion::V2,
31//!     &mut HashMap::new(),
32//! );
33//! ```
34//!
35//! ## Feature Flags
36//!
37//! | Feature     | Default | Description |
38//! |-------------|:-------:|-------------|
39//! | `std`       | ✓       | Enables `std::collections::{HashMap, HashSet}` and thread-parallel lattice resolution. |
40//! | `alloc`     | ✓       | Bare `alloc` support for `no_std` targets (implied by `std`). |
41//! | `cli`       | ✗       | Builds the `rezzy` CLI binary and the `merge` module. |
42//! | `hashing`   | ✗       | SHA-256 content-hashing for events missing an `event_id`. |
43//! | `mock-ruma` | ✗       | Enables Ruma SDK interop for upstream parity testing. |
44//! | `regen`     | ✗       | Builds the `regen_oracles` snapshot regeneration binary. |
45//!
46//! ## Spec References
47//!
48//! - [Matrix Spec — Server-Server API §3: State Resolution (V1)](https://spec.matrix.org/v1.13/server-server-api/#room-state-resolution)
49//! - [Matrix Spec — Room Versions](https://spec.matrix.org/v1.13/rooms/)
50//! - [MSC1693 — State Resolution V2][MSC1693]
51//! - [MSC4297 — State Resolution V2.1][MSC4297]
52//! - [MSC4242 — State DAGs (V2.2)][MSC4242]
53//!
54//! [MSC1693]: https://github.com/matrix-org/matrix-spec-proposals/pull/1693
55//! [MSC4297]: https://github.com/matrix-org/matrix-spec-proposals/pull/4297

113/// One resolved-state entry in `(type, state_key, event_id)` form.
114#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
115pub struct ResolvedStateEntry {
116    pub event_type: String,
117    pub state_key: String,
118    pub event_id: String,
119}
120
121/// Converts a resolved state map into a stable, sorted list of entries.
122#[must_use]
123pub fn resolved_state_entries<Id: basespec::rezzy_types::EventId>(
124    final_state_map: &crate::state::at::SharedState<Id>,
125) -> Vec<ResolvedStateEntry> {
126    let mut entries = final_state_map
127        .iter()
128        .map(|((event_type, state_key), event_id)| ResolvedStateEntry {
129            event_type: event_type.to_string(),
130            state_key: state_key.clone(),
131            event_id: event_id.to_string(),
132        })
133        .collect::<Vec<_>>();
134    entries.sort_by(|a, b| {
135        a.event_type
136            .cmp(&b.event_type)
137            .then_with(|| a.state_key.cmp(&b.state_key))
138    });
139    entries
140}

142/// Re-exported hashmap and hashset — uses `std::collections` when `std` is
143/// enabled, falls back to `hashbrown` for `no_std` targets.
144///
145/// All resolution functions are generic over `BuildHasher`, so this is
146/// purely a convenience for callers who don't need a specific hasher.
147#[cfg(feature = "std")]
148pub use std::collections::{HashMap, HashSet};
149
150/// See the `std` variant's documentation.
151#[cfg(not(feature = "std"))]
152pub use hashbrown::{HashMap, HashSet};
153
154/// Internal-only hashmap/hashset for maps rezzy builds and owns entirely
155/// (never a caller-supplied parameter or a public field), keyed with
156/// `hashbrown`'s `foldhash`-based `DefaultHashBuilder` instead of
157/// `std::collections`'s `SipHash13`.
158///
159/// State keys and event IDs handled by resolution originate from federation
160/// input an attacker can choose, so these maps still benefit from a randomized
161/// hasher. `foldhash` gives us a fast, minimally DoS-resistant, non-cryptographic
162/// default; that is appropriate here because the maps are internal-only, but we
163/// do not treat it as a security boundary. Not part of the public API:
164/// swapping the hasher on a caller-visible `HashMap<K, V, S>` parameter would
165/// silently break every external caller that passes a
166/// `std::collections::HashMap` there, since `S: BuildHasher` genericity only
167/// covers the hasher, not the underlying map type. See `benches/state_backend.rs`
168/// and the perf investigation this followed for how that was found out the hard
169/// way.
170pub(crate) type FastMap<K, V> = hashbrown::HashMap<K, V, hashbrown::DefaultHashBuilder>;
171
172/// See [`FastMap`]'s documentation.
173#[allow(dead_code)]
174pub(crate) type FastSet<K> = hashbrown::HashSet<K, hashbrown::DefaultHashBuilder>;

178mod tests {
179    use super::*;
180    use alloc::vec;
181
182    #[test]
183    fn resolved_state_entries_orders_by_type_then_state_key() {
184        let mut state: state::at::SharedState<String> = state::at::SharedState::new();
185        state.insert(("m.room.member".into(), "@bob:x".into()), "$b".into());
186        state.insert(("m.room.create".into(), String::new()), "$c".into());
187        state.insert(("m.room.member".into(), "@alice:x".into()), "$a".into());
188
189        let entries = resolved_state_entries(&state);
190        assert_eq!(
191            entries,
192            vec![
193                ResolvedStateEntry {
194                    event_type: "m.room.create".into(),
195                    state_key: String::new(),
196                    event_id: "$c".into(),
197                },
198                ResolvedStateEntry {
199                    event_type: "m.room.member".into(),
200                    state_key: "@alice:x".into(),
201                    event_id: "$a".into(),
202                },
203                ResolvedStateEntry {
204                    event_type: "m.room.member".into(),
205                    state_key: "@bob:x".into(),
206                    event_id: "$b".into(),
207                },
208            ]
209        );
210    }
