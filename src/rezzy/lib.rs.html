3//! # Rezzy — Matrix State Resolution Engine
4//!
5//! Spec-compliant implementation of Matrix state resolution versions
6//! **V1**, **V2**, **V2.1** ([MSC4297]), **V2.1.1**, and **V2.2** ([MSC4242]).
7//! Runs in `#![no_std]` environments with `alloc`.
8//!
9//! ## Feature Flags
10//!
11//! | Feature     | Default | Description |
12//! |-------------|:-------:|-------------|
13//! | `std`       | ✓       | Enables `std::collections::{HashMap, HashSet}` and thread-parallel lattice resolution. |
14//! | `cli`       | ✗       | Builds the `rezzy` CLI binary and merge utilities. |
15//! | `mock-ruma` | ✗       | Enables Ruma SDK interop for upstream parity testing. |
16//! | `regen`     | ✗       | Builds the `regen-oracles` snapshot regeneration binary. |
17//! | `signing`   | ✗       | Signature-verification traits (`SignatureVerifier` et al.), backend-agnostic. |
18//! | `signing-core` | ✗    | Backend-agnostic signature verification (`SignatureVerifier` et al.). |
19//! | `signing-consensus` | ✗   | `ed25519-zebra`-backed (ZIP 215) `SignatureVerifier` implementation. |
20//!
21//! Canonical-JSON SHA-256 hashing is always compiled in — see [`reference_hash`]
22//! and [`verify_content_hash`].

91/// Selects the presentation shape for resolved room data.
92///
93/// This is a library-level input so downstream callers can choose between
94/// timeline-oriented output and the raw resolved-state view without depending
95/// on the CLI binary.
96#[derive(Debug, Clone, Copy, Default)]
97pub enum OutputFormat {
98    #[default]
99    Events,
100    Default,
101    Deltas,
102    Federation,
103    Summary,
104    Timeline,
105    TimelineChronological,
106    ResolveState,
107    Hamt,
108}

110#[cfg(feature = "cli")]
111impl clap::ValueEnum for OutputFormat {
112    fn value_variants<'a>() -> &'a [Self] {
113        &[
114            Self::Events,
115            Self::Default,
116            Self::Deltas,
117            Self::Federation,
118            Self::Summary,
119            Self::Timeline,
120            Self::TimelineChronological,
121            Self::ResolveState,
122            Self::Hamt,
123        ]
124    }
125
126    fn to_possible_value(&self) -> Option<clap::builder::PossibleValue> {
127        Some(match self {
128            Self::Events => clap::builder::PossibleValue::new("events"),
129            Self::Default => clap::builder::PossibleValue::new("default"),
130            Self::Deltas => clap::builder::PossibleValue::new("deltas"),
131            Self::Federation => clap::builder::PossibleValue::new("federation"),
132            Self::Summary => clap::builder::PossibleValue::new("summary"),
133            Self::Timeline => clap::builder::PossibleValue::new("timeline"),
134            Self::TimelineChronological => {
135                clap::builder::PossibleValue::new("timeline-chronological")
136            }
137            Self::ResolveState => {
138                clap::builder::PossibleValue::new("resolve-state").alias("resolve_state")
139            }
140            Self::Hamt => clap::builder::PossibleValue::new("hamt"),
141        })
142    }
143}

145/// One resolved-state entry in `(type, state_key, event_id)` form.
146#[derive(Debug, PartialEq, Eq)]
147pub struct ResolvedStateEntry<Id = String, K = String> {
148    pub event_type: EventType,
149    pub state_key: K,
150    pub event_id: Id,
151}
152
153/// Converts a resolved state map into a stable, sorted list of entries.
154#[must_use]
155pub fn resolved_state_entries<Id, K>(
156    final_state_map: &crate::state::at::SharedState<Id, K>,
157) -> Vec<ResolvedStateEntry<Id, K>>
158where
159    Id: basespec::rezzy_types::EventId,
160    K: basespec::rezzy_types::StateKey,
161{
162    let mut entries = final_state_map
163        .iter()
164        .map(|((event_type, state_key), event_id)| ResolvedStateEntry {
165            event_type: event_type.clone(),
166            state_key: state_key.clone(),
167            event_id: event_id.clone(),
168        })
169        .collect::<Vec<_>>();
170    entries.sort_by(|a, b| {
171        a.event_type
172            .cmp(&b.event_type)
173            .then_with(|| a.state_key.cmp(&b.state_key))
174    });
175    entries
176}

178/// Re-exported hashmap and hashset — uses `std::collections` when `std` is
179/// enabled, falls back to `hashbrown` for `no_std` targets.
180///
181/// All resolution functions are generic over `BuildHasher`, so this is
182/// purely a convenience for callers who don't need a specific hasher.
183#[cfg(feature = "std")]
184pub use std::collections::{HashMap, HashSet};
185
186/// See the `std` variant's documentation.
187#[cfg(not(feature = "std"))]
188pub use hashbrown::{HashMap, HashSet};
189
190/// Internal-only hashmap/hashset for maps rezzy builds and owns entirely
191/// (never a caller-supplied parameter or a public field), keyed with
192/// `hashbrown`'s `foldhash`-based `DefaultHashBuilder` instead of
193/// `std::collections`'s `SipHash13`.
194///
195/// State keys and event IDs handled by resolution originate from federation
196/// input an attacker can choose, so these maps still benefit from a randomized
197/// hasher. `foldhash` gives us a fast, minimally DoS-resistant, non-cryptographic
198/// default; that is appropriate here because the maps are internal-only, but we
199/// do not treat it as a security boundary. Not part of the public API:
200/// swapping the hasher on a caller-visible `HashMap<K, V, S>` parameter would
201/// silently break every external caller that passes a
202/// `std::collections::HashMap` there, since `S: BuildHasher` genericity only
203/// covers the hasher, not the underlying map type. See `benches/state_backend.rs`
204/// and the perf investigation this followed for how that was found out the hard
205/// way.
206pub(crate) type FastMap<K, V> = hashbrown::HashMap<K, V, hashbrown::DefaultHashBuilder>;
207
208/// A hash set using rezzy's default randomized hasher.
209///
210/// This is public because the narrow-conflict override APIs accept it. Most
211/// callers should not need it: their state resolution entry point derives the
212/// conflicted-key set itself.
213pub type FastSet<K> = hashbrown::HashSet<K, hashbrown::DefaultHashBuilder>;

221    #[test]
222    fn resolved_state_entries_orders_by_type_then_state_key() {
223        let mut state: state::at::SharedState<String> = state::at::SharedState::new();
224        state.insert(("m.room.member".into(), "@bob:x".into()), "$b".into());
225        state.insert(("m.room.create".into(), String::new()), "$c".into());
226        state.insert(("m.room.member".into(), "@alice:x".into()), "$a".into());
227
228        let entries = resolved_state_entries(&state);
229        assert_eq!(
230            entries,
231            vec![
232                ResolvedStateEntry {
233                    event_type: EventType::RoomCreate,
234                    state_key: String::new(),
235                    event_id: "$c".into(),
236                },
237                ResolvedStateEntry {
238                    event_type: EventType::RoomMember,
239                    state_key: "@alice:x".into(),
240                    event_id: "$a".into(),
241                },
242                ResolvedStateEntry {
243                    event_type: EventType::RoomMember,
244                    state_key: "@bob:x".into(),
245                    event_id: "$b".into(),
246                },
247            ]
248        );
249    }
