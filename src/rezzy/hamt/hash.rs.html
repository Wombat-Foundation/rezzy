5/// A 256-bit structural hash for HAMT nodes.
6///
7/// This is a local storage/cache key used to skip identical subtrees within a
8/// caller-selected structural-key namespace; it is not a wire format. The
9/// structural key is included in both node identity and routing, so callers
10/// using distinct per-room keys intentionally produce disjoint node hashes.
11///
12/// The structural key separates the caller-selected namespaces and is included
13/// in both routing and node identity. Its public nature does not raise the
14/// cost of a collision within a namespace, so a full 256-bit digest is retained
15/// to provide a 128-bit generic collision-security margin.
16///
17/// # Threat model
18///
19/// The `structural_key` is the room's `room_id`: fully public, not a secret,
20/// and known in advance to anyone who can address the room. This is safe for
21/// HAMT routing because:
22///
23/// - **BLAKE3 is collision-resistant** at 128-bit security. Grinding
24///   a shallow-prefix collision (k levels of 5-bit agreement) costs
25///   `2^(5k)` hash evaluations; for k ≤ ~10 this is practical (seconds),
26///   but only causes O(depth) slowdown — the tree still terminates.
27/// - **Full-depth exhaustion** (52 levels = 2^260 hashes) is
28///   computationally infeasible.
29/// - `HamtBuildError::HashCollision` at max depth is a safe error return,
30///   not a panic or data corruption.
31///
32/// The threat model assumes:
33/// 1. The structural key is per-room (`room_id`), so precomputed collisions
34///    for one room cannot be transferred to another.
35/// 2. Callers of `build_hamt_with_key_hash` do **not** feed wire-derived
36///    path hashes through the custom `key_hash` closure without local
37///    re-keying via `key_path_hash(structural_key, key)`.
38///
39/// Because the key is public and fixed to `room_id`, an attacker can precompute
40/// shallow collisions against a known room. That residual cost (seconds of compute
41/// for a handful of extra tree levels) is bounded and accepted; defending against
42/// deliberate depth spam within a single room is an admission / rate-limiting
43/// concern at the homeserver level.
44pub type StructuralHash = [u8; 32];
45
46/// A 32-byte state-group identifier derived from the full root lattice.
47///
48/// This is the cross-server, deduplicable identifier for a resolved root. It
49/// must not be confused with the local-only `StructuralHash`.
50pub type StateGroupId = [u8; 32];

57/// Why a JSON document is not a valid [`RootHandle`].
58#[derive(Debug, Clone, Copy, PartialEq, Eq)]
59pub enum RootHandleJsonError {
60    /// The document is not a JSON object.
61    NotAnObject,
62    /// A required field is missing or has the wrong shape.
63    InvalidField(&'static str),
64}
65
66/// A resolved root handle carrying the local structural hash, global state-group identifier,
67/// and explicit codec/routing version metadata.
68///
69/// # Persistence contract
70///
71/// `RootHandle` is designed for **JSON persistence only**. Its `[u8; 32]` fields
72/// map naturally to JSON number arrays. External JSON adapters should default
73/// missing version metadata when reading documents written before those fields
74/// were added.
75///
76/// **Do not use bincode or other positional binary formats** with this struct.
77/// The field layout has changed since initial design (`StructuralHash` widened from
78/// `[u8; 16]` to `[u8; 32]`, version fields were prepended), and bincode's
79/// position-dependent decoding would silently misparse legacy payloads. If binary
80/// persistence is needed, use a versioned envelope with an explicit format tag.
81#[derive(Clone, PartialEq, Eq, Hash)]
82pub struct RootHandle {
83    pub codec_version: u8,
84    pub routing_version: u8,
85    pub routing_params: [u8; 4],
86    pub structural_hash: StructuralHash,
87    pub state_group_id: StateGroupId,
88}
89
90impl RootHandle {
91    /// Serializes to the JSON persistence form (byte arrays as number arrays).
92    #[must_use]
93    pub fn to_json(&self) -> rezzy_json::Value {
94        fn bytes(b: &[u8]) -> rezzy_json::Value {
95            rezzy_json::Value::Array(
96                b.iter()
97                    .map(|x| rezzy_json::Value::from(u64::from(*x)))
98                    .collect(),
99            )
100        }
101        rezzy_json::json!({
102            "codec_version": u64::from(self.codec_version),
103            "routing_version": u64::from(self.routing_version),
104            "routing_params": bytes(&self.routing_params),
105            "structural_hash": bytes(&self.structural_hash),
106            "state_group_id": bytes(&self.state_group_id),
107        })
108    }
109
110    /// Reads the JSON persistence form.
111    ///
112    /// Documents written before version metadata existed omit `codec_version`,
113    /// `routing_version` and `routing_params`; those default to codec 1,
114    /// routing 1 and zero params.
115    ///
116    /// # Errors
117    /// Returns an error if the value is not an object or a field is malformed.
118    pub fn from_json(value: &rezzy_json::Value) -> Result<Self, RootHandleJsonError> {
119        fn byte_array<const N: usize>(
120            value: &rezzy_json::Value,
121            name: &'static str,
122        ) -> Result<[u8; N], RootHandleJsonError> {
123            let items = value
124                .as_array()
125                .filter(|a| a.len() == N)
126                .ok_or(RootHandleJsonError::InvalidField(name))?;
127            let mut out = [0_u8; N];
128            for (slot, item) in out.iter_mut().zip(items) {
129                *slot = item
130                    .as_u64()
131                    .and_then(|v| u8::try_from(v).ok())
132                    .ok_or(RootHandleJsonError::InvalidField(name))?;
133            }
134            Ok(out)
135        }
136        fn version(
137            value: &rezzy_json::Value,
138            name: &'static str,
139            default: u8,
140        ) -> Result<u8, RootHandleJsonError> {
141            value.get(name).map_or(Ok(default), |v| {
142                v.as_u64()
143                    .and_then(|v| u8::try_from(v).ok())
144                    .ok_or(RootHandleJsonError::InvalidField(name))
145            })
146        }
147        if value.as_object().is_none() {
148            return Err(RootHandleJsonError::NotAnObject);
149        }
150        let required = |name: &'static str| {
151            value
152                .get(name)
153                .ok_or(RootHandleJsonError::InvalidField(name))
154        };
155        Ok(Self {
156            codec_version: version(value, "codec_version", 1)?,
157            routing_version: version(value, "routing_version", 1)?,
158            routing_params: match value.get("routing_params") {
159                Some(v) => byte_array(v, "routing_params")?,
160                None => [0; 4],
161            },
162            structural_hash: byte_array(required("structural_hash")?, "structural_hash")?,
163            state_group_id: byte_array(required("state_group_id")?, "state_group_id")?,
164        })
165    }
166
167    /// Builds a root handle with the current codec and v1 routing from a precomputed
168    /// structural hash and a state lattice.
169    #[must_use]
170    pub fn from_lthash(
171        structural_hash: StructuralHash,
172        lattice: &crate::incremental::LtHash,
173    ) -> Self {
174        Self::with_versions(
175            HAMT_CODEC_VERSION,
176            HAMT_ROUTING_VERSION,
177            [0; 4],
178            structural_hash,
179            lattice,
180        )
181    }
182
183    /// Builds a root handle with explicit codec and routing versioning.
184    #[must_use]
185    pub fn with_versions(
186        codec_version: u8,
187        routing_version: u8,
188        routing_params: [u8; 4],
189        structural_hash: StructuralHash,
190        lattice: &crate::incremental::LtHash,
191    ) -> Self {
192        Self {
193            codec_version,
194            routing_version,
195            routing_params,
196            structural_hash,
197            state_group_id: state_group_id_from_lthash(lattice),
198        }
199    }
200}

202/// The structural-key-prefixed BLAKE3 builder.
203///
204/// BLAKE3's digest is a fixed 32 bytes, which is the width the persisted
205/// structural hash already used, so there is no truncation step and no
206/// parameterized digest type to carry around.
207pub(crate) struct StructuralHashBuilder(blake3::Hasher);
208
209impl StructuralHashBuilder {
210    pub(crate) fn new(key: &[u8]) -> Self {
211        let mut hasher = blake3::Hasher::new();
212        hasher.update(&(key.len() as u64).to_le_bytes());
213        hasher.update(key);
214        Self(hasher)
215    }
216
217    pub(crate) fn finalize(self) -> StructuralHash {
218        *self.0.finalize().as_bytes()
219    }
220}
221
222impl Hasher for StructuralHashBuilder {
223    fn finish(&self) -> u64 {
224        0
225    }
226
227    fn write(&mut self, bytes: &[u8]) {
228        self.0.update(bytes);
229    }
230}

232/// Computes the 32-byte state-group identifier from the full resolved lattice.
233///
234/// This uses the `LtHash` digest, which is `BLAKE3(lattice)`.
235#[must_use]
236pub fn state_group_id_from_lthash(lattice: &crate::incremental::LtHash) -> StateGroupId {
237    lattice.digest()
238}

248        let handle = RootHandle {
249            codec_version: HAMT_CODEC_VERSION,
250            routing_version: HAMT_ROUTING_VERSION,
251            routing_params: [0; 4],

255        let mut set = HashSet::new();
256        set.insert(handle.clone());
257        assert!(set.contains(&handle));
258    }

265            routing_params: [1, 2, 3, 4],
266            structural_hash: [7; 32],

272    #[test]
273    fn root_handle_legacy_json_defaults_version_metadata() {
274        let legacy = rezzy_json::Value::parse(&std::format!(
275            r#"{{"structural_hash":{:?},"state_group_id":{:?}}}"#,
276            [1; 32],
277            [2; 32]
278        ))
279        .unwrap();
280        let handle = RootHandle::from_json(&legacy).unwrap();
281        assert_eq!(handle.codec_version, 1);
282        assert_eq!(handle.routing_version, 1);
283        assert_eq!(handle.routing_params, [0; 4]);
284        assert_eq!(handle.structural_hash, [1; 32]);
285        assert!(RootHandle::from_json(&rezzy_json::json!([])).is_err());
286    }
