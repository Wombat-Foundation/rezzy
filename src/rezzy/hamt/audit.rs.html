1//! Reachability audit for content-addressed HAMT storage.
2//!
3//! This module answers exactly one question, in-memory and storage-agnostic:
4//! given a set of live roots and the full universe of node hashes a storage
5//! backend currently holds, which of those node hashes are *not* reachable
6//! from any root?
7//!
8//! It deliberately does not decide what to do with the answer. Root-set
9//! completeness (is the caller's root list actually every live root?),
10//! scan/snapshot consistency (was `universe` collected consistently with the
11//! roots?), and any notion of quarantine, age cutoffs, or hard deletion are
12//! the storage backend's responsibility, not this module's. Treat the
13//! `unreachable` side of a [`ReachabilityAudit`] as a candidate list for
14//! further safety checks, never as a delete list on its own.
15

28/// [`IndexedUniverse::try_build`] was given more than `u32::MAX` distinct
29/// hashes, so no dense index could be assigned to all of them.
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub struct UniverseTooLarge {
32    /// The number of distinct hashes that overflowed a `u32` index.
33    pub distinct_count: usize,
34}
35
36impl fmt::Display for UniverseTooLarge {
37    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
38        write!(
39            f,
40            "universe has {} distinct hashes, more than u32::MAX can index",
41            self.distinct_count
42        )
43    }
44}
45
46#[cfg(feature = "std")]
47impl std::error::Error for UniverseTooLarge {}

49/// A `universe` of node hashes assigned dense `u32` indexes, in the order the
50/// hashes were given.
51///
52/// This is the compaction step a `RoaringBitmap`-backed audit needs:
53/// `StructuralHash` (16 bytes, high-entropy, not locally dense) cannot be
54/// used as a roaring index directly, so every hash in `universe` is given a
55/// stable position instead. Identity always resolves back through
56/// [`Self::hash_at`]/`hashes` to the full hash — the dense index is a
57/// local, single-call addressing scheme, not an identifier of its own.
58#[derive(Debug, Clone, PartialEq, Eq)]
59pub struct IndexedUniverse {
60    /// `hashes[i]` is the `StructuralHash` assigned to dense index `i`.
61    hashes: Vec<StructuralHash>,
62    index_by_hash: HashMap<StructuralHash, u32>,
63}
64
65impl IndexedUniverse {
66    /// Assigns each hash in `universe` a dense `u32` index, first-seen order.
67    /// Duplicate hashes collapse onto the same index.
68    ///
69    /// # Errors
70    /// Returns [`UniverseTooLarge`] if `universe` contains more than
71    /// `u32::MAX` distinct hashes.
72    ///
73    /// # Panics
74    /// Panics only if the internal overflow guard is violated and a dense
75    /// index can no longer be represented as `u32`.
76    pub fn try_build(
77        universe: impl IntoIterator<Item = StructuralHash>,
78    ) -> Result<Self, UniverseTooLarge> {
79        let mut hashes: Vec<StructuralHash> = Vec::new();
80        let mut index_by_hash: HashMap<StructuralHash, u32> = HashMap::new();
81        for hash in universe {
82            if index_by_hash.contains_key(&hash) {
83                continue;
84            }
85            if hashes.len() >= u32::MAX as usize {
86                return Err(UniverseTooLarge {
87                    distinct_count: hashes.len().saturating_add(1),
88                });
89            }
90            let idx = u32::try_from(hashes.len()).expect(
91                "IndexedUniverse::try_build bounds-checks the dense index before converting it",
92            );
93            index_by_hash.insert(hash, idx);
94            hashes.push(hash);
95        }
96        Ok(Self {
97            hashes,
98            index_by_hash,
99        })
100    }
101
102    /// The number of distinct hashes indexed.
103    #[must_use]
104    pub fn len(&self) -> usize {
105        self.hashes.len()
106    }
107
108    /// True if no hashes are indexed.
109    #[must_use]
110    pub fn is_empty(&self) -> bool {
111        self.hashes.is_empty()
112    }
113
114    /// The dense index assigned to `hash`, if it was part of `universe`.
115    #[must_use]
116    pub fn index_of(&self, hash: &StructuralHash) -> Option<u32> {
117        self.index_by_hash.get(hash).copied()
118    }
119
120    /// The full hash assigned to dense index `idx`, if in range.
121    #[must_use]
122    pub fn hash_at(&self, idx: u32) -> Option<StructuralHash> {
123        self.hashes.get(idx as usize).copied()
124    }
125
126    /// All indexed hashes, in dense-index order.
127    #[must_use]
128    pub fn hashes(&self) -> &[StructuralHash] {
129        &self.hashes
130    }
131}

133/// The result of a multi-root reachability audit, expressed as
134/// [`RoaringBitmap`]s over an [`IndexedUniverse`] rather than as
135/// `StructuralHash` collections.
136///
137/// Use this instead of [`ReachabilityAudit`] when the caller needs to keep
138/// many audits in memory, diff them, or intersect/union them repeatedly —
139/// operations `RoaringBitmap` is built for and a `HashSet<StructuralHash>`
140/// is not. `universe` is the only place `StructuralHash` identity lives;
141/// `reachable`/`unreachable` are addressed purely through its dense indexes.
142#[cfg(feature = "std")]
143#[derive(Debug, Clone, PartialEq)]
144pub struct BitmapReachabilityAudit {
145    /// Dense-index universe backing the bitmap partitions.
146    pub universe: IndexedUniverse,
147    /// Bitmap of dense indices reachable from at least one root.
148    pub reachable: RoaringBitmap,
149    /// Bitmap of dense indices not reachable from any root.
150    pub unreachable: RoaringBitmap,
151}
152
153/// Errors from [`bitmap_reachability_audit`]: either the traversal itself
154/// failed, or `universe` could not be given a dense index.
155#[cfg(feature = "std")]
156#[derive(Debug, Clone, PartialEq)]
157pub enum BitmapAuditError<E> {
158    /// `universe` had more than `u32::MAX` distinct hashes.
159    Universe(UniverseTooLarge),
160    /// The reachability walk itself failed.
161    Traversal(HamtTraversalError<E>),
162}
163
164#[cfg(feature = "std")]
165impl<E: fmt::Display> fmt::Display for BitmapAuditError<E> {
166    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
167        match self {
168            Self::Universe(err) => write!(f, "{err}"),
169            Self::Traversal(err) => write!(f, "{err}"),
170        }
171    }
172}
173
174#[cfg(feature = "std")]
175impl<E> std::error::Error for BitmapAuditError<E>
176where
177    E: std::error::Error + fmt::Debug + 'static,
178{
179    fn source(&self) -> Option<&(dyn std::error::Error + 'static)> {
180        match self {
181            Self::Universe(err) => Some(err),
182            Self::Traversal(err) => <HamtTraversalError<E> as std::error::Error>::source(err),
183        }
184    }
185}

187#[cfg(feature = "std")]
188impl<E> From<UniverseTooLarge> for BitmapAuditError<E> {
189    fn from(err: UniverseTooLarge) -> Self {
190        Self::Universe(err)
191    }
192}

194#[cfg(feature = "std")]
195impl<E> From<HamtTraversalError<E>> for BitmapAuditError<E> {
196    fn from(err: HamtTraversalError<E>) -> Self {
197        Self::Traversal(err)
198    }

201/// Partitions `universe` into reachable/unreachable [`RoaringBitmap`]s over a
202/// freshly built [`IndexedUniverse`].
203///
204/// Same traversal and semantics as [`reachability_audit`], but marks
205/// directly into a `RoaringBitmap` via `universe`'s dense index instead of
206/// accumulating a `HashSet<StructuralHash>` mark set first — this is the
207/// version worth using when the caller actually wants the roaring
208/// representation, not `reachability_audit`'s result reshaped afterward.
209/// Hashes the walk reaches that are outside `universe` are marked but never
210/// materialize a bitmap entry, matching `reachability_audit`'s handling of
211/// the same case.
212///
213/// # Errors
214/// Returns [`BitmapAuditError::Universe`] if `universe` has more than
215/// `u32::MAX` distinct hashes, or [`BitmapAuditError::Traversal`] on the
216/// same conditions as [`reachability_audit`].
217///
218/// # Panics
219/// Does not panic on any caller-controlled input — an oversized `universe`
220/// is reported as [`BitmapAuditError::Universe`], not a panic. The one
221/// `.expect()` in this function's body re-derives `universe.len()` as a
222/// `u32` for bitmap construction, which [`IndexedUniverse::try_build`]
223/// (called just above it, and propagated with `?` on failure) already
224/// guarantees fits.
225#[cfg(feature = "std")]
226pub fn bitmap_reachability_audit<K, V, F, E>(
227    roots: impl IntoIterator<Item = Arc<HamtNode<K, V>>>,
228    universe: impl IntoIterator<Item = StructuralHash>,
229    resolver: &mut F,
230) -> Result<BitmapReachabilityAudit, BitmapAuditError<E>>
231where
232    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
233{
234    let universe = IndexedUniverse::try_build(universe)?;
235
236    let mut reachable = RoaringBitmap::new();
237    // Hashes outside `universe` still need dedup so a subtree shared across
238    // roots (or reachable from inside and outside `universe`) is walked
239    // once, same as `reachability_audit`. `reachable`'s own membership
240    // check covers dedup for anything actually in `universe`, so this set
241    // only ever grows for hashes the caller's `universe` scan missed.
242    let mut visited_outside_universe: HashSet<StructuralHash> = HashSet::new();
243    for root in roots {
244        walk_reachable_node_hashes(&root, resolver, &mut |hash| {
245            if let Some(idx) = universe.index_of(&hash) {
246                reachable.insert(idx)
247            } else {
248                visited_outside_universe.insert(hash)
249            }
250        })
251        .map_err(BitmapAuditError::Traversal)?;
252    }
253
254    let universe_len = u32::try_from(universe.len())
255        .expect("IndexedUniverse::try_build already bounds-checked this");
256    let full_range: RoaringBitmap = (0..universe_len).collect();
257    let unreachable: RoaringBitmap = [&full_range, &reachable].difference();
258
259    Ok(BitmapReachabilityAudit {
260        universe,
261        reachable,
262        unreachable,
263    })
264}

266/// The result of a multi-root reachability audit: `universe` partitioned
267/// into hashes reachable from at least one of the audited roots and hashes
268/// that are not.
269///
270/// `reachable` is exposed (not just `unreachable`) so a caller can reuse the
271/// same mark set for a second-pass confirmation check right before acting
272/// on `unreachable` (e.g. `audit.reachable.contains(&hash)`), instead of
273/// re-deriving it downstream as `universe - unreachable` or re-walking the
274/// roots a second time.
275#[derive(Debug, Clone)]
276pub struct ReachabilityAudit {
277    /// Hashes in `universe` reachable from at least one audited root.
278    pub reachable: HashSet<StructuralHash>,
279    /// Hashes in `universe` reachable from none of the audited roots.
280    pub unreachable: Vec<StructuralHash>,
281}
282
283/// Partitions `universe` into hashes reachable from `roots` and hashes that
284/// are not.
285///
286/// Walks each root with [`walk_reachable_node_hashes`] against one shared
287/// `HashSet`, so subtrees shared between roots are resolved and walked only
288/// once in total. `resolver` is called once per distinct node hash
289/// encountered across all roots combined (not once per root), same as the
290/// underlying walk.
291///
292/// `reachable` is the full mark set intersected with `universe`; nodes the
293/// walk reaches that are outside `universe` (e.g. a caller's scan missed
294/// them) are not included in either field.
295///
296/// # Errors
297/// Returns [`HamtTraversalError::Resolve`] if `resolver` fails, or
298/// [`HamtTraversalError::MaxDepthExceeded`] if a walk recurses past the
299/// deepest depth a legitimately-built HAMT can have.
300///
301/// On error, the in-progress mark set may be partially populated (see
302/// [`walk_reachable_node_hashes`]'s error-recovery note); this function
303/// discards it and returns the error rather than a partial answer, so a
304/// caller never mistakes a partial reachable set for a complete one.
305pub fn reachability_audit<K, V, F, E>(
306    roots: impl IntoIterator<Item = Arc<HamtNode<K, V>>>,
307    universe: impl IntoIterator<Item = StructuralHash>,
308    resolver: &mut F,
309) -> Result<ReachabilityAudit, HamtTraversalError<E>>
310where
311    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
312{
313    let mut marked: HashSet<StructuralHash> = HashSet::new();
314    for root in roots {
315        walk_reachable_node_hashes(&root, resolver, &mut |hash| marked.insert(hash))?;
316    }
317
318    let mut reachable: HashSet<StructuralHash> = HashSet::new();
319    let mut unreachable: Vec<StructuralHash> = Vec::new();
320    let mut seen_universe: HashSet<StructuralHash> = HashSet::new();
321    for hash in universe {
322        if !seen_universe.insert(hash) {
323            continue;
324        }
325        if marked.contains(&hash) {
326            reachable.insert(hash);
327        } else {
328            unreachable.push(hash);
329        }
330    }
331
332    Ok(ReachabilityAudit {
333        reachable,
334        unreachable,
335    })
336}

338/// Computes the node hashes in `universe` that are not reachable from any of
339/// `roots`.
340///
341/// Convenience wrapper over [`reachability_audit`] for callers that only
342/// need the candidate list; see [`ReachabilityAudit`] if the reachable side
343/// is also useful (e.g. for a second-pass confirmation check).
344///
345/// # Errors
346/// See [`reachability_audit`].
347pub fn unreachable_node_hashes<K, V, F, E>(
348    roots: impl IntoIterator<Item = Arc<HamtNode<K, V>>>,
349    universe: impl IntoIterator<Item = StructuralHash>,
350    resolver: &mut F,
351) -> Result<Vec<StructuralHash>, HamtTraversalError<E>>
352where
353    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
354{
355    reachability_audit(roots, universe, resolver).map(|audit| audit.unreachable)
356}
