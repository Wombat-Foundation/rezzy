14type Polynomial = Vec<u64>;

16const FIELD_BITS: usize = 64;
17const MIXED_FACTOR_TRIALS: usize = 8;
18const FACTOR_TRIALS: usize = MIXED_FACTOR_TRIALS + FIELD_BITS;
19const FACTOR_PARAMETER_SEED: u64 = 0x9e37_79b9_7f4a_7c15;
20const TRACE_SQUARES: usize = 63;
21const MAX_FACTOR_WORK: usize = 8_000_000;
22
23pub(crate) fn decode(
24    odd_syndromes: &[u64],
25    max_elements: usize,
26) -> Result<Vec<u64>, AlgebraicError> {
27    let all = reconstruct_syndromes(odd_syndromes);
28    let mut locator = berlekamp_massey(&all, max_elements).ok_or(AlgebraicError::DecodeFailure)?;
29    if locator.len() == 1 {
30        return Ok(Vec::new());
31    }
32    locator.reverse();
33    let expected = locator
34        .len()
35        .checked_sub(1)
36        .ok_or(AlgebraicError::DecodeFailure)?;
37    let mut roots = Vec::with_capacity(expected);
38    find_roots(locator, &mut roots)?;
39    if roots.len() != expected || roots.contains(&0) {
40        return Err(AlgebraicError::DecodeFailure);
41    }
42    roots.sort_unstable();
43    Ok(roots)
44}

46fn reconstruct_syndromes(odd: &[u64]) -> Vec<u64> {
47    let mut all = vec![0; odd.len().saturating_mul(2)];
48    for (index, value) in odd.iter().copied().enumerate() {
49        all[index.saturating_mul(2)] = value;
50        // Earlier iterations have already reconstructed the source at `index`.
51        all[index.saturating_mul(2).saturating_add(1)] = gf64_mul(all[index], all[index]);
52    }
53    all
54}

56fn berlekamp_massey(syndromes: &[u64], max_degree: usize) -> Option<Polynomial> {
57    let mut current = vec![1];
58    let mut previous = vec![1];
59    let mut previous_discrepancy = 1;
60
61    for (n, syndrome) in syndromes.iter().copied().enumerate() {
62        let mut discrepancy = syndrome;
63        for (i, coefficient) in current.iter().copied().enumerate().skip(1) {
64            let Some(syndrome_index) = n.checked_sub(i) else {
65                continue;
66            };
67            discrepancy ^= gf64_mul(syndromes[syndrome_index], coefficient);
68        }
69        if discrepancy == 0 {
70            continue;
71        }
72
73        let current_degree = current.len().checked_sub(1)?;
74        let previous_degree = previous.len().checked_sub(1)?;
75        let shift = n
76            .checked_add(1)?
77            .checked_sub(current_degree)?
78            .checked_sub(previous_degree)?;
79        let swap = current_degree.checked_mul(2)? <= n;
80        let old_current = current.clone();
81        if swap {
82            let new_len = previous.len().checked_add(shift)?;
83            if new_len.checked_sub(1)? > max_degree {
84                return None;
85            }
86            current.resize(new_len, 0);
87        }
88        let scale = gf64_mul(discrepancy, gf64_inv(previous_discrepancy)?);
89        for (i, coefficient) in previous.iter().copied().enumerate() {
90            let target = i.checked_add(shift)?;
91            current[target] ^= gf64_mul(scale, coefficient);
92        }
93        if swap {
94            previous = old_current;
95            previous_discrepancy = discrepancy;
96        }
97    }
98    (current.last().copied().unwrap_or(0) != 0).then_some(current)
99}

101fn gf64_inv(value: u64) -> Option<u64> {
102    if value == 0 {
103        return None;
104    }
105    // a^(2^64-2), using a fixed square-and-multiply schedule.
106    let mut result = 1;
107    let exponent = u64::MAX - 1;
108    for bit in (0..64).rev() {
109        result = gf64_mul(result, result);
110        if exponent & (1_u64 << bit) != 0 {
111            result = gf64_mul(result, value);
112        }
113    }
114    Some(result)
115}

117fn trim(poly: &mut Polynomial) {
118    while poly.last() == Some(&0) {
119        poly.pop();
120    }
121}

123fn make_monic(poly: &mut Polynomial) -> Option<()> {
124    let leading = *poly.last()?;
125    if leading == 1 {
126        return Some(());
127    }
128    let inverse = gf64_inv(leading)?;
129    for coefficient in poly {
130        *coefficient = gf64_mul(*coefficient, inverse);
131    }
132    Some(())
133}

135fn poly_mod(modulus: &[u64], value: &mut Polynomial) -> Option<()> {
136    let modulus_degree = modulus.len().checked_sub(1)?;
137    if modulus.last() != Some(&1) {
138        return None;
139    }
140    while value.len() >= modulus.len() {
141        let term = value.pop()?;
142        if term != 0 {
143            let offset = value.len().checked_sub(modulus_degree)?;
144            let target = &mut value[offset..offset.checked_add(modulus_degree)?];
145            let source = &modulus[..modulus_degree];
146            let evaluator = crate::reconcile::gf64_simd::get_evaluator();
147
148            match evaluator {
149                #[cfg(all(target_arch = "x86_64", has_avx512_support))]
150                crate::reconcile::gf64_simd::EvaluatorBackend::Avx512 => {
151                    crate::reconcile::gf64_simd::Avx512Evaluator::poly_mac(term, source, target);
152                }
153                #[cfg(target_arch = "x86_64")]
154                crate::reconcile::gf64_simd::EvaluatorBackend::Sse => {
155                    crate::reconcile::gf64_simd::SseEvaluator::poly_mac(term, source, target);
156                }
157                crate::reconcile::gf64_simd::EvaluatorBackend::Scalar => {
158                    crate::reconcile::gf64_simd::ScalarEvaluator::poly_mac(term, source, target);
159                }
160            }
161        }
162    }
163    trim(value);
164    Some(())
165}

167fn poly_div(mut dividend: Polynomial, divisor: &[u64]) -> Option<Polynomial> {
168    if divisor.last() != Some(&1) || dividend.len() < divisor.len() {
169        return None;
170    }
171    let mut quotient = vec![0; dividend.len().checked_sub(divisor.len())?.checked_add(1)?];
172    let divisor_degree = divisor.len().checked_sub(1)?;
173    while dividend.len() >= divisor.len() {
174        let term = dividend.pop()?;
175        let position = dividend.len().checked_sub(divisor_degree)?;
176        quotient[position] = term;
177        if term != 0 {
178            let target = &mut dividend[position..position.checked_add(divisor_degree)?];
179            let source = &divisor[..divisor_degree];
180            let evaluator = crate::reconcile::gf64_simd::get_evaluator();
181
182            match evaluator {
183                #[cfg(all(target_arch = "x86_64", has_avx512_support))]
184                crate::reconcile::gf64_simd::EvaluatorBackend::Avx512 => {
185                    crate::reconcile::gf64_simd::Avx512Evaluator::poly_mac(term, source, target);
186                }
187                #[cfg(target_arch = "x86_64")]
188                crate::reconcile::gf64_simd::EvaluatorBackend::Sse => {
189                    crate::reconcile::gf64_simd::SseEvaluator::poly_mac(term, source, target);
190                }
191                crate::reconcile::gf64_simd::EvaluatorBackend::Scalar => {
192                    crate::reconcile::gf64_simd::ScalarEvaluator::poly_mac(term, source, target);
193                }
194            }
195        }
196    }
197    trim(&mut quotient);
198    Some(quotient)
199}

201fn poly_gcd(mut left: Polynomial, mut right: Polynomial) -> Option<Polynomial> {
202    if left.len() < right.len() {
203        core::mem::swap(&mut left, &mut right);
204    }
205    while !right.is_empty() {
206        make_monic(&mut right)?;
207        poly_mod(&right, &mut left)?;
208        core::mem::swap(&mut left, &mut right);
209    }
210    make_monic(&mut left)?;
211    Some(left)
212}

214fn poly_square(poly: &mut Polynomial) -> Option<()> {
215    if poly.is_empty() {
216        return Some(());
217    }
218    let new_len = poly.len().checked_mul(2)?.checked_sub(1)?;
219    let old = core::mem::take(poly);
220    poly.resize(new_len, 0);
221    for (target, coefficient) in (0..new_len).step_by(2).zip(old) {
222        poly[target] = gf64_mul(coefficient, coefficient);
223    }
224    Some(())
225}

227fn trace_mod(modulus: &[u64], parameter: u64) -> Option<Polynomial> {
228    let mut trace = vec![0, parameter];
229    for _ in 0..TRACE_SQUARES {
230        poly_square(&mut trace)?;
231        if trace.len() < 2 {
232            trace.resize(2, 0);
233        }
234        trace[1] = parameter;
235        poly_mod(modulus, &mut trace)?;
236    }
237    Some(trace)
238}

240const fn next_factor_parameter(mut state: u64) -> u64 {
241    state ^= state << 13;
242    state ^= state >> 7;
243    state ^ (state << 17)
244}

246fn solve_quadratic_form(target: u64) -> Option<u64> {
247    // Solve the GF(2)-linear map z -> z^2 + z using a reduced 64x65 matrix.
248    #[derive(Clone, Copy)]
249    struct Row {
250        coefficients: u64,
251        rhs: bool,
252    }
253
254    let mut rows = [Row {
255        coefficients: 0,
256        rhs: false,
257    }; 64];
258    for column in 0..64 {
259        let basis = 1_u64 << column;
260        let image = gf64_mul(basis, basis) ^ basis;
261        for (row, equation) in rows.iter_mut().enumerate() {
262            if image & (1_u64 << row) != 0 {
263                equation.coefficients |= 1_u64 << column;
264            }
265        }
266    }
267    for (row, equation) in rows.iter_mut().enumerate() {
268        if target & (1_u64 << row) != 0 {
269            equation.rhs = true;
270        }
271    }
272
273    let mut rank = 0;
274    for column in 0..64 {
275        let pivot = (rank..64).find(|row| rows[*row].coefficients & (1_u64 << column) != 0);
276        let Some(pivot) = pivot else { continue };
277        rows.swap(rank, pivot);
278        let pivot_row = rows[rank];
279        for (row, equation) in rows.iter_mut().enumerate() {
280            if row != rank && equation.coefficients & (1_u64 << column) != 0 {
281                equation.coefficients ^= pivot_row.coefficients;
282                equation.rhs ^= pivot_row.rhs;
283            }
284        }
285        rank = rank.checked_add(1)?;
286    }
287    if rows.iter().any(|row| row.coefficients == 0 && row.rhs) {
288        return None;
289    }
290    let mut solution = 0_u64;
291    for row in rows.iter().take(rank) {
292        if row.coefficients == 0 {
293            return None;
294        }
295        let pivot = row.coefficients.trailing_zeros();
296        if row.rhs {
297            solution |= 1_u64 << pivot;
298        }
299    }
300    (gf64_mul(solution, solution) ^ solution == target).then_some(solution)
301}

303fn find_roots(poly: Polynomial, roots: &mut Vec<u64>) -> Result<(), AlgebraicError> {
304    let mut work = MAX_FACTOR_WORK;
305    find_roots_with_budget(poly, roots, &mut work)
306}
307
308fn factor_trial_cost(degree: usize) -> Option<usize> {
309    degree.checked_mul(degree)?.checked_mul(TRACE_SQUARES)
310}
311
312fn find_roots_with_budget(
313    poly: Polynomial,
314    roots: &mut Vec<u64>,
315    work: &mut usize,
316) -> Result<(), AlgebraicError> {
317    let mut pending = vec![poly];
318    while let Some(poly) = pending.pop() {
319        let degree = poly
320            .len()
321            .checked_sub(1)
322            .ok_or(AlgebraicError::DecodeFailure)?;
323        if degree == 0 {
324            continue;
325        }
326        if degree == 1 {
327            roots.push(poly[0]);
328            continue;
329        }
330        if degree == 2 {
331            let linear = poly[1];
332            if linear == 0 {
333                return Err(AlgebraicError::DecodeFailure);
334            }
335            let inverse = gf64_inv(linear).ok_or(AlgebraicError::DecodeFailure)?;
336            let normalized = gf64_mul(poly[0], gf64_mul(inverse, inverse));
337            let root = gf64_mul(
338                solve_quadratic_form(normalized).ok_or(AlgebraicError::DecodeFailure)?,
339                linear,
340            );
341            roots.push(root);
342            roots.push(root ^ linear);
343            continue;
344        }
345
346        let mut split = None;
347        let mut parameter = FACTOR_PARAMETER_SEED;
348        for trial in 0..FACTOR_TRIALS {
349            if trial >= MIXED_FACTOR_TRIALS {
350                let basis_bit = trial
351                    .checked_sub(MIXED_FACTOR_TRIALS)
352                    .ok_or(AlgebraicError::DecodeFailure)?;
353                let basis_bit =
354                    u32::try_from(basis_bit).map_err(|_| AlgebraicError::DecodeFailure)?;
355                parameter = 1_u64
356                    .checked_shl(basis_bit)
357                    .ok_or(AlgebraicError::DecodeFailure)?;
358            }
359            let cost = factor_trial_cost(degree).ok_or(AlgebraicError::DecodeFailure)?;
360            *work = work
361                .checked_sub(cost)
362                .ok_or(AlgebraicError::BudgetExhausted)?;
363            let trace = trace_mod(&poly, parameter).ok_or(AlgebraicError::DecodeFailure)?;
364            let factor = poly_gcd(poly.clone(), trace).ok_or(AlgebraicError::DecodeFailure)?;
365            if factor.len() > 1 && factor.len() < poly.len() {
366                let quotient = poly_div(poly, &factor).ok_or(AlgebraicError::DecodeFailure)?;
367                split = Some((factor, quotient));
368                break;
369            }
370            if trial < MIXED_FACTOR_TRIALS {
371                parameter = next_factor_parameter(parameter);
372            }
373        }
374        let (factor, quotient) = split.ok_or(AlgebraicError::DecodeFailure)?;
375        pending.push(quotient);
376        pending.push(factor);
377    }
378    Ok(())
379}

387    fn inverses_roundtrip() {
388        for value in [1, 2, 3, 0xdead_beef, u64::MAX] {

393    #[test]
394    fn decodes_small_sets() {
395        for expected in [vec![1], vec![1, 2], vec![1, 2, 3], vec![1, 2, 3, 4]] {
396            let mut odd = vec![0; expected.len()];
397            for value in &expected {
398                let squared = gf64_mul(*value, *value);
399                let mut power = *value;
400                for syndrome in &mut odd {
401                    *syndrome ^= power;
402                    power = gf64_mul(power, squared);
403                }
404            }
405            assert_eq!(decode(&odd, expected.len()), Ok(expected));
406        }
407    }

421            let mut expected = Vec::new();
422            while expected.len() < size {
423                state ^= state << 13;
424                state ^= state >> 7;
425                state ^= state << 17;
426                if state != 0 && !expected.contains(&state) {
427                    expected.push(state);
428                }
429            }
430            expected.sort_unstable();
431            let mut odd = vec![0; size];
432            for value in &expected {
433                let squared = gf64_mul(*value, *value);
434                let mut power = *value;
435                for syndrome in &mut odd {
436                    *syndrome ^= power;
437                    power = gf64_mul(power, squared);
438                }
439            }
440            assert_eq!(decode(&odd, size), Ok(expected));
441        }
