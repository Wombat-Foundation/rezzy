15//! State delta compression — efficient incremental state storage/checkpoint chains.
16//!
17//! Instead of storing the full `state_map` at *every* event, homeservers can
18//! store a base snapshot and a chain of deltas.
19//! This module provides the primitives for computing and applying those deltas.

26/// Which phase of state resolution produced a delta.
27#[derive(Clone, Copy, PartialEq, Eq)]
28pub enum ResolvePhase {
29    /// Power events: `m.room.create`, `m.room.power_levels`, `m.room.join_rules`,
30    /// bans, and kicks. Sorted by reverse topological order (Kahn's algorithm).
31    Power,
32    /// Non-power events: everything else. Sorted by mainline proximity.
33    NonPower,
34}

36/// A per-event record of what changed in the resolved state during resolution.
37///
38/// [`resolve_iterative_sort_with_deltas`](crate::resolve_iterative_sort_with_deltas) emits one of
39/// these for every conflicted event that is auth-checked, regardless of whether
40/// it was accepted or rejected.
41pub struct ResolutionDelta<Id: crate::basespec::rezzy_types::EventId = String, K = String> {
42    /// The event that was auth-checked.
43    pub event_id: Id,
44    /// Whether the event passed the iterative auth check.
45    pub accepted: bool,
46    /// The `(event_type, state_key)` slot this event targets.
47    pub key: (crate::basespec::event_types::EventType, K),
48    /// The event ID that was previously in this slot (if any).
49    /// `None` if the slot was empty before this event.
50    pub replaced: Option<Id>,
51    /// Whether this was processed in the power phase or non-power phase.
52    pub phase: ResolvePhase,
53}

55/// A single state delta entry — an addition, modification, or deletion.
56#[derive(Debug, PartialEq, Eq)]
57pub struct StateDelta<Id, K = String>
58where
59    Id: crate::basespec::rezzy_types::EventId,
60    K: crate::basespec::rezzy_types::StateKey + Clone,
61{
62    /// The event type (e.g. `"m.room.member"`).
63    pub event_type: String,
64    /// The state key (e.g. `"@alice:example.com"` or `""`).
65    pub state_key: K,
66    /// The new event ID, or `None` if this key was deleted.
67    pub event_id: Option<Id>,
68}

70/// Computes the delta between a parent state and the current state.
71///
72/// Returns a list of [`StateDelta`] entries representing:
73/// - **Additions**: keys present in `current` but not in `parent`.
74/// - **Modifications**: keys present in both but with different event IDs.
75/// - **Deletions**: keys present in `parent` but not in `current`.
76///
77/// If the two states are identical, returns an empty `Vec`.
78#[must_use]
79pub fn compute_state_delta<Id, K>(
80    parent: &crate::state::at::SharedState<Id, K>,
81    current: &crate::state::at::SharedState<Id, K>,
82) -> Vec<StateDelta<Id, K>>
83where
84    Id: crate::basespec::rezzy_types::EventId,
85    K: crate::basespec::rezzy_types::StateKey + Clone,
86{
87    let mut deltas = Vec::new();
88
89    // Additions and modifications
90    for ((event_type, state_key), event_id) in current {
91        match parent.get(&(event_type.clone(), state_key.clone())) {
92            Some(parent_event_id) if parent_event_id == event_id => {}
93            _ => {
94                deltas.push(StateDelta {
95                    event_type: event_type.to_string(),
96                    state_key: state_key.clone(),
97                    event_id: Some(event_id.clone()),
98                });
99            }
100        }
101    }
102
103    // Deletions
104    for key in parent.keys() {
105        if !current.contains_key(key) {
106            deltas.push(StateDelta {
107                event_type: key.0.to_string(),
108                state_key: key.1.clone(),
109                event_id: None,
110            });
111        }
112    }
113
114    deltas
115}

117/// Applies a single delta entry to `state`: inserts `event_id` when present,
118/// otherwise removes the addressed slot.
119fn apply_delta_to_state<Id: crate::basespec::rezzy_types::EventId>(
120    state: &mut crate::state::at::SharedState<Id>,
121    delta: &StateDelta<Id>,
122) {
123    let key = (
124        crate::basespec::event_types::EventType::from(delta.event_type.as_str()),
125        delta.state_key.clone(),
126    );
127    if let Some(ref event_id) = delta.event_id {
128        state.insert(key, event_id.clone());
129    } else {
130        state.remove(&key);
131    }
132}

134/// Applies a list of deltas to a base state, producing the reconstructed state.
135///
136/// - Entries with `event_id = Some(id)` are inserted/overwritten.
137/// - Entries with `event_id = None` are removed from the base.
138#[must_use]
139pub fn apply_state_delta<Id: crate::basespec::rezzy_types::EventId>(
140    base: &crate::state::at::SharedState<Id>,
141    deltas: &[StateDelta<Id>],
142) -> crate::state::at::SharedState<Id> {
143    let mut result = base.clone();
144    for delta in deltas {
145        apply_delta_to_state(&mut result, delta);
146    }
147    result
148}

150/// Decodes a 64-character ASCII hex string into a 32-byte array.
151///
152/// Returns `Err` with a static message if the input is not exactly 64 ASCII
153/// hex characters. This rejects non-ASCII UTF-8 up front, preventing panics
154/// from slicing inside multibyte character boundaries.
155#[cfg(test)]
156fn decode_hex_32(s: &str) -> Result<[u8; 32], &'static str> {
157    if s.len() != 64 {
158        return Err("expected 64-character hex string");
159    }
160    if !s.is_ascii() {
161        return Err("hex string contains non-ASCII characters");
162    }
163    let mut bytes = [0u8; 32];
164    for (i, byte) in bytes.iter_mut().enumerate() {
165        let start = i.saturating_mul(2);
166        let end = start.saturating_add(2);
167        // SAFETY: s.is_ascii() guarantees every byte is a single-byte character,
168        // so slicing at any byte offset is always on a character boundary.
169        let byte_str = &s[start..end];
170        *byte = u8::from_str_radix(byte_str, 16).map_err(|_| "invalid hex character")?;
171    }
172    Ok(bytes)
173}

175/// Maximum number of delta hops before a full snapshot is inserted (default: 100,
176/// configurable: true).
177///
178/// Matches Synapse's `MAX_STATE_DELTA_HOPS`. When a delta chain would exceed
179/// this length, [`compute_compacted_delta_chain_from_resolved`] inserts a full
180/// base snapshot instead of another delta, bounding reconstruction cost to at
181/// most this many hops.
182pub const MAX_DELTA_CHAIN_HOPS: usize = 100;
183
184/// A checkpoint that may be either a delta from a parent or a full snapshot.
185///
186/// When chain compaction triggers (every [`MAX_DELTA_CHAIN_HOPS`] events),
187/// the checkpoint stores the full state map as `snapshot` instead of a delta.
188/// Readers walk backwards from any checkpoint, applying deltas, until they
189/// hit a snapshot.
190pub struct CompactedCheckpoint<Id: crate::basespec::rezzy_types::EventId = String> {
191    /// 256-bit hash of the state map at this point.
192    pub state_hash: [u8; 32],
193    /// Hash of the parent checkpoint (if any).
194    pub parent_hash: Option<[u8; 32]>,
195    /// The event ID that produced this checkpoint.
196    pub event_id: Id,
197    /// Deltas from the parent state. Empty when `snapshot` is `Some`.
198    pub deltas: Vec<StateDelta<Id>>,
199    /// Full state snapshot, present every [`MAX_DELTA_CHAIN_HOPS`] checkpoints.
200    /// When this is `Some`, `deltas` is empty and reconstruction starts here.
201    pub snapshot: Option<crate::state::at::SharedState<Id>>,
202}
203
204/// Builds a compacted delta chain from pre-resolved `(event_id, state_map)` pairs,
205/// inserting full snapshots every `max_hops` events to bound reconstruction cost.
206///
207/// This is the recommended API for homeserver state storage. The caller supplies
208/// pre-resolved states (from [`compute_state_at_batch`](crate::compute_state_at_batch)
209/// / [`resolve_iterative_sort`](crate::resolve_iterative_sort)), and this function handles only the
210/// storage compression concern.
211///
212/// A custom `max_hops` can be provided; pass `None` to use the default
213/// [`MAX_DELTA_CHAIN_HOPS`] (100).
214#[must_use]
215pub fn compute_compacted_delta_chain_from_resolved<Id: crate::basespec::rezzy_types::EventId>(
216    resolved_states: impl IntoIterator<Item = (Id, crate::state::at::SharedState<Id>)>,
217    max_hops: Option<usize>,
218) -> Vec<CompactedCheckpoint<Id>> {
219    let max_hops = max_hops.unwrap_or(MAX_DELTA_CHAIN_HOPS);
220    let mut checkpoints = Vec::new();
221    let mut prev_state: Option<crate::state::at::SharedState<Id>> = None;
222    let mut prev_hash: Option<[u8; 32]> = None;
223    let mut hops_since_snapshot: usize = 0;
224
225    for (event_id, state) in resolved_states {
226        let state_hash = compute_state_hash(&state);
227        let current_hops = hops_since_snapshot.saturating_add(1);
228
229        let (deltas, snapshot, recorded_hops) = match prev_state {
230            Some(ref base) if current_hops < max_hops => {
231                let deltas = compute_state_delta(base, &state);
232                (deltas, None, current_hops)
233            }
234            _ => {
235                // First event or chain exceeded max_hops — insert a
236                // full snapshot to reset the chain.
237                (Vec::new(), Some(state.clone()), 0)
238            }
239        };
240
241        checkpoints.push(CompactedCheckpoint {
242            state_hash,
243            parent_hash: prev_hash,
244            event_id,
245            deltas,
246            snapshot,
247        });
248
249        prev_hash = Some(state_hash);
250        prev_state = Some(state);
251        hops_since_snapshot = recorded_hops;
252    }
253
254    checkpoints
255}

257/// Reconstruct the full state map by looking up a checkpoint by event ID.
258///
259/// This is a convenience wrapper around [`reconstruct_state_at`] that finds
260/// the checkpoint index for the given `event_id` and reconstructs the state.
261///
262/// Returns `None` if the event ID is not found in the checkpoint chain or
263/// the chain is broken.
264#[must_use]
265pub fn reconstruct_state_at_by_event_id<Id: crate::basespec::rezzy_types::EventId>(
266    checkpoints: &[CompactedCheckpoint<Id>],
267    event_id: &Id,
268) -> Option<crate::state::at::SharedState<Id>> {
269    let idx = checkpoints.iter().position(|cp| &cp.event_id == event_id)?;
270    reconstruct_state_at(checkpoints, idx)
271}

273/// Reconstructs the full state map from a stored delta chain.
274///
275/// Walks `checkpoints` **backwards** from `target_index`, applying deltas
276/// until a snapshot is found. Returns the reconstructed state at that index.
277///
278/// # Arguments
279///
280/// * `checkpoints` — a slice of [`CompactedCheckpoint`]s in chronological order.
281/// * `target_index` — the index of the checkpoint to reconstruct.
282///
283/// # Returns
284///
285/// `Some(state_map)` if a snapshot base was found, `None` if the chain is
286/// broken (no snapshot ancestor exists).
287#[must_use]
288pub fn reconstruct_state_at<Id: crate::basespec::rezzy_types::EventId>(
289    checkpoints: &[CompactedCheckpoint<Id>],
290    target_index: usize,
291) -> Option<crate::state::at::SharedState<Id>> {
292    if target_index >= checkpoints.len() {
293        return None;
294    }
295
296    // Collect deltas by walking backwards until we hit a snapshot
297    let mut delta_stack: Vec<&[StateDelta<Id>]> = Vec::new();
298    let mut current_idx = target_index;
299
300    loop {
301        let cp = &checkpoints[current_idx];
302
303        if let Some(ref snapshot) = cp.snapshot {
304            // Found a base snapshot — apply accumulated deltas forward
305            let mut state = snapshot.clone();
306            while let Some(deltas) = delta_stack.pop() {
307                for delta in deltas {
308                    apply_delta_to_state(&mut state, delta);
309                }
310            }
311            return Some(state);
312        }
313
314        // No snapshot — push deltas and walk to parent
315        delta_stack.push(&cp.deltas);
316
317        let parent_hash = cp.parent_hash.as_ref()?;
318        // Prefer immediate predecessor; fall back to scanning earlier checkpoints.
319        if let Some(prev) = current_idx
320            .checked_sub(1)
321            .filter(|&pi| checkpoints[pi].state_hash == *parent_hash)
322        {
323            current_idx = prev;
324        } else {
325            current_idx = checkpoints[..current_idx]
326                .iter()
327                .rposition(|cp| cp.state_hash == *parent_hash)?;
328        }
329    }
330}

332/// Reconstructs the full state map at multiple checkpoint indices in one pass.
333///
334/// Instead of calling [`reconstruct_state_at`] N times (which re-walks shared
335/// ancestors), this function finds the nearest snapshot before the earliest
336/// target and walks forward once, capturing state at each requested index.
337///
338/// **Complexity**: `O(span)` where `span = max(targets) - snapshot_base`,
339/// versus `O(N * span)` for N individual calls.
340///
341/// # Arguments
342///
343/// * `checkpoints` — a slice of [`CompactedCheckpoint`]s in chronological order.
344/// * `target_indices` — the indices to reconstruct (duplicates and out-of-bounds
345///   are silently ignored).
346///
347/// # Returns
348///
349/// A persistent ordered map keyed by the requested indices. Missing
350/// entries indicate broken chains or out-of-bounds indices.
351///
352/// # Panics
353///
354/// Panics if an internal index increment overflows `usize` (should never
355/// happen with valid checkpoint data).
356#[must_use]
357pub fn reconstruct_state_batch<Id: crate::basespec::rezzy_types::EventId>(
358    checkpoints: &[CompactedCheckpoint<Id>],
359    target_indices: &[usize],
360) -> crate::state::PersistentOrdMap<usize, crate::state::at::SharedState<Id>> {
361    use crate::HashMap;
362
363    let mut sorted_targets: Vec<usize> = target_indices
364        .iter()
365        .copied()
366        .filter(|&i| i < checkpoints.len())
367        .collect();
368    sorted_targets.sort_unstable();
369    sorted_targets.dedup();
370
371    if sorted_targets.is_empty() {
372        return crate::state::PersistentOrdMap::new();
373    }
374
375    // Backward-only parent lookup: prefer immediate predecessor, fall back to
376    // scanning earlier checkpoints. This avoids the HashMap shadowing bug where
377    // duplicate hashes overwrite earlier entries.
378    let find_parent_idx = |idx: usize, parent_hash: &[u8; 32]| -> Option<usize> {
379        idx.checked_sub(1)
380            .filter(|&pi| checkpoints[pi].state_hash == *parent_hash)
381            .or_else(|| {
382                checkpoints[..idx]
383                    .iter()
384                    .rposition(|cp| cp.state_hash == *parent_hash)
385            })
386    };
387
388    // Walk backwards from all targets to find all required ancestors
389    let mut required_indices = alloc::collections::BTreeSet::new();
390    let mut queue: Vec<usize> = sorted_targets.clone();
391    while let Some(idx) = queue.pop() {
392        if required_indices.insert(idx) && checkpoints[idx].snapshot.is_none() {
393            if let Some(parent_hash) = &checkpoints[idx].parent_hash {
394                if let Some(p_idx) = find_parent_idx(idx, parent_hash) {
395                    queue.push(p_idx);
396                }
397            }
398        }
399    }
400
401    let mut known_states = HashMap::new();
402    let mut results = crate::state::PersistentOrdMap::new();
403
404    // Iterate forward only through the required indices
405    for idx in required_indices {
406        let cp = &checkpoints[idx];
407        let state = if let Some(ref snapshot) = cp.snapshot {
408            snapshot.clone()
409        } else if let Some(parent_hash) = &cp.parent_hash {
410            if let Some(p_idx) = find_parent_idx(idx, parent_hash) {
411                if let Some(parent_state) = known_states.get(&p_idx) {
412                    apply_state_delta(parent_state, &cp.deltas)
413                } else {
414                    #[cfg(feature = "std")]
415                    std::eprintln!(
416                        "state_delta: checkpoint {idx} parent {p_idx} not reconstructed"
417                    );
418                    continue; // BROKEN CHAIN: Parent was required but not reconstructed (gap in chain)
419                }
420            } else {
421                #[cfg(feature = "std")]
422                std::eprintln!(
423                    "state_delta: checkpoint {idx} has no earlier match for parent_hash {parent_hash:?}"
424                );
425                continue; // BROKEN CHAIN: No earlier checkpoint matches parent_hash
426            }
427        } else {
428            #[cfg(feature = "std")]
429            std::eprintln!(
430                "state_delta: checkpoint {idx} is an orphan (no snapshot, no parent_hash)"
431            );
432            continue; // BROKEN CHAIN: No snapshot and no parent_hash — orphan checkpoint
433        };
434
435        if sorted_targets.binary_search(&idx).is_ok() {
436            results.insert(idx, state.clone());
437        }
438        known_states.insert(idx, state);
439    }
440
441    results
442}

534    #[test]
535    fn test_compaction_inserts_snapshots() {
536        // NOTE: This test creates `O(N^2)` state items as it builds the DAG.
537        // `LtHash` computes a BLAKE3 XOF expansion per item, making large `N` values
538        // extremely slow in debug builds. We keep `N=35` to ensure fast tests.
539        // Build 35 pre-resolved states — should trigger snapshots at 0, 10, 20, 30
540        let states = build_states(35);
541
542        let checkpoints = compute_compacted_delta_chain_from_resolved(states, Some(10));
543        assert_eq!(checkpoints.len(), 35);
544
545        // First event always gets a snapshot (no parent)
546        assert!(
547            checkpoints[0].snapshot.is_some(),
548            "first event must have snapshot"
549        );
550
551        // Count snapshots
552        let snapshot_count = checkpoints
553            .iter()
554            .filter(|cp| cp.snapshot.is_some())
555            .count();
556        assert!(
557            snapshot_count >= 3,
558            "35 events with max_hops=10 should produce at least 3 snapshots, got {snapshot_count}"
559        );
560
561        // No delta chain should exceed max_hops
562        for (i, cp) in checkpoints.iter().enumerate() {
563            if cp.snapshot.is_some() {
564                continue;
565            }
566            // Walk back to find the nearest snapshot
567            let mut hops = 0;
568            let mut idx = i;
569            while checkpoints[idx].snapshot.is_none() {
570                hops += 1;
571                // Find parent by hash
572                let parent_hash = checkpoints[idx].parent_hash.as_ref().unwrap();
573                idx = checkpoints
574                    .iter()
575                    .position(|c| c.state_hash == *parent_hash)
576                    .unwrap();
577            }
578            assert!(
579                hops < 10,
580                "delta chain at index {i} has {hops} hops, exceeds max 10"
581            );
582        }
583    }

840    /// Regression: `hash_to_idx` `HashMap` shadowing — when duplicate hashes exist,
841    /// `.collect()` only stores the LAST index, erasing the valid earlier parent.
842    /// This test constructs a chain where `HASH_A` appears at indices 0 and 4, and
843    /// verifies that index 3 (whose parent is `HASH_A`) correctly resolves to
844    /// index 0, not silently fails because index 4 shadowed the lookup.
845    #[test]
846    fn test_batch_hash_shadowing_regression() {
847        let hash_a = [0xAA; 32];
848        let hash_b = [0xBB; 32];
849        let hash_c = [0xCC; 32];
850        let hash_d = [0xDD; 32];
851
852        let state_a = {
853            let mut s = StateMap::new();
854            s.insert(("m.room.create".into(), String::new()), "$c".into());
855            s
856        };
857
858        // Chain: [0:A] -> [1:B] -> [2:C] -> [3:D] -> [4:A]
859        // Index 3's parent_hash is HASH_C (index 2) — normal.
860        // Index 1's parent_hash is HASH_A — should resolve to index 0.
861        // Index 4's parent_hash is HASH_D (index 3) — normal.
862        // The trap: hash_to_idx["HASH_A"] would be 4 (last), not 0.
863        // Reconstructing index 1 needs HASH_A → index 0, but HashMap returns 4.
864        let checkpoints = alloc::vec![
865            CompactedCheckpoint::<String> {
866                state_hash: hash_a,
867                parent_hash: None,
868                event_id: "$0".into(),
869                deltas: alloc::vec![],
870                snapshot: Some(state_a.clone()),
871            },
872            CompactedCheckpoint::<String> {
873                state_hash: hash_b,
874                parent_hash: Some(hash_a),
875                event_id: "$1".into(),
876                deltas: alloc::vec![StateDelta {
877                    event_type: "m.room.topic".into(),
878                    state_key: String::new(),
879                    event_id: Some("$t1".into()),
880                }],
881                snapshot: None,
882            },
883            CompactedCheckpoint::<String> {
884                state_hash: hash_c,
885                parent_hash: Some(hash_b),
886                event_id: "$2".into(),
887                deltas: alloc::vec![StateDelta {
888                    event_type: "m.room.topic".into(),
889                    state_key: String::new(),
890                    event_id: Some("$t2".into()),
891                }],
892                snapshot: None,
893            },
894            CompactedCheckpoint::<String> {
895                state_hash: hash_d,
896                parent_hash: Some(hash_c),
897                event_id: "$3".into(),
898                deltas: alloc::vec![StateDelta {
899                    event_type: "m.room.topic".into(),
900                    state_key: String::new(),
901                    event_id: Some("$t3".into()),
902                }],
903                snapshot: None,
904            },
905            CompactedCheckpoint::<String> {
906                state_hash: hash_a, // Duplicate of index 0!
907                parent_hash: Some(hash_d),
908                event_id: "$4".into(),
909                deltas: alloc::vec![StateDelta {
910                    event_type: "m.room.topic".into(),
911                    state_key: String::new(),
912                    event_id: None, // deletion — reverts to state_a
913                }],
914                snapshot: None,
915            },
916        ];
917
918        // Batch reconstruct all — with HashMap shadowing, indices 1-3 would
919        // silently fail because HASH_A maps to index 4 (future), not index 0.
920        let results = reconstruct_state_batch(&checkpoints, &[0, 1, 2, 3, 4]);
921        assert_eq!(results.len(), 5, "all 5 checkpoints should reconstruct");
922        assert_eq!(results[&0], state_a, "index 0 (snapshot)");
923        assert_eq!(results[&4], state_a, "index 4 (reverted to state_a)");
924
925        // Also verify single-target reconstruction matches batch
926        for i in 0..5 {
927            let single = reconstruct_state_at(&checkpoints, i)
928                .unwrap_or_else(|| panic!("single reconstruct failed at {i}"));
929            assert_eq!(single, results[&i], "single vs batch mismatch at index {i}");
930        }
931    }

933    /// Coverage: exercises the `rposition` fallback in `reconstruct_state_at`
934    /// (lines 325-327) when the parent hash doesn't match the immediate
935    /// predecessor — e.g. after a fork/merge or gap in the chain.
936    #[test]
937    fn test_reconstruct_rposition_fallback() {
938        let mut state_a = StateMap::new();
939        state_a.insert(("m.room.create".into(), String::new()), "$c".into());
940        let hash_a = compute_state_hash(&state_a);
941
942        let mut state_b = StateMap::new();
943        state_b.insert(("m.room.create".into(), String::new()), "$c".into());
944        state_b.insert(("m.room.member".into(), "@alice:x".into()), "$j".into());
945        let hash_b = compute_state_hash(&state_b);
946
947        let mut state_c = StateMap::new();
948        state_c.insert(("m.room.create".into(), String::new()), "$c".into());
949        state_c.insert(("m.room.member".into(), "@bob:x".into()), "$k".into());
950        let hash_c = compute_state_hash(&state_c);
951
952        // Chain: [0]=snapshot(A), [1]=delta->A(B), [2]=delta->A(C)
953        // Index 2's parent_hash points to A (index 0), but index 1 has hash_b ≠ hash_a,
954        // so the immediate-predecessor check fails and rposition scans backwards.
955        let delta_c_from_a = compute_state_delta(&state_a, &state_c);
956        let checkpoints = alloc::vec![
957            CompactedCheckpoint::<String> {
958                state_hash: hash_a,
959                parent_hash: None,
960                event_id: "$e0".into(),
961                deltas: alloc::vec![],
962                snapshot: Some(state_a.clone()),
963            },
964            CompactedCheckpoint::<String> {
965                state_hash: hash_b,
966                parent_hash: Some(hash_a),
967                event_id: "$e1".into(),
968                deltas: compute_state_delta(&state_a, &state_b),
969                snapshot: None,
970            },
971            CompactedCheckpoint::<String> {
972                state_hash: hash_c,
973                parent_hash: Some(hash_a), // points to index 0, NOT index 1
974                event_id: "$e2".into(),
975                deltas: delta_c_from_a,
976                snapshot: None,
977            },
978        ];
979
980        // Reconstruct index 2 — must use rposition to find parent at index 0
981        let reconstructed = reconstruct_state_at(&checkpoints, 2)
982            .expect("should reconstruct via rposition fallback");
983        assert_eq!(reconstructed, state_c);
984
985        // Verify single-event lookup also works
986        let by_id = reconstruct_state_at_by_event_id(&checkpoints, &String::from("$e2"))
987            .expect("event ID lookup should work");
988        assert_eq!(by_id, state_c);
