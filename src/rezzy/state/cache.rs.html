15//! LRU cache for pre-constructed [`LeanEvent`]s, eliminating repeated
16//! deserialization and string copying on hot state resolution paths.
17//!
18//! # Motivation
19//!
20//! State resolution is a CPU-critical hot path in Matrix homeservers. Every
21//! resolution call typically requires converting native PDU types into
22//! [`LeanEvent`] structures — cloning strings for `event_type`, `sender`,
23//! `state_key`, and deep-copying JSON content. For active rooms, the same
24//! events are converted hundreds of times.
25//!
26//! `LeanEventCache` amortizes this cost to once-per-event by caching
27//! `Arc<LeanEvent>` with LRU eviction. It implements [`EventProvider`](crate::basespec::rezzy_types::EventProvider) so
28//! it plugs directly into [`resolve_state_maps_lazy_with_diff`](crate::resolve::multi::resolve_state_maps_lazy_with_diff).
29//!
30//! # Example
31//!
32//! ```rust,no_run
33//! use rezzy::state::cache::LeanEventCache;
34//! use rezzy::{LeanEvent, HashMap};
35//!
36//! let mut cache = LeanEventCache::<String>::new(10_000);
37//!
38//! // Insert a pre-built LeanEvent
39//! let event = LeanEvent {
40//!     event_id: "$abc".into(),
41//!     event_type: "m.room.member".into(),
42//!     state_key: Some("@alice:x".into()),
43//!     sender: "@alice:x".into(),
44//!     depth: 1,
45//!     ..Default::default()
46//! };
47//! cache.insert(event);
48//!
49//! // Retrieve (O(1) lookup, updates LRU order)
50//! let cached = cache.get("$abc").unwrap();
51//! assert_eq!(cached.sender, "@alice:x");
52//! ```
53

60/// A fixed-capacity LRU cache for pre-constructed [`LeanEvent`]s.
61///
62/// Events are stored as `Arc<LeanEvent<Id, C>>` for cheap cloning. When the
63/// cache exceeds capacity, the least-recently-used entry is evicted.
64///
65/// The cache is **not** internally synchronized — callers must wrap it in a
66/// `Mutex` or `RwLock` for concurrent access. This keeps the core `no_std`
67/// compatible while allowing `std` users to choose their synchronization
68/// strategy.
69///
70/// # Implementation
71///
72/// Uses a `HashMap` for O(1) key lookup plus a `BTreeMap<u64, Id>` side-index
73/// for **O(log n)** LRU eviction. The `BTreeMap` maps generation ticks to
74/// event IDs, so `pop_first()` instantly yields the least-recently-used entry.
75///
76/// # Interior mutability
77///
78/// The `BTreeMap` side-index is wrapped in [`RefCell`] and per-entry
79/// `last_access` fields use [`Cell<u64>`] so that the
80/// [`EventProvider`](crate::basespec::rezzy_types::EventProvider)
81/// implementation (which takes `&self`) can update LRU state. This ensures
82/// events accessed through the lazy resolver path are properly marked as
83/// recently used and not prematurely evicted.
84pub struct LeanEventCache<Id: EventId, C: EventContent = serde_json::Value> {
85    map: HashMap<Id, CacheEntry<Id, C>>,
86    /// Sorted index: generation → event ID. Enables O(log n) eviction via
87    /// `pop_first()`. Wrapped in `RefCell` for interior mutability through
88    /// `&self` (the `EventProvider` path).
89    access_order: RefCell<BTreeMap<u64, Id>>,
90    generation: Cell<u64>,
91    capacity: usize,
92    hits: Cell<u32>,
93    misses: Cell<u32>,
94    evictions: Cell<u32>,
95}
96
97struct CacheEntry<Id: EventId, C: EventContent> {
98    event: Arc<LeanEvent<Id, C>>,
99    last_access: Cell<u64>,
100}
101
102impl<Id: EventId, C: EventContent> LeanEventCache<Id, C> {
103    /// Creates a new cache with the given maximum capacity.
104    ///
105    /// # Panics
106    ///
107    /// Panics if `capacity` is zero.
108    #[must_use]
109    pub fn new(capacity: usize) -> Self {
110        assert!(capacity > 0, "LeanEventCache capacity must be > 0");
111        Self {
112            map: HashMap::with_capacity(capacity),
113            access_order: RefCell::new(BTreeMap::new()),
114            generation: Cell::new(0),
115            capacity,
116            hits: Cell::new(0),
117            misses: Cell::new(0),
118            evictions: Cell::new(0),
119        }
120    }
121
122    /// Returns the number of cached events.
123    #[must_use]
124    pub fn len(&self) -> usize {
125        self.map.len()
126    }
127
128    /// Returns `true` if the cache is empty.
129    #[must_use]
130    pub fn is_empty(&self) -> bool {
131        self.map.is_empty()
132    }
133
134    /// Returns the maximum capacity.
135    #[must_use]
136    pub fn capacity(&self) -> usize {
137        self.capacity
138    }
139
140    /// Returns a snapshot of the current cache statistics.
141    #[must_use]
142    pub fn stats(&self) -> CacheStats {
143        CacheStats {
144            hits: self.hits.get(),
145            misses: self.misses.get(),
146            evictions: self.evictions.get(),
147        }
148    }
149
150    /// Shared LRU bump — increments generation, updates `last_access`,
151    /// and rotates the `BTreeMap` side-index entry. Called from both
152    /// `touch` (returns `Arc`) and `EventProvider::get_event` (returns `&`).
153    fn bump_entry(&self, entry: &CacheEntry<Id, C>) {
154        let old_gen = entry.last_access.get();
155        let new_gen = self.generation.get().wrapping_add(1);
156        self.generation.set(new_gen);
157        entry.last_access.set(new_gen);
158
159        let mut order = self.access_order.borrow_mut();
160        order.remove(&old_gen);
161        order.insert(new_gen, entry.event.event_id.clone());
162    }
163
164    /// Internal LRU touch — bumps generation and updates `last_access`
165    /// plus the `BTreeMap` side-index through interior mutability, so it
166    /// works from both `&self` and `&mut self` contexts.
167    fn touch<Q>(&self, id: &Q) -> Option<Arc<LeanEvent<Id, C>>>
168    where
169        Id: core::borrow::Borrow<Q>,
170        Q: ?Sized + Eq + core::hash::Hash,
171    {
172        if let Some(entry) = self.map.get(id) {
173            self.bump_entry(entry);
174            self.hits.set(self.hits.get().saturating_add(1));
175            Some(Arc::clone(&entry.event))
176        } else {
177            self.misses.set(self.misses.get().saturating_add(1));
178            None
179        }
180    }
181
182    /// Looks up an event by ID, returning a shared reference and updating
183    /// the LRU generation.
184    pub fn get<Q>(&mut self, id: &Q) -> Option<Arc<LeanEvent<Id, C>>>
185    where
186        Id: core::borrow::Borrow<Q>,
187        Q: ?Sized + Eq + core::hash::Hash,
188    {
189        self.touch(id)
190    }
191
192    /// Inserts a `LeanEvent`, wrapping it in `Arc`. If the cache is at
193    /// capacity, the least-recently-used entry is evicted first.
194    ///
195    /// If an event with the same ID already exists, it is replaced.
196    pub fn insert(&mut self, event: LeanEvent<Id, C>) -> Arc<LeanEvent<Id, C>> {
197        let arc = Arc::new(event);
198        self.insert_arc(Arc::clone(&arc));
199        arc
200    }
201
202    /// Inserts a pre-wrapped `Arc<LeanEvent>`. Useful when the caller already
203    /// has an `Arc` (e.g., from a shared database layer).
204    pub fn insert_arc(&mut self, event: Arc<LeanEvent<Id, C>>) {
205        let r#gen = self.generation.get().wrapping_add(1);
206        self.generation.set(r#gen);
207        let id = event.event_id.clone();
208
209        // Replace existing entry — remove old generation from side-index
210        if let Some(existing) = self.map.get_mut(&id) {
211            let old_gen = existing.last_access.get();
212            self.access_order.borrow_mut().remove(&old_gen);
213            existing.event = event;
214            existing.last_access.set(r#gen);
215            self.access_order.borrow_mut().insert(r#gen, id);
216            return;
217        }
218
219        // New entry — evict if at capacity
220        if self.map.len() >= self.capacity {
221            self.evict_lru();
222        }
223
224        let entry = CacheEntry {
225            event,
226            last_access: Cell::new(r#gen),
227        };
228        self.access_order.borrow_mut().insert(r#gen, id.clone());
229        self.map.insert(id, entry);
230    }
231
232    /// Gets an event by ID, or inserts one created by the closure if missing.
233    ///
234    /// This is the primary API for homeserver integration — the closure
235    /// typically converts a native PDU type into a `LeanEvent`:
236    ///
237    /// ```rust,no_run
238    /// # use rezzy::state::cache::LeanEventCache;
239    /// # use rezzy::LeanEvent;
240    /// # let mut cache = LeanEventCache::<String>::new(100);
241    /// let event = cache.get_or_insert("$abc", || {
242    ///     // Convert from native PDU type
243    ///     LeanEvent {
244    ///         event_id: "$abc".into(),
245    ///         event_type: "m.room.member".into(),
246    ///         sender: "@alice:x".into(),
247    ///         state_key: Some("@alice:x".into()),
248    ///         depth: 1,
249    ///         ..Default::default()
250    ///     }
251    /// });
252    /// ```
253    ///
254    /// # Panics (debug only)
255    ///
256    /// In debug builds, panics if the closure returns a `LeanEvent` whose
257    /// `event_id` does not match the requested `id`.
258    pub fn get_or_insert<Q>(
259        &mut self,
260        id: &Q,
261        f: impl FnOnce() -> LeanEvent<Id, C>,
262    ) -> Arc<LeanEvent<Id, C>>
263    where
264        Id: core::borrow::Borrow<Q>,
265        Q: ?Sized + Eq + core::hash::Hash,
266    {
267        if let Some(arc) = self.get(id) {
268            return arc;
269        }
270        let event = f();
271        debug_assert!(
272            event.event_id.borrow() == id,
273            "get_or_insert: closure produced event with mismatched event_id"
274        );
275        self.insert(event)
276    }
277
278    /// Inserts a batch of events, returning a `HashMap` of `Arc`s.
279    ///
280    /// Events already in the cache are returned from cache (updating LRU).
281    /// Missing events are inserted.
282    pub fn insert_batch(
283        &mut self,
284        events: impl IntoIterator<Item = LeanEvent<Id, C>>,
285    ) -> HashMap<Id, Arc<LeanEvent<Id, C>>> {
286        let mut result = HashMap::new();
287        for event in events {
288            let id = event.event_id.clone();
289            let arc = match self.get(&id) {
290                Some(cached) => cached,
291                _ => self.insert(event),
292            };
293            result.insert(id, arc);
294        }
295        result
296    }
297
298    /// Removes all entries from the cache.
299    pub fn clear(&mut self) {
300        self.map.clear();
301        self.access_order.borrow_mut().clear();
302        self.generation.set(0);
303    }
304
305    /// Evicts the least-recently-used entry in **O(log n)** via the
306    /// `BTreeMap` side-index.
307    fn evict_lru(&mut self) {
308        if let Some((_gen, id)) = self.access_order.borrow_mut().pop_first() {
309            self.map.remove(&id);
310            self.evictions.set(self.evictions.get().saturating_add(1));
311        }
312    }
313}

315/// `LeanEventCache` implements [`EventProvider`](crate::basespec::rezzy_types::EventProvider) so it can be passed directly
316/// to [`resolve_state_maps_lazy_with_diff`](crate::resolve::multi::resolve_state_maps_lazy_with_diff).
317///
318/// Unlike a plain `HashMap` provider, this implementation updates the LRU
319/// generation for every access through interior mutability (`Cell` + `RefCell`),
320/// ensuring that events heavily used during lazy resolution are not prematurely
321/// evicted.
322impl<Id: EventId, C: EventContent> crate::basespec::rezzy_types::EventProvider<Id, C>
323    for LeanEventCache<Id, C>
324{
325    fn get_event(&self, id: &Id) -> Option<&LeanEvent<Id, C>> {
326        if let Some(entry) = self.map.get(id) {
327            self.bump_entry(entry);
328            self.hits.set(self.hits.get().saturating_add(1));
329            Some(entry.event.as_ref())
330        } else {
331            self.misses.set(self.misses.get().saturating_add(1));
332            None
333        }
334    }
335}

337/// Collects cache hit/miss statistics for monitoring.
338#[derive(Debug, Clone, Default)]
339pub struct CacheStats {
340    /// Number of cache hits.
341    pub hits: u32,
342    /// Number of cache misses.
343    pub misses: u32,
344    /// Number of evictions.
345    pub evictions: u32,
346}
347
348impl CacheStats {
349    /// Returns the hit rate as a percentage (0.0–100.0).
350    #[must_use]
351    pub fn hit_rate(&self) -> f64 {
352        let total = self.hits.saturating_add(self.misses);
353        if total == 0 {
354            return 0.0;
355        }
356        (f64::from(self.hits) / f64::from(total)) * 100.0
357    }
358}
