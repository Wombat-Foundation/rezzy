15//! State resolution entry point — the [`resolve_iterative_sort`] function.
16//!
17//! This module implements the full Matrix state resolution pipeline:
18//!
19//! 1. **CDO pre-filter** (V2.1.1 only): removes causally dominated events.
20//! 2. **Power phase**: classifies events as power vs. non-power, expands auth
21//!    chains, then iteratively auth-checks power events in reverse topological order.
22//! 3. **Non-power phase**: sorts remaining events by mainline distance and
23//!    iteratively auth-checks them against the progressively-built resolved state.
24//!
25//! For the lattice-coordinatized variant (parallel, `O(1)` projection), see
26//! [`crate::resolve::lattice::resolve_lattice_fold`].
27

37/// Prepares the conflicted events map and tracks original conflicted keys before CDO pre-filtering.
38pub(crate) fn prepare_conflicted_and_keys<
39    Id: crate::basespec::rezzy_types::EventId,
40    C: crate::basespec::rezzy_types::EventContent,
41    S1: core::hash::BuildHasher,
42    S2: core::hash::BuildHasher,
43    K,
44>(
45    conflicted_events: &mut HashMap<Id, LeanEvent<Id, C, K>, S1>,
46    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
47    version: StateResVersion,
48) -> alloc::collections::BTreeSet<Id>
49where
50    K: AsRef<str> + Clone,
51{
52    let original_conflicted_keys = conflicted_events.keys().cloned().collect();
53    if version == StateResVersion::V2_1_1 {
54        let filtered = crate::resolve::cdo::apply_cdo_filter(conflicted_events, auth_context);
55        conflicted_events.clear();
56        for (k, v) in filtered {
57            conflicted_events.insert(k, v);
58        }
59    }
60    original_conflicted_keys
61}

63/// Derives a genuine-conflicted-key set by treating every event in
64/// `conflicted_events` as genuinely conflicting (as opposed to being present
65/// only as auth-chain context for a *different* key's genuine conflict).
66///
67/// This is the default used by the public `resolve_iterative_sort*` entry
68/// points, which take a single flat `conflicted_events` map and have no way
69/// to know which of its entries came from a real per-key state-map diff
70/// versus a supplemental auth-diff/MSC4297-subgraph walk — so they preserve
71/// the pre-existing (pre-fix) behavior: nothing in `conflicted_events` is
72/// blocked from deciding its own key. Callers that *do* know the real
73/// distinction (`resolve_multiple_prev_states` in `state::at`, and
74/// `resolve_state_maps`/`resolve_state_maps_lazy_with_diff` in
75/// `resolve::multi`) bypass this default and call
76/// `resolve_iterative_sort_with_all_caches` directly with the narrower,
77/// real set captured *before* their own supplementation step.
78// TODO(perf): this calls `EventType::from(ev.event_type.as_str())` once per
79// conflicted event just to build the gate set, and the power/non-power
80// phase loops below call it *again* on the same event when actually
81// inserting into `resolved`. For well-known types (the common case:
82// create/power_levels/join_rules/member) that's just a few cheap string
83// comparisons, but for `EventType::Custom` it's a duplicate `Box<str>` heap
84// allocation per conflicted event. Fixing this properly means threading the
85// already-interned `EventType` alongside each event through
86// `route_power_events`/`power_events`/`non_power_events` (currently keyed by
87// `Id` only) instead of re-deriving it — a real (if narrow) restructure, not
88// attempted here since it's a constant-factor cost bounded by the conflicted
89// set size, not the full event set.
90pub(crate) fn derive_all_conflicted_keys<Id, C, S, K>(
91    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S>,
92) -> crate::FastSet<(EventType, K)>
93where
94    Id: crate::basespec::rezzy_types::EventId,
95    K: Clone + Default + Eq + core::hash::Hash,
96{
97    conflicted_events
98        .values()
99        .map(|ev| {
100            (
101                EventType::from(ev.event_type.as_str()),
102                ev.state_key.clone().unwrap_or_default(),
103            )
104        })
105        .collect()
106}

108/// State Resolution V2+ auth-chain expansion (room versions 2 - 11+, Spec [§State Resolution]).
109///
110/// After the initial power/non-power classification, this function recursively
111/// walks the `auth_events` of each power event. Any event found in the
112/// conflicted set (`sort_set`) is promoted from `non_power_events` to
113/// `power_events`. This ensures that the auth chain dependencies of power
114/// events are resolved in the correct (power) phase.
115///
116/// This is specified in the [V2 state resolution algorithm][v2-spec], Step 3,
117/// and applies to all versions that use V2-derived resolution: V2, V2.1, V2.1.1, and V2.2.
118///
119/// [v2-spec]: https://spec.matrix.org/v1.13/rooms/v2/#state-resolution
120/// Expands the auth chains for a set of V2 power events, building an auth context.
121pub fn expand_v2_power_events_auth_chains<
122    Id: crate::basespec::rezzy_types::EventId,
123    C: Clone,
124    S1: core::hash::BuildHasher,
125    S2: core::hash::BuildHasher,
126    S3: core::hash::BuildHasher,
127    K: Clone,
128>(
129    power_events: &mut HashMap<Id, LeanEvent<Id, C, K>, S1>,
130    non_power_events: &mut HashMap<Id, LeanEvent<Id, C, K>, S2>,
131    sort_set: &HashMap<Id, LeanEvent<Id, C, K>, S3>,
132) {
133    let mut queue: alloc::collections::VecDeque<Id> = power_events.keys().cloned().collect();
134    while let Some(id) = queue.pop_front() {
135        if let Some(ev) = sort_set.get(&id) {
136            for aid in &ev.auth_events {
137                if !power_events.contains_key(aid) {
138                    if let Some(aev) = sort_set.get(aid) {
139                        power_events.insert(aid.clone(), aev.clone());
140                        non_power_events.remove(aid);
141                        queue.push_back(aid.clone());
142                    }
143                }
144            }
145        }
146    }
147}

149/// MSC4297 (v2.1+): Routes administrative ancestral power events from `auth_context` into `power_events`.
150pub(crate) fn route_msc4297_ancestral_power_events<
151    Id: crate::basespec::rezzy_types::EventId,
152    C: crate::basespec::rezzy_types::EventContent + Clone,
153    S1: core::hash::BuildHasher,
154    S2: core::hash::BuildHasher,
155    K: Clone,
156>(
157    power_events: &mut HashMap<Id, LeanEvent<Id, C, K>, S1>,
158    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
159    original_conflicted_keys: &alloc::collections::BTreeSet<Id>,
160    version: StateResVersion,
161) {
162    if matches!(
163        version,
164        StateResVersion::V2_1 | StateResVersion::V2_1_1 | StateResVersion::V2_2
165    ) {
166        let mut conflicted_power_ancestry = alloc::collections::BTreeSet::new();
167        let mut queue = alloc::collections::VecDeque::new();
168        for ev in power_events.values() {
169            for aid in &ev.auth_events {
170                queue.push_back(aid.clone());
171            }
172        }
173        while let Some(aid) = queue.pop_front() {
174            if conflicted_power_ancestry.insert(aid.clone()) {
175                if let Some(aev) = auth_context.get(&aid) {
176                    for parent_id in &aev.auth_events {
177                        queue.push_back(parent_id.clone());
178                    }
179                }
180            }
181        }
182
183        for id in &conflicted_power_ancestry {
184            if original_conflicted_keys.contains(id) {
185                continue;
186            }
187            if let Some(ev) = auth_context.get(id) {
188                // NOTE: V2.1.1+ strictly isolates the supplemental merge to PLs and creates.
189                // V2.1 (MSC4297) supplemented `m.room.join_rules`, which inadvertently caused the
190                // Invite Lock bug (evaluating historical joins against newer invite-only rules).
191                let is_join_rules_allowed =
192                    ev.event_type == "m.room.join_rules" && version == StateResVersion::V2_1;
193
194                if ev.event_type == "m.room.power_levels"
195                    || ev.event_type == "m.room.create"
196                    || is_join_rules_allowed
197                {
198                    power_events.insert(id.clone(), ev.clone());
199                }
200            }
201        }
202    }
203}

205/// Runs the sequential power phase iterative auth checks to establish the authoritative administrative framework.
206#[allow(clippy::too_many_arguments)]
207pub(crate) fn run_power_phase_iterative_checks<Id, C, S2, S3, S4, K>(
208    resolved: &mut crate::state::at::SharedState<Id, K>,
209    power_events: &HashMap<Id, LeanEvent<Id, C, K>, S4>,
210    sort_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, LeanEvent<Id, C, K>>,
211    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
212    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S3>,
213    version: StateResVersion,
214    local_auth_cache: &mut LocalAuthCache<Id, C, K>,
215    create_ev: Option<&LeanEvent<Id, C, K>>,
216    pl_cache: &mut HashMap<Id, i64>,
217    conflicted_keys: &crate::FastSet<(EventType, K)>,
218) where
219    Id: crate::basespec::rezzy_types::EventId,
220    S2: core::hash::BuildHasher,
221    S3: core::hash::BuildHasher,
222    S4: core::hash::BuildHasher,
223    C: crate::basespec::rezzy_types::EventContent,
224    K: crate::basespec::rezzy_types::StateKey,
225    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
226{
227    let sorted_power_ids = lean_kahn_sort(power_events, sort_context, create_ev, version, pl_cache);
228    for id in &sorted_power_ids {
229        if let Some(event) = conflicted_events.get(id).or_else(|| auth_context.get(id)) {
230            let local_auth = compute_local_auth(
231                event,
232                auth_context,
233                conflicted_events,
234                local_auth_cache,
235                version,
236            );
237            if iterative_auth_ok(
238                event,
239                resolved,
240                auth_context,
241                conflicted_events,
242                local_auth,
243                create_ev,
244                version,
245                true,
246            ) {
247                if let Some(sk) = &event.state_key {
248                    let key = (EventType::from(event.event_type.as_str()), sk.clone());
249                    // Only a genuinely conflicted key may be decided by the
250                    // power phase. `power_events` can also contain events
251                    // pulled in purely as auth-chain context (the
252                    // `auth(C) \ auth(U)` supplement, or the MSC4297
253                    // conflicted subgraph) — those exist so *other*, actually
254                    // conflicting events' auth chains can be validated, not
255                    // so their own (possibly stale, superseded) key can win
256                    // over a value every merge parent already agreed on.
257                    if !conflicted_keys.contains(&key) {
258                        continue;
259                    }
260                    resolved.insert(key, event.event_id.clone());
261                }
262            }
263        }
264    }
265}

267/// Returns the starting point for state resolution based on the algorithm version.
268/// V1 and V2 inherit the unconflicted state as their base, whereas V2.1+ starts from an empty set.
269pub(crate) fn get_initial_resolved_state<Id, K>(
270    unconflicted_state: &crate::state::at::SharedState<Id, K>,
271    version: StateResVersion,
272) -> crate::state::at::SharedState<Id, K>
273where
274    Id: Clone,
275    K: Ord + Clone,
276{
277    if version.is_v2_1_plus() {
278        imbl::OrdMap::new()
279    } else {
280        unconflicted_state.clone()
281    }

284pub(crate) fn merge_unconflicted_power_events<Id, K>(
285    version: StateResVersion,
286    unconflicted_state: &crate::state::at::SharedState<Id, K>,
287    resolved: &mut crate::state::at::SharedState<Id, K>,
288) where
289    Id: Clone,
290    K: Ord + Clone + Default,
291{
292    use crate::basespec::event_types::{M_ROOM_CREATE, M_ROOM_JOIN_RULES, M_ROOM_POWER_LEVELS};
293
294    // Under V2.1+, progressive state resolution starts empty, meaning unconflicted power
295    // events are missing from `resolved`. We must merge unconflicted power events (like power levels)
296    // into `resolved` before building the mainline, so they are visible to `build_mainline` and sorting.
297    if version.is_v2_1_plus() {
298        for event_type in [M_ROOM_POWER_LEVELS, M_ROOM_JOIN_RULES, M_ROOM_CREATE] {
299            let key = (EventType::from(event_type), K::default());
300            if let Some(v) = unconflicted_state.get(&key) {
301                resolved.entry(key).or_insert_with(|| v.clone());
302            }
303        }
304    }
305}

307/// Executes the first half of the Matrix State Resolution algorithm (the power phase).
308///
309/// This involves setting up the sorting context, dividing events into power and non-power events,
310/// tracking the deterministically chosen `m.room.create` event, and yielding the separated subsets
311/// for subsequent Kahn sorting and iterative auth checks.
312#[allow(clippy::type_complexity)]
313pub(crate) fn execute_power_phase<'a, Id, C, S1, S2, K>(
314    unconflicted_state: &crate::state::at::SharedState<Id, K>,
315    conflicted_events: &'a HashMap<Id, LeanEvent<Id, C, K>, S1>,
316    auth_context: &'a HashMap<Id, LeanEvent<Id, C, K>, S2>,
317    original_conflicted_keys: &alloc::collections::BTreeSet<Id>,
318    version: StateResVersion,
319) -> (
320    crate::basespec::rezzy_types::SortContext<'a, Id, C, S1, S2, LeanEvent<Id, C, K>>, // sort_context
321    HashMap<Id, LeanEvent<Id, C, K>>, // power_events
322    HashMap<Id, LeanEvent<Id, C, K>>, // non_power_events
323    Option<&'a LeanEvent<Id, C, K>>,  // m.room.create event
324)
325where
326    Id: crate::basespec::rezzy_types::EventId,
327    S1: core::hash::BuildHasher,
328    S2: core::hash::BuildHasher,
329    C: crate::basespec::rezzy_types::EventContent,
330    K: Ord + Clone + Default + AsRef<str>,
331{
332    let sort_context = crate::basespec::rezzy_types::SortContext {
333        primary: conflicted_events,
334        secondary: auth_context,
335        _marker: core::marker::PhantomData,
336    };
337
338    let mut power_events = HashMap::new();
339    let mut non_power_events = HashMap::new();
340    crate::resolve::lattice::route_power_events(
341        conflicted_events,
342        &mut power_events,
343        &mut non_power_events,
344        version,
345    );
346
347    if version != StateResVersion::V1 {
348        expand_v2_power_events_auth_chains(
349            &mut power_events,
350            &mut non_power_events,
351            conflicted_events,
352        );
353    }
354
355    route_msc4297_ancestral_power_events(
356        &mut power_events,
357        auth_context,
358        original_conflicted_keys,
359        version,
360    );
361
362    let create_key = (
363        EventType::from(crate::basespec::event_types::M_ROOM_CREATE),
364        K::default(),
365    );
366
367    let create_ev = unconflicted_state
368        .get(&create_key)
369        // 1. O(1) Fast Path: It's already in the agreed unconflicted state
370        .and_then(|id| auth_context.get(id).or_else(|| conflicted_events.get(id)))
371        // 2. Slow Path: It's currently in conflict (e.g. root of DAG)
372        // We only scan the tiny `conflicted_events` set, NEVER the massive `auth_context`!
373        .or_else(|| {
374            conflicted_events
375                .values()
376                .filter(|ev| ev.event_type == crate::basespec::event_types::M_ROOM_CREATE)
377                .min_by_key(|ev| &ev.event_id)
378        });
379
380    // Return updated refs
381    (sort_context, power_events, non_power_events, create_ev)
382}

384/// Resolves conflicted Matrix room state using the specified algorithm version.
385///
386/// This is the primary entry point for state resolution. Given the set of
387/// unconflicted state (agreed upon by all forks), the conflicted events
388/// (present in some forks but not others), and the full auth context,
389/// it produces the single deterministic resolved state map.
390///
391/// # Parameters
392///
393/// - `unconflicted_state`: State entries that all forks agree on, keyed by
394///   `(event_type, state_key) -> event_id`. For **partial joins**, pass the
395///   trusted state snapshot from the join response — this serves as the
396///   checkpoint base. See _Checkpoint / Partial-Join_ below.
397/// - `conflicted_events`: Events that differ across forks. These will be
398///   sorted, auth-checked, and selectively applied.
399/// - `auth_context`: The full set of events reachable via `auth_events`
400///   from the conflicted set. Must include all power-level, membership,
401///   and join-rules events needed for authorization.
402/// - `version`: Which resolution algorithm to use (see [`StateResVersion`]).
403///
404/// # Returns
405///
406/// A `imbl::OrdMap<(event_type, state_key), event_id>` representing the resolved
407/// room state — the union of unconflicted state and the winners from the
408/// conflicted set.
409///
410/// # Checkpoint / Partial-Join
411///
412/// For partial joins (federated rooms where a server doesn't have full
413/// history), pass the trusted state snapshot as `unconflicted_state`:
414///
415/// ```rust,no_run
416/// # use rezzy::{resolve_iterative_sort, LeanEvent, StateResVersion, HashMap};
417/// # use rezzy::basespec::event_types::EventType;
418/// # use imbl::OrdMap;
419/// // State snapshot from /send_join response
420/// let checkpoint: imbl::OrdMap<(EventType, String), String> = /* ... */
421/// # imbl::OrdMap::new();
422/// let new_events: HashMap<String, LeanEvent> = /* events since join */
423/// # HashMap::new();
424/// let auth_ctx: HashMap<String, LeanEvent> = /* auth chain for new_events */
425/// # HashMap::new();
426///
427/// let resolved = resolve_iterative_sort(checkpoint, new_events, &auth_ctx, StateResVersion::V2, &mut std::collections::HashMap::new());
428/// ```
429///
430/// # Auth Chain Safety
431///
432/// **The auth chain for conflicted events must be complete.** You can trust a
433/// snapshot for the unconflicted base, but truncating the auth chain for
434/// conflicted events causes:
435///
436/// - **Sorting failures**: cannot establish mainline order without the full
437///   power-level chain.
438/// - **Auth check failures**: missing historical power levels or membership
439///   events cause events to be incorrectly rejected.
440/// - **State reset attacks**: an adversary can craft events whose truncated
441///   auth chain makes an illegitimate power grab appear valid
442///   (ref: CVE-2025-49090).
443///
444/// # Panics
445///
446/// Will panic if an event referenced in `auth_events` or `prev_events` by
447/// a conflicted event is missing from both `conflicted_events` and
448/// `auth_context`.
449///
450/// # Algorithm overview
451///
452/// 1. Classify conflicted events into **power events** (create, PL, join rules,
453///    bans/kicks) and **non-power events**.
454/// 2. Sort power events via [`lean_kahn_sort`] and iteratively auth-check them
455///    to build the authoritative administrative state.
456/// 3. Sort non-power events via [`mainline_sort`] (by proximity to the resolved
457///    power-levels chain) and iteratively auth-check them.
458/// 4. Merge winners into the unconflicted base.
459#[must_use]
460#[allow(clippy::implicit_hasher)]
461pub fn resolve_iterative_sort<
462    Id: crate::basespec::rezzy_types::EventId,
463    C: crate::basespec::rezzy_types::EventContent + Clone,
464    S1: core::hash::BuildHasher,
465    S2: core::hash::BuildHasher,
466    K,
467>(
468    unconflicted_state: crate::state::at::SharedState<Id, K>,
469    conflicted_events: HashMap<Id, LeanEvent<Id, C, K>, S1>,
470    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
471    version: StateResVersion,
472    pl_cache: &mut HashMap<Id, i64>,
473) -> crate::state::at::SharedState<Id, K>
474where
475    K: crate::basespec::rezzy_types::StateKey,
476    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
477{
478    resolve_iterative_sort_with_cache::<Id, C, S1, S2, K>(
479        unconflicted_state,
480        conflicted_events,
481        auth_context,
482        None,
483        version,
484        pl_cache,
485    )
486}

523/// Like [`resolve_iterative_sort_with_cache`], but additionally accepts a

529#[allow(clippy::too_many_arguments)]
530pub(crate) fn resolve_iterative_sort_with_all_caches<
531    Id: crate::basespec::rezzy_types::EventId,
532    C: crate::basespec::rezzy_types::EventContent + Clone,
533    S1: core::hash::BuildHasher,
534    S2: core::hash::BuildHasher,
535    K,
536>(
537    unconflicted_state: crate::state::at::SharedState<Id, K>,
538    mut conflicted_events: HashMap<Id, LeanEvent<Id, C, K>, S1>,
539    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
540    external_auth_cache: Option<&mut LocalAuthCache<Id, C, K>>,
541    version: StateResVersion,
542    pl_cache: &mut HashMap<Id, i64>,
543    mainline_cache: &mut FastMap<Id, Option<Id>>,
544    conflicted_keys: &crate::FastSet<(EventType, K)>,
545) -> crate::state::at::SharedState<Id, K>
546where
547    K: crate::basespec::rezzy_types::StateKey,
548    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
549{
550    let original_conflicted_keys =
551        prepare_conflicted_and_keys(&mut conflicted_events, auth_context, version);
552
553    // MSC4297 (v2.1+): The algorithm starts from an empty set of state.
554    let mut resolved = get_initial_resolved_state(&unconflicted_state, version);
555
556    let (sort_context, power_events, non_power_events, create_ev) = execute_power_phase(
557        &unconflicted_state,
558        &conflicted_events,
559        auth_context,
560        &original_conflicted_keys,
561        version,
562    );
563
564    let mut fallback_cache = crate::state::at::LocalAuthCache::<Id, C, K>::new(version);
565    let local_auth_cache = match external_auth_cache {
566        Some(cache) => cache,
567        None => &mut fallback_cache,
568    };
569    if local_auth_cache.version != version {
570        local_auth_cache.map.clear();
571        local_auth_cache.version = version;
572    }
573
574    run_power_phase_iterative_checks(
575        &mut resolved,
576        &power_events,
577        &sort_context,
578        auth_context,
579        &conflicted_events,
580        version,
581        local_auth_cache,
582        create_ev,
583        pl_cache,
584        conflicted_keys,
585    );
586
587    let sort_set = &conflicted_events;
588
589    merge_unconflicted_power_events(version, &unconflicted_state, &mut resolved);
590
591    // Step 3: Build the power-level mainline for mainline sort
592    let mainline = build_mainline_with_cache(&resolved, &sort_context, mainline_cache);
593
594    // Step 4: Sort non-power events by mainline ordering + iterative auth check
595    let mut non_power_list: Vec<&LeanEvent<Id, C, K>> = non_power_events.values().collect();
596    mainline_sort(&mut non_power_list, &mainline, &sort_context);
597
598    for ev in non_power_list {
599        let local_auth = compute_local_auth(ev, auth_context, sort_set, local_auth_cache, version);
600        if iterative_auth_ok(
601            ev,
602            &resolved,
603            auth_context,
604            sort_set,
605            local_auth,
606            create_ev,
607            version,
608            false,
609        ) {
610            if let Some(sk) = &ev.state_key {
611                let key = (EventType::from(ev.event_type.as_str()), sk.clone());
612                // Same guard as the power phase: only a genuinely conflicted
613                // key may be decided here.
614                if conflicted_keys.contains(&key) {
615                    resolved.insert(key, ev.event_id.clone());
616                }
617            }
618        }
619    }
620
621    // Final step (Matrix v2 spec): "Update the result of step 5 with the
622    // unconflicted state." The correct merge direction depends on what
623    // `resolved` already contains, which differs by version:
624    //
625    // - V1/V2: `get_initial_resolved_state` seeds `resolved` as a *clone* of
626    //   `unconflicted_state`, so every unconflicted value is already present.
627    //   Applying unconflicted last is then a no-op for every legitimately
628    //   resolved key, and only matters where an auth-diff-supplied power
629    //   event (pulled into `conflicted_events` purely to supply auth context
630    //   for a genuinely conflicting *other* key) incorrectly overwrote a key
631    //   that was never actually in conflict — which it must not be allowed
632    //   to do. So unconflicted must win here.
633    // - V2.1+: `resolved` starts *empty*, and `merge_unconflicted_power_events`
634    //   deliberately re-admits only power_levels/join_rules/create (via
635    //   `or_insert_with`, never overwriting). Other keys — notably
636    //   membership — are intentionally left for the power/non-power phases
637    //   to populate from scratch, which is how MSC4297 ban/kick
638    //   supplementation works: a validly-authorized ban pulled in via the
639    //   auth diff must be able to override a stale "unconflicted" join,
640    //   and federation convergence requires matching other MSC4297
641    //   implementations here. So `resolved` must win.
642    let final_resolved = if version.is_v2_1_plus() {
643        let mut f = unconflicted_state;
644        for (k, v) in resolved {
645            f.insert(k, v);
646        }
647        f
648    } else {
649        let mut f = resolved;
650        for (k, v) in unconflicted_state {
651            f.insert(k, v);
652        }
653        f
654    };
655    drop(conflicted_events);
656    final_resolved
657}

707/// Internal helper combining the functionality of [`resolve_iterative_sort_with_deltas`] and
708/// [`resolve_iterative_sort_with_cache`].
709#[must_use]
710#[allow(clippy::type_complexity, clippy::too_many_lines)]
711#[allow(clippy::implicit_hasher)]
712pub fn resolve_iterative_sort_with_cache_and_deltas<
713    Id: crate::basespec::rezzy_types::EventId,
714    C: crate::basespec::rezzy_types::EventContent + Clone,
715    S1: core::hash::BuildHasher,
716    S2: core::hash::BuildHasher,
717    K,
718>(
719    unconflicted_state: crate::state::at::SharedState<Id, K>,
720    mut conflicted_events: HashMap<Id, LeanEvent<Id, C, K>, S1>,
721    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
722    external_auth_cache: Option<&mut LocalAuthCache<Id, C, K>>,
723    version: StateResVersion,
724    pl_cache: &mut HashMap<Id, i64>,
725) -> (
726    crate::state::at::SharedState<Id, K>,
727    alloc::vec::Vec<crate::state::delta::ResolutionDelta<Id, K>>,
728)
729where
730    K: crate::basespec::rezzy_types::StateKey,
731    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
732{
733    use crate::state::delta::{ResolutionDelta, ResolvePhase};
734
735    let conflicted_keys = derive_all_conflicted_keys(&conflicted_events);
736    let original_conflicted_keys =
737        prepare_conflicted_and_keys(&mut conflicted_events, auth_context, version);
738
739    let mut resolved = get_initial_resolved_state(&unconflicted_state, version);
740    let mut deltas = alloc::vec::Vec::new();
741
742    // --- Power phase (with delta tracking) ---
743
744    let (sort_context, power_events, non_power_events, create_ev) = execute_power_phase(
745        &unconflicted_state,
746        &conflicted_events,
747        auth_context,
748        &original_conflicted_keys,
749        version,
750    );
751
752    let mut fallback_cache = LocalAuthCache::new(version);
753    let local_auth_cache = match external_auth_cache {
754        Some(cache) => cache,
755        None => &mut fallback_cache,
756    };
757    if local_auth_cache.version != version {
758        local_auth_cache.map.clear();
759        local_auth_cache.version = version;
760    }
761
762    let sort_set = &conflicted_events;
763
764    let sorted_power_ids =
765        lean_kahn_sort(&power_events, &sort_context, create_ev, version, pl_cache);
766    for id in &sorted_power_ids {
767        if let Some(event) = sort_set.get(id).or_else(|| auth_context.get(id)) {
768            let Some(sk) = &event.state_key else { continue };
769            let key = (EventType::from(event.event_type.as_str()), sk.clone());
770            let local_auth =
771                compute_local_auth(event, auth_context, sort_set, local_auth_cache, version);
772            let accepted = iterative_auth_ok(
773                event,
774                &resolved,
775                auth_context,
776                sort_set,
777                local_auth,
778                create_ev,
779                version,
780                true,
781            );
782            let replaced = if accepted && conflicted_keys.contains(&key) {
783                let old = resolved.get(&key).cloned();
784                resolved.insert(key.clone(), event.event_id.clone());
785                old
786            } else {
787                resolved.get(&key).cloned()
788            };
789            if original_conflicted_keys.contains(&event.event_id) {
790                deltas.push(ResolutionDelta {
791                    event_id: event.event_id.clone(),
792                    accepted,
793                    key: key.clone(),
794                    replaced,
795                    phase: ResolvePhase::Power,
796                });
797            }
798        }
799    }
800
801    // --- Non-power phase (with delta tracking) ---
802
803    merge_unconflicted_power_events(version, &unconflicted_state, &mut resolved);
804
805    let mainline = build_mainline(&resolved, &sort_context);
806    let mut non_power_list: alloc::vec::Vec<&LeanEvent<Id, C, K>> =
807        non_power_events.values().collect();
808    mainline_sort(&mut non_power_list, &mainline, &sort_context);
809
810    for ev in non_power_list {
811        let Some(sk) = &ev.state_key else { continue };
812        let key = (EventType::from(ev.event_type.as_str()), sk.clone());
813        let local_auth = compute_local_auth(ev, auth_context, sort_set, local_auth_cache, version);
814        let accepted = iterative_auth_ok(
815            ev,
816            &resolved,
817            auth_context,
818            sort_set,
819            local_auth,
820            create_ev,
821            version,
822            false,
823        );
824        let replaced = if accepted && conflicted_keys.contains(&key) {
825            let old = resolved.get(&key).cloned();
826            resolved.insert(key.clone(), ev.event_id.clone());
827            old
828        } else {
829            resolved.get(&key).cloned()
830        };
831        deltas.push(ResolutionDelta {
832            event_id: ev.event_id.clone(),
833            accepted,
834            key: key.clone(),
835            replaced,
836            phase: ResolvePhase::NonPower,
837        });
838    }
839
840    // Same version-gated fix as resolve_iterative_sort_with_all_caches: see
841    // the comment there for why V1/V2 and V2.1+ need opposite merge
842    // directions for this final step.
843    let final_resolved = if version.is_v2_1_plus() {
844        let mut f = unconflicted_state;
845        for (k, v) in resolved {
846            f.insert(k, v);
847        }
848        f
849    } else {
850        let mut f = resolved;
851        for (k, v) in unconflicted_state {
852            f.insert(k, v);
853        }
854        f
855    };
856    drop(conflicted_events);
857    (final_resolved, deltas)
858}
