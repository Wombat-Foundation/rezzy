26/// Dynamically fetches the sender's power level by inspecting the event's immediate `auth_events`.
27/// Recursive traversal of the auth chain is avoided to prevent bypassing immediate restrictions.
28pub(crate) fn get_power_level_from_auth_chain<Id, C, E>(
29    event: &E,
30    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
31    create_ev: Option<&E>,
32    version: StateResVersion,
33) -> i64
34where
35    Id: crate::basespec::rezzy_types::EventId,
36    C: crate::basespec::rezzy_types::EventContent,
37    E: EventLike<Id = Id, Content = C>,
38{
39    let mut pl_event = None;
40
41    // Spec compliance: only check immediate auth_events.
42    for aid in event.auth_events() {
43        if let Some(aev) = auth_context.get_event(aid) {
44            if aev.event_type().as_ref() == M_ROOM_POWER_LEVELS && aev.state_key() == Some("") {
45                pl_event = Some(aev);
46                break;
47            }
48        }
49    }
50
51    // V12+ (MSC4289): creators have spec-mandated infinite power level.
52    if matches!(
53        version,
54        StateResVersion::V2_1 | StateResVersion::V2_1_1 | StateResVersion::V2_2
55    ) {
56        let is_creator = create_ev.is_some_and(|ev| {
57            ev.sender() == event.sender() || ev.content().has_additional_creator(event.sender())
58        });
59        if is_creator {
60            return MAX_POWER_LEVEL_RUST;
61        }
62    }
63
64    if let Some(pl_ev) = pl_event {
65        if let Some(pl) = pl_ev.get_user_power_level(event.sender()) {
66            return pl;
67        }
68
69        if let Some(default_pl) = pl_ev.get_users_default() {
70            return default_pl;
71        }
72        return 0; // Default if PL event exists but no users_default
73    }
74
75    // No PL event in the auth chain — fall back to the pre-computed
76    // power_level field. This handles the bootstrap PL event (which only
77    // auths against $create) and simple unit-test events.
78    event.power_level()
79}

81/// Computes the shortest distance from the event to the m.room.create event via `auth_events`.
82/// Safely avoids stack overflow on deep DAGs using an iterative post-order traversal with memoization.
83pub(crate) fn compute_auth_distance_iterative<'a, Id, C, E>(
84    curr_id: &'a Id,
85    auth_context: &'a impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
86    create_id: Option<&'a Id>,
87    memo: &mut FastMap<&'a Id, u64>,
88) -> u64
89where
90    Id: crate::basespec::rezzy_types::EventId + 'a,
91    C: 'a,
92    E: EventLike<Id = Id, Content = C> + 'a,
93{
94    if Some(curr_id) == create_id {
95        return 0;
96    }
97    if let Some(&dist) = memo.get(curr_id) {
98        if dist != u64::MAX - 1 {
99            return dist;
100        }
101    }
102
103    let mut stack = Vec::new();
104    stack.push(curr_id);
105
106    while let Some(&top) = stack.last() {
107        if let Some(&dist) = memo.get(top) {
108            if dist != u64::MAX - 1 {
109                stack.pop();
110                continue;
111            }
112        } else {
113            memo.insert(top, u64::MAX - 1);
114        }
115
116        let mut all_children_done = true;
117        let mut min_dist = u64::MAX;
118
119        if let Some(ev) = auth_context.get_event(top) {
120            if !ev.auth_events().is_empty() {
121                for aid in ev.auth_events() {
122                    if Some(aid) == create_id {
123                        min_dist = min_dist.min(1);
124                    } else if let Some(&c_dist) = memo.get(aid) {
125                        if c_dist != u64::MAX - 1 {
126                            min_dist = min_dist.min(c_dist.saturating_add(1));
127                        }
128                    } else {
129                        all_children_done = false;
130                        stack.push(aid);
131                    }
132                }
133            }
134        }
135
136        if all_children_done {
137            memo.insert(top, min_dist);
138            stack.pop();
139        }
140    }
141
142    memo.get(curr_id).copied().unwrap_or(u64::MAX)
143}

145/// Detailed Kahn's Topological Sort algorithm for event power resolution.
146///
147/// This function performs a reverse topological sort on a set of events, placing
148/// descendants before their ancestors. It returns diagnostic details about any cycles
149/// if they are detected.
150///
151/// # Panics
152///
153/// Will panic if graph invariants are violated during topological sorting (specifically, if
154/// the in-degree map lacks an entry for a child event during the queue processing phase).
155#[allow(clippy::implicit_hasher)]
156pub fn lean_kahn_sort_with_cycle_diagnostics<Id, C, E, S1>(
157    events: &HashMap<Id, E, S1>,
158    sort_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
159    create_ev: Option<&E>,
160    version: StateResVersion,
161    pl_cache: &mut HashMap<Id, i64>,
162) -> KahnSortResult<Id>
163where
164    Id: crate::basespec::rezzy_types::EventId,
165    S1: core::hash::BuildHasher,
166    C: Clone + crate::basespec::rezzy_types::EventContent,
167    E: EventLike<Id = Id, Content = C>,
168{
169    pl_cache.clear();
170
171    let mut in_degree: FastMap<Id, usize> = FastMap::default();
172    let mut adjacency: FastMap<Id, Vec<Id>> = FastMap::default();
173
174    for (id, event) in events {
175        in_degree.entry(id.clone()).or_insert(0);
176        for auth in event.auth_events() {
177            if events.contains_key(auth) {
178                // Topological sort: ancestors come BEFORE descendants.
179                // But we want a REVERSE topological sort: descendants BEFORE ancestors.
180                // So we add edges from ancestors to descendants.
181                adjacency.entry(auth.clone()).or_default().push(id.clone());
182                let val = in_degree.entry(id.clone()).or_insert(0);
183                *val = val.saturating_add(1);
184            }
185        }
186    }
187
188    // Pre-compute power levels once per event to avoid redundant auth chain walks
189    // inside the hot BinaryHeap push path.
190    for (id, ev) in events {
191        if !pl_cache.contains_key(id) {
192            pl_cache.insert(
193                id.clone(),
194                get_power_level_from_auth_chain(ev, sort_context, create_ev, version),
195            );
196        }
197    }
198
199    let depth_cache: FastMap<Id, u64> = if version == StateResVersion::V2_2 {
200        let mut memo = FastMap::default();
201        let create_id = create_ev.map(super::super::basespec::rezzy_types::DagNode::event_id);
202        events
203            .keys()
204            .map(|id| {
205                (
206                    id.clone(),
207                    compute_auth_distance_iterative(id, sort_context, create_id, &mut memo),
208                )
209            })
210            .collect()
211    } else {
212        FastMap::default()
213    };
214
215    let mut queue: BinaryHeap<SortPriority<'_, E>> = BinaryHeap::new();
216    for (id, &degree) in &in_degree {
217        if degree == 0 {
218            if let Some(event) = events.get(id) {
219                queue.push(SortPriority {
220                    event,
221                    power_level: pl_cache.get(id).copied().unwrap_or(0),
222                    auth_chain_distance: depth_cache.get(id).copied().unwrap_or(0),
223                    version,
224                });
225            }
226        }
227    }
228
229    let mut result = Vec::new();
230    while let Some(priority) = queue.pop() {
231        let event = priority.event;
232
233        result.push(event.event_id().clone());
234        if let Some(neighbors) = adjacency.get(event.event_id()) {
235            for next_id in neighbors {
236                let degree = in_degree.get_mut(next_id).unwrap();
237                *degree = degree.saturating_sub(1);
238                if *degree == 0 {
239                    let next_ev = events.get(next_id).unwrap();
240                    queue.push(SortPriority {
241                        event: next_ev,
242                        power_level: pl_cache.get(next_id).copied().unwrap_or(0),
243                        auth_chain_distance: depth_cache.get(next_id).copied().unwrap_or(0),
244                        version,
245                    });
246                }
247            }
248        }
249    }
250
251    // Detect cycles: events that never reached in-degree 0.
252    if result.len() != events.len() {
253        let sorted_set: alloc::collections::BTreeSet<&Id> = result.iter().collect();
254        let stuck: Vec<Id> = events
255            .keys()
256            .filter(|id| !sorted_set.contains(id))
257            .cloned()
258            .collect();
259        return KahnSortResult::CycleDetected {
260            sorted: result,
261            stuck,
262        };
263    }
264
265    KahnSortResult::Ok(result)
266}

268/// A simplified implementation of Kahn's Topological Sort.
269/// Backward-compatible wrapper that falls back to standard tie-breaking on cycles.
270///
271/// # Panics
272///
273/// Will panic if graph invariants are violated (specifically, if an event returned
274/// in the cycle-breaking list of stuck nodes is missing from the input `events` map).
275// jscpd:ignore-start
276#[must_use]
277#[allow(clippy::implicit_hasher)]
278pub fn lean_kahn_sort<Id, C, E, S1>(
279    events: &HashMap<Id, E, S1>,
280    sort_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
281    create_ev: Option<&E>,
282    version: StateResVersion,
283    pl_cache: &mut HashMap<Id, i64>,
284) -> Vec<Id>
285where
286    Id: crate::basespec::rezzy_types::EventId,
287    S1: core::hash::BuildHasher,
288    C: Clone + crate::basespec::rezzy_types::EventContent,
289    E: EventLike<Id = Id, Content = C>,
290{
291    // jscpd:ignore-end
292    match lean_kahn_sort_with_cycle_diagnostics(events, sort_context, create_ev, version, pl_cache)
293    {
294        KahnSortResult::Ok(sorted) => sorted,
295        KahnSortResult::CycleDetected {
296            mut sorted,
297            mut stuck,
298        } => {
299            #[cfg(feature = "std")]
300            std::eprintln!("KAHN CYCLE DETECTED! Stuck: {stuck:?}");
301            stuck.sort_by(|a, b| {
302                let ev_a = events.get(a).unwrap();
303                let ev_b = events.get(b).unwrap();
304                // Standard tie-breaking fallback (origin_server_ts ascending, then event_id ascending)
305                ev_a.origin_server_ts()
306                    .cmp(&ev_b.origin_server_ts())
307                    .then_with(|| a.cmp(b))
308            });
309            sorted.append(&mut stuck);
310            sorted
311        }
312    }
313}

315pub(crate) fn build_mainline<Id, C, E, K>(
316    resolved: &crate::state::at::SharedState<Id, K>,
317    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
318) -> Vec<Id>
319where
320    Id: crate::basespec::rezzy_types::EventId,
321    C: Clone + crate::basespec::rezzy_types::EventContent,
322    E: EventLike<Id = Id, Content = C>,
323    K: Ord + Clone + Default,
324{
325    // The hot path (compute_state_at's fork-merge loop) now threads a persistent
326    // cache through `resolve_iterative_sort_with_all_caches`, so this fresh-cache
327    // fallback only matters for one-shot callers (e.g. resolve_lattice_fold's V2
328    // path, which calls build_mainline exactly once per resolution).
329    build_mainline_with_cache(resolved, auth_context, &mut FastMap::default())
330}

332/// Like [`build_mainline`], but populates a `pl_parent_cache` mapping each
333/// event ID to its nearest `m.room.power_levels` ancestor in the auth chain.
334///
335/// Subsequent calls sharing the same cache skip BFS entirely for events
336/// already resolved, turning the mainline walk from `O(M × B)` (M = mainline
337/// length, B = auth chain breadth) to `O(M)` on cache hits.
338pub(crate) fn build_mainline_with_cache<Id, C, E, K>(
339    resolved: &crate::state::at::SharedState<Id, K>,
340    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
341    pl_parent_cache: &mut FastMap<Id, Option<Id>>,
342) -> Vec<Id>
343where
344    Id: crate::basespec::rezzy_types::EventId,
345    C: Clone + crate::basespec::rezzy_types::EventContent,
346    E: EventLike<Id = Id, Content = C>,
347    K: Ord + Clone + Default,
348{
349    let mut mainline = Vec::new();
350    let mut seen_in_mainline = crate::FastSet::default();
351    let pl_key = (
352        crate::basespec::event_types::EventType::from(M_ROOM_POWER_LEVELS),
353        K::default(),
354    );
355    let mut current = resolved.get(&pl_key).cloned();
356
357    while let Some(eid) = current {
358        if !seen_in_mainline.insert(eid.clone()) {
359            #[cfg(feature = "std")]
360            std::eprintln!("REZZY_WARN: MAINLINE CYCLE DETECTED!");
361            break; // Cycle detected in the power-levels mainline!
362        }
363        mainline.push(eid.clone());
364
365        // Check cache first
366        if let Some(cached) = pl_parent_cache.get(&eid) {
367            current = None;
368            current.clone_from(cached);
369            continue;
370        }
371
372        // BFS to find the nearest PL ancestor
373        let mut found = None;
374        if let Some(ev) = auth_context.get_event(&eid) {
375            let mut queue = VecDeque::new();
376            for auth_id in ev.auth_events() {
377                queue.push_back(auth_id.clone());
378            }
379            let mut visited = crate::FastSet::default();
380            while let Some(q_id) = queue.pop_front() {
381                if !visited.insert(q_id.clone()) {
382                    continue;
383                }
384                if let Some(auth_ev) = auth_context.get_event(&q_id) {
385                    if auth_ev.event_type().as_ref() == M_ROOM_POWER_LEVELS {
386                        found = Some(q_id);
387                        break;
388                    }
389                    for aid in auth_ev.auth_events() {
390                        queue.push_back(aid.clone());
391                    }
392                }
393            }
394        }
395
396        pl_parent_cache.insert(eid, found.clone());
397        current = found;
398    }
399
400    mainline
401}

403/// Precompute the closest mainline position for every target event reachable via
404/// `auth_events` using a stack-safe O(V+E) iterative DFS upward search.
405///
406/// This entirely avoids `O(N)` cloning of the DAG, and prevents stack overflow
407/// by using an explicit stack to simulate recursion while memoizing distances.
408pub(crate) fn compute_closest_mainline_positions<Id, C, E>(
409    events: &mut [&E],
410    mainline: &[Id],
411    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
412) -> HashMap<Id, usize>
413where
414    Id: crate::basespec::rezzy_types::EventId,
415    C: Clone + crate::basespec::rezzy_types::EventContent,
416    E: EventLike<Id = Id, Content = C>,
417{
418    let mut memo = HashMap::new();
419
420    // Pre-populate mainline events with their exact index position
421    for (pos, id) in mainline.iter().enumerate() {
422        memo.insert(id.clone(), pos);
423    }
424
425    let mut stack = Vec::new();
426
427    for ev in events.iter() {
428        stack.push(ev.event_id());
429
430        while let Some(&top) = stack.last() {
431            if let Some(&val) = memo.get(top) {
432                if val != usize::MAX - 1 {
433                    stack.pop();
434                    continue;
435                }
436            } else {
437                memo.insert(top.clone(), usize::MAX - 1);
438            }
439
440            let mut all_children_done = true;
441            let mut min_pos = usize::MAX;
442
443            if let Some(node) = auth_context.get_event(top) {
444                for aid in node.auth_events() {
445                    if let Some(&child_pos) = memo.get(aid) {
446                        if child_pos != usize::MAX - 1 {
447                            min_pos = min_pos.min(child_pos);
448                        }
449                    } else {
450                        all_children_done = false;
451                        stack.push(aid);
452                    }
453                }
454            }
455
456            if all_children_done {
457                // Clamp "no path found" to mainline.len() so callers
458                // don't see a raw usize::MAX leaking into comparisons.
459                let resolved_pos = if min_pos == usize::MAX {
460                    mainline.len()
461                } else {
462                    min_pos
463                };
464                memo.insert(top.clone(), resolved_pos);
465                stack.pop();
466            }
467        }
468    }
469
470    memo
471}

473/// Sorts non-power events by mainline ordering per the Matrix spec.
474///
475/// The mainline is the chain of `m.room.power_levels` events reachable from
476/// the currently resolved PL state. Each event's "mainline position" is the
477/// closest PL event in its auth chain. The sort order is:
478///
479/// 1. **Mainline position** descending (farther = worse = applied first).
480/// 2. **`origin_server_ts`** ascending (earlier = applied first, later wins).
481/// 3. **`event_id`** ascending (lexicographic tie-break).
482///
483/// Events closer to the current power-levels event are applied **last**
484/// and therefore win for same-key conflicts (last-write-wins).
485pub fn mainline_sort<Id, C, E>(
486    events: &mut [&E],
487    mainline: &[Id],
488    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
489) where
490    Id: crate::basespec::rezzy_types::EventId,
491    C: Clone + crate::basespec::rezzy_types::EventContent,
492    E: EventLike<Id = Id, Content = C>,
493{
494    #[cfg(all(feature = "std", debug_assertions, not(test)))]
495    std::eprintln!(
496        "[DEBUG] mainline_sort: sorting {} non-power events against mainline of length {}",
497        events.len(),
498        mainline.len()
499    );
500    // O(V+E) iterative DFS to find the closest mainline index for all non-power events
501    let dist = compute_closest_mainline_positions(events, mainline, auth_context);
502
503    events.sort_by(|a, b| {
504        // Hopefully safe to unwrap. DFS guarantees all events are in `dist`.
505        let pos_a = dist[a.event_id()];
506        let pos_b = dist[b.event_id()];
507
508        // Larger mainline position = farther from current PL = worse = comes first
509        // (so it gets overwritten by closer events via last-write-wins)
510        match pos_b.cmp(&pos_a) {
511            Ordering::Equal => {
512                // Earlier timestamp comes first (later wins via last-write)
513                match a.origin_server_ts().cmp(&b.origin_server_ts()) {
514                    Ordering::Equal => a.event_id().cmp(b.event_id()),
515                    ord => ord,
516                }
517            }
518            ord => ord,
519        }
520    });
521}

784        let pl0 = LeanEvent::<String> {

786            event_type: "m.room.power_levels".into(),
787            auth_events: alloc::vec![],

792            event_type: "m.room.power_levels".into(),
793            auth_events: alloc::vec!["PL0".into()],
794            ..Default::default()
795        };
796        let pl2 = LeanEvent::<String> {
797            event_id: "PL2".into(),
798            event_type: "m.room.power_levels".into(),
799            auth_events: alloc::vec!["PL1".into()],
800            ..Default::default()
801        };
802
803        let mut ctx = HashMap::new();
804        ctx.insert("PL0".into(), pl0);
805        ctx.insert("PL1".into(), pl1);
806        ctx.insert("PL2".into(), pl2);
807

819        assert_eq!(ml2, ml1, "cached mainline must match original");
820    }
