15//! Lattice-coordinatized state resolution.
16//!
17//! This module provides [`resolve_lattice_fold`], an alternative to
18//! [`crate::resolve::iterative::resolve_iterative_sort`] that replaces the sequential mainline sort
19//! with a parallel causal coordinatization projection and commutative
20//! join-semilattice fold. Note that this is currently an exploratory hypothesis
21//! that approximates iterative auth-checks using a fixed terminal snapshot.
22//!
23//! ## How it works
24//!
25//! 1. The **power phase** is identical to [`resolve_iterative_sort`](crate::resolve::iterative::resolve_iterative_sort).
26//! 2. Instead of sorting non-power events, each event is assigned a **mainline
27//!    coordinate** (its closest position on the power-levels chain).
28//! 3. Events are folded per `(type, state_key)` using a commutative **Least
29//!    Upper Bound** (LUB) operator — the event with the best coordinate wins.
30//! 4. The fold is embarrassingly parallel and runs on `std::thread::scope`
31//!    when the `std` feature is enabled.
32//!
33//! # Panics
34//!
35//! Panics if the resolved mainline does not begin with the resolved
36//! `m.room.power_levels` event.
37//!
38//! Also contains [`route_power_events`], which classifies events into
39//! power vs. non-power buckets.
40//!
41//! ### Internal Pipeline
42//!
43//! The non-power phase is implemented by three internal functions:
44//!
45//! 1. **`fold_lattice_chunk`** — processes a slice of events in a single thread,
46//!    auth-checking each and folding per-`(type, state_key)` winners via the LUB operator.
47//! 2. **`merge_lattice_winners`** — merges thread-local winner maps back into
48//!    the global result using the same LUB comparator.
49//! 3. **`compute_lattice_coordinatized_winners`** — orchestrates the parallel
50//!    fan-out via `std::thread::scope`, splits events into chunks, and coordinates
51//!    the fold-then-merge pipeline.

62/// Determines whether `ev` beats `current_winner` under the Least Upper Bound (LUB)
63/// tie-breaking rules.
64///
65/// The comparison cascade is:
66/// 1. **Mainline position**: closer to the current PL event (smaller index) wins.
67/// 2. **`origin_server_ts`**: later timestamp wins.
68/// 3. **`event_id`**: lexicographically largest ID wins.
69///
70/// This operator is **commutative** and **associative**, which is what allows
71/// the fold to be parallelized without affecting the result.
72#[must_use]
73pub fn is_lattice_winner_better<Id, C, S: core::hash::BuildHasher>(
74    ev: &LeanEvent<Id, C>,
75    current_winner: &LeanEvent<Id, C>,
76    mainline_distances: &HashMap<Id, usize, S>,
77    mainline_len: usize,
78) -> bool
79where
80    Id: crate::basespec::rezzy_types::EventId,
81    C: crate::basespec::rezzy_types::EventContent,
82{
83    let ev_pos = mainline_distances
84        .get(&ev.event_id)
85        .copied()
86        .unwrap_or(mainline_len);
87    let winner_pos = mainline_distances
88        .get(&current_winner.event_id)
89        .copied()
90        .unwrap_or(mainline_len);
91
92    // The Commutative Join Operator (Least Upper Bound):
93    if ev_pos < winner_pos {
94        true // Closer to mainline wins
95    } else if ev_pos > winner_pos {
96        false
97    } else if ev.origin_server_ts > current_winner.origin_server_ts {
98        true // Later timestamp wins
99    } else if ev.origin_server_ts < current_winner.origin_server_ts {
100        false
101    } else {
102        // Lexicographical sort: LARGEST string wins.
103        ev.event_id > current_winner.event_id
104    }
105}

107fn update_winner_if_better<'a, Id, C>(
108    winners: &mut HashMap<(EventType, String), &'a LeanEvent<Id, C>>,
109    key: (EventType, String),
110    ev: &'a LeanEvent<Id, C>,
111    mainline_distances: &HashMap<Id, usize>,
112    mainline_len: usize,
113) where
114    Id: crate::basespec::rezzy_types::EventId,
115    C: crate::basespec::rezzy_types::EventContent,
116{
117    let is_better = if let Some(current_winner) = winners.get(&key) {
118        is_lattice_winner_better(ev, current_winner, mainline_distances, mainline_len)
119    } else {
120        true // First event for this state key inherently wins
121    };
122
123    if is_better {
124        winners.insert(key, ev);
125    }
126}

128#[allow(clippy::too_many_arguments)]
129fn fold_lattice_chunk<'a, Id, C, S2: core::hash::BuildHasher, S3: core::hash::BuildHasher>(
130    // jscpd:ignore-start
131    chunk: &[&'a LeanEvent<Id, C>],
132    mainline_distances: &HashMap<Id, usize>,
133    mainline_len: usize,
134    terminal_power_state: &crate::state::at::SharedState<Id>,
135    auth_context: &HashMap<Id, LeanEvent<Id, C>, S2>,
136    sort_set: &HashMap<Id, LeanEvent<Id, C>, S3>,
137    version: StateResVersion,
138    create_ev: Option<&LeanEvent<Id, C>>,
139    // jscpd:ignore-end
140) -> HashMap<(EventType, String), &'a LeanEvent<Id, C>>
141where
142    Id: crate::basespec::rezzy_types::EventId,
143    C: crate::basespec::rezzy_types::EventContent + Clone,
144{
145    let mut thread_res: HashMap<(EventType, String), &'a LeanEvent<Id, C>> = HashMap::new();
146    let mut local_auth_cache = crate::state::at::LocalAuthCache::<Id, C>::new(version);
147
148    for &ev in chunk {
149        // VALIDATE FIRST (filters out Byzantine garbage/supremum deletion attacks)
150        let local_auth =
151            compute_local_auth(ev, auth_context, sort_set, &mut local_auth_cache, version);
152
153        if !iterative_auth_ok(
154            ev,
155            terminal_power_state,
156            auth_context,
157            sort_set,
158            local_auth,
159            create_ev,
160            version,
161            false,
162        ) {
163            continue; // Drop unauthorized events before they can compete for the LUB!
164        }
165
166        // Skip events with no `state_key` (e.g. `m.room.redaction`)
167        if ev.state_key.is_none() {
168            continue;
169        }
170
171        // NOW COMPETE FOR LUB
172        let key = (
173            EventType::from(ev.event_type.as_str()),
174            ev.state_key.clone().unwrap(),
175        );
176        update_winner_if_better(&mut thread_res, key, ev, mainline_distances, mainline_len);
177    }
178    thread_res
179}

181fn merge_lattice_winners<'a, Id, C>(
182    key_winners: &mut HashMap<(EventType, String), &'a LeanEvent<Id, C>>,
183    thread_res: HashMap<(EventType, String), &'a LeanEvent<Id, C>>,
184    mainline_distances: &HashMap<Id, usize>,
185    mainline_len: usize,
186) where
187    Id: crate::basespec::rezzy_types::EventId,
188    C: crate::basespec::rezzy_types::EventContent,
189{
190    for (key, ev) in thread_res {
191        update_winner_if_better(key_winners, key, ev, mainline_distances, mainline_len);
192    }
193}

195#[allow(clippy::too_many_arguments)]
196fn compute_lattice_coordinatized_winners<
197    'a,
198    Id,
199    C,
200    S1: core::hash::BuildHasher + Sync + Send,
201    S2: core::hash::BuildHasher + Sync + Send,
202    S3: core::hash::BuildHasher + Sync + Send,
203>(
204    // jscpd:ignore-start
205    non_power_events: &'a HashMap<Id, LeanEvent<Id, C>, S1>,
206    mainline_distances: &HashMap<Id, usize>,
207    mainline_len: usize,
208    terminal_power_state: &crate::state::at::SharedState<Id>,
209    auth_context: &HashMap<Id, LeanEvent<Id, C>, S2>,
210    sort_set: &HashMap<Id, LeanEvent<Id, C>, S3>,
211    version: StateResVersion,
212    create_ev: Option<&LeanEvent<Id, C>>,
213    // jscpd:ignore-end
214    key_winners: &mut HashMap<(EventType, String), &'a LeanEvent<Id, C>>,
215) where
216    Id: crate::basespec::rezzy_types::EventId + Sync + Send,
217    C: crate::basespec::rezzy_types::EventContent + Clone + Sync + Send,
218{
219    let v: Vec<&'a LeanEvent<Id, C>> = non_power_events.values().collect();
220
221    #[cfg(feature = "std")]
222    {
223        let num_threads =
224            std::thread::available_parallelism().map_or(4, core::num::NonZeroUsize::get);
225        let chunk_size = (non_power_events
226            .len()
227            .saturating_add(num_threads)
228            .saturating_sub(1))
229        .checked_div(num_threads)
230        .unwrap_or(1)
231        .max(1);
232        let chunks: Vec<&[&'a LeanEvent<Id, C>]> = v.chunks(chunk_size).collect();
233
234        let winners = std::sync::Mutex::new(HashMap::new());
235        std::thread::scope(|s| {
236            let mut handles = Vec::with_capacity(chunks.len());
237            for chunk in chunks {
238                let handle = s.spawn(|| {
239                    fold_lattice_chunk(
240                        chunk,
241                        mainline_distances,
242                        mainline_len,
243                        terminal_power_state,
244                        auth_context,
245                        sort_set,
246                        version,
247                        create_ev,
248                    )
249                });
250                handles.push(handle);
251            }
252            for handle in handles {
253                let thread_res = handle.join().unwrap();
254                let mut guard = winners.lock().unwrap();
255                merge_lattice_winners(&mut guard, thread_res, mainline_distances, mainline_len);
256            }
257        });
258        *key_winners = winners.into_inner().unwrap();
259    }
260    #[cfg(not(feature = "std"))]
261    {
262        *key_winners = fold_lattice_chunk(
263            &v,
264            mainline_distances,
265            mainline_len,
266            terminal_power_state,
267            auth_context,
268            sort_set,
269            version,
270            create_ev,
271        );
272    }
273}

275/// Classifies conflicted events into power events and non-power events.
276///
277/// Power events are those that affect the room's administrative state:
278/// - `m.room.create`
279/// - `m.room.power_levels`
280/// - `m.room.join_rules`
281/// - `m.room.member` events that are bans or kicks (V2.1+ only; V2 treats
282///   **all** member events as power events)
283///
284/// Non-power events are everything else (messages, topics, `m.room.third_party_invite`, etc.).
285pub fn route_power_events<
286    Id: crate::basespec::rezzy_types::EventId,
287    C: Clone + crate::basespec::rezzy_types::EventContent,
288    S1: core::hash::BuildHasher,
289    S2: core::hash::BuildHasher,
290    S3: core::hash::BuildHasher,
291    K: Clone + AsRef<str>,
292>(
293    sort_set: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
294    power_events: &mut HashMap<Id, LeanEvent<Id, C, K>, S2>,
295    non_power_events: &mut HashMap<Id, LeanEvent<Id, C, K>, S3>,
296    version: crate::StateResVersion,
297) {
298    for (id, ev) in sort_set {
299        if ev.is_power_event(version) {
300            power_events.insert(id.clone(), ev.clone());
301        } else {
302            non_power_events.insert(id.clone(), ev.clone());
303        }
304    }
305}

307/// Resolves conflicted state using causal coordinatization projection
308/// and commutative join-semilattice folding.
309///
310/// This is an exploratory hypothesis that approximates [`crate::resolve::iterative::resolve_iterative_sort`] but
311/// replaces the sequential mainline sort + iterative auth-check loop with a
312/// parallel per-key fold. Each non-power event competes for its `(type, state_key)`
313/// slot via the [`is_lattice_winner_better`] LUB operator.
314///
315/// Use this variant when:
316/// - The conflicted set is large (thousands of events).
317/// - The `std` feature is enabled (to benefit from thread parallelism).
318///
319/// The power phase (Steps 1–2) is shared with `resolve_iterative_sort`.
320///
321/// **Note:** V2.1+ rooms delegate entirely to [`resolve_iterative_sort`](crate::resolve::iterative::resolve_iterative_sort)
322/// because the lattice fold does not support MSC4297's conflicted subgraph. This
323/// changes the parallelism characteristics for V2.1+ callers.
324///
325/// # Panics
326///
327/// Panics if the resolved mainline does not begin with the resolved
328/// `m.room.power_levels` event.
329// jscpd:ignore-start
330#[must_use]
331pub fn resolve_lattice_fold<
332    Id,
333    C,
334    S1: core::hash::BuildHasher + Sync + Send,
335    S2: core::hash::BuildHasher + Sync + Send,
336>(
337    unconflicted_state: crate::state::at::SharedState<Id>,
338    mut conflicted_events: HashMap<Id, LeanEvent<Id, C>, S1>,
339    auth_context: &HashMap<Id, LeanEvent<Id, C>, S2>,
340    version: StateResVersion,
341) -> crate::state::at::SharedState<Id>
342where
343    Id: crate::basespec::rezzy_types::EventId + Sync + Send,
344    C: crate::basespec::rezzy_types::EventContent + Sync + Send + Clone,
345{
346    // jscpd:ignore-end
347    let mut pl_cache = HashMap::new();
348
349    if version.is_v2_1_plus() {
350        return crate::resolve::iterative::resolve_iterative_sort(
351            unconflicted_state,
352            conflicted_events,
353            auth_context,
354            version,
355            &mut pl_cache,
356        );
357    }
358
359    // This exploratory path doesn't add auth-diff/subgraph-context events to
360    // `conflicted_events` beyond what its own caller supplied, so every
361    // entry is treated as genuinely conflicted (matching the pre-existing
362    // behavior of this experimental resolver).
363    let conflicted_keys = crate::resolve::iterative::derive_all_conflicted_keys(&conflicted_events);
364
365    let original_conflicted_keys = crate::resolve::iterative::prepare_conflicted_and_keys(
366        &mut conflicted_events,
367        auth_context,
368        version,
369    );
370
371    let mut resolved =
372        crate::resolve::iterative::get_initial_resolved_state(&unconflicted_state, version);
373
374    let (sort_context, power_events, non_power_events, create_ev) =
375        crate::resolve::iterative::execute_power_phase(
376            &unconflicted_state,
377            &conflicted_events,
378            auth_context,
379            &original_conflicted_keys,
380            version,
381        );
382
383    // Initialize local auth cache for power-phase checks
384    let mut local_auth_cache = crate::state::at::LocalAuthCache::<Id, C>::new(version);
385
386    crate::resolve::iterative::run_power_phase_iterative_checks(
387        &mut resolved,
388        &power_events,
389        &sort_context,
390        auth_context,
391        &conflicted_events,
392        version,
393        &mut local_auth_cache,
394        create_ev,
395        &mut pl_cache,
396        &conflicted_keys,
397    );
398
399    let sort_set = &conflicted_events;
400
401    // Coordinate Projection Phase (Mainline distance mapping)
402    let mainline = build_mainline(&resolved, &sort_context);
403    let mut target_events: alloc::vec::Vec<&LeanEvent<Id, C>> = non_power_events.values().collect();
404    let mainline_distances =
405        compute_closest_mainline_positions(&mut target_events, &mainline, &sort_context);
406    let mainline_len = mainline.len();
407
408    // Semilattice Fold Phase
409    let mut key_winners = HashMap::new();
410    compute_lattice_coordinatized_winners(
411        &non_power_events,
412        &mainline_distances,
413        mainline_len,
414        &resolved,
415        auth_context,
416        sort_set,
417        version,
418        create_ev,
419        &mut key_winners,
420    );
421
422    // Merge Winners into Final Resolved State
423    let mut final_resolved = unconflicted_state;
424    for (k, v) in resolved {
425        final_resolved.insert(k, v);
426    }
427    for (k, ev) in key_winners {
428        final_resolved.insert(k, ev.event_id.clone());
429    }
430
431    drop(conflicted_events);
432    final_resolved
