3/// Adapter that lets `core::fmt::Display` values be streamed into a BLAKE3 hasher.
4pub(super) struct HashWriter<'a> {
5    pub(super) hasher: &'a mut blake3::Hasher,
6}
7
8impl core::fmt::Write for HashWriter<'_> {
9    #[inline]
10    fn write_str(&mut self, s: &str) -> core::fmt::Result {
11        self.hasher.update(s.as_bytes());
12        Ok(())
13    }
14}

16/// Truncate a string to fit within a `u16` length prefix (65535 bytes).
17///
18/// Valid Matrix events are capped at 64KiB total, so real event types and state keys
19/// can never reach this limit. Truncation only applies to malformed/adversarial input.
20#[inline]
21pub(crate) fn truncate_to_u16_limit(s: &str) -> (&str, u16) {
22    let limit = usize::from(u16::MAX);
23    let s_len = s.len();
24    if s_len > limit {
25        let mut end = limit;
26        while !s.is_char_boundary(end) {
27            end = end.saturating_sub(1);
28        }
29        (&s[..end], u16::try_from(end).unwrap())
30    } else {
31        (s, u16::try_from(s_len).unwrap())
32    }
33}

35/// Squeezes `LANES * 2` bytes out of a fresh BLAKE3 hasher and unpacks them into
36/// little-endian 16-bit lanes.
37///
38/// Lanes are unpacked in 128-lane (256-byte) windows so each [`blake3::OutputReader::fill`]
39/// call is a whole number of 64-byte blocks; the scratch buffer is fixed-size so no
40/// heap allocation happens on the hot insert path regardless of `LANES`.
41#[must_use]
42pub(crate) fn expand_lanes<const LANES: usize>(
43    feed: impl FnOnce(&mut blake3::Hasher),
44) -> [u16; LANES] {
45    let mut hasher = blake3::Hasher::new();
46    feed(&mut hasher);
47
48    let mut reader = hasher.finalize_xof();
49    let mut out = [0u16; LANES];
50    let mut scratch = [0u8; 256];
51    for chunk in out.chunks_mut(128) {
52        let n = chunk.len().wrapping_mul(2);
53        reader.fill(&mut scratch[..n]);
54        for (lane, pair) in chunk.iter_mut().zip(scratch[..n].chunks_exact(2)) {
55            *lane = u16::from_le_bytes([pair[0], pair[1]]);
56        }
57    }
58    out
59}

61/// Computes the lane expansion for a single state entry under the given
62/// domain-separation tag.
63///
64/// Input encoding: `len(type) || type || len(state_key) || state_key || event_id`
65/// where each `len()` is an unsigned 16-bit little-endian byte count.
66#[must_use]
67pub(crate) fn seed_lattice<const LANES: usize>(
68    dst: &[u8],
69    event_type: &str,
70    state_key: &str,
71    event_id: &dyn core::fmt::Display,
72) -> [u16; LANES] {
73    let (event_type, type_len) = truncate_to_u16_limit(event_type);
74    let (state_key, sk_len) = truncate_to_u16_limit(state_key);
75
76    expand_lanes(|hasher| {
77        hasher.update(dst);
78        hasher.update(&type_len.to_le_bytes());
79        hasher.update(event_type.as_bytes());
80        hasher.update(&sk_len.to_le_bytes());
81        hasher.update(state_key.as_bytes());
82
83        // `HashWriter::write_str` only forwards to `Hasher::update`, so the
84        // formatting result can never be an error.
85        let mut writer = HashWriter { hasher };
86        let _ = core::fmt::write(&mut writer, format_args!("{event_id}"));
87    })
88}

90/// Expands an opaque byte string under `dst`: `dst || bytes`.
91///
92/// Unlike the field and state-entry encodings this one carries no length prefix, so the
93/// caller must supply a byte string that cannot be confused with a neighboring element
94/// by an alternate split (for example by hashing a framed length itself).
95#[must_use]
96pub(crate) fn seed_bytes_lattice<const LANES: usize>(dst: &[u8], bytes: &[u8]) -> [u16; LANES] {
97    expand_lanes(|hasher| {
98        hasher.update(dst);
99        hasher.update(bytes);
100    })
101}

103/// Expands one key-value field under `dst`:
104/// `dst || len(key) || key || len(val) || val`.
105///
106/// Both lengths are unsigned 16-bit little-endian byte counts and both halves are
107/// truncated to that limit exactly like a state entry.
108#[must_use]
109pub(crate) fn seed_field_lattice<const LANES: usize>(
110    dst: &[u8],
111    key: &str,
112    val: &str,
113) -> [u16; LANES] {
114    let (key, key_len) = truncate_to_u16_limit(key);
115    let (val, val_len) = truncate_to_u16_limit(val);
116
117    expand_lanes(|hasher| {
118        hasher.update(dst);
119        hasher.update(&key_len.to_le_bytes());
120        hasher.update(key.as_bytes());
121        hasher.update(&val_len.to_le_bytes());
122        hasher.update(val.as_bytes());
123    })
124}

126/// Adds `src` into `dst` lane-wise with wrapping addition.
127///
128/// Processed in 8-lane chunks to assist SIMD auto-vectorization; any leftover lanes
129/// (only reachable for a `LANES` that is not a multiple of 8) are handled scalar-wise.
130#[inline]
131pub(crate) fn add_lattice<const LANES: usize>(dst: &mut [u16; LANES], src: &[u16; LANES]) {
132    let mut dst_chunks = dst.chunks_exact_mut(8);
133    let mut src_chunks = src.chunks_exact(8);
134    for (a, b) in (&mut dst_chunks).zip(&mut src_chunks) {
135        a[0] = a[0].wrapping_add(b[0]);
136        a[1] = a[1].wrapping_add(b[1]);
137        a[2] = a[2].wrapping_add(b[2]);
138        a[3] = a[3].wrapping_add(b[3]);
139        a[4] = a[4].wrapping_add(b[4]);
140        a[5] = a[5].wrapping_add(b[5]);
141        a[6] = a[6].wrapping_add(b[6]);
142        a[7] = a[7].wrapping_add(b[7]);
143    }
144    for (a, b) in dst_chunks
145        .into_remainder()
146        .iter_mut()
147        .zip(src_chunks.remainder())
148    {
149        *a = a.wrapping_add(*b);
150    }
151}

153/// Subtracts `src` from `dst` lane-wise with wrapping subtraction.
154///
155/// Processed in 8-lane chunks to assist SIMD auto-vectorization; see [`add_lattice`].
156#[inline]
157pub(crate) fn sub_lattice<const LANES: usize>(dst: &mut [u16; LANES], src: &[u16; LANES]) {
158    let mut dst_chunks = dst.chunks_exact_mut(8);
159    let mut src_chunks = src.chunks_exact(8);
160    for (a, b) in (&mut dst_chunks).zip(&mut src_chunks) {
161        a[0] = a[0].wrapping_sub(b[0]);
162        a[1] = a[1].wrapping_sub(b[1]);
163        a[2] = a[2].wrapping_sub(b[2]);
164        a[3] = a[3].wrapping_sub(b[3]);
165        a[4] = a[4].wrapping_sub(b[4]);
166        a[5] = a[5].wrapping_sub(b[5]);
167        a[6] = a[6].wrapping_sub(b[6]);
168        a[7] = a[7].wrapping_sub(b[7]);
169    }
170    for (a, b) in dst_chunks
171        .into_remainder()
172        .iter_mut()
173        .zip(src_chunks.remainder())
174    {
175        *a = a.wrapping_sub(*b);
176    }
177}

179/// Collapses a lattice into its 32-byte wire digest: `BLAKE3(S)`, where `S` is the
180/// little-endian serialization of the `LANES` 16-bit values.
181#[must_use]
182pub(crate) fn lattice_digest<const LANES: usize>(lattice: &[u16; LANES]) -> [u8; 32] {
183    let mut hasher = blake3::Hasher::new();
184    let mut scratch = [0u8; 256];
185    for chunk in lattice.chunks(128) {
186        let n = chunk.len().wrapping_mul(2);
187        for (pair, lane) in scratch[..n].chunks_exact_mut(2).zip(chunk) {
188            pair.copy_from_slice(&lane.to_le_bytes());
189        }
190        hasher.update(&scratch[..n]);
191    }
192    *hasher.finalize().as_bytes()
193}
