1//! Generic HAMT primitives used by higher-level state handling.
2//!
3//! Layout:
4//! - `hash`: keyed structural hashing for subtree identity
5//! - `codec`: dense on-disk encoding for persisted internal nodes
6//! - `delta`: subtree differencing for set isolation
7//! - `audit`: multi-root reachability audit for storage GC
8//! - `gc`: experimental incremental refcount bookkeeping for integrations
9//!   with a strictly linear root history (behind `unstable-refcount-gc`)
10//! - `tests`: regression coverage for the generic HAMT core
11//!

137/// A node in the 32-way CHAMP (Compressed Hash Array Mapped Prefix) trie.

143    pub nodemap: u32,

155    pub fn compute_structural_hash(
156        key: &[u8],
157        datamap: u32,
158        nodemap: u32,
159        leaves: &[(K, V)],
160        children: &[NodeRef<K, V>],
161    ) -> StructuralHash
162    where
163        K: HamtCodec,
164        V: HamtCodec,
165    {
166        let child_hashes = children
167            .iter()
168            .map(NodeRef::structural_hash)
169            .collect::<Vec<_>>();
170        let canonical_bytes =
171            PersistedInternalNode::encode_v1_parts(datamap, nodemap, leaves, &child_hashes);
172        let mut mac = StructuralHashBuilder::new(key);
173        mac.write(STRUCTURAL_HASH_DOMAIN_V1);
174        mac.write(&canonical_bytes);
175        mac.finalize()
176    }

305    /// Visits every key/value pair in the HAMT, resolving lazy children as
306    /// needed.
307    ///
308    /// This is the reusable traversal primitive for callers that need to
309    /// stream or collect the full contents of a subtree.
310    ///
311    /// # Errors
312    /// Returns the error from either `resolver` or `visitor`.
313    pub fn visit_entries<F, E>(
314        &self,
315        resolver: &mut F,
316        visitor: &mut impl FnMut(&K, &V) -> Result<(), E>,
317    ) -> Result<(), E>
318    where
319        F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
320    {
321        for (key, value) in &self.leaves {
322            visitor(key, value)?;
323        }
324
325        for child in &self.children {
326            let child_node = match child {
327                NodeRef::Resolved(node) => node.clone(),
328                NodeRef::Lazy(hash) => resolver(hash)?,
329            };
330            child_node.visit_entries(resolver, visitor)?;
331        }
332
333        Ok(())
334    }

344    /// Checks if any key-value entry in the HAMT satisfies a predicate.
345    ///
346    /// Traversal proceeds lazily across child nodes, returning `Ok(true)` and
347    /// stopping immediately on the first entry for which `predicate(&key, &value)`
348    /// returns `Ok(true)`. Returns `Ok(false)` if no matching entry is found.

350    /// # Errors
351    /// Returns [`HamtTraversalError::Resolve`] with the error emitted by
352    /// `resolver` or `predicate`, or [`HamtTraversalError::MaxDepthExceeded`]
353    /// if the walk recurses past the deepest depth a legitimately-built HAMT
354    /// can have.
355    pub fn any_entry<F, E>(
356        &self,
357        resolver: &mut F,
358        predicate: &mut impl FnMut(&K, &V) -> Result<bool, E>,
359    ) -> Result<bool, HamtTraversalError<E>>
360    where
361        F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
362    {
363        self.any_entry_inner(resolver, predicate, 0)
364    }

366    fn any_entry_inner<F, E>(
367        &self,
368        resolver: &mut F,
369        predicate: &mut impl FnMut(&K, &V) -> Result<bool, E>,
370        depth: usize,
371    ) -> Result<bool, HamtTraversalError<E>>
372    where
373        F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
374    {
375        if depth >= HAMT_MAX_DEPTH {
376            return Err(HamtTraversalError::MaxDepthExceeded { depth });
377        }
378        let next_depth = depth.saturating_add(1);
379
380        for (key, value) in &self.leaves {
381            if predicate(key, value).map_err(HamtTraversalError::Resolve)? {
382                return Ok(true);
383            }
384        }
385
386        for child in &self.children {
387            let child_node = match child {
388                NodeRef::Resolved(node) => node.clone(),
389                NodeRef::Lazy(hash) => resolver(hash).map_err(HamtTraversalError::Resolve)?,
390            };
391            if child_node.any_entry_inner(resolver, predicate, next_depth)? {
392                return Ok(true);
393            }
394        }
395
396        Ok(false)
397    }

411        &self,
412        resolver: &mut F,
413        predicate: &mut impl FnMut(&K, &V) -> Result<bool, E>,

417        V: Clone,

423    fn find_entry_inner<F, E>(
424        &self,
425        resolver: &mut F,
426        predicate: &mut impl FnMut(&K, &V) -> Result<bool, E>,
427        depth: usize,
428    ) -> Result<Option<(K, V)>, HamtTraversalError<E>>
429    where
430        K: Clone,
431        V: Clone,
432        F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
433    {
434        if depth >= HAMT_MAX_DEPTH {
435            return Err(HamtTraversalError::MaxDepthExceeded { depth });
436        }
437        let next_depth = depth.saturating_add(1);
438
439        for (key, value) in &self.leaves {
440            if predicate(key, value).map_err(HamtTraversalError::Resolve)? {
441                return Ok(Some((key.clone(), value.clone())));
442            }
443        }
444
445        for child in &self.children {
446            let child_node = match child {
447                NodeRef::Resolved(node) => node.clone(),
448                NodeRef::Lazy(hash) => resolver(hash).map_err(HamtTraversalError::Resolve)?,
449            };
450            if let Some(entry) = child_node.find_entry_inner(resolver, predicate, next_depth)? {
451                return Ok(Some(entry));
452            }
453        }
454
455        Ok(None)
456    }

458    /// Resolves which slot `path_hash` routes to at `depth` and returns what
459    /// occupies it, if anything.
460    ///
461    /// This is the single shared CHAMP descent step: it computes the slot,
462    /// tests it against `datamap`/`nodemap`, and resolves the compressed
463    /// index. Both [`get_by_path_hash`](Self::get_by_path_hash) and
464    /// [`search_by_path_hash`](Self::search_by_path_hash) build on it and
465    /// only differ in how they terminate at a leaf or follow a child.
466    fn slot_at(&self, path_hash: &StructuralHash, depth: usize) -> Slot<'_, K, V> {
467        let slot = bucket_index(path_hash, depth);
468        let bit = 1_u32 << slot;
469
470        if (self.datamap & bit) != 0 {
471            Slot::Leaf(&self.leaves[map_index(self.datamap, slot)])
472        } else if (self.nodemap & bit) != 0 {
473            Slot::Child(&self.children[map_index(self.nodemap, slot)])
474        } else {
475            Slot::Empty
476        }
477    }

479    fn get_by_path_hash_inner<Q>(
480        &self,
481        key: &Q,
482        path_hash: &StructuralHash,
483        depth: usize,
484    ) -> Option<&V>
485    where
486        K: Eq + Borrow<Q>,
487        Q: Eq + ?Sized,
488    {
489        if depth >= HAMT_MAX_DEPTH {
490            return None;
491        }
492        let next_depth = depth.saturating_add(1);
493
494        match self.slot_at(path_hash, depth) {
495            Slot::Leaf((stored_key, value)) => (stored_key.borrow() == key).then_some(value),
496            Slot::Child(NodeRef::Resolved(child)) => {
497                child.get_by_path_hash_inner(key, path_hash, next_depth)
498            }
499            Slot::Child(NodeRef::Lazy(_)) | Slot::Empty => None,
500        }
501    }

503    fn search_by_path_hash_inner<Q, F, E>(
504        &self,
505        key: &Q,
506        path_hash: &StructuralHash,
507        depth: usize,
508        resolver: &mut F,
509    ) -> Result<Option<V>, E>
510    where
511        K: Eq + Borrow<Q>,
512        V: Clone,
513        Q: Eq + ?Sized,
514        F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
515    {
516        if depth >= HAMT_MAX_DEPTH {
517            return Ok(None);
518        }
519        let next_depth = depth.saturating_add(1);
520
521        match self.slot_at(path_hash, depth) {
522            Slot::Leaf((stored_key, value)) => {
523                Ok((stored_key.borrow() == key).then(|| value.clone()))
524            }
525            Slot::Child(NodeRef::Resolved(child)) => {
526                child.search_by_path_hash_inner(key, path_hash, next_depth, resolver)
527            }
528            Slot::Child(NodeRef::Lazy(hash)) => {
529                let child = resolver(hash)?;
530                child.search_by_path_hash_inner(key, path_hash, next_depth, resolver)
531            }
532            Slot::Empty => Ok(None),
533        }
534    }

618        "bucket_index called at or beyond HAMT_MAX_DEPTH ({depth} >= {HAMT_MAX_DEPTH})"
619    );
620    let bit_offset = depth.saturating_mul(HAMT_BRANCH_BITS);
621    let byte_index = bit_offset / 8;
622    let bit_shift = bit_offset % 8;
623    let hash_len = hash.len();
624    debug_assert!(
625        byte_index < hash_len,
626        "byte_index out of bounds for StructuralHash ({byte_index} >= {hash_len})",
627    );
628
629    let mut word = u16::from(hash[byte_index]);
630    // No checked_add: byte_index < hash_len (32, asserted above) always, so
631    // byte_index + 1 <= 32 never overflows usize. A checked_add here can
632    // never observe None -- it's not a real safety margin, just an untestable
633    // dead branch (a coverage tool will flag its closing brace as unreached,
634    // correctly: reaching it would require byte_index near usize::MAX, which
635    // contradicts the precondition this function already asserts).
636    if let Some(next) = hash.get(byte_index.saturating_add(1)) {
637        word |= u16::from(*next) << 8;
638    }
639    usize::from((word >> bit_shift) & HAMT_BRANCH_MASK)
640}

642struct BuildEntry<K, V> {
643    key: K,
644    value: V,
645    path_hash: StructuralHash,
646}
647
648fn build_node<K, V>(
649    structural_key: &[u8],
650    entries: Vec<BuildEntry<K, V>>,
651    depth: usize,
652) -> Result<Arc<HamtNode<K, V>>, HamtBuildError>
653where
654    K: Hash + HamtCodec,
655    V: HamtCodec,
656{
657    if depth >= HAMT_MAX_DEPTH {
658        return Err(HamtBuildError::HashCollision {
659            depth,
660            bucket_size: entries.len(),
661        });
662    }
663
664    let mut buckets: Vec<Vec<BuildEntry<K, V>>> =
665        (0..HAMT_BRANCH_FACTOR).map(|_| Vec::new()).collect();
666    for entry in entries {
667        let slot = bucket_index(&entry.path_hash, depth);
668        buckets[slot].push(entry);
669    }
670
671    let mut datamap = 0_u32;
672    let mut nodemap = 0_u32;
673    let mut leaves = Vec::new();
674    let mut children = Vec::new();
675
676    for (slot, mut bucket) in buckets.into_iter().enumerate() {
677        if bucket.is_empty() {
678            continue;
679        }
680
681        let bit = 1_u32 << slot;
682        if bucket.len() == 1 {
683            datamap |= bit;
684            let entry = bucket
685                .pop()
686                .expect("singleton bucket must contain one entry");
687            leaves.push((entry.key, entry.value));
688            continue;
689        }
690
691        let next_depth = depth.saturating_add(1);
692        if next_depth >= HAMT_MAX_DEPTH {
693            return Err(HamtBuildError::HashCollision {
694                depth,
695                bucket_size: bucket.len(),
696            });
697        }
698
699        nodemap |= bit;
700        let child = build_node(structural_key, bucket, next_depth)?;
701        children.push(NodeRef::Resolved(child));
702    }
703
704    let structural_hash =
705        HamtNode::compute_structural_hash(structural_key, datamap, nodemap, &leaves, &children);
706
707    Ok(Arc::new(HamtNode {
708        datamap,
709        nodemap,
710        leaves,
711        children,
712        structural_hash,
713    }))
714}

739/// Builds a full HAMT from an iterator of key/value entries using a custom
740/// per-key path hash function.
741///
742/// Callers that build a tree with this function must use the matching custom
743/// lookup and mutation APIs:
744/// - [`HamtNode::get_with_key_hash`]
745/// - [`HamtNode::search_with_key_hash`]
746/// - [`insert_with_key_hash`]
747/// - [`remove_with_key_hash`]
748/// - [`HamtNode::get_by_path_hash`]
749/// - [`HamtNode::search_by_path_hash`]
750///
751/// The default [`HamtNode::get`], [`HamtNode::search`], [`insert`], and
752/// [`remove`] helpers derive their own routing hash using the same keyed
753/// structural hash used by [`build_hamt`], and are only correct for trees
754/// built by [`build_hamt`].
755///
756/// This is the most general builder entry point and is useful when a caller
757/// already has a pre-hashed key stream.
758///
759/// # Errors
760/// Returns [`HamtBuildError::HashCollision`] if the input exhausts the
761/// available trie depth before entries can be separated into distinct slots.
762pub fn build_hamt_with_key_hash<K, V, I, F>(
763    structural_key: &[u8],
764    entries: I,
765    mut key_hash: F,
766) -> Result<Arc<HamtNode<K, V>>, HamtBuildError>
767where
768    K: Hash + HamtCodec,
769    V: HamtCodec,
770    I: IntoIterator<Item = (K, V)>,
771    F: FnMut(&K) -> StructuralHash,
772{
773    let entries = entries
774        .into_iter()
775        .map(|(key, value)| {
776            let path_hash = key_hash(&key);
777            BuildEntry {
778                key,
779                value,
780                path_hash,
781            }
782        })
783        .collect::<Vec<_>>();
784    build_node(structural_key, entries, 0)
785}

787/// Builds a root handle for a freshly constructed HAMT.
788///
789/// This is a convenience helper for downstream code that already knows the
790/// resolved lattice and wants both the tree and the root identity in one call.
791/// # Errors
792/// Returns the same build errors as [`build_hamt`].
793pub fn build_hamt_root_handle<K, V, I>(
794    structural_key: &[u8],
795    lattice: &crate::state::LtHash,

808/// Errors that can occur while incrementally mutating a HAMT via
809/// [`insert`] or [`remove`].
810#[derive(Clone, Debug, PartialEq, Eq)]
811pub enum HamtMutateError<E> {
812    /// Too many entries collided into the same slot after exhausting the
813    /// available hash depth.
814    HashCollision { depth: usize, bucket_size: usize },
815    /// Traversal exceeded maximum allowable HAMT depth.
816    MaxDepthExceeded { depth: usize },
817    /// The resolver failed to load a lazy child that the mutation needed to
818    /// descend into.
819    Resolve(E),
820}
821
822impl<E> fmt::Display for HamtMutateError<E>
823where
824    E: fmt::Display,
825{
826    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
827        match self {
828            Self::HashCollision { depth, bucket_size } => write!(
829                f,
830                "hamt mutation hash collision at depth {depth} with bucket size {bucket_size}"
831            ),
832            Self::MaxDepthExceeded { depth } => {
833                write!(f, "hamt mutation max depth exceeded at depth {depth}")
834            }
835            Self::Resolve(err) => write!(f, "hamt mutation resolver failed: {err}"),
836        }
837    }
838}

916fn insert_node<K, V, F, E>(
917    node: &Arc<HamtNode<K, V>>,
918    structural_key: &[u8],
919    key: K,
920    value: V,
921    path_hash: &StructuralHash,

948struct InsertStep<K, V> {
949    key: K,
950    value: V,
951    path_hash: StructuralHash,
952    slot: usize,
953    bit: u32,
954}
955
956fn insert_node_with_ctx<K, V, KeyHash, F, E>(
957    node: &Arc<HamtNode<K, V>>,
958    key: K,
959    value: V,
960    path_hash: StructuralHash,
961    depth: usize,
962    ctx: &mut InsertCtx<'_, K, V, KeyHash, F>,
963) -> MutateResult<K, V, E>
964where
965    K: Hash + Eq + Clone + HamtCodec,
966    V: Clone + HamtCodec,
967    KeyHash: FnMut(&K) -> StructuralHash,
968    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
969{
970    if depth >= HAMT_MAX_DEPTH {
971        return Err(HamtMutateError::HashCollision {
972            depth,
973            bucket_size: node.leaves.len().saturating_add(1),
974        });
975    }
976
977    let structural_key = ctx.structural_key;
978    let slot = bucket_index(&path_hash, depth);
979    let bit = 1_u32 << slot;
980    let step = InsertStep {
981        key,
982        value,
983        path_hash,
984        slot,
985        bit,
986    };
987
988    if (node.datamap & bit) != 0 {
989        let idx = map_index(node.datamap, slot);
990        if node.leaves[idx].0 == step.key {
991            let mut leaves = node.leaves.clone();
992            let old_value = core::mem::replace(&mut leaves[idx].1, step.value);
993            let new_node = rebuild_node(
994                structural_key,
995                node.datamap,
996                node.nodemap,
997                leaves,
998                node.children.clone(),
999            );
1000            emit_node_to_sink(&new_node, ctx.sink);
1001            return Ok((new_node, Some(old_value)));
1002        }
1003
1004        return insert_into_leaf_slot(node, structural_key, step, depth.saturating_add(1), ctx);
1005    }
1006
1007    if (node.nodemap & bit) != 0 {
1008        return insert_into_child_slot(node, structural_key, step, depth.saturating_add(1), ctx);
1009    }
1010
1011    // Empty slot: insert directly as a new leaf.
1012    let mut leaves = node.leaves.clone();
1013    let idx = map_index(node.datamap, step.slot);
1014    leaves.insert(idx, (step.key, step.value));
1015    let new_datamap = node.datamap | step.bit;
1016    let new_node = rebuild_node(
1017        structural_key,
1018        new_datamap,
1019        node.nodemap,
1020        leaves,
1021        node.children.clone(),
1022    );
1023    emit_node_to_sink(&new_node, ctx.sink);
1024    Ok((new_node, None))
1025}

1027fn insert_into_leaf_slot<K, V, KeyHash, F, E>(
1028    node: &Arc<HamtNode<K, V>>,
1029    structural_key: &[u8],
1030    step: InsertStep<K, V>,
1031    next_depth: usize,
1032    ctx: &mut InsertCtx<'_, K, V, KeyHash, F>,
1033) -> MutateResult<K, V, E>
1034where
1035    K: Hash + Eq + Clone + HamtCodec,
1036    V: Clone + HamtCodec,
1037    KeyHash: FnMut(&K) -> StructuralHash,
1038    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
1039{
1040    let InsertStep {
1041        key,
1042        value,
1043        path_hash,
1044        slot,
1045        bit,
1046    } = step;
1047    let idx = map_index(node.datamap, slot);
1048
1049    let (existing_key, existing_value) = node.leaves[idx].clone();
1050    let existing_path_hash = (ctx.key_hash)(&existing_key);
1051    let split_entries = vec![
1052        BuildEntry {
1053            key: existing_key,
1054            value: existing_value,
1055            path_hash: existing_path_hash,
1056        },
1057        BuildEntry {
1058            key,
1059            value,
1060            path_hash,
1061        },
1062    ];
1063    let child = build_node(structural_key, split_entries, next_depth)?;
1064    emit_split_nodes_to_sink(&child, ctx.sink);
1065
1066    let mut leaves = node.leaves.clone();
1067    leaves.remove(idx);
1068    let new_datamap = node.datamap & !bit;
1069    let new_nodemap = node.nodemap | bit;
1070    let child_idx = map_index(new_nodemap, slot);
1071    let mut children = node.children.clone();
1072    children.insert(child_idx, NodeRef::Resolved(child));
1073    let new_node = rebuild_node(structural_key, new_datamap, new_nodemap, leaves, children);
1074    emit_node_to_sink(&new_node, ctx.sink);
1075    Ok((new_node, None))
1076}

1078fn insert_into_child_slot<K, V, KeyHash, F, E>(
1079    node: &Arc<HamtNode<K, V>>,
1080    structural_key: &[u8],
1081    step: InsertStep<K, V>,
1082    next_depth: usize,
1083    ctx: &mut InsertCtx<'_, K, V, KeyHash, F>,
1084) -> MutateResult<K, V, E>
1085where
1086    K: Hash + Eq + Clone + HamtCodec,
1087    V: Clone + HamtCodec,
1088    KeyHash: FnMut(&K) -> StructuralHash,
1089    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
1090{
1091    let InsertStep {
1092        key,
1093        value,
1094        path_hash,
1095        slot,
1096        ..
1097    } = step;
1098    let idx = map_index(node.nodemap, slot);
1099    let child = match &node.children[idx] {
1100        NodeRef::Resolved(child) => child.clone(),
1101        NodeRef::Lazy(hash) => (ctx.resolver)(hash).map_err(HamtMutateError::Resolve)?,
1102    };
1103    let (new_child, old_value) =
1104        insert_node_with_ctx(&child, key, value, path_hash, next_depth, ctx)?;
1105    let mut children = node.children.clone();
1106    children[idx] = NodeRef::Resolved(new_child);
1107    let new_node = rebuild_node(
1108        structural_key,
1109        node.datamap,
1110        node.nodemap,
1111        node.leaves.clone(),
1112        children,
1113    );
1114    emit_node_to_sink(&new_node, ctx.sink);
1115    Ok((new_node, old_value))
1116}

1190/// What remains of a subtree after a leaf was removed from it.
1191#[derive(Debug)]
1192enum RemoveOutcome<K, V> {
1193    /// The subtree has no entries left.
1194    Empty,
1195    /// The subtree collapsed to a single leaf, which must be inlined into
1196    /// the parent's `datamap` rather than kept as a child node — the same
1197    /// invariant `build_node` enforces when building from scratch.
1198    Leaf(K, V),
1199    /// The subtree still has two or more entries and remains a node.
1200    Node(Arc<HamtNode<K, V>>),
1201}

1203/// Builds the outcome for a node whose contents just changed, collapsing it
1204/// to `Leaf`/`Empty` if it no longer has enough entries to justify being a
1205/// node, so the structural hash always matches what `build_node` would have
1206/// produced for the same final key set.
1207fn finalize_after_removal<K, V>(
1208    structural_key: &[u8],
1209    datamap: u32,
1210    nodemap: u32,
1211    leaves: Vec<(K, V)>,
1212    children: Vec<NodeRef<K, V>>,
1213) -> RemoveOutcome<K, V>
1214where
1215    K: Hash + HamtCodec,
1216    V: HamtCodec,
1217{
1218    if nodemap == 0 {
1219        if leaves.is_empty() {
1220            return RemoveOutcome::Empty;
1221        }
1222        if leaves.len() == 1 {
1223            let (key, value) = leaves
1224                .into_iter()
1225                .next()
1226                .expect("checked leaves.len() == 1 above");
1227            return RemoveOutcome::Leaf(key, value);
1228        }
1229    }
1230    RemoveOutcome::Node(rebuild_node(
1231        structural_key,
1232        datamap,
1233        nodemap,
1234        leaves,
1235        children,
1236    ))
1237}

1245/// Removes a key from a HAMT via `O(log S)` path-copying, without
1246/// rebuilding the whole tree.
1247///
1248/// Collapses any subtree that drops to a single leaf back into its
1249/// parent's `datamap`, cascading as needed, so the resulting tree's
1250/// `structural_hash` always matches what [`build_hamt`] would have
1251/// produced from the same final key set.
1252///
1253/// Returns the new root and the removed value, if `key` was present.
1254///
1255/// This helper is only correct for trees built with [`build_hamt`]. If the
1256/// tree was built with [`build_hamt_with_key_hash`], use
1257/// [`remove_with_key_hash`] with the same routing function instead.
1258///
1259/// # Errors
1260/// Returns [`HamtMutateError::Resolve`] if `resolver` fails to load a lazy
1261/// child on the path to `key`.
1262pub fn remove<K, V, Q, F, E>(
1263    node: &Arc<HamtNode<K, V>>,
1264    structural_key: &[u8],
1265    key: &Q,
1266    resolver: &mut F,
1267) -> MutateResult<K, V, E>
1268where
1269    K: Hash + Eq + Borrow<Q> + Clone + HamtCodec,
1270    V: Clone + HamtCodec,
1271    Q: Hash + Eq + ?Sized,
1272    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
1273{
1274    let path_hash = key_path_hash(structural_key, key);
1275    let mut sink = |_: &Arc<HamtNode<K, V>>| {};
1276    let (outcome, old_value) = {
1277        let mut ctx = RemoveCtx {
1278            structural_key,
1279            resolver,
1280            sink: &mut sink,
1281        };
1282        remove_node_with_ctx(node, key, &path_hash, 0, &mut ctx)?
1283    };
1284    let new_root = finalize_remove_root(
1285        structural_key,
1286        outcome,
1287        |leaf_key| bucket_index(&key_path_hash(structural_key, leaf_key), 0),
1288        &mut sink,
1289    );
1290    Ok((new_root, old_value))
1291}

1293fn remove_node_with_ctx<K, V, Q, F, E>(
1294    node: &Arc<HamtNode<K, V>>,
1295    key: &Q,
1296    path_hash: &StructuralHash,
1297    depth: usize,
1298    ctx: &mut RemoveCtx<'_, K, V, F>,
1299) -> RemoveStepResult<K, V, E>
1300where
1301    K: Hash + Eq + Borrow<Q> + Clone + HamtCodec,
1302    V: Clone + HamtCodec,
1303    Q: Eq + ?Sized,
1304    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
1305{
1306    if depth >= HAMT_MAX_DEPTH {
1307        return Err(HamtMutateError::MaxDepthExceeded { depth });
1308    }
1309
1310    let slot = bucket_index(path_hash, depth);
1311    let bit = 1_u32 << slot;
1312    let structural_key = ctx.structural_key;
1313
1314    if (node.datamap & bit) != 0 {
1315        let idx = map_index(node.datamap, slot);
1316        if node.leaves[idx].0.borrow() != key {
1317            return Ok((RemoveOutcome::Node(node.clone()), None));
1318        }
1319
1320        let old_value = node.leaves[idx].1.clone();
1321        let mut leaves = node.leaves.clone();
1322        leaves.remove(idx);
1323        let new_datamap = node.datamap & !bit;
1324        let outcome = finalize_after_removal(
1325            structural_key,
1326            new_datamap,
1327            node.nodemap,
1328            leaves,
1329            node.children.clone(),
1330        );
1331        if let RemoveOutcome::Node(ref new_node) = outcome {
1332            emit_node_to_sink(new_node, ctx.sink);
1333        }
1334        return Ok((outcome, Some(old_value)));
1335    }
1336
1337    if (node.nodemap & bit) == 0 {
1338        return Ok((RemoveOutcome::Node(node.clone()), None));
1339    }
1340
1341    let idx = map_index(node.nodemap, slot);
1342    let child = match &node.children[idx] {
1343        NodeRef::Resolved(child) => child.clone(),
1344        NodeRef::Lazy(hash) => (ctx.resolver)(hash).map_err(HamtMutateError::Resolve)?,
1345    };
1346
1347    let next_depth = depth.saturating_add(1);
1348    if next_depth >= HAMT_MAX_DEPTH {
1349        return Err(HamtMutateError::MaxDepthExceeded { depth: next_depth });
1350    }
1351    let (child_outcome, old_value) = remove_node_with_ctx(&child, key, path_hash, next_depth, ctx)?;
1352    if old_value.is_none() {
1353        return Ok((RemoveOutcome::Node(node.clone()), None));
1354    }
1355
1356    match child_outcome {
1357        RemoveOutcome::Empty => {
1358            let mut children = node.children.clone();
1359            children.remove(idx);
1360            let new_nodemap = node.nodemap & !bit;
1361            let outcome = finalize_after_removal(
1362                structural_key,
1363                node.datamap,
1364                new_nodemap,
1365                node.leaves.clone(),
1366                children,
1367            );
1368            if let RemoveOutcome::Node(ref new_node) = outcome {
1369                emit_node_to_sink(new_node, ctx.sink);
1370            }
1371            Ok((outcome, old_value))
1372        }
1373        RemoveOutcome::Leaf(leaf_key, leaf_value) => {
1374            let mut children = node.children.clone();
1375            children.remove(idx);
1376            let new_nodemap = node.nodemap & !bit;
1377            let new_datamap = node.datamap | bit;
1378            let leaf_idx = map_index(node.datamap, slot);
1379            let mut leaves = node.leaves.clone();
1380            leaves.insert(leaf_idx, (leaf_key, leaf_value));
1381            let outcome =
1382                finalize_after_removal(structural_key, new_datamap, new_nodemap, leaves, children);
1383            if let RemoveOutcome::Node(ref new_node) = outcome {
1384                emit_node_to_sink(new_node, ctx.sink);
1385            }
1386            Ok((outcome, old_value))
1387        }
1388        RemoveOutcome::Node(new_child) => {
1389            let mut children = node.children.clone();
1390            children[idx] = NodeRef::Resolved(new_child);
1391            let new_node = rebuild_node(
1392                structural_key,
1393                node.datamap,
1394                node.nodemap,
1395                node.leaves.clone(),
1396                children,
1397            );
1398            emit_node_to_sink(&new_node, ctx.sink);
1399            Ok((RemoveOutcome::Node(new_node), old_value))
1400        }
1401    }
1402}

1404fn finalize_remove_root<K, V, F>(
1405    structural_key: &[u8],
1406    outcome: RemoveOutcome<K, V>,
1407    mut root_slot_for_leaf: F,
1408    sink: &mut MutationSink<'_, K, V>,
1409) -> Arc<HamtNode<K, V>>
1410where
1411    K: Hash + Clone + HamtCodec,
1412    V: Clone + HamtCodec,
1413    F: FnMut(&K) -> usize,
1414{
1415    match outcome {
1416        RemoveOutcome::Empty => {
1417            let root = rebuild_node(structural_key, 0, 0, Vec::new(), Vec::new());
1418            emit_node_to_sink(&root, sink);
1419            root
1420        }
1421        RemoveOutcome::Leaf(leaf_key, leaf_value) => {
1422            let root_slot = root_slot_for_leaf(&leaf_key);
1423            let leaves = vec![(leaf_key, leaf_value)];
1424            let root = rebuild_node(structural_key, 1_u32 << root_slot, 0, leaves, Vec::new());
1425            emit_node_to_sink(&root, sink);
1426            root
1427        }
1428        RemoveOutcome::Node(new_root) => new_root,
1429    }
1430}

1501/// Persists a single mutation (insert if `Some`, remove if `None`) on top of `prev_root`.
1502///
1503/// Emits all created spine nodes inline during the mutation descent without a second
1504/// post-hoc tree traversal or redundant lazy-child resolutions.
1505///
1506/// Returns:
1507/// - The new root `Arc<HamtNode<K, V>>`
1508/// - The displaced / removed value, if any (needed for lattice subtraction)

1511/// # Publication Contract

1541        let mut sink = |node: &Arc<HamtNode<K, V>>| {

1543                node.structural_hash,

1545            ));
1546        };
1547        if let Some(val) = value {
1548            let path_hash = key_path_hash(structural_key, &key);
1549            let mut key_hash = |k: &K| key_path_hash(structural_key, k);
1550            let mut ctx = InsertCtx {
1551                structural_key,

1651/// Persists a batch of mutations producing a SINGLE published final root.
1652///
1653/// # Publication Contract — READ BEFORE USE
1654/// This function is ONLY correct when **no intermediate roots are published** (e.g.
1655/// bulk initial state build or a single coalesced update). If any intermediate state group
1656/// is published or referenced elsewhere, you MUST use [`persist_chain`] instead; otherwise,
1657/// intermediate state groups will point to nodes that were never persisted.
1658///
1659/// Returns the final root, all displaced values per mutation (for lattice arithmetic),
1660/// and the encoded new nodes.
1661///
1662/// # Performance Note
1663/// Callers using lazy resolvers should pass a memoizing resolver (or use [`persist_chain`])
1664/// to avoid repeated backend fetches during the end-of-batch diff pass.
1665///
1666/// # Custom-hash roots
1667/// Same default-hash-only routing as [`persist_mutation`]; see its
1668/// "Custom-hash roots" section — do not call this on a
1669/// [`build_hamt_with_key_hash`] tree. Use
1670/// [`persist_mutations_with_key_hash`] instead.
1671///
1672/// # Errors
1673/// Returns [`HamtMutateError`] if `resolver` fails to load a lazy child or depth is exhausted.
1674pub fn persist_mutations<K, V, I, F, E>(
1675    prev_root: &Arc<HamtNode<K, V>>,
1676    structural_key: &[u8],
1677    mutations: I,
1678    resolver: &mut F,
1679) -> PersistMutationsResult<K, V, E>
1680where
1681    K: Hash + Eq + Clone + HamtCodec,
1682    V: Hash + Clone + HamtCodec,
1683    I: IntoIterator<Item = (K, Option<V>)>,
1684    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
1685{
1686    let mut current_root = prev_root.clone();
1687    let mut displaced_vec = Vec::new();
1688
1689    for (key, opt_val) in mutations {
1690        let (next_root, displaced) = if let Some(val) = opt_val {
1691            insert(&current_root, structural_key, key, val, resolver)?
1692        } else {
1693            remove(&current_root, structural_key, &key, resolver)?
1694        };
1695        displaced_vec.push(displaced);
1696        current_root = next_root;
1697    }
1698
1699    if current_root.structural_hash == prev_root.structural_hash {
1700        return Ok((current_root, displaced_vec, Vec::new()));
1701    }
1702
1703    let delta = diff_node_hashes(prev_root, &current_root, resolver)?;
1704    let new_set: crate::HashSet<StructuralHash> = delta.new_node_hashes.into_iter().collect();
1705
1706    let mut created = Vec::with_capacity(new_set.len());
1707    let mut stack = vec![current_root.clone()];
1708    while let Some(node) = stack.pop() {
1709        if new_set.contains(&node.structural_hash) {
1710            let bytes = PersistedInternalNode::from(node.as_ref()).encode_v1();
1711            created.push((node.structural_hash, bytes));
1712            for child in &node.children {
1713                if let NodeRef::Resolved(child_node) = child {
1714                    if new_set.contains(&child_node.structural_hash) {
1715                        stack.push(child_node.clone());
1716                    }
1717                }
1718            }
1719        }
1720    }
1721
1722    Ok((current_root, displaced_vec, created))
1723}

1811/// Returns [`HamtMutateError`] if `resolver` fails to load a lazy child or depth is exhausted.
1812pub fn persist_chain<K, V, I, F, E>(
1813    prev_root: &Arc<HamtNode<K, V>>,
1814    structural_key: &[u8],
1815    mutations: I,
1816    resolver: &mut F,
1817) -> Result<Vec<ChainStep<K, V>>, HamtMutateError<E>>
1818where
1819    K: Hash + Eq + Clone + HamtCodec,
1820    V: Hash + Clone + HamtCodec,
1821    I: IntoIterator<Item = (K, Option<V>)>,
1822    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
1823{
1824    let mut steps = Vec::new();
1825    let mut current_root = prev_root.clone();
1826
1827    for (key, opt_val) in mutations {
1828        let (next_root, displaced, created) =
1829            persist_mutation(&current_root, structural_key, key, opt_val, resolver)?;
1830        steps.push(ChainStep {
1831            root_hash: next_root.structural_hash,
1832            root: next_root.clone(),
1833            displaced,
1834            created,
1835        });
1836        current_root = next_root;
1837    }
1838
1839    Ok(steps)
1840}

1889/// Descends into a batch of persisted node byte slices for their targeted keys.
1890///
1891/// `nodes_and_keys` contains the expected node hash, encoded bytes, depth, and
1892/// target key hashes for each fetched node. All entries in a single `nodes_and_keys`
1893/// call must share the same `depth` (the current level of breadth-first descent).
1894/// Each node is evaluated only against the keys that routed to it at that depth.
1895/// `key_hash_fn` computes the `KeyPathHash` of a decoded leaf key to verify exact matches.
1896///
1897/// # Depth Contract
1898/// All tuples in `nodes_and_keys` must specify the same `depth` level. Successful
1899/// descent produces [`DescendResult::pending`] entries representing child nodes at `depth + 1`.
1900///
1901/// # Absence Invariant
1902/// A key is reported in `absent` ONLY when:
1903/// 1. Its routed slot bit is clear in both `datamap` and `nodemap`.
1904/// 2. Its routed slot bit is set in `datamap`, but the decoded leaf key at that slot has a
1905///    different path hash. (Note: this is sound because this HAMT rejects multi-leaf bucket
1906///    overflow with [`HamtMutateError::HashCollision`] rather than storing collision lists).
1907///
1908/// # Errors
1909/// Returns [`DescendError::Decode`] if any node byte slice fails v1 parsing, or
1910/// [`DescendError::CorruptNode`] if internal bitmap invariants are violated.
1911pub fn descend_level<K, V, KeyHash>(
1912    structural_key: &[u8],
1913    nodes_and_keys: &[(StructuralHash, &[u8], usize, &[KeyPathHash])],
1914    mut key_hash_fn: KeyHash,
1915) -> Result<DescendResult<V>, DescendError>
1916where
1917    K: HamtCodec + Eq + Hash,
1918    V: HamtCodec + Clone + Hash,
1919    KeyHash: FnMut(&K) -> KeyPathHash,
1920{
1921    if let Some(&(_, _, first_depth, _)) = nodes_and_keys.first() {
1922        // Runtime check: mixed-depth entries would merge requests from different
1923        // levels into one frontier, misrouting subsequent lookups. The
1924        // debug_assert! was previously only checked in debug builds.
1925        if !nodes_and_keys.iter().all(|&(_, _, d, _)| d == first_depth) {
1926            return Err(DescendError::CorruptNode(
1927                "mixed-depth entries in a single descend_level call",
1928            ));
1929        }
1930    }
1931
1932    let mut found = Vec::new();
1933    let mut absent = Vec::new();
1934    let mut pending_map: crate::HashMap<StructuralHash, Vec<KeyPathHash>> =
1935        crate::HashMap::default();
1936
1937    for &(expected_hash, node_bytes, depth, target_keys) in nodes_and_keys {
1938        let node = PersistedInternalNode::<K, V>::decode_v1_unverified(node_bytes)
1939            .map_err(DescendError::Decode)?
1940            .into_hamt_node_verified(structural_key, expected_hash)
1941            .map_err(|_| {
1942                DescendError::CorruptNode("decoded node contents do not match requested node hash")
1943            })?;
1944
1945        for &req_hash in target_keys {
1946            if depth >= HAMT_MAX_DEPTH {
1947                return Err(DescendError::CorruptNode(
1948                    "HAMT depth exceeds routing limit",
1949                ));
1950            }
1951            let slot = bucket_index(&req_hash, depth);
1952            let bit = 1_u32 << slot;
1953
1954            if (node.datamap & bit) != 0 {
1955                let leaf_idx = map_index(node.datamap, slot);
1956                let (leaf_key, val) = node
1957                    .leaves
1958                    .get(leaf_idx)
1959                    .ok_or(DescendError::CorruptNode("leaf index out of bounds"))?;
1960                // Lazy slot validation: only check the leaf being read, not
1961                // the entire datamap. In a correct CHAMP HAMT, a leaf at
1962                // slot `s` must satisfy bucket_index(key_hash(leaf_key), depth) == s.
1963                if depth < HAMT_MAX_DEPTH {
1964                    let expected_slot = bucket_index(&key_hash_fn(leaf_key), depth);
1965                    if expected_slot != slot {
1966                        return Err(DescendError::CorruptNode("leaf routed to wrong slot"));
1967                    }
1968                }
1969                if key_hash_fn(leaf_key) == req_hash {
1970                    found.push((req_hash, val.clone()));
1971                } else {
1972                    // Different leaf occupies this exact slot: key is proven absent
1973                    // (Sound because this HAMT rejects bucket overflow with HashCollision)
1974                    absent.push(req_hash);
1975                }
1976            } else if (node.nodemap & bit) != 0 {
1977                let child_idx = map_index(node.nodemap, slot);
1978                let child_hash = node
1979                    .children
1980                    .get(child_idx)
1981                    .ok_or(DescendError::CorruptNode("child index out of bounds"))?
1982                    .structural_hash();
1983                pending_map.entry(child_hash).or_default().push(req_hash);
1984            } else {
1985                // Bitmap slot is clear: proven, terminal absence
1986                absent.push(req_hash);
1987            }
1988        }
1989    }
1990
1991    let pending = pending_map.into_iter().collect();
1992
1993    Ok(DescendResult {
1994        found,
1995        absent,
1996        pending,
1997    })
