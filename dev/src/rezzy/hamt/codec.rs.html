7/// Wire version of the current persisted-node layout (32-byte structural
8/// hashes, inline leaves, then child hashes in nodemap order).
9///
10/// Recorded in the first byte of every encoded node. The storage key
11/// selected by a lookup commits to the exact bytes written, so a node can
12/// only ever be decoded under the version that produced it.
13pub(crate) const HAMT_WIRE_VERSION: u8 = 0x02;
14
15/// Wire version of the original persisted-node layout, which used 16-byte
16/// structural hashes (pre-`e349d0f`).
17///
18/// The layout change from 16- to 32-byte hashes is not recoverable from the
19/// bytes alone, so `0x01` records are rejected explicitly instead of being
20/// silently misparsed. [`decode_v1_legacy_unverified`] exists only for
21/// diagnostics; it cannot re-derive the storage key the record was written
22/// under, so its output is not verifiable.
23pub(crate) const LEGACY_WIRE_VERSION_16_BYTE_HASHES: u8 = 0x01;
24
25/// Custom binary codec for HAMT leaf payloads.
26///
27/// This stays explicit and versioned instead of delegating persistence
28/// semantics to a generic serializer.
29pub trait HamtCodec: Sized {
30    /// Encodes the value into the provided buffer.
31    fn encode_hamt(&self, out: &mut Vec<u8>);
32
33    /// Decodes the value from the provided buffer cursor.
34    ///
35    /// # Errors
36    /// Returns an error if the input buffer is too short or the encoded
37    /// payload is malformed.
38    fn decode_hamt(input: &[u8], cursor: &mut usize) -> Result<Self, &'static str>;
39}
40
41macro_rules! impl_fixed_hamt_codec {
42    ($($ty:ty),* $(,)?) => {
43        $(
44            impl HamtCodec for $ty {
45                #[inline]
46                fn encode_hamt(&self, out: &mut Vec<u8>) {
47                    out.extend_from_slice(&self.to_le_bytes());
48                }
49
50                #[inline]
51                fn decode_hamt(input: &[u8], cursor: &mut usize) -> Result<Self, &'static str> {
52                    let width = core::mem::size_of::<$ty>();
53                    let end = cursor
54                        .checked_add(width)
55                        .ok_or("HAMT codec cursor overflow")?;
56                    let bytes = input
57                        .get(*cursor..end)
58                        .ok_or("HAMT codec buffer too short")?;
59                    let mut raw = [0u8; core::mem::size_of::<$ty>()];
60                    raw.copy_from_slice(bytes);
61                    *cursor = end;
62                    Ok(<$ty>::from_le_bytes(raw))
63                }
64            }
65        )*
66    };
67}
68
69impl_fixed_hamt_codec!(u8, u16, u32, u64, u128, i8, i16, i32, i64, i128);

109impl HamtCodec for bool {
110    #[inline]
111    fn encode_hamt(&self, out: &mut Vec<u8>) {
112        out.push(u8::from(*self));
113    }
114
115    #[inline]
116    fn decode_hamt(input: &[u8], cursor: &mut usize) -> Result<Self, &'static str> {
117        let end = cursor.checked_add(1).ok_or("HAMT codec cursor overflow")?;
118        let byte = *input.get(*cursor).ok_or("HAMT codec buffer too short")?;
119        *cursor = end;
120        match byte {
121            0 => Ok(false),
122            1 => Ok(true),
123            _ => Err("HAMT codec invalid bool encoding"),
124        }
125    }
126}

128impl HamtCodec for String {
129    fn encode_hamt(&self, out: &mut Vec<u8>) {
130        let len = u32::try_from(self.len()).expect("string too long for HAMT codec");
131        out.extend_from_slice(&len.to_le_bytes());
132        out.extend_from_slice(self.as_bytes());
133    }
134
135    fn decode_hamt(input: &[u8], cursor: &mut usize) -> Result<Self, &'static str> {
136        let len_end = cursor.checked_add(4).ok_or("HAMT codec cursor overflow")?;
137        let len_bytes = input
138            .get(*cursor..len_end)
139            .ok_or("HAMT codec buffer too short")?;
140        let mut raw = [0u8; 4];
141        raw.copy_from_slice(len_bytes);
142        *cursor = len_end;
143
144        let len = u32::from_le_bytes(raw) as usize;
145        let end = cursor
146            .checked_add(len)
147            .ok_or("HAMT codec cursor overflow")?;
148        let bytes = input
149            .get(*cursor..end)
150            .ok_or("HAMT codec buffer too short")?;
151        *cursor = end;
152        String::from_utf8(bytes.to_vec()).map_err(|_| "HAMT codec invalid UTF-8")
153    }
154}

184impl<A: HamtCodec, B: HamtCodec> HamtCodec for (A, B) {
185    fn encode_hamt(&self, out: &mut Vec<u8>) {
186        self.0.encode_hamt(out);
187        self.1.encode_hamt(out);
188    }
189
190    fn decode_hamt(input: &[u8], cursor: &mut usize) -> Result<Self, &'static str> {
191        let a = A::decode_hamt(input, cursor)?;
192        let b = B::decode_hamt(input, cursor)?;
193        Ok((a, b))
194    }
195}

211/// A representation of an internal node that is safe to persist to disk.
212///
213/// Leaves are stored inline as `(K, V)` pairs in datamap order, while child
214/// references are stored separately in nodemap order.
215#[derive(Clone, Debug, PartialEq, Eq)]
216pub struct PersistedInternalNode<K, V> {
217    pub datamap: u32,
218    pub nodemap: u32,
219    pub leaves: Vec<(K, V)>,
220    pub child_hashes: Vec<StructuralHash>,
221}
222
223impl<K, V> PersistedInternalNode<K, V>
224where
225    K: HamtCodec,
226    V: HamtCodec,
227{
228    /// Encodes the canonical v1 representation shared by persistence and
229    /// structural hashing. Keeping this in one place ensures a node's
230    /// storage key commits to precisely the bytes stored under that key.
231    pub(crate) fn encode_v1_parts(
232        datamap: u32,
233        nodemap: u32,
234        leaves: &[(K, V)],
235        child_hashes: &[StructuralHash],
236    ) -> Vec<u8> {
237        let leaf_slots = datamap.count_ones() as usize;
238        let child_slots = nodemap.count_ones() as usize;
239        assert_eq!(
240            leaves.len(),
241            leaf_slots,
242            "leaf count must match datamap bits"
243        );
244        assert_eq!(
245            child_hashes.len(),
246            child_slots,
247            "child count must match nodemap bits"
248        );
249        assert_eq!(datamap & nodemap, 0, "datamap and nodemap must not overlap");
250
251        let mut body = Vec::new();
252        for (key, value) in leaves {
253            key.encode_hamt(&mut body);
254            value.encode_hamt(&mut body);
255        }
256        for hash in child_hashes {
257            body.extend_from_slice(hash);
258        }
259
260        let leaf_count = u32::try_from(leaves.len()).expect("too many leaves for v1 encoding");
261        let child_count =
262            u32::try_from(child_hashes.len()).expect("too many child hashes for v1 encoding");
263        let capacity = 1_usize
264            .checked_add(4)
265            .and_then(|value| value.checked_add(4))
266            .and_then(|value| value.checked_add(4))
267            .and_then(|value| value.checked_add(4))
268            .and_then(|value| value.checked_add(body.len()))
269            .expect("encoded node size overflows usize");
270
271        let mut buf = Vec::with_capacity(capacity);
272        buf.push(HAMT_WIRE_VERSION);
273        buf.extend_from_slice(&datamap.to_le_bytes());
274        buf.extend_from_slice(&nodemap.to_le_bytes());
275        buf.extend_from_slice(&leaf_count.to_le_bytes());
276        buf.extend_from_slice(&child_count.to_le_bytes());
277        buf.extend_from_slice(&body);
278        buf
279    }

281    /// Encodes the node to a dense binary format.
282    ///
283    /// Layout:
284    /// - Version (1 byte): `0x02`
285    /// - Datamap (4 bytes, LE)
286    /// - Nodemap (4 bytes, LE)
287    /// - Leaf count (4 bytes, LE)
288    /// - Child count (4 bytes, LE)
289    /// - Inline leaves (`K`, `V` pairs in datamap order)
290    /// - Child hashes (`32 * child_count` bytes in nodemap order)
291    ///
292    /// # Panics
293    /// Panics if the payload lengths disagree with the bitmaps, if `datamap`
294    /// and `nodemap` overlap, or if the payloads would overflow `usize`.
295    #[must_use]
296    pub fn encode_v1(&self) -> Vec<u8> {
297        Self::encode_v1_parts(self.datamap, self.nodemap, &self.leaves, &self.child_hashes)
298    }
299
300    /// Decodes the node from a dense binary format.
301    ///
302    /// # Errors
303    /// Returns an error when the version byte is invalid or the buffer is too
304    /// short for the declared payload.
305    pub fn decode_v1_unverified(buf: &[u8]) -> Result<Self, &'static str> {
306        match buf.first().copied() {
307            Some(HAMT_WIRE_VERSION) => {}
308            Some(LEGACY_WIRE_VERSION_16_BYTE_HASHES) => {
309                return Err(
310                    "Legacy v1 node (16-byte structural hashes) is unsupported; \
311                     re-persist or migrate before decoding",
312                );
313            }
314            _ => return Err("Invalid version byte"),
315        }
316        if buf.len() < 17 {
317            return Err("Buffer too short for v1 header");
318        }
319
320        let datamap = u32::from_le_bytes(
321            buf.get(1..5)
322                .ok_or("Buffer too short for datamap")?
323                .try_into()
324                .map_err(|_| "Buffer too short for datamap")?,
325        );
326        let nodemap = u32::from_le_bytes(
327            buf.get(5..9)
328                .ok_or("Buffer too short for nodemap")?
329                .try_into()
330                .map_err(|_| "Buffer too short for nodemap")?,
331        );
332
333        if (datamap & nodemap) != 0 {
334            return Err("Datamap and nodemap overlap: node is corrupt");
335        }
336
337        let leaf_count = u32::from_le_bytes(
338            buf.get(9..13)
339                .ok_or("Buffer too short for leaf count")?
340                .try_into()
341                .map_err(|_| "Buffer too short for leaf count")?,
342        ) as usize;
343        let child_count = u32::from_le_bytes(
344            buf.get(13..17)
345                .ok_or("Buffer too short for child count")?
346                .try_into()
347                .map_err(|_| "Buffer too short for child count")?,
348        ) as usize;
349
350        let expected_leaves = datamap.count_ones() as usize;
351        let expected_children = nodemap.count_ones() as usize;
352        if leaf_count != expected_leaves {
353            return Err("Leaf count does not match datamap");
354        }
355        if child_count != expected_children {
356            return Err("Child count does not match nodemap");
357        }
358
359        let mut cursor = 17_usize;
360        let mut leaves = Vec::with_capacity(leaf_count);
361        for _ in 0..leaf_count {
362            let key = K::decode_hamt(buf, &mut cursor)?;
363            let value = V::decode_hamt(buf, &mut cursor)?;
364            leaves.push((key, value));
365        }
366
367        let child_bytes = child_count
368            .checked_mul(core::mem::size_of::<StructuralHash>())
369            .ok_or("Child hash payload size overflows usize")?;
370        let total_len = cursor
371            .checked_add(child_bytes)
372            .ok_or("Child hash payload size overflows usize")?;
373        if buf.len() < total_len {
374            return Err("Buffer too short for child hashes");
375        }
376        if buf.len() > total_len {
377            return Err("Buffer contains trailing bytes");
378        }
379
380        let mut child_hashes = Vec::with_capacity(child_count);
381        for i in 0..child_count {
382            let start = cursor
383                .checked_add(
384                    i.checked_mul(core::mem::size_of::<StructuralHash>())
385                        .ok_or("Child hash index overflows usize")?,
386                )
387                .ok_or("Child hash index overflows usize")?;
388            let end = start
389                .checked_add(core::mem::size_of::<StructuralHash>())
390                .ok_or("Child hash index overflows usize")?;
391            let mut hash = [0u8; core::mem::size_of::<StructuralHash>()];
392            hash.copy_from_slice(&buf[start..end]);
393            child_hashes.push(hash);
394        }
395
396        Ok(Self {
397            datamap,
398            nodemap,
399            leaves,
400            child_hashes,
401        })
402    }

404    /// Parse-only decoder for pre-`e349d0f` `0x01` records that carried
405    /// 16-byte structural hashes.
406    ///
407    /// This exists solely so legacy bytes can be inspected or migrated: the
408    /// current layout cannot recover the storage key those records were
409    /// written under (the key committed to the 16-byte representation), so a
410    /// legacy node can never satisfy [`Self::into_hamt_node_verified`].
411    /// Decoded child hashes hold their legacy 16 bytes left-aligned in the
412    /// 32-byte [`StructuralHash`] array; the trailing half is zero.
413    ///
414    /// # Errors
415    /// Returns an error when the buffer lacks the `0x01` version byte or
416    /// does not match the legacy layout.
417    pub fn decode_v1_legacy_unverified(buf: &[u8]) -> Result<Self, &'static str> {
418        const LEGACY_HASH_WIDTH: usize = 16;
419
420        if !matches!(
421            buf.first().copied(),
422            Some(LEGACY_WIRE_VERSION_16_BYTE_HASHES)
423        ) {
424            return Err("Invalid version byte");
425        }
426        if buf.len() < 17 {
427            return Err("Buffer too short for v1 header");
428        }
429
430        let datamap = u32::from_le_bytes(
431            buf.get(1..5)
432                .ok_or("Buffer too short for datamap")?
433                .try_into()
434                .map_err(|_| "Buffer too short for datamap")?,
435        );
436        let nodemap = u32::from_le_bytes(
437            buf.get(5..9)
438                .ok_or("Buffer too short for nodemap")?
439                .try_into()
440                .map_err(|_| "Buffer too short for nodemap")?,
441        );
442
443        if (datamap & nodemap) != 0 {
444            return Err("Datamap and nodemap overlap: node is corrupt");
445        }
446
447        let leaf_count = u32::from_le_bytes(
448            buf.get(9..13)
449                .ok_or("Buffer too short for leaf count")?
450                .try_into()
451                .map_err(|_| "Buffer too short for leaf count")?,
452        ) as usize;
453        let child_count = u32::from_le_bytes(
454            buf.get(13..17)
455                .ok_or("Buffer too short for child count")?
456                .try_into()
457                .map_err(|_| "Buffer too short for child count")?,
458        ) as usize;
459
460        let expected_leaves = datamap.count_ones() as usize;
461        let expected_children = nodemap.count_ones() as usize;
462        if leaf_count != expected_leaves {
463            return Err("Leaf count does not match datamap");
464        }
465        if child_count != expected_children {
466            return Err("Child count does not match nodemap");
467        }
468
469        let mut cursor = 17_usize;
470        let mut leaves = Vec::with_capacity(leaf_count);
471        for _ in 0..leaf_count {
472            let key = K::decode_hamt(buf, &mut cursor)?;
473            let value = V::decode_hamt(buf, &mut cursor)?;
474            leaves.push((key, value));
475        }
476
477        let child_bytes = child_count
478            .checked_mul(LEGACY_HASH_WIDTH)
479            .ok_or("Child hash payload size overflows usize")?;
480        let total_len = cursor
481            .checked_add(child_bytes)
482            .ok_or("Child hash payload size overflows usize")?;
483        if buf.len() < total_len {
484            return Err("Buffer too short for child hashes");
485        }
486        if buf.len() > total_len {
487            return Err("Buffer contains trailing bytes");
488        }
489
490        let mut child_hashes = Vec::with_capacity(child_count);
491        for _ in 0..child_count {
492            let end = cursor
493                .checked_add(LEGACY_HASH_WIDTH)
494                .ok_or("Child hash index overflows usize")?;
495            let mut hash = [0u8; core::mem::size_of::<StructuralHash>()];
496            hash[..LEGACY_HASH_WIDTH].copy_from_slice(&buf[cursor..end]);
497            child_hashes.push(hash);
498            cursor = end;
499        }
500
501        Ok(Self {
502            datamap,
503            nodemap,
504            leaves,
505            child_hashes,
506        })
507    }

509    /// Validates this decoded node against the storage key that selected it.
510    ///
511    /// Persisted v1 nodes deliberately do not carry a duplicate structural
512    /// hash. Callers must supply the hash used to fetch the bytes; this
513    /// recomputes the node identity under `structural_key` and rejects a
514    /// shape-valid node stored under the wrong key.
515    ///
516    /// # Errors
517    /// Returns an error if the node's contents do not reproduce
518    /// `expected_hash`.
519    pub fn into_hamt_node_verified(
520        self,
521        structural_key: &[u8],
522        expected_hash: StructuralHash,
523    ) -> Result<crate::hamt::HamtNode<K, V>, &'static str>
524    where
525        K: Hash,
526        V: Hash,
527    {
528        if (self.datamap & self.nodemap) != 0 {
529            return Err("PersistedInternalNode datamap and nodemap overlap");
530        }
531        if self.leaves.len() != self.datamap.count_ones() as usize {
532            return Err("PersistedInternalNode leaf count does not match datamap");
533        }
534        if self.child_hashes.len() != self.nodemap.count_ones() as usize {
535            return Err("PersistedInternalNode child count does not match nodemap");
536        }
537
538        let children: Vec<_> = self
539            .child_hashes
540            .into_iter()
541            .map(crate::hamt::NodeRef::Lazy)
542            .collect();
543        let recomputed = crate::hamt::HamtNode::compute_structural_hash(
544            structural_key,
545            self.datamap,
546            self.nodemap,
547            &self.leaves,
548            &children,
549        );
550        if recomputed != expected_hash {
551            return Err("persisted node contents do not match expected structural hash");
552        }
553
554        Ok(crate::hamt::HamtNode {
555            datamap: self.datamap,
556            nodemap: self.nodemap,
557            leaves: self.leaves,
558            children,
559            structural_hash: expected_hash,
560        })
561    }
562
563    /// Decodes a v1 node and verifies it against the storage key that selected
564    /// the bytes.
565    ///
566    /// This is the normal cold-storage load path.
567    /// [`Self::decode_v1_unverified`] is only the syntactic decoder for
568    /// callers that need to inspect persisted data before choosing an expected
569    /// hash.
570    ///
571    /// # Errors
572    /// Returns an error when the bytes are malformed or their recomputed
573    /// structural hash does not match `expected_hash`.
574    pub fn decode_v1_verified(
575        buf: &[u8],
576        structural_key: &[u8],
577        expected_hash: StructuralHash,
578    ) -> Result<crate::hamt::HamtNode<K, V>, &'static str>
579    where
580        K: Hash,
581        V: Hash,
582    {
583        Self::decode_v1_unverified(buf)?.into_hamt_node_verified(structural_key, expected_hash)
584    }
585}

587impl<K: Clone, V: Clone> From<&crate::hamt::HamtNode<K, V>> for PersistedInternalNode<K, V> {
588    fn from(node: &crate::hamt::HamtNode<K, V>) -> Self {
589        let child_hashes = node
590            .children
591            .iter()
592            .map(super::NodeRef::structural_hash)
593            .collect();
594
595        Self {
596            datamap: node.datamap,
597            nodemap: node.nodemap,
598            leaves: node.leaves.clone(),
599            child_hashes,
600        }
601    }
