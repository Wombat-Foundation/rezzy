1//! Reachability audit for content-addressed HAMT storage.
2//!
3//! This module answers exactly one question, in-memory and storage-agnostic:
4//! given a set of live roots and the full universe of node hashes a storage
5//! backend currently holds, which of those node hashes are *not* reachable
6//! from any root?
7//!
8//! It deliberately does not decide what to do with the answer. Root-set
9//! completeness (is the caller's root list actually every live root?),
10//! scan/snapshot consistency (was `universe` collected consistently with the
11//! roots?), and any notion of quarantine, age cutoffs, or hard deletion are
12//! the storage backend's responsibility, not this module's. Treat the
13//! `unreachable` side of a [`NodeReachabilityAudit`] as a candidate list for
14//! further safety checks, never as a delete list on its own.
15

28/// [`IndexedUniverse::try_build`] was given more than `u32::MAX` distinct
29/// hashes, so no dense index could be assigned to all of them.
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub struct UniverseTooLarge {
32    /// The number of distinct hashes counted before construction stopped.
33    /// When `allocation_failed` is false, this is the true total (which
34    /// exceeds `u32::MAX`). When `allocation_failed` is true, this is a
35    /// partial count — the builder stopped before scanning the whole universe.
36    pub distinct_count: usize,
37    /// True when construction stopped because memory allocation failed.
38    /// In this case, `distinct_count` is a partial count, not the true total.
39    pub allocation_failed: bool,
40}
41
42impl fmt::Display for UniverseTooLarge {
43    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
44        if self.allocation_failed {
45            write!(
46                f,
47                "universe has at least {} distinct hashes (allocation failed before scanning complete), more than u32::MAX can index",
48                self.distinct_count
49            )
50        } else {
51            write!(
52                f,
53                "universe has {} distinct hashes, more than u32::MAX can index",
54                self.distinct_count
55            )
56        }
57    }
58}

60impl core::error::Error for UniverseTooLarge {}

62/// A `universe` of node hashes assigned dense `u32` indexes, in the order the
63/// hashes were given.
64///
65/// This is the compaction step a `RoaringBitmap`-backed audit needs:
66/// `StructuralHash` (32 bytes, high-entropy, not locally dense) cannot be
67/// used as a roaring index directly, so every hash in `universe` is given a
68/// stable position instead. Identity always resolves back through
69/// [`Self::hash_at`]/`hashes` to the full hash — the dense index is a
70/// local, single-call addressing scheme, not an identifier of its own.
71///
72/// Backed by the crate-wide generic [`DenseIndex`] primitive (indexed type
73/// `StructuralHash`, `u32` width); this wrapper keeps the hash-specific
74/// [`hash_at`](Self::hash_at) naming and the [`UniverseTooLarge`] error type.
75#[derive(Debug, Clone, PartialEq, Eq)]
76pub struct IndexedUniverse(DenseIndex<StructuralHash>);
77
78impl From<IndexTooLarge> for UniverseTooLarge {
79    fn from(err: IndexTooLarge) -> Self {
80        Self {
81            distinct_count: err.distinct_count,
82            allocation_failed: err.allocation_failed,
83        }
84    }
85}

87impl IndexedUniverse {
88    /// Assigns each hash in `universe` a dense `u32` index, first-seen order.
89    /// Duplicate hashes collapse onto the same index.
90    ///
91    /// # Errors
92    /// Returns [`UniverseTooLarge`] if `universe` contains more than
93    /// `u32::MAX` distinct hashes (the bitmap audit converts `len()` to
94    /// `u32` for roaring indexing; exactly `u32::MAX` succeeds).
95    pub fn try_build(
96        universe: impl IntoIterator<Item = StructuralHash>,
97    ) -> Result<Self, UniverseTooLarge> {
98        // `bitmap_node_reachability_audit` converts `universe.len()` to `u32`
99        // for roaring bitmap indexing, so cap at `u32::MAX` — one less than
100        // the raw `DenseIndex<StructuralHash>` addressable slot count.
101        DenseIndex::try_build_bounded(universe, (u32::MAX as usize).saturating_add(1))
102            .map(Self)
103            .map_err(Into::into)
104    }
105
106    /// The number of distinct hashes indexed.
107    #[must_use]
108    pub fn len(&self) -> usize {
109        self.0.len()
110    }
111
112    /// True if no hashes are indexed.
113    #[must_use]
114    pub fn is_empty(&self) -> bool {
115        self.0.is_empty()
116    }
117
118    /// The dense index assigned to `hash`, if it was part of `universe`.
119    #[must_use]
120    pub fn index_of(&self, hash: &StructuralHash) -> Option<u32> {
121        self.0.index_of(hash)
122    }
123
124    /// The full hash assigned to dense index `idx`, if in range.
125    #[must_use]
126    pub fn hash_at(&self, idx: u32) -> Option<StructuralHash> {
127        self.0.item_at(idx as usize).copied()
128    }
129
130    /// All indexed hashes, in dense-index order.
131    #[must_use]
132    pub fn hashes(&self) -> &[StructuralHash] {
133        self.0.items()
134    }
135}

137/// The result of a multi-root reachability audit, expressed as
138/// [`RoaringBitmap`]s over an [`IndexedUniverse`] rather than as
139/// `StructuralHash` collections.
140///
141/// Use this instead of [`NodeReachabilityAudit`] when the caller needs to keep
142/// many audits in memory, diff them, or intersect/union them repeatedly —
143/// operations `RoaringBitmap` is built for and a `HashSet<StructuralHash>`
144/// is not. `universe` is the only place `StructuralHash` identity lives;
145/// `reachable`/`unreachable` are addressed purely through its dense indexes.
146#[derive(Debug, Clone, PartialEq)]
147pub struct BitmapNodeReachabilityAudit {
148    pub universe: IndexedUniverse,
149    pub reachable: RoaringBitmap,
150    pub unreachable: RoaringBitmap,
151}
152
153/// Errors from [`bitmap_node_reachability_audit`]: either the traversal itself
154/// failed, or `universe` could not be given a dense index.
155#[derive(Debug, Clone, PartialEq)]
156pub enum BitmapAuditError<E> {
157    /// `universe` had more than `u32::MAX` distinct hashes.
158    Universe(UniverseTooLarge),
159    /// The reachability walk itself failed.
160    Traversal(HamtTraversalError<E>),
161}
162
163impl<E: fmt::Display> fmt::Display for BitmapAuditError<E> {
164    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
165        match self {
166            Self::Universe(err) => write!(f, "{err}"),
167            Self::Traversal(err) => write!(f, "{err}"),
168        }
169    }
170}
171
172impl<E> core::error::Error for BitmapAuditError<E>
173where
174    E: core::error::Error + fmt::Debug + 'static,
175{
176    fn source(&self) -> Option<&(dyn core::error::Error + 'static)> {
177        match self {
178            Self::Universe(err) => Some(err),
179            Self::Traversal(err) => Some(err),
180        }
181    }
182}

184impl<E> From<UniverseTooLarge> for BitmapAuditError<E> {
185    fn from(err: UniverseTooLarge) -> Self {
186        Self::Universe(err)
187    }
188}

190impl<E> From<HamtTraversalError<E>> for BitmapAuditError<E> {

193    }

196/// Partitions `universe` into reachable/unreachable [`RoaringBitmap`]s over a
197/// freshly built [`IndexedUniverse`].
198///
199/// Same traversal and semantics as [`node_reachability_audit`], but marks
200/// directly into a `RoaringBitmap` via `universe`'s dense index instead of
201/// accumulating a `HashSet<StructuralHash>` mark set first — this is the
202/// version worth using when the caller actually wants the roaring
203/// representation, not `node_reachability_audit`'s result reshaped afterward.
204/// Hashes the walk reaches that are outside `universe` are marked but never
205/// materialize a bitmap entry, matching `node_reachability_audit`'s handling of
206/// the same case.
207///
208/// # Errors
209/// Returns [`BitmapAuditError::Universe`] if `universe` has more than
210/// `u32::MAX` distinct hashes, or [`BitmapAuditError::Traversal`] on the
211/// same conditions as [`node_reachability_audit`].
212///
213/// # Panics
214/// Does not panic on any caller-controlled input — an oversized `universe`
215/// is reported as [`BitmapAuditError::Universe`], not a panic. The
216/// `u32::try_from(universe.len())` fallback below re-derives that length,
217/// which [`IndexedUniverse::try_build`] (called just above it, and
218/// propagated with `?` on failure) already guarantees fits.
219pub fn bitmap_node_reachability_audit<K, V, F, E>(
220    roots: impl IntoIterator<Item = Arc<HamtNode<K, V>>>,
221    universe: impl IntoIterator<Item = StructuralHash>,
222    resolver: &mut F,
223) -> Result<BitmapNodeReachabilityAudit, BitmapAuditError<E>>
224where
225    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
226{
227    // Unreachable in practice: `IndexedUniverse::try_build` above already
228    // guarantees `universe.len()` fits `u32`. Local fn (not a closure) so
229    // `coverage(off)` — item-only — can mark it dead without hiding the
230    // reachable code around it.
231    #[cfg_attr(coverage_nightly, coverage(off))]
232    fn universe_overflow<E>(universe: &IndexedUniverse) -> BitmapAuditError<E> {
233        BitmapAuditError::Universe(UniverseTooLarge {
234            distinct_count: universe.len(),
235            allocation_failed: false,
236        })
237    }
238
239    let universe = IndexedUniverse::try_build(universe)?;
240
241    let mut reachable = RoaringBitmap::new();
242    // Hashes outside `universe` still need dedup so a subtree shared across
243    // roots (or reachable from inside and outside `universe`) is walked
244    // once, same as `node_reachability_audit`. `reachable`'s own membership
245    // check covers dedup for anything actually in `universe`, so this set
246    // only ever grows for hashes the caller's `universe` scan missed.
247    let mut visited_outside_universe: HashSet<StructuralHash> = HashSet::default();
248    for root in roots {
249        walk_reachable_node_hashes(&root, resolver, &mut |hash| {
250            if let Some(idx) = universe.index_of(&hash) {
251                reachable.insert(idx)
252            } else {
253                visited_outside_universe.insert(hash)
254            }
255        })
256        .map_err(BitmapAuditError::Traversal)?;
257    }
258
259    let universe_len = u32::try_from(universe.len()).map_err(|_| universe_overflow(&universe))?;
260    // `unreachable` is the full index range minus `reachable`; build the full
261    // range as a bitmap and subtract. `MultiOps::difference` reduces over many
262    // bitmaps; for a pair, call `Sub::sub` by name to sidestep clippy's
263    // `arithmetic_side_effects` (a false positive for set-difference).
264    let full_range: RoaringBitmap = (0..universe_len).collect();
265    let unreachable: RoaringBitmap = core::ops::Sub::sub(full_range, &reachable);
266
267    Ok(BitmapNodeReachabilityAudit {
268        universe,
269        reachable,
270        unreachable,
271    })
272}

274/// The result of a multi-root reachability audit: `universe` partitioned
275/// into hashes reachable from at least one of the audited roots and hashes
276/// that are not.
277///
278/// `reachable` is exposed (not just `unreachable`) so a caller can reuse the
279/// same mark set for a second-pass confirmation check right before acting
280/// on `unreachable` (e.g. `audit.reachable.contains(&hash)`), instead of
281/// re-deriving it downstream as `universe - unreachable` or re-walking the
282/// roots a second time.
283#[derive(Debug, Clone)]
284pub struct NodeReachabilityAudit {
285    /// Hashes in `universe` reachable from at least one audited root.
286    pub reachable: HashSet<StructuralHash>,
287    /// Hashes in `universe` reachable from none of the audited roots.
288    pub unreachable: Vec<StructuralHash>,
289}
290
291/// Partitions `universe` into hashes reachable from `roots` and hashes that
292/// are not.
293///
294/// Walks each root with [`walk_reachable_node_hashes`] against one shared
295/// `HashSet`, so subtrees shared between roots are resolved and walked only
296/// once in total. `resolver` is called once per distinct node hash
297/// encountered across all roots combined (not once per root), same as the
298/// underlying walk.
299///
300/// `reachable` is the full mark set intersected with `universe`; nodes the
301/// walk reaches that are outside `universe` (e.g. a caller's scan missed
302/// them) are not included in either field.
303///
304/// # Errors
305/// Returns [`HamtTraversalError::Resolve`] if `resolver` fails, or
306/// [`HamtTraversalError::MaxDepthExceeded`] if a walk recurses past the
307/// deepest depth a legitimately-built HAMT can have.
308///
309/// On error, the in-progress mark set may be partially populated (see
310/// [`walk_reachable_node_hashes`]'s error-recovery note); this function
311/// discards it and returns the error rather than a partial answer, so a
312/// caller never mistakes a partial reachable set for a complete one.
313pub fn node_reachability_audit<K, V, F, E>(
314    roots: impl IntoIterator<Item = Arc<HamtNode<K, V>>>,
315    universe: impl IntoIterator<Item = StructuralHash>,
316    resolver: &mut F,
317) -> Result<NodeReachabilityAudit, HamtTraversalError<E>>
318where
319    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
320{
321    let mut marked: HashSet<StructuralHash> = HashSet::default();
322    for root in roots {
323        walk_reachable_node_hashes(&root, resolver, &mut |hash| marked.insert(hash))?;
324    }
325
326    let mut reachable: HashSet<StructuralHash> = HashSet::default();
327    // `unreachable` is a partition of `universe`, so a hash that appears
328    // multiple times in `universe` must be emitted only once -- matching the
329    // dedup that `IndexedUniverse` (and thus the bitmap variant) provides.
330    let mut seen_unreachable: HashSet<StructuralHash> = HashSet::default();
331    let mut unreachable: Vec<StructuralHash> = Vec::new();
332    for hash in universe {
333        if marked.contains(&hash) {
334            reachable.insert(hash);
335        } else if seen_unreachable.insert(hash) {
336            unreachable.push(hash);
337        }
338    }
339
340    Ok(NodeReachabilityAudit {
341        reachable,
342        unreachable,
343    })
344}

346/// Computes the node hashes in `universe` that are not reachable from any of
347/// `roots`.
348///
349/// Convenience wrapper over [`node_reachability_audit`] for callers that only
350/// need the candidate list; see [`NodeReachabilityAudit`] if the reachable side
351/// is also useful (e.g. for a second-pass confirmation check).
352///
353/// # Errors
354/// See [`node_reachability_audit`].
355pub fn unreachable_node_hashes<K, V, F, E>(
356    roots: impl IntoIterator<Item = Arc<HamtNode<K, V>>>,
357    universe: impl IntoIterator<Item = StructuralHash>,
358    resolver: &mut F,
359) -> Result<Vec<StructuralHash>, HamtTraversalError<E>>
360where
361    F: FnMut(&StructuralHash) -> Result<Arc<HamtNode<K, V>>, E>,
362{
363    node_reachability_audit(roots, universe, resolver).map(|audit| audit.unreachable)
364}
