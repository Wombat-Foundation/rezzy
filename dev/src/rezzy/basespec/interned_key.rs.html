1//! SPIKE: can a dedicated `StateProvider` impl let a borrowing `InternId<'a>`
2//! be used as a room-state key WITHOUT the `for<'q> (String, K): Borrow<dyn
3//! StateKeyDyn + 'q>` / `'static` wall?
4//!
5//! The shared blanket impl routes lookups through `Borrow<dyn StateKeyDyn>`
6//! (zero-alloc `&str` queries against an arbitrary `K`), which forces `K:
7//! 'static`. This module instead builds a wrapper that holds `&'a Interner`
8//! and converts the query `&str` into an OWNED key on the spot via the
9//! interner — no borrowed trait-object lookup, no `for<'q>`, no `'static`.
10//!
11//! The map key is `(InternId<'a>, InternId<'a>)` — BOTH halves are `Copy`, so
12//! `get_event` resolves both `&str` halves through `id_of` with ZERO
13//! allocation (the same cost profile as the `Borrow<dyn StateKeyDyn>` path it
14//! replaces). `InternId`'s `Ord` always compares the resolved strings (not
15//! the interner's discovery-order index), so the tuple's `Ord` agrees with
16//! the `StateKeyDyn` `(ev_type, state_key)` lexicographic ordering contract
17//! unconditionally — independent of interning order.
18

25/// Owns the string arena + `string -> id` map. Slot 0 reserved for "".
26///
27/// Each interned string is allocated exactly once, as an `Rc<str>`; both
28/// `id_to_str` and `str_to_id` hold clones of that same `Rc` (a refcount
29/// bump, not a fresh allocation).
30#[derive(Debug, Clone)]
31pub struct Interner {
32    id_to_str: alloc::vec::Vec<Rc<str>>,
33    str_to_id: crate::HashMap<Rc<str>, u32>,
34}
35
36impl Default for Interner {
37    /// Delegates to [`Self::new`] so slot 0 is always reserved for "" —
38    /// `#[derive(Default)]` would instead produce an empty interner,
39    /// breaking the invariant every other constructor upholds.
40    fn default() -> Self {
41        Self::new()
42    }
43}
44
45impl Interner {
46    #[must_use]
47    pub fn new() -> Self {
48        let empty: Rc<str> = Rc::from("");
49        let mut str_to_id = crate::HashMap::new();
50        str_to_id.insert(Rc::clone(&empty), 0);
51        Self {
52            id_to_str: alloc::vec![empty],
53            str_to_id,
54        }
55    }
56
57    /// Interns `s`, returning its dense index (idempotent).
58    ///
59    /// # Panics
60    ///
61    /// Panics if more than `u32::MAX` distinct strings are interned.
62    pub fn intern(&mut self, s: &str) -> u32 {
63        if let Some(&id) = self.str_to_id.get(s) {
64            return id;
65        }
66        let id = u32::try_from(self.id_to_str.len()).expect("interner overflow");
67        // Single allocation: `s` is converted to an `Rc<str>` once, and
68        // `id_to_str`/`str_to_id` each hold a cheap `Rc::clone` (refcount
69        // bump) of it rather than a second independent copy of the string.
70        let rc: Rc<str> = Rc::from(s);
71        self.id_to_str.push(Rc::clone(&rc));
72        self.str_to_id.insert(rc, id);
73        id
74    }
75
76    /// The index for `s`, if it has been interned.
77    #[must_use]
78    pub fn id_of(&self, s: &str) -> Option<u32> {
79        self.str_to_id.get(s).copied()
80    }
81
82    /// Resolves an index back to its string.
83    ///
84    /// # Panics
85    ///
86    /// Panics if `id` was not issued by this interner.
87    #[must_use]
88    pub fn get(&self, id: u32) -> &str {
89        &self.id_to_str[usize::try_from(id).expect("non-negative")]
90    }
91
92    /// Number of distinct strings interned so far (including the reserved
93    /// slot 0 for `""`).
94    #[must_use]
95    pub fn len(&self) -> usize {
96        self.id_to_str.len()
97    }
98
99    /// Whether any strings beyond the reserved slot 0 have been interned.
100    ///
101    /// Slot 0 (`""`) is always present after [`Self::new`]/[`Self::default`],
102    /// so `id_to_str` itself is never actually empty — this compares against
103    /// `1`, not `0`, to match the doc's "beyond slot 0" contract.
104    #[must_use]
105    pub fn is_empty(&self) -> bool {
106        self.id_to_str.len() <= 1
107    }
108}

110/// A `Copy` interned state key: `u32` into an [`Interner`]. `Ord` by rank.
111#[derive(Debug, Clone, Copy)]
112pub struct InternId<'a> {
113    idx: u32,
114    interner: &'a Interner,
115}
116
117// RESOLVED: Cross-interner collision
118// Two different `Interner`s can hand out the same dense index to different
119// strings, so comparing indices alone would let unrelated `InternId`s from
120// different arenas compare equal and collapse map entries. `InternId` now
121// carries the `&Interner` it was minted from (it always did, via
122// `from_index`/`interner()`), so equality takes arena identity into account:
123// same arena compares by index (cheap, and unambiguous because one arena
124// never assigns the same index to two different strings); different arenas
125// fall back to comparing the resolved strings, which is always correct
126// regardless of which arena assigned which index.
127//
128// RESOLVED: Ordering mismatch
129// `Interner::intern` assigns indices in first-use (discovery) order, not
130// sorted string order, so index order need not agree with string order.
131// Relying on `debug_assert!`-checked "callers must sort before inserting"
132// would silently misorder `BTreeMap` keys in release builds whenever that
133// invariant slipped. Instead `Ord` always compares the resolved strings
134// (never the index), so the ordering is correct unconditionally, independent
135// of interning order or which arena(s) the operands came from — no invariant
136// to violate.
137impl PartialEq for InternId<'_> {
138    fn eq(&self, other: &Self) -> bool {
139        if core::ptr::eq(self.interner, other.interner) {
140            // Same arena: `Interner::intern` never assigns the same index to
141            // two different strings, so index equality is exactly string
142            // equality here, without resolving either side.
143            self.idx == other.idx
144        } else {
145            self.as_ref() == other.as_ref()
146        }
147    }
148}
149impl Eq for InternId<'_> {}
150impl PartialOrd for InternId<'_> {
151    fn partial_cmp(&self, other: &Self) -> Option<core::cmp::Ordering> {
152        Some(self.cmp(other))
153    }
154}
155impl Ord for InternId<'_> {
156    fn cmp(&self, other: &Self) -> core::cmp::Ordering {
157        // Always compare the resolved strings: index order tracks discovery
158        // (first-use) order, not string order, so it cannot be trusted for
159        // `Ord` even within a single arena.
160        self.as_ref().cmp(other.as_ref())
161    }
162}
163impl core::hash::Hash for InternId<'_> {
164    fn hash<H: core::hash::Hasher>(&self, state: &mut H) {
165        // Must hash the same value `eq` compares on some path for any pair
166        // of (possibly cross-arena) equal `InternId`s, so hash the resolved
167        // string unconditionally rather than the index.
168        self.as_ref().hash(state);
169    }
170}

172impl<'a> InternId<'a> {
173    /// # Panics
174    ///
175    /// Panics if `idx` was not issued by `interner` (i.e. is out of bounds
176    /// for it).
177    #[must_use]
178    pub fn from_index(interner: &'a Interner, idx: u32) -> Self {
179        let index =
180            usize::try_from(idx).expect("InternId::from_index: u32 index does not fit usize");
181        assert!(
182            index < interner.len(),
183            "InternId::from_index: idx {idx} out of bounds for interner of len {}",
184            interner.len()
185        );
186        Self { idx, interner }
187    }
188
189    #[must_use]
190    pub fn index(self) -> u32 {
191        self.idx
192    }
193
194    #[must_use]
195    pub fn interner(self) -> &'a Interner {
196        self.interner
197    }
198}
199
200impl AsRef<str> for InternId<'_> {
201    fn as_ref(&self) -> &str {
202        self.interner.get(self.idx)
203    }
204}
205
206impl fmt::Display for InternId<'_> {
207    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
208        f.write_str(self.as_ref())
209    }
210}

212/// A room-state map keyed by `(InternId<'a>, InternId<'a>)` — both halves
213/// `Copy` — whose `StateProvider` impl converts a `&str` query into the key via
214/// the interner with no allocation. No `Borrow<dyn StateKeyDyn>`, no `'static`.
215pub struct InternedRoomState<'a, Id = alloc::string::String, C = serde_json::Value> {
216    interner: &'a Interner,
217    map: alloc::collections::BTreeMap<(InternId<'a>, InternId<'a>), LeanEvent<Id, C, InternId<'a>>>,
218}
219
220impl<'a, Id, C> InternedRoomState<'a, Id, C> {
221    /// Builds a room state directly from an already-interned map. `benches/`
222    /// uses this to construct a realistic room for the `get_event` micro-bench
223    /// without reaching into private fields.
224    ///
225    /// # Panics
226    /// `InternId`'s `Eq`/`Ord`/`Hash` fall back to comparing resolved strings
227    /// across different arenas (see the note beside those impls), so mixing
228    /// arenas is no longer a correctness hazard by itself -- but it silently
229    /// forfeits the zero-cost index fast path on every comparison and usually
230    /// indicates a bug (an `InternId` escaping the arena it was meant to
231    /// belong to). Panics if any key or state-key value in `map` was not
232    /// produced by `interner`.
233    #[must_use]
234    pub fn new(
235        interner: &'a Interner,
236        map: alloc::collections::BTreeMap<
237            (InternId<'a>, InternId<'a>),
238            LeanEvent<Id, C, InternId<'a>>,
239        >,
240    ) -> Self {
241        for ((et, sk), ev) in &map {
242            assert!(
243                core::ptr::eq(et.interner(), interner)
244                    && core::ptr::eq(sk.interner(), interner)
245                    && ev
246                        .state_key
247                        .map_or(true, |k| core::ptr::eq(k.interner(), interner)),
248                "InternedRoomState::new: map contains InternId(s) from a \
249                 different Interner than the one supplied"
250            );
251        }
252        // No sortedness invariant to check here: `InternId::cmp` always
253        // compares resolved strings (never the interner's discovery-order
254        // index), so `BTreeMap`'s key order already agrees with the
255        // `StateKeyDyn` lexicographic contract unconditionally.
256        Self { interner, map }
257    }
258}
259
260impl<'a, Id, C> StateProvider<Id, C, LeanEvent<Id, C, InternId<'a>>>
261    for InternedRoomState<'a, Id, C>
262where
263    Id: crate::basespec::rezzy_types::EventId,
264    C: crate::basespec::rezzy_types::EventContent,
265{
266    fn get_event(
267        &self,
268        event_type: &str,
269        state_key: &str,
270    ) -> Option<&LeanEvent<Id, C, InternId<'a>>> {
271        let et = self.interner.id_of(event_type)?;
272        let sk = self.interner.id_of(state_key)?;
273        let key = (
274            InternId::from_index(self.interner, et),
275            InternId::from_index(self.interner, sk),
276        );
277        self.map.get(&key)
278    }
279}

447    /// Covers the `InternId` accessors and trait impls not exercised by the
448    /// state-provider lookup: the `intern` idempotent early-return, `index()`,
449    /// `interner()`, `AsRef<str>`, `Display`, `PartialOrd` (the `<` path, which

456        let idx_bob = interner.intern("@bob:x");
457        assert_eq!(idx1, idx2, "intern is idempotent");
458
459        let interner_ref = &interner;
460        let key = InternId::from_index(interner_ref, idx1);
461        let key2 = InternId::from_index(interner_ref, idx2);

464        // Accessors.

466        assert_eq!(key.interner().get(idx1), "@alice:x");

479        let build_hasher = crate::HashSet::<()>::default().hasher().clone();

498    /// `Interner::intern` assigns ids in first-use (discovery) order, not
499    /// sorted string order. `InternId::Ord` must still agree with plain
500    /// string ordering even though the indices here are deliberately
501    /// out of string order (unlike the sorted-insertion spike test above).
502    #[test]
503    fn ord_follows_string_order_even_when_indices_are_unsorted() {
504        let mut interner = Interner::new();
505        // Discovery order gives "zebra" index 1 and "apple" index 2, i.e.
506        // index order is the OPPOSITE of string order.
507        let idx_zebra = interner.intern("zebra");
508        let idx_apple = interner.intern("apple");

512        let apple = InternId::from_index(&interner, idx_apple);

515        assert!(apple < zebra);
516        assert_eq!(apple.cmp(&zebra), core::cmp::Ordering::Less);
517    }

519    /// Two different `Interner`s can (and, with discovery-order assignment,
520    /// routinely do) hand out the same dense index to different strings.
521    /// `InternId`'s `Eq`/`Ord`/`Hash` must resolve this via the strings
522    /// rather than treating same-index `InternId`s from different arenas as
523    /// interchangeable.
524    #[test]
525    fn cross_interner_same_index_does_not_collide() {
526        let mut arena_a = Interner::new();
527        let idx_a = arena_a.intern("aaa"); // idx 1
528
529        let mut arena_b = Interner::new();
530        let idx_b = arena_b.intern("zzz"); // also idx 1
531
532        assert_eq!(idx_a, idx_b, "sanity: same dense index in both arenas");
533
534        let a = InternId::from_index(&arena_a, idx_a);
535        let b = InternId::from_index(&arena_b, idx_b);
536
537        // Must NOT compare equal just because the raw indices match.
538        assert_ne!(a, b);
539        assert_eq!(a.cmp(&b), core::cmp::Ordering::Less, "\"aaa\" < \"zzz\"");
540        assert!(a < b);
541
542        let build_hasher = crate::HashSet::<()>::default().hasher().clone();
543        assert_ne!(
544            build_hasher.hash_one(a),
545            build_hasher.hash_one(b),
546            "unequal cross-arena InternIds should (almost certainly) hash \
547             differently"
548        );
549
550        // Same string, different arenas, same index -> still equal, since
551        // equality falls back to comparing resolved strings across arenas.
552        let mut arena_c = Interner::new();
553        let idx_c = arena_c.intern("aaa");
554        let c = InternId::from_index(&arena_c, idx_c);
555        assert_eq!(
556            a, c,
557            "equal strings from different arenas must compare equal"
558        );
559        assert_eq!(
560            build_hasher.hash_one(a),
561            build_hasher.hash_one(c),
562            "equal cross-arena InternIds must hash identically"
563        );
