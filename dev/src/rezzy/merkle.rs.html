28/// A Merkle root that nobody has signed.
29///
30/// Some root-computing functions in this crate return this wrapper
31/// ([`header_root`], [`CausalSet::unsigned_root`], [`StateMap::unsigned_root`]);
32/// others return a bare [`type@Hash`] ([`root`], [`CausalSet::root`],
33/// [`StateMap::root`]). The wrapper exists so a caller at a return site that
34/// produces `UnsignedRoot` is reminded that the value is only a *proof* of
35/// anything when it is either (a) folded into an `event_root` an event's
36/// sender actually signed (a true MSC4511C Part C proof), or (b) signed
37/// after the fact by whoever computed it, standing behind it as a responder
38/// (a Part B attestation -- see `crate::signing::attest` when the
39/// `signing-dalek` feature is enabled). Neither case is automatic: this type
40/// exists so a caller cannot accidentally hand a bare computed root to code
41/// that presents it as authoritative without having gone through one of
42/// those two steps.
43///
44/// A root computed over event IDs from a room that never adopted MSC4511C
45/// (any room in existence today) can *never* reach case (a) -- nothing signs
46/// a `causal_set`/`state_root` field on those events. It can still reach
47/// case (b): sign this value yourself via `crate::signing::attest::sign_attestation`
48/// and the result is a real, checkable claim -- just a Part B one, only as
49/// trustworthy as that one signer, not a room-participant-committed
50/// guarantee.
51///
52/// Callers who receive a root over federation should prefer
53/// [`verify_causal_inclusion`] / [`verify_inclusion`] only after confirming
54/// the root's provenance (e.g. extracted from a signature-checked event).
55/// A bare `Hash` passed to a verifier proves nothing about who stands
56/// behind it.
57#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
58pub struct UnsignedRoot(pub Hash);
59
60impl UnsignedRoot {
61    /// Unwraps the raw digest.
62    #[must_use]
63    pub const fn into_inner(self) -> Hash {
64        self.0
65    }
66}
67
68impl From<Hash> for UnsignedRoot {
69    fn from(hash: Hash) -> Self {
70        Self(hash)
71    }
72}

74/// Errors returned by MSC4511 Merkle and canonical JSON operations.

76pub enum MerkleError {
77    EmptyFieldName,
78    InvalidFieldName,
79    DuplicateField(String),

84}

86impl fmt::Display for MerkleError {
87    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
88        match self {
89            Self::EmptyFieldName => f.write_str("merkle: empty field name"),
90            Self::InvalidFieldName => f.write_str("merkle: invalid field name"),
91            Self::DuplicateField(name) => write!(f, "merkle: duplicate field: {name}"),
92            Self::FieldNotFound(name) => write!(f, "merkle: field not found: {name}"),
93            Self::NoLeaves => f.write_str("merkle: no leaves"),
94            Self::IntegerRange => f.write_str("canonical json integer out of range"),
95            Self::UnsupportedNumber => f.write_str("unsupported canonical json number"),
96        }
97    }
98}

359/// Computes the ordered (leaf-to-root) sibling path proving `field_name`'s
360/// leaf is included in the RFC 6962-shaped root over `fields`, along with
361/// that root. This is the `leaf_paths` construction MSC4511's "Cryptographic
362/// proof responses" section describes.
363///
364/// # Errors
365///
366/// Returns a [`MerkleError`] if `fields` cannot be canonicalized or contains
367/// a duplicate field name, or [`MerkleError::FieldNotFound`] if no field
368/// named `field_name` is present.
369pub fn leaf_path(
370    fields: &[Field],
371    field_name: &str,
372) -> Result<(Vec<ProofStep>, Hash), MerkleError> {
373    let ls = leaves(fields)?;
374    let idx = ls
375        .iter()
376        .position(|l| l.name == field_name)
377        .ok_or_else(|| MerkleError::FieldNotFound(field_name.into()))?;
378    let hashes = ls.iter().map(|l| l.hash).collect::<Vec<_>>();
379    let (root, path) = merkle_root_and_path(&hashes, idx).ok_or(MerkleError::NoLeaves)?;
380    Ok((path, root))
381}

383/// Recomputes the root from `leaf_hash` and `path` (leaf-to-root ordered
384/// siblings) and reports whether it matches `root`.
385#[must_use]
386pub fn verify_leaf_path(leaf_hash: Hash, path: &[ProofStep], root: Hash) -> bool {
387    let mut cur = leaf_hash;
388    for step in path {
389        cur = match step.side {
390            Side::Left => inner_hash(step.hash, cur),
391            Side::Right => inner_hash(cur, step.hash),
392        };
393    }
394    cur == root
395}

397/// Computes the RFC 6962 root over `hashes` and the ordered (leaf-to-root)
398/// sibling path for `hashes[target]`, mirroring [`merkle_root`]'s
399/// largest-power-of-two split so the two stay consistent.
400fn merkle_root_and_path(hashes: &[Hash], target: usize) -> Option<(Hash, Vec<ProofStep>)> {
401    match hashes.len() {
402        0 => None,
403        1 => Some((hashes[0], Vec::new())),
404        2 => {
405            if target == 0 {
406                Some((
407                    inner_hash(hashes[0], hashes[1]),
408                    alloc::vec![ProofStep {
409                        side: Side::Right,
410                        hash: hashes[1]
411                    }],
412                ))
413            } else {
414                Some((
415                    inner_hash(hashes[0], hashes[1]),
416                    alloc::vec![ProofStep {
417                        side: Side::Left,
418                        hash: hashes[0]
419                    }],
420                ))
421            }
422        }
423        len => {
424            let k = largest_power_of_two_less_than(len);
425            if target < k {
426                let (left_root, mut path) = merkle_root_and_path(&hashes[..k], target)?;
427                let right_root = merkle_root(&hashes[k..])?;
428                path.push(ProofStep {
429                    side: Side::Right,
430                    hash: right_root,
431                });
432                Some((inner_hash(left_root, right_root), path))
433            } else {
434                // `target >= k` here (the `target < k` branch above already
435                // handled the other case), so this never saturates.
436                let (right_root, mut path) =
437                    merkle_root_and_path(&hashes[k..], target.saturating_sub(k))?;
438                let left_root = merkle_root(&hashes[..k])?;
439                path.push(ProofStep {
440                    side: Side::Left,
441                    hash: left_root,
442                });
443                Some((inner_hash(left_root, right_root), path))
444            }
445        }
446    }
447}

449fn leaves(fields: &[Field]) -> Result<Vec<Leaf>, MerkleError> {
450    let mut leaves = fields
451        .iter()
452        .map(field_leaf)
453        .collect::<Result<Vec<_>, _>>()?;
454    leaves.sort_by(|a, b| a.name.as_bytes().cmp(b.name.as_bytes()));
455    for pair in leaves.windows(2) {
456        if pair[0].name == pair[1].name {
457            return Err(MerkleError::DuplicateField(pair[0].name.clone()));
458        }
459    }
460    Ok(leaves)
461}

478fn merkle_root(hashes: &[Hash]) -> Option<Hash> {
479    match hashes.len() {
480        0 => None,
481        1 => Some(hashes[0]),
482        2 => Some(inner_hash(hashes[0], hashes[1])),
483        len => {
484            let k = largest_power_of_two_less_than(len);
485            let left = merkle_root(&hashes[..k])?;
486            let right = merkle_root(&hashes[k..])?;
487            Some(inner_hash(left, right))
488        }
489    }
490}
491
492fn largest_power_of_two_less_than(n: usize) -> usize {
493    let mut k = 1;
494    while k << 1 < n {
495        k <<= 1;
496    }
497    k
498}

531fn append_canonical_value(out: &mut Vec<u8>, value: &Value) -> Result<(), MerkleError> {
532    match value {
533        Value::Null => out.extend_from_slice(b"null"),
534        Value::Bool(true) => out.extend_from_slice(b"true"),
535        Value::Bool(false) => out.extend_from_slice(b"false"),
536        Value::Number(number) => append_number(out, number)?,
537        Value::String(string) => append_string(out, string),
538        Value::Array(items) => {
539            out.push(b'[');
540            for (index, item) in items.iter().enumerate() {
541                if index > 0 {
542                    out.push(b',');
543                }
544                append_canonical_value(out, item)?;
545            }
546            out.push(b']');
547        }
548        Value::Object(object) => {
549            let mut keys = object.keys().collect::<Vec<_>>();
550            keys.sort_by(|a, b| a.as_bytes().cmp(b.as_bytes()));
551            out.push(b'{');
552            for (index, key) in keys.iter().enumerate() {
553                if index > 0 {
554                    out.push(b',');
555                }
556                append_string(out, key);
557                out.push(b':');
558                append_canonical_value(out, &object[*key])?;
559            }
560            out.push(b'}');
561        }
562    }
563    Ok(())
564}

566fn append_number(out: &mut Vec<u8>, number: &serde_json::Number) -> Result<(), MerkleError> {
567    if let Some(n) = number.as_i64() {
568        if !(MIN_CANONICAL_INT..=MAX_CANONICAL_INT).contains(&n) {
569            return Err(MerkleError::IntegerRange);
570        }
571        out.extend_from_slice(n.to_string().as_bytes());
572        return Ok(());
573    }
574
575    if number.as_u64().is_some() {
576        return Err(MerkleError::IntegerRange);
577    }
578
579    Err(MerkleError::UnsupportedNumber)
580}

582fn append_string(out: &mut Vec<u8>, string: &str) {
583    out.push(b'"');
584    for ch in string.chars() {
585        match ch {
586            '"' => out.extend_from_slice(br#"\""#),
587            '\\' => out.extend_from_slice(br"\\"),
588            '\u{08}' => out.extend_from_slice(br"\b"),
589            '\u{0c}' => out.extend_from_slice(br"\f"),
590            '\n' => out.extend_from_slice(br"\n"),
591            '\r' => out.extend_from_slice(br"\r"),
592            '\t' => out.extend_from_slice(br"\t"),
593            '\u{00}'..='\u{1f}' => {
594                let code = ch as usize;
595                out.extend_from_slice(b"\\u00");
596                out.push(HEX_LOWER[(code >> 4) & 0x0f]);
597                out.push(HEX_LOWER[code & 0x0f]);
598            }
599            _ => {
600                let mut buf = [0; 4];
601                out.extend_from_slice(ch.encode_utf8(&mut buf).as_bytes());
602            }
603        }
604    }
605    out.push(b'"');
606}

608fn hash_parts(parts: &[&[u8]]) -> Hash {
609    let mut hasher = Sha3_256::new();
610    for part in parts {
611        hasher.update(part);
612    }
613    hasher.finalize().into()
614}

624    /// The number of bit-levels in the causal sparse Merkle sum trie: one
625    /// level per bit of a 32-byte (256-bit) event-ID digest key.
626    pub const CAUSAL_DEPTH: usize = 256;
627
628    const CAUSAL_LEAF_DST: &[u8] = b"msc4511:causal-leaf:v1";
629    const CAUSAL_NODE_DST: &[u8] = b"msc4511:causal-node:v1";
630    const CAUSAL_EMPTY_LEAF_DST: &[u8] = b"msc4511:causal-empty-leaf:v1";
631
632    /// Computes SHA3-256("msc4511:causal-leaf:v1" || `key`).
633    fn causal_leaf(key: Hash) -> Hash {
634        hash_parts(&[CAUSAL_LEAF_DST, &key])
635    }
636
637    /// Computes SHA3-256("msc4511:causal-node:v1" || `u16be(depth)` ||
638    /// `left_hash` || `u64be(left_count)` || `right_hash` ||
639    /// `u64be(right_count)`).
640    fn causal_node(
641        depth: u16,
642        left_hash: Hash,
643        left_count: u64,
644        right_hash: Hash,
645        right_count: u64,
646    ) -> Hash {
647        hash_parts(&[
648            CAUSAL_NODE_DST,
649            &depth.to_be_bytes(),
650            &left_hash,
651            &left_count.to_be_bytes(),
652            &right_hash,
653            &right_count.to_be_bytes(),
654        ])
655    }

657    /// Returns the bit of `key` at depth `d` (0 = most significant bit of
658    /// byte 0), matching the MSB-to-LSB traversal defined for the causal
659    /// trie. `d % 8` is always in `0..=7`, so the subtraction from 7 never
660    /// underflows; `saturating_sub` documents that instead of asserting it.
661    fn causal_bit(key: &Hash, d: usize) -> u8 {
662        let byte_idx = d / 8;
663        let bit_idx = 7_usize.saturating_sub(d % 8);
664        (key[byte_idx] >> bit_idx) & 1
665    }

667    /// Converts a trie depth (always `< CAUSAL_DEPTH`, i.e. `<= 255`) to the
668    /// `u16` `causal_node` hashes over. `unwrap_or(u16::MAX)` is unreachable
669    /// in practice but keeps this a checked, non-panicking conversion rather
670    /// than a silent truncating `as` cast.
671    fn depth_u16(depth: usize) -> u16 {
672        u16::try_from(depth).unwrap_or(u16::MAX)
673    }

675    /// The canonical empty-subtree hash at every depth in `[0, CAUSAL_DEPTH]`,
676    /// indexed by depth. Index `CAUSAL_DEPTH` is the distinguished empty
677    /// leaf; every other index is derived from `causal_node` of two empty
678    /// children at the next depth.
679    ///
680    /// Built once per top-level call (see [`empty_table`]) rather than
681    /// recomputed recursively per lookup: a naive `empty_hash(depth)` that
682    /// recurses to `CAUSAL_DEPTH` on every call is called at nearly every
683    /// level of [`subtree_root`]'s own recursion, which blows up
684    /// to roughly `CAUSAL_DEPTH^2` hash calls for one root computation.
685    /// Building this table bottom-up costs exactly `CAUSAL_DEPTH` hash calls
686    /// total.
687    type EmptyTable = [Hash; CAUSAL_DEPTH.saturating_add(1)];
688
689    /// Builds [`EmptyTable`] bottom-up: one pass, `CAUSAL_DEPTH` `causal_node`
690    /// calls plus one leaf hash.
691    fn build_empty_table() -> EmptyTable {
692        let mut table = [[0_u8; super::HASH_SIZE]; CAUSAL_DEPTH.saturating_add(1)];
693        table[CAUSAL_DEPTH] = hash_parts(&[CAUSAL_EMPTY_LEAF_DST]);
694        let mut depth = CAUSAL_DEPTH;
695        while depth > 0 {
696            depth = depth.saturating_sub(1);
697            let child = table[depth.saturating_add(1)];
698            table[depth] = causal_node(depth_u16(depth), child, 0, child, 0);
699        }
700        table
701    }

707    }

722    /// An in-memory population of event-ID keys committed by an MSC4511
723    /// 256-level sparse Merkle sum trie.
724    ///
725    /// The trie is maintained incrementally: each `insert` / `extend` updates
726    /// only the O(256) nodes along the inserted key's path, so `root()`,
727    /// `inclusion_proof()`, and `non_inclusion_proof()` are O(1) / O(256)
728    /// rather than O(n·256).
729    #[derive(Debug, Clone, Default, PartialEq, Eq)]
730    pub struct CausalSet {
731        keys: BTreeSet<Hash>,
732        /// `(depth, key_prefix) → (hash, count)`. `key_prefix` at depth `d`
733        /// is the first `d` bits of a key, stored in the low `d` bits of a
734        /// 32-byte array (MSB-first). This is the full node cache: root is
735        /// at depth 0, leaves at depth 256.
736        nodes: BTreeMap<(u16, [u8; 32]), (Hash, u64)>,
737    }

751    /// One sibling in a causal sparse Merkle sum trie path, ordered
752    /// leaf-to-root: applying each step in order (combining the running
753    /// hash/count with `hash`/`count` on the named side, via
754    /// `causal_node`) reconstructs the trie root and count.
755    ///
756    /// The side (left/right) is not stored here — it is deterministically
757    /// derived from the key bit at each depth during verification
758    /// ([`verify_causal_path`]). This removes a redundant field from the
759    /// wire format and eliminates an entire class of forgery.
760    #[derive(Debug, Clone, Copy, PartialEq, Eq)]
761    pub struct CausalProofStep {
762        pub hash: Hash,
763        pub count: u64,
764    }

766    impl CausalSet {
767        /// Returns the canonical empty causal set: root `empty_hash(0)`,
768        /// count 0.
769        #[must_use]
770        pub fn empty() -> Self {
771            Self {
772                keys: BTreeSet::new(),
773                nodes: BTreeMap::new(),
774            }
775        }
776
777        /// Inserts a key into `self` in place without cloning.
778        ///
779        /// Updates the node cache incrementally: walks the path from root to
780        /// leaf, creating any missing nodes, then recomputes hashes bottom-up.
781        /// O(256) per insert.
782        pub fn insert_mut(&mut self, key: Hash) -> bool {
783            if !self.keys.insert(key) {
784                return false;
785            }
786            let empty = empty_table();
787
788            // Phase 1: walk down, creating all nodes along the path.
789            let mut prefix = [0u8; 32];
790            for d in 0..CAUSAL_DEPTH {
791                let depth = depth_u16(d);
792                self.nodes.entry((depth, prefix)).or_insert((empty[d], 0));
793                if causal_bit(&key, d) == 0 {
794                    // go left: clear bit d in prefix
795                    prefix[d / 8] &= !(1 << (7_usize.wrapping_sub(d % 8)));
796                } else {
797                    // go right: set bit d in prefix
798                    prefix[d / 8] |= 1 << (7_usize.wrapping_sub(d % 8));
799                }
800            }
801            // Leaf at depth 256
802            let leaf_depth = depth_u16(CAUSAL_DEPTH);
803            self.nodes
804                .insert((leaf_depth, prefix), (causal_leaf(key), 1));
805
806            // Phase 2: walk back up, recomputing hashes from cached children.
807            // `child_prefix` is maintained to have bits 0..d set from the
808            // key and bits d+1..255 = 0 at each iteration. This matches
809            // how Phase 1 stored child nodes at depth d+1.
810            let mut child_prefix = prefix;
811            for d in (0..CAUSAL_DEPTH).rev() {
812                let depth = depth_u16(d);
813                let child_depth = depth_u16(d.wrapping_add(1));
814
815                // Strip child_prefix to only have bits 0..d set.
816                let byte_idx = d / 8;
817                let bit_idx = d % 8;
818                child_prefix[byte_idx] &= 0xFF << (7_usize.wrapping_sub(bit_idx));
819                for byte in child_prefix.iter_mut().skip(byte_idx.wrapping_add(1)) {
820                    *byte = 0;
821                }
822
823                // Left child prefix: bit d = 0.
824                let mut left_prefix = child_prefix;
825                left_prefix[byte_idx] &= !(1 << (7_usize.wrapping_sub(bit_idx)));
826                let (left_hash, left_count) = self
827                    .nodes
828                    .get(&(child_depth, left_prefix))
829                    .copied()
830                    .unwrap_or((empty[d.wrapping_add(1)], 0));
831
832                // Right child prefix: bit d = 1.
833                let mut right_prefix = child_prefix;
834                right_prefix[byte_idx] |= 1 << (7_usize.wrapping_sub(bit_idx));
835                let (right_hash, right_count) = self
836                    .nodes
837                    .get(&(child_depth, right_prefix))
838                    .copied()
839                    .unwrap_or((empty[d.wrapping_add(1)], 0));
840
841                let node = causal_node(depth, left_hash, left_count, right_hash, right_count);
842                // Node at depth d is stored with prefix bits 0..d-1 set
843                // (= child_prefix with bit d cleared = left_prefix).
844                self.nodes.insert(
845                    (depth, left_prefix),
846                    (node, count_sum(left_count, right_count)),
847                );
848            }
849            true
850        }

894        /// Computes the canonical sparse Merkle sum trie root for `self`.
895        /// O(1) — reads from the maintained node cache.

897        pub fn root(&self) -> Hash {
898            if self.keys.is_empty() {
899                return empty_table()[0];
900            }
901            let root_depth: u16 = 0;
902            let root_prefix = [0u8; 32];
903            self.nodes
904                .get(&(root_depth, root_prefix))
905                .map_or_else(|| empty_table()[0], |(h, _)| *h)
906        }

919    /// A single step in a compressed causal-trie proof path.
920    ///
921    /// Runs of consecutive canonical-empty siblings are collapsed into a
922    /// single `EmptyRun` entry; non-empty steps are emitted individually
923    /// as `Step`.
924    #[derive(Debug, Clone, PartialEq, Eq)]
925    pub enum CompressedCausalStep {
926        /// A non-empty sibling step (hash and count are explicit).
927        Step(CausalProofStep),
928        /// A run of `length` consecutive canonical-empty siblings.
929        /// `start_depth` is the sibling depth of the first empty step
930        /// in the run (the step at path position 0 has sibling depth `T`
931        /// where `T = terminal_depth`).  Expansion proceeds in proof
932        /// order: `empty[start_depth]`, `empty[start_depth - 1]`, …,
933        /// `empty[start_depth - length + 1]`, each with count 0 and side
934        /// derived from `causal_bit(key, sibling_depth - 1)`.
935        EmptyRun { start_depth: u16, length: u16 },
936    }

938    /// Error type for compressed causal-trie proof operations.
939    #[derive(Debug, Clone, PartialEq, Eq)]
940    pub enum CausalProofError {
941        /// A compressed step references a sibling depth outside
942        /// `1..=CAUSAL_DEPTH`, or `terminal_depth` itself exceeds
943        /// `CAUSAL_DEPTH`.
944        ///
945        /// `EmptyRun` carries no hash or count of its own — it is expanded
946        /// directly from `empty_table()` with count 0 — so there is no
947        /// attacker-supplied empty-step value to validate; this variant is
948        /// the only depth-shaped rejection the format has.
949        InvalidDepth(u16),
950        /// The decompressed path length does not match `terminal_depth`.
951        PathLengthMismatch {
952            decompressed: usize,
953            expected: usize,
954        },
955        /// The compressed encoding is truncated (fewer steps than
956        /// `terminal_depth`).
957        Truncated,
958        /// The compressed encoding has data after the complete path.
959        ExcessData,
960        /// An empty run would extend below sibling depth 1.
961        RunBelowRoot,
962        /// An empty run's start depth does not match the expected sibling
963        /// depth for the current path position (non-contiguous).
964        NonContiguousRun {
965            expected_sibling_depth: usize,
966            actual_start: usize,
967        },
968        /// Two consecutive `EmptyRun` entries: the second is redundant
969        /// because the first could have been extended to cover the same
970        /// range. This is a canonicity violation — `compress_causal_path`
971        /// never emits adjacent runs.
972        NonMaximalRun,
973        /// A `Step` entry whose `count == 0` and `hash` equals the
974        /// canonical empty subtree at the expected sibling depth. This
975        /// should have been collapsed into an `EmptyRun` by the
976        /// compressor. Rejecting it enforces the same canonicity rule
977        /// that adjacent `EmptyRun` rejection does: every step in a
978        /// compressed path is either non-empty or part of a maximal run.
979        NonCanonicalStep,
980    }

1020    /// Compresses a causal-trie proof path by collapsing runs of
1021    /// consecutive canonical-empty siblings into `EmptyRun` entries.
1022    ///
1023    /// Each step at position `i` in `path` is combined at parent depth
1024    /// `T - 1 - i` (where `T = terminal_depth`); the sibling subtree it
1025    /// carries is therefore rooted at depth `T - i`, and its canonical-empty
1026    /// value is `empty_table()[T - i]`. A step is canonical-empty when its
1027    /// hash equals that value and its count is 0. Consecutive canonical-empty
1028    /// steps are merged into a single `EmptyRun { start_depth, length }`,
1029    /// where `start_depth` is the *sibling* depth (`T - i`) of the first
1030    /// empty step in the run.
1031    #[must_use]
1032    pub fn compress_causal_path(
1033        terminal_depth: usize,
1034        path: &[CausalProofStep],
1035    ) -> Vec<CompressedCausalStep> {
1036        let empty = empty_table();
1037        let mut out: Vec<CompressedCausalStep> = Vec::new();
1038        let mut run_start: Option<usize> = None;
1039        let mut run_len: u16 = 0;
1040
1041        for (i, step) in path.iter().enumerate() {
1042            // Sibling depth: one above the parent depth `T - 1 - i`.
1043            let sibling_depth = terminal_depth.saturating_sub(i);
1044            let is_empty =
1045                step.count == 0 && sibling_depth < empty.len() && step.hash == empty[sibling_depth];
1046
1047            if is_empty {
1048                if run_start.is_none() {
1049                    run_start = Some(sibling_depth);
1050                    run_len = 1;
1051                } else {
1052                    run_len = run_len.saturating_add(1);
1053                }
1054            } else {
1055                // Flush any active run (sibling depths decrease as `i`
1056                // increases, so `start_depth` is the deepest/first point
1057                // of the run and the run expands toward the root).
1058                if let Some(start) = run_start.take() {
1059                    out.push(CompressedCausalStep::EmptyRun {
1060                        start_depth: depth_u16(start),
1061                        length: run_len,
1062                    });
1063                    run_len = 0;
1064                }
1065                out.push(CompressedCausalStep::Step(*step));
1066            }
1067        }
1068
1069        // Flush trailing run.
1070        if let Some(start) = run_start.take() {
1071            out.push(CompressedCausalStep::EmptyRun {
1072                start_depth: depth_u16(start),
1073                length: run_len,
1074            });
1075        }
1076
1077        out
1078    }

1080    /// Decompresses a compressed causal-trie proof path, validating that
1081    /// every `EmptyRun` expands to the correct canonical empty hashes with
1082    /// count 0.
1083    ///
1084    /// Returns the expanded `Vec<CausalProofStep>` on success, or a
1085    /// [`CausalProofError`] if the compressed encoding is malformed,
1086    /// non-canonical, or has the wrong total length.
1087    ///
1088    /// # Errors
1089    ///
1090    /// Returns [`CausalProofError`] on any validation failure.
1091    pub fn decompress_causal_path(
1092        terminal_depth: usize,
1093        compressed: &[CompressedCausalStep],
1094    ) -> Result<Vec<CausalProofStep>, CausalProofError> {
1095        if terminal_depth > CAUSAL_DEPTH {
1096            return Err(CausalProofError::InvalidDepth(depth_u16(terminal_depth)));
1097        }
1098        let empty = empty_table();
1099        let mut out: Vec<CausalProofStep> = Vec::new();
1100        let mut prev_was_empty_run = false;
1101
1102        for step in compressed {
1103            match step {
1104                CompressedCausalStep::Step(s) => {
1105                    if out.len() >= terminal_depth {
1106                        return Err(CausalProofError::ExcessData);
1107                    }
1108                    // Reject a Step that carries a canonical-empty value:
1109                    // the compressor would have collapsed this into an
1110                    // EmptyRun, so emitting it as a Step is non-canonical.
1111                    let sibling_depth = terminal_depth.saturating_sub(out.len());
1112                    if s.count == 0 && sibling_depth < empty.len() && s.hash == empty[sibling_depth]
1113                    {
1114                        return Err(CausalProofError::NonCanonicalStep);
1115                    }
1116                    prev_was_empty_run = false;
1117                    out.push(*s);
1118                }
1119                CompressedCausalStep::EmptyRun {
1120                    start_depth,
1121                    length,
1122                } => {
1123                    // Reject consecutive EmptyRuns: the second should
1124                    // have been merged into the first by the compressor.
1125                    if prev_was_empty_run {
1126                        return Err(CausalProofError::NonMaximalRun);
1127                    }
1128                    prev_was_empty_run = true;
1129                    let start = *start_depth as usize;
1130                    let len = *length as usize;
1131
1132                    if len == 0 {
1133                        return Err(CausalProofError::Truncated);
1134                    }
1135                    // Sibling depths only exist in `1..=CAUSAL_DEPTH` (depth
1136                    // 0 is the root, which has no parent to be a sibling
1137                    // of), so `start` outside that range is structurally
1138                    // invalid regardless of where it sits in the path.
1139                    if start == 0 || start > CAUSAL_DEPTH {
1140                        return Err(CausalProofError::InvalidDepth(*start_depth));
1141                    }
1142                    // The run expands toward the root: sibling depths
1143                    // `start, start - 1, ..., start - len + 1`, which must
1144                    // not go below depth 1. Checked before contiguity so a
1145                    // self-evidently out-of-bounds run is reported as such
1146                    // even when it's also misaligned with the current
1147                    // position.
1148                    if len > start {
1149                        return Err(CausalProofError::RunBelowRoot);
1150                    }
1151                    // `start` is the sibling depth of the first expanded
1152                    // step, i.e. `terminal_depth - out.len()`. A mismatch
1153                    // means the encoding's run boundaries don't line up
1154                    // with its actual position in the path.
1155                    let expected_start = terminal_depth.saturating_sub(out.len());
1156                    if start != expected_start {
1157                        return Err(CausalProofError::NonContiguousRun {
1158                            expected_sibling_depth: expected_start,
1159                            actual_start: start,
1160                        });
1161                    }
1162                    for j in 0..len {
1163                        // `j < len <= start` (checked above) and
1164                        // `sibling_depth >= 1` (since `len <= start`
1165                        // guarantees `start - len + 1 >= 1`), so both
1166                        // subtractions stay in range; `saturating_sub`
1167                        // avoids the raw `-` clippy flags regardless.
1168                        let sibling_depth = start.saturating_sub(j);
1169                        out.push(CausalProofStep {
1170                            hash: empty[sibling_depth],
1171                            count: 0,
1172                        });
1173                    }
1174                }
1175            }
1176        }
1177
1178        if out.len() != terminal_depth {
1179            return Err(CausalProofError::PathLengthMismatch {
1180                decompressed: out.len(),
1181                expected: terminal_depth,
1182            });
1183        }
1184
1185        Ok(out)
1186    }

1231        /// Splits `prefix` (bits `0..d` already set to `key`'s path, the
1232        /// rest zero) into `key`'s two depth-`d+1` children: `(sibling,
1233        /// own)`. `sibling` is the subtree `key` does *not* descend into;
1234        /// `own` is the one it does. Shared by [`Self::inclusion_proof`]
1235        /// and [`Self::non_inclusion_proof`], whose only difference is what
1236        /// they do with `own` (walk it unconditionally vs. stop at the
1237        /// first empty one).
1238        fn step_prefixes(mut prefix: [u8; 32], key: &Hash, d: usize) -> ([u8; 32], [u8; 32]) {
1239            let mut sibling = prefix;
1240            let byte_idx = d / 8;
1241            let bit = 1_u8 << 7_usize.wrapping_sub(d % 8);
1242            if causal_bit(key, d) == 0 {
1243                // key goes left; sibling is right.
1244                sibling[byte_idx] |= bit;
1245                prefix[byte_idx] &= !bit;
1246            } else {
1247                // key goes right; sibling is left.
1248                sibling[byte_idx] &= !bit;
1249                prefix[byte_idx] |= bit;
1250            }
1251            (sibling, prefix)
1252        }
1253
1254        /// Looks up the sibling at `(child_depth, sibling_prefix)`, falling
1255        /// back to the canonical empty node for that depth, and packages it
1256        /// as a proof step.
1257        fn sibling_step(
1258            &self,
1259            empty: &EmptyTable,
1260            child_depth: u16,
1261            sibling_prefix: [u8; 32],
1262            d: usize,
1263        ) -> CausalProofStep {
1264            let (sib_hash, sib_count) = self
1265                .nodes
1266                .get(&(child_depth, sibling_prefix))
1267                .copied()
1268                .unwrap_or((empty[d.wrapping_add(1)], 0));
1269            CausalProofStep {
1270                hash: sib_hash,
1271                count: sib_count,
1272            }
1273        }
1274
1275        /// Returns the ordered (leaf-to-root) sibling path proving `key` is a
1276        /// member of `self`, along with `self`'s root and count. Returns
1277        /// [`None`] if `key` is not a member; there is no inclusion proof for
1278        /// a non-member. O(256) — walks the node cache.
1279        #[must_use]
1280        pub fn inclusion_proof(&self, key: &Hash) -> Option<(Vec<CausalProofStep>, Hash, u64)> {
1281            if self.keys.is_empty() || !self.keys.contains(key) {
1282                return None;
1283            }
1284            let empty = empty_table();
1285            let mut path = Vec::with_capacity(CAUSAL_DEPTH);
1286            let mut prefix = [0u8; 32];
1287            for d in 0..CAUSAL_DEPTH {
1288                let child_depth = depth_u16(d.wrapping_add(1));
1289                let (sibling_prefix, own_prefix) = Self::step_prefixes(prefix, key, d);
1290                path.push(self.sibling_step(&empty, child_depth, sibling_prefix, d));
1291                prefix = own_prefix;
1292            }
1293            let root_hash = self.root();
1294            let root_count = self.count();
1295            path.reverse();
1296            Some((path, root_hash, root_count))
1297        }
1298
1299        /// Returns the ordered (leaf-to-root) sibling path proving `key` is
1300        /// NOT a member of `self` (the key-directed path terminates in a
1301        /// canonical empty subtree at the returned depth), along with
1302        /// `self`'s root and count. Returns [`None`] if `key` IS a member; no
1303        /// non-inclusion proof exists for a member. O(256) — walks the node
1304        /// cache.
1305        #[must_use]
1306        pub fn non_inclusion_proof(
1307            &self,
1308            key: &Hash,
1309        ) -> Option<(Vec<CausalProofStep>, usize, Hash, u64)> {
1310            let empty = empty_table();
1311            if self.keys.is_empty() {
1312                return Some((Vec::new(), 0, empty[0], 0));
1313            }
1314            // Walk down the key-directed path, collecting siblings, until we
1315            // hit an empty node (the key is not in the set).
1316            let mut path = Vec::new();
1317            let mut prefix = [0u8; 32];
1318            for d in 0..CAUSAL_DEPTH {
1319                let child_depth = depth_u16(d.wrapping_add(1));
1320                let (sibling_prefix, child_prefix) = Self::step_prefixes(prefix, key, d);
1321                path.push(self.sibling_step(&empty, child_depth, sibling_prefix, d));
1322                // Check if the child node on the key-directed path exists
1323                // and is non-empty.
1324                let child = self.nodes.get(&(child_depth, child_prefix));
1325                let child_is_empty = child.map_or(true, |(_, c)| *c == 0);
1326                if child_is_empty {
1327                    // Found the terminal: an empty subtree at depth d+1.
1328                    let root_hash = self.root();
1329                    let root_count = self.count();
1330                    path.reverse();
1331                    return Some((path, d.wrapping_add(1), root_hash, root_count));
1332                }
1333                prefix = child_prefix;
1334            }
1335            // All 256 levels are non-empty and key was found — this is
1336            // actually an inclusion case, but the caller asked for
1337            // non_inclusion_proof which returns None for members.
1338            None
1339        }
1340    }

1355    /// Recomputes a root from the canonical empty hash at `terminal_depth`
1356    /// and `path` (leaf-to-root ordered siblings) and reports whether it
1357    /// matches `root` and `count`.
1358    #[must_use]
1359    pub fn verify_causal_non_inclusion(
1360        key: &Hash,
1361        terminal_depth: usize,
1362        path: &[CausalProofStep],
1363        root: Hash,
1364        count: u64,
1365    ) -> bool {
1366        if terminal_depth > CAUSAL_DEPTH {
1367            return false;
1368        }
1369        if path.len() != terminal_depth {
1370            return false;
1371        }
1372        // Minimality: if the sibling at the terminal depth were also
1373        // canonical-empty, the parent at terminal_depth - 1 would have two
1374        // empty children and so be canonical-empty itself — the descent
1375        // would have stopped a level higher. Biconditional, so this is
1376        // exact, not a heuristic.
1377        if let Some(s0) = path.first() {
1378            if s0.count == 0 && s0.hash == empty_table()[terminal_depth] {
1379                return false;
1380            }
1381        }
1382        // Under `std`, `empty_table` is built once and reused by root, proof,
1383        // and verification operations. The no_std fallback retains allocation-
1384        // free portability without requiring a synchronization primitive.
1385        verify_causal_path(
1386            empty_table()[terminal_depth],
1387            0,
1388            terminal_depth,
1389            key,
1390            path,
1391            root,
1392            count,
1393        )
1394    }

1396    /// Sums two subtree/sibling counts. A locally built causal set's
1397    /// population is bounded by the number of distinct 32-byte keys held in
1398    /// memory, so this sum cannot reach `u64::MAX`; `saturating_add` avoids a
1399    /// panic path without changing any reachable result. Verification of an
1400    /// untrusted path uses `checked_add` in `verify_causal_path` and rejects
1401    /// overflow, as the draft's "Room-version validity" section requires.
1402    fn count_sum(a: u64, b: u64) -> u64 {
1403        a.saturating_add(b)
1404    }

1406    /// Recomputes a causal trie root from a terminal node (either a
1407    /// `causal_leaf` and count 1, or a canonical empty hash and count 0) by
1408    /// applying `path`'s siblings from the level just above the terminal
1409    /// depth up to the root. `path` is ordered leaf-to-root (deepest sibling
1410    /// first), so `depth` walks downward from `terminal_depth - 1` to 0;
1411    /// `path.len() == terminal_depth` is checked first, so the decrement
1412    /// below never underflows past 0.
1413    ///
1414    /// The side (left/right) at each level is derived from `causal_bit(key,
1415    /// depth)`, not stored in the path — this removes a redundant wire field
1416    /// and eliminates an entire class of forgery.
1417    fn verify_causal_path(
1418        terminal_hash: Hash,
1419        terminal_count: u64,
1420        terminal_depth: usize,
1421        key: &Hash,
1422        path: &[CausalProofStep],
1423        root: Hash,
1424        count: u64,
1425    ) -> bool {
1426        if path.len() != terminal_depth {
1427            return false;
1428        }
1429        let mut cur_hash = terminal_hash;
1430        let mut cur_count = terminal_count;
1431        let mut depth = terminal_depth;
1432        for step in path {
1433            depth = depth.saturating_sub(1);
1434            let side = if causal_bit(key, depth) == 0 {
1435                CausalSide::Right
1436            } else {
1437                CausalSide::Left
1438            };
1439            cur_hash = match side {
1440                CausalSide::Left => {
1441                    causal_node(depth_u16(depth), step.hash, step.count, cur_hash, cur_count)
1442                }
1443                CausalSide::Right => {
1444                    causal_node(depth_u16(depth), cur_hash, cur_count, step.hash, step.count)
1445                }
1446            };
1447            cur_count = match cur_count.checked_add(step.count) {
1448                Some(sum) => sum,
1449                None => return false,
1450            };
1451        }
1452        cur_hash == root && cur_count == count
1453    }

1474        /// Splits `keys` into (left, right) by their bit at `depth`, matching
1475        /// [`causal_bit`]'s MSB-first convention.
1476        fn partition_by_bit(keys: &[Hash], depth: usize) -> (Vec<Hash>, Vec<Hash>) {
1477            let mut left = Vec::new();
1478            let mut right = Vec::new();
1479            for k in keys {
1480                if causal_bit(k, depth) == 0 {
1481                    left.push(*k);
1482                } else {
1483                    right.push(*k);
1484                }
1485            }

1489        /// Independent recursive computation of the causal trie root for a
1490        /// given key set. Used as a differential oracle against the
1491        /// incremental node-cache implementation.
1492        pub(crate) fn subtree_root(keys: &[Hash], depth: usize) -> (Hash, u64) {
1493            if depth == CAUSAL_DEPTH {
1494                return (causal_leaf(keys[0]), 1);
1495            }
1496            let (left, right) = partition_by_bit(keys, depth);
1497            let next_depth = depth.saturating_add(1);
1498            let (left_hash, left_count) = subtree_root_or_empty(&left, next_depth);
1499            let (right_hash, right_count) = subtree_root_or_empty(&right, next_depth);
1500            (
1501                causal_node(
1502                    depth_u16(depth),
1503                    left_hash,
1504                    left_count,
1505                    right_hash,
1506                    right_count,
1507                ),
1508                count_sum(left_count, right_count),
1509            )
1510        }

1522        /// Memoized causal-trie oracle.
1523        ///
1524        /// Builds every non-empty subtree of the trie over `keys` exactly
1525        /// once, keyed by `(depth, prefix)` the same way the production node
1526        /// cache is — `O(n·depth)` nodes total. A proof for any key is then
1527        /// just `O(depth)` memo lookups, instead of the from-scratch
1528        /// `O(n·depth²)` recompute the old `descend` paid per key (and
1529        /// `O(n²·depth²)` across the set). This is what lets the dense tests
1530        /// run `n` up to 64 at negligible cost.
1531        pub(crate) struct CausalOracle {
1532            nodes: std::collections::HashMap<(u16, [u8; 32]), (Hash, u64)>,
1533        }
1534
1535        impl CausalOracle {
1536            /// Builds the memoized subtree table for `keys`.
1537            pub(crate) fn new(keys: &[Hash]) -> Self {
1538                let mut nodes = std::collections::HashMap::new();
1539                let empty = empty_table();
1540                Self::build(&mut nodes, keys, 0, [0u8; 32], &empty);
1541                Self { nodes }
1542            }
1543
1544            fn build(
1545                nodes: &mut std::collections::HashMap<(u16, [u8; 32]), (Hash, u64)>,
1546                keys: &[Hash],
1547                depth: usize,
1548                prefix: [u8; 32],
1549                empty: &EmptyTable,
1550            ) {
1551                let Some(first) = keys.first() else {
1552                    return;
1553                };
1554                if depth == CAUSAL_DEPTH {
1555                    nodes.insert((depth_u16(depth), prefix), (causal_leaf(*first), 1));
1556                    return;
1557                }
1558                let (left, right) = partition_by_bit(keys, depth);
1559                let next_depth = depth.saturating_add(1);
1560                let mut left_prefix = prefix;
1561                let mut right_prefix = prefix;
1562                clear_bit(&mut left_prefix, depth);
1563                set_bit(&mut right_prefix, depth);
1564                Self::build(nodes, &left, next_depth, left_prefix, empty);
1565                Self::build(nodes, &right, next_depth, right_prefix, empty);
1566                let (left_hash, left_count) = nodes
1567                    .get(&(depth_u16(next_depth), left_prefix))
1568                    .copied()
1569                    .unwrap_or((empty[next_depth], 0));
1570                let (right_hash, right_count) = nodes
1571                    .get(&(depth_u16(next_depth), right_prefix))
1572                    .copied()
1573                    .unwrap_or((empty[next_depth], 0));
1574                nodes.insert(
1575                    (depth_u16(depth), prefix),
1576                    (
1577                        causal_node(
1578                            depth_u16(depth),
1579                            left_hash,
1580                            left_count,
1581                            right_hash,
1582                            right_count,
1583                        ),
1584                        count_sum(left_count, right_count),
1585                    ),
1586                );
1587            }
1588
1589            /// Root (`hash`, `count`) of the whole key set — the node at
1590            /// depth 0, or the canonical empty root.
1591            pub(crate) fn root(&self) -> (Hash, u64) {
1592                self.nodes
1593                    .get(&(depth_u16(0), [0u8; 32]))
1594                    .copied()
1595                    .unwrap_or((empty_table()[0], 0))
1596            }
1597
1598            /// The root (`hash`, `count`) of the whole key set, plus the
1599            /// leaf-to-root sibling path along `target`'s bit-directed
1600            /// descent. A member ends in [`TerminalKind::Leaf`] at
1601            /// [`CAUSAL_DEPTH`]; a non-member ends in [`TerminalKind::Empty`]
1602            /// at the first depth whose target-directed subtree is empty.
1603            pub(crate) fn descend(
1604                &self,
1605                target: &Hash,
1606            ) -> (Hash, u64, Vec<CausalProofStep>, TerminalKind, usize) {
1607                let empty = empty_table();
1608                let mut path = Vec::with_capacity(CAUSAL_DEPTH);
1609                let mut prefix = [0u8; 32];
1610                let mut depth = 0;
1611                let (kind, term_depth) = loop {
1612                    if !self.nodes.contains_key(&(depth_u16(depth), prefix)) {
1613                        break (TerminalKind::Empty, depth);
1614                    }
1615                    if depth == CAUSAL_DEPTH {
1616                        break (TerminalKind::Leaf, depth);
1617                    }
1618                    let next_depth = depth.saturating_add(1);
1619                    let mut sibling_prefix = prefix;
1620                    if causal_bit(target, depth) == 0 {
1621                        set_bit(&mut sibling_prefix, depth);
1622                        clear_bit(&mut prefix, depth);
1623                    } else {
1624                        clear_bit(&mut sibling_prefix, depth);
1625                        set_bit(&mut prefix, depth);
1626                    }
1627                    let (sibling_hash, sibling_count) = self
1628                        .nodes
1629                        .get(&(depth_u16(next_depth), sibling_prefix))
1630                        .copied()
1631                        .unwrap_or((empty[next_depth], 0));
1632                    path.push(CausalProofStep {
1633                        hash: sibling_hash,
1634                        count: sibling_count,
1635                    });
1636                    depth = next_depth;
1637                };
1638                path.reverse();
1639                let (root_hash, root_count) = self.root();
1640                (root_hash, root_count, path, kind, term_depth)
1641            }
1642        }
