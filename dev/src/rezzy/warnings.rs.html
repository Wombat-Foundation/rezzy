1//! Diagnostic warnings for spec-undefined or implementation-deferred conditions.
2//!
3//! Unlike [`crate::auth::AuthError`] (which strictly invalidates events per the
4//! spec), a [`Warning`] reports non-fatal anomalies — such as missing DAG
5//! references or legacy oversized fields — without enforcing a rejection policy.
6//!
7//! Every variant provides a stable [`Warning::code`] for programmatic matching and logging.
8
9use alloc::string::String;
10use alloc::vec::Vec;
11
12/// A non-fatal condition rezzy detected but left to the caller's own policy.
13///
14/// See the module docs for the distinction from [`crate::auth::AuthError`].
15#[derive(Debug, Clone, PartialEq, Eq)]
16pub enum Warning<Id = String> {
17    /// An event's `prev_events` cite one or more IDs not found in the local DAG.
18    UnknownPrevEvent {
19        /// The event whose `prev_events` reference missing parents.
20        event_id: Id,
21        /// The specific parent IDs that are unknown locally.
22        missing_ids: Vec<Id>,
23    },
24    /// A pre-v11 event exceeds the 255-byte field length limit.
25    OversizedFieldPreV11 {
26        /// The event with the oversized field.
27        event_id: Id,
28        /// Which field exceeded the limit (`"event_id"`, `"sender"`, `"event_type"`, or `"state_key"`).
29        field: &'static str,
30        /// The field's actual byte length.
31        len: usize,
32        /// The limit it exceeded (255, per spec).
33        limit: usize,
34    },
35}
36
37impl<Id> Warning<Id> {
38    /// A short, stable code for this warning, safe to match on or log
39    /// across rezzy versions independent of the variant's exact fields.
40    #[must_use]
41    pub const fn code(&self) -> &'static str {
42        match self {
43            Self::UnknownPrevEvent { .. } => "W001_UNKNOWN_PREV_EVENT",
44            Self::OversizedFieldPreV11 { .. } => "W002_OVERSIZED_FIELD_PRE_V11",
45        }
46    }
47}
48
49impl<Id: core::fmt::Display> core::fmt::Display for Warning<Id> {
50    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
51        match self {
52            Self::UnknownPrevEvent {
53                event_id,
54                missing_ids,
55            } => {
56                write!(
57                    f,
58                    "[{}] event {} references {} missing prev_events",
59                    self.code(),
60                    event_id,
61                    missing_ids.len()
62                )
63            }
64            Self::OversizedFieldPreV11 {
65                event_id,
66                field,
67                len,
68                limit,
69            } => {
70                write!(
71                    f,
72                    "[{}] event {} field '{}' length {} exceeds limit {}",
73                    self.code(),
74                    event_id,
75                    field,
76                    len,
77                    limit
78                )
79            }
80        }
81    }
82}

84impl<Id> From<crate::state::at::BackwardExtremity<Id>> for Warning<Id> {
85    fn from(gap: crate::state::at::BackwardExtremity<Id>) -> Self {
86        Self::UnknownPrevEvent {
87            event_id: gap.event_id,
88            missing_ids: gap.missing_prev_events,
89        }
90    }
91}

93/// A successful result bundled with any non-fatal [`Warning`]s collected
94/// along the way.
95///
96/// Hard failures stay in the `Err` side of whatever `Result` a function
97/// returns; `Outcome` only ever wraps the `Ok` payload, so a caller that
98/// doesn't care about warnings can ignore `.warnings` entirely and still
99/// get correct pass/fail behavior from the `Result`.
100#[derive(Debug, Clone, PartialEq, Eq)]
101pub struct Outcome<T, Id> {
102    /// The successful result.
103    pub value: T,
104    /// Non-fatal conditions collected while producing `value`. Empty in the
105    /// common case; never affects whether `value` itself is trustworthy --
106    /// only whether the caller wants to act on what's in here (log it,
107    /// apply local policy, surface it to an operator, ...).
108    pub warnings: Vec<Warning<Id>>,
109}
110
111impl<T, Id> Outcome<T, Id> {
112    /// Wraps `value` with no warnings.
113    #[must_use]
114    pub const fn new(value: T) -> Self {
115        Self {
116            value,
117            warnings: Vec::new(),
118        }
119    }
120
121    /// Wraps `value` with an already-collected list of warnings.
122    #[must_use]
123    pub const fn with_warnings(value: T, warnings: Vec<Warning<Id>>) -> Self {
124        Self { value, warnings }
125    }
126}

128#[cfg(test)]
129#[cfg_attr(coverage_nightly, coverage(off))]
130mod tests {
131    use super::{Outcome, Warning};
132    use alloc::{
133        string::{String, ToString},
134        vec,
135        vec::Vec,
136    };
137
138    #[test]
139    fn test_warning_codes_are_stable() {
140        let unknown_prev: Warning<String> = Warning::UnknownPrevEvent {
141            event_id: "$a".into(),
142            missing_ids: vec!["$missing".into()],
143        };
144        assert_eq!(unknown_prev.code(), "W001_UNKNOWN_PREV_EVENT");
145        assert_eq!(
146            unknown_prev.to_string(),
147            "[W001_UNKNOWN_PREV_EVENT] event $a references 1 missing prev_events"
148        );
149
150        let oversized: Warning<String> = Warning::OversizedFieldPreV11 {
151            event_id: "$a".into(),
152            field: "sender",
153            len: 300,
154            limit: 255,
155        };
156        assert_eq!(oversized.code(), "W002_OVERSIZED_FIELD_PRE_V11");
157        assert_eq!(
158            oversized.to_string(),
159            "[W002_OVERSIZED_FIELD_PRE_V11] event $a field 'sender' length 300 exceeds limit 255"
160        );
161    }
162
163    #[test]
164    fn test_backward_extremity_converts_to_unknown_prev_event_warning() {
165        let gap = crate::state::at::BackwardExtremity {
166            event_id: "$a".to_string(),
167            missing_prev_events: vec!["$b".to_string(), "$c".to_string()],
168        };
169        let warning: Warning<String> = gap.into();
170        assert_eq!(
171            warning,
172            Warning::UnknownPrevEvent {
173                event_id: "$a".into(),
174                missing_ids: vec!["$b".into(), "$c".into()],
175            }
176        );
177        assert_eq!(warning.code(), "W001_UNKNOWN_PREV_EVENT");
178    }
179
180    #[test]
181    fn test_outcome_new_has_no_warnings() {
182        let outcome: Outcome<i32, String> = Outcome::new(42);
183        assert_eq!(outcome.value, 42);
184        assert_eq!(outcome.warnings, [] as [Warning<String>; 0]);
185    }
186
187    #[test]
188    fn test_outcome_with_warnings() {
189        let warnings: Vec<Warning<String>> = vec![Warning::OversizedFieldPreV11 {
190            event_id: "$a".into(),
191            field: "event_id",
192            len: 300,
193            limit: 255,
194        }];
195        let outcome = Outcome::with_warnings(42, warnings.clone());
196        assert_eq!(outcome.value, 42);
197        assert_eq!(outcome.warnings, warnings);
198    }
199}
