15//! Causal Domination Operator (CDO) — vectorized conflicted-event filter.
16//! **Retired from the live path and retained as design history.**
17//!
18//! This module is no longer called by resolution: the V2.1.1 pre-filter was
19//! removed from `prepare_conflicted_and_keys` (see the soundness note there)
20//! because it was unsound. It is kept here, with its tests, as the record of
21//! what was tried and why it failed, and as a reference for the replacement
22//! (a resolved-state screening pass that *applies* auth predicates instead of
23//! approximating domination).
24//!
25//! ## What the CDO was
26//!
27//! A V2.1.1 optimization that ran *before* the main resolution algorithm. It
28//! identified conflicted events **causally dominated** by a higher-priority
29//! administrative action (ban, kick, PL demotion, or join-rules lockdown) and
30//! removed them from the conflicted set. An event was "causally dominated" if
31//! a higher-priority admin action *restricted* it (see
32//! [`LeanEvent::restricts_event`]) and was **not** an ancestor or descendant
33//! of it (independent causal branches). Ancestor/descendant relationships were
34//! computed via SWAR bitmask sweeps over a topologically-sorted event array.
35//!
36//! ## Why it is unsound
37//!
38//! **A drop is sound iff `IterativeAuthChecks` would have rejected the event.**
39//! The CDO ran pre-auth, before `resolved` held the authoritative power state,
40//! so it could never establish that. Three independent violations were found,
41//! each by an adversarial fixture rather than by reasoning:
42//! - **Dominator-validity (the decisive one):** the CDO trusted a dominator's
43//!   *structural* shape (ban/kick/lockdown/demotion) without verifying the
44//!   dominator itself passes auth. An auth-invalid, low-power user's forged
45//!   ban erases a legitimate membership on CDO-running servers while non-CDO
46//!   (Synapse) servers keep it — a permanent federation fork. This falsifies
47//!   the earlier claim (below) that `is_ban_or_kick` domination "did not
48//!   reproduce a divergence." Reachable across all three admin classes; see
49//!   `cdo_dominator_validity_gap_scope_inverted` in the differential harness.
50//! - **Join-rules lockdown:** `is_lockdown()` fired on independent branches
51//!   without checking the target was actually unauthorized (mitigated for a
52//!   target that cites prior authorization, but bypassable).
53//! - **Transitive propagation:** an event dropped via a dropped dominator
54//!   cascaded drops transitively (fixed, then the whole filter retired).
55//! - **Forgeable priority:** `sort_cdo_events` ordered dominators by the
56//!   author-supplied `event.power_level` (see the note on that function),
57//!   never validated against auth — a second, independent trust-of-input hole,
58//!   the same class the resolver refuses to tolerate for `depth`.
59//!
60//! ## What the replacement must be
61//!
62//! Not "the CDO moved later." Once `resolved` holds authoritative
63//! `power_levels`/`join_rules`/`bans`, the sound predicate is the auth rule itself
64//! ("is this sender banned / under-powered in `resolved`?"), applied directly
65//! as an O(N) screening pass over remaining candidates — not a domination test.
66//! Concurrency-implies-domination was the unsound core; it does not come back
67//! post-power-phase, it stops being needed. The replacement must also check
68//! what phase-2 (non-power) acceptance can still mutate that another event's
69//! auth reads, and must not change mainline ordering of the surviving set.

80/// Returns `true` if `possible_ancestor_id` is an ancestor of `child_id`.
81///
82/// Missing event IDs in the lookup context return `false`, except when
83/// `child_id == possible_ancestor_id`, which is `true` regardless of
84/// context membership.
85#[must_use]
86pub fn is_ancestor<Id, C: Clone, Q, S: core::hash::BuildHasher, K>(
87    child_id: &Q,
88    possible_ancestor_id: &Q,
89    context: &HashMap<Id, LeanEvent<Id, C, K>, S>,
90) -> bool
91where
92    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
93    Q: ?Sized + Eq + core::hash::Hash + Ord,
94{
95    if child_id == possible_ancestor_id {
96        return true;
97    }
98    let Some((actual_child, _)) = context.get_key_value(child_id) else {
99        return false;
100    };
101    let Some((actual_ancestor, _)) = context.get_key_value(possible_ancestor_id) else {
102        return false;
103    };
104
105    // No pruning on `depth` here: it is an author-supplied field on the
106    // event, not verified against the graph before this runs, so a forged
107    // depth (e.g. an ancestor claiming a depth >= the child's) could make a
108    // real ancestor relationship come back `false`. Correctness over the
109    // pruning shortcut — walk the actual prev_events/auth_events edges.
110    let mut stack = Vec::new();
111    stack.push(actual_child);
112    let mut visited = BTreeSet::new();
113    visited.insert(actual_child);
114
115    while let Some(current) = stack.pop() {
116        if current == actual_ancestor {
117            return true;
118        }
119        if let Some(ev) = context.get(current.borrow()) {
120            for parent in ev.prev_events.iter().chain(ev.auth_events.iter()) {
121                if visited.insert(parent) {
122                    stack.push(parent);
123                }
124            }
125        }
126    }
127    false
128}

130/// Number of `u64` words per bitmask chunk (8 × 64 = 512 bits on AVX-512).
131#[cfg(target_feature = "avx512f")]
132const WORDS_PER_CHUNK: usize = 8;
133
134/// Number of `u64` words per bitmask chunk (4 × 64 = 256 bits on AVX2/NEON).
135#[cfg(not(target_feature = "avx512f"))]
136const WORDS_PER_CHUNK: usize = 4;
137
138fn compute_cdo_bit_masks_chunk<Id, C, S: core::hash::BuildHasher, K>(
139    admin_chunk: &[Id],
140    id_to_idx: &HashMap<Id, usize, S>,
141    sorted_events: &[(usize, &LeanEvent<Id, C, K>)],
142    parents: &[Vec<usize>],
143    children: &[Vec<usize>],
144    and_masks: &mut [u64],
145    desc_masks: &mut [u64],
146) where
147    Id: crate::basespec::rezzy_types::EventId,
148{
149    and_masks.fill(0);
150    desc_masks.fill(0);
151
152    for (i, admin_id) in admin_chunk.iter().enumerate() {
153        if let Some(&idx) = id_to_idx.get(admin_id) {
154            let word = i >> 6;
155            let bit = 1u64 << (i & 63);
156            let target_idx = idx.saturating_mul(WORDS_PER_CHUNK).saturating_add(word);
157            and_masks[target_idx] |= bit;
158            desc_masks[target_idx] |= bit;
159        }
160    }
161
162    // Forward Sweep (Ancestors) - Pure array iteration
163    for &(u, _) in sorted_events {
164        let u_base = u.saturating_mul(WORDS_PER_CHUNK);
165        for &p in &parents[u] {
166            let p_base = p.saturating_mul(WORDS_PER_CHUNK);
167            for w in 0..WORDS_PER_CHUNK {
168                and_masks[u_base.saturating_add(w)] |= and_masks[p_base.saturating_add(w)];
169            }
170        }
171    }
172
173    // Backward Sweep (Descendants) - Pure array iteration
174    for &(u, _) in sorted_events.iter().rev() {
175        let u_base = u.saturating_mul(WORDS_PER_CHUNK);
176        for &c in &children[u] {
177            let c_base = c.saturating_mul(WORDS_PER_CHUNK);
178            for w in 0..WORDS_PER_CHUNK {
179                desc_masks[u_base.saturating_add(w)] |= desc_masks[c_base.saturating_add(w)];
180            }
181        }
182    }
183}

185fn sort_cdo_events<'a, Id: Ord + Clone, C: Clone, K>(
186    events: &[&'a LeanEvent<Id, C, K>],
187) -> Vec<&'a LeanEvent<Id, C, K>> {
188    // Trusts the author-supplied `event.power_level` (cdo.rs) rather than the
189    // auth-derived sender level the resolver uses via `SortPriority.power_level`
190    // (rezzy_types.rs). This field is never populated from auth before the
191    // filter runs (see prepare_conflicted_and_keys), so an attacker can forge a
192    // high `power_level` to hoist their structural admin event ahead of the
193    // victim's — a second, independent trust-of-input hole, the same class the
194    // Kahn sort below refuses to tolerate for `depth`. Retained as design
195    // history; the replacement must order by the auth-derived level.
196    let mut sorted = events.to_vec();
197    sorted.sort_by(|a, b| {
198        let type_priority = |t: &str| match t {
199            M_ROOM_POWER_LEVELS => 0,
200            M_ROOM_JOIN_RULES => 1,
201            _ => 2,
202        };
203
204        let cmp_pl = b.power_level.cmp(&a.power_level);
205        if cmp_pl != Ordering::Equal {
206            return cmp_pl;
207        }
208
209        let cmp_type = type_priority(&a.event_type).cmp(&type_priority(&b.event_type));
210        if cmp_type != Ordering::Equal {
211            return cmp_type;
212        }
213
214        let cmp_ts = a.origin_server_ts.cmp(&b.origin_server_ts);
215        if cmp_ts != Ordering::Equal {
216            return cmp_ts;
217        }
218
219        a.event_id.cmp(&b.event_id)
220    });
221    sorted
222}

224struct AdjacencyStructures<'a, Id, C, K> {
225    id_to_idx: HashMap<Id, usize>,
226    sorted_events: Vec<(usize, &'a LeanEvent<Id, C, K>)>,
227    parents: Vec<Vec<usize>>,
228    children: Vec<Vec<usize>>,
229    /// IDs appended after Kahn's algorithm failed to reach in-degree zero
230    /// (i.e. the input contained a cycle). These have no true topological
231    /// position, so the SWAR ancestor/descendant bitmasks are unreliable for
232    /// them; domination decisions must not be made from their ordering.
233    unordered_ids: BTreeSet<Id>,
234}

236/// Builds the adjacency lists and rank maps used by the domination sweep.
237fn build_adjacency_structures<'a, Id, C: Clone, S1, S2, K>(
238    conflicted_events: &'a HashMap<Id, LeanEvent<Id, C, K>, S1>,
239    auth_context: &'a HashMap<Id, LeanEvent<Id, C, K>, S2>,
240) -> AdjacencyStructures<'a, Id, C, K>
241where
242    Id: crate::basespec::rezzy_types::EventId,
243    S1: core::hash::BuildHasher,
244    S2: core::hash::BuildHasher,
245{
246    let mut relevant_events = HashMap::new();
247    let mut visited = BTreeSet::new();
248    let mut queue = alloc::collections::VecDeque::new();
249
250    for (id, ev) in conflicted_events {
251        relevant_events.insert(id.clone(), ev);
252        visited.insert(id.clone());
253        for aid in ev.prev_events.iter().chain(ev.auth_events.iter()) {
254            if visited.insert(aid.clone()) {
255                queue.push_back(aid.clone());
256            }
257        }
258    }
259
260    while let Some(aid) = queue.pop_front() {
261        if let Some(aev) = auth_context.get(&aid) {
262            relevant_events.insert(aid.clone(), aev);
263            for parent_id in aev.prev_events.iter().chain(aev.auth_events.iter()) {
264                if visited.insert(parent_id.clone()) {
265                    queue.push_back(parent_id.clone());
266                }
267            }
268        }
269    }
270
271    // Topologically order `relevant_events` via Kahn's algorithm over the
272    // actual prev_events/auth_events edges, so the SWAR sweeps below
273    // (`compute_cdo_bit_masks_chunk`) can rely on every parent sitting at a
274    // strictly lower array index than every child. Do not sort by
275    // `event.depth` here: `depth` is an author-supplied field that is not
276    // verified against the graph before this filter runs, so a forged depth
277    // can desynchronize array order from true causal order. The forward and
278    // backward sweeps are each a single O(N) pass that accumulates
279    // transitive ancestor/descendant bitmasks assuming strict topological
280    // order; a desynchronized order corrupts those bitmasks silently
281    // instead of erroring.
282    let mut in_degree: HashMap<Id, usize> = HashMap::with_capacity(relevant_events.len());
283    let mut children_of: HashMap<Id, Vec<Id>> = HashMap::with_capacity(relevant_events.len());
284    for id in relevant_events.keys() {
285        in_degree.insert(id.clone(), 0);
286    }
287    for (id, &ev) in &relevant_events {
288        for parent_id in ev.prev_events.iter().chain(ev.auth_events.iter()) {
289            if relevant_events.contains_key(parent_id) {
290                if let Some(deg) = in_degree.get_mut(id) {
291                    *deg = deg.saturating_add(1);
292                }
293                children_of
294                    .entry(parent_id.clone())
295                    .or_default()
296                    .push(id.clone());
297            }
298        }
299    }
300
301    // Deterministic Kahn's algorithm: always expand the lexicographically
302    // smallest ready id first, so output order stays reproducible across
303    // identical inputs (matching the previous sort's event_id tie-break).
304    let mut ready: BTreeSet<Id> = in_degree
305        .iter()
306        .filter(|&(_, &deg)| deg == 0)
307        .map(|(id, _)| id.clone())
308        .collect();
309    let mut sorted_ids: Vec<Id> = Vec::with_capacity(relevant_events.len());
310    let mut included: BTreeSet<Id> = BTreeSet::new();
311    while let Some(id) = ready.iter().next().cloned() {
312        ready.remove(&id);
313        included.insert(id.clone());
314        if let Some(children) = children_of.get(&id) {
315            for child in children {
316                if let Some(deg) = in_degree.get_mut(child) {
317                    *deg = deg.saturating_sub(1);
318                    if *deg == 0 {
319                        ready.insert(child.clone());
320                    }
321                }
322            }
323        }
324        sorted_ids.push(id);
325    }
326    // Defensive fallback: a genuine event DAG cannot contain a cycle, but if
327    // referential integrity is somehow violated (e.g. a maliciously crafted
328    // cycle) and some ids never reach in-degree zero, append them in a
329    // deterministic order rather than silently dropping them from the sweep.
330    // Track them as `unordered_ids`: their array position is not a real
331    // topological order, so the domination sweeps must not draw conclusions
332    // from it (they treat them as neither dominator nor dominatee).
333    let mut unordered_ids = BTreeSet::new();
334    if sorted_ids.len() < relevant_events.len() {
335        let mut leftover: Vec<Id> = relevant_events
336            .keys()
337            .filter(|id| !included.contains(*id))
338            .cloned()
339            .collect();
340        leftover.sort();
341        unordered_ids.extend(leftover.iter().cloned());
342        sorted_ids.extend(leftover);
343    }
344
345    let n = relevant_events.len();
346    let mut id_to_idx = HashMap::with_capacity(n);
347    let mut sorted_events = Vec::with_capacity(n);
348    for (i, id) in sorted_ids.into_iter().enumerate() {
349        let ev = relevant_events[&id];
350        id_to_idx.insert(id, i);
351        sorted_events.push((i, ev));
352    }
353
354    let mut parents = alloc::vec![Vec::new(); sorted_events.len()];
355    let mut children = alloc::vec![Vec::new(); sorted_events.len()];
356
357    for (child_idx, ev) in &sorted_events {
358        for parent_id in ev.prev_events.iter().chain(ev.auth_events.iter()) {
359            if let Some(&parent_idx) = id_to_idx.get(parent_id) {
360                parents[*child_idx].push(parent_idx);
361                children[parent_idx].push(*child_idx);
362            }
363        }
364    }
365
366    AdjacencyStructures {
367        id_to_idx,
368        sorted_events,
369        parents,
370        children,
371        unordered_ids,
372    }
373}

375struct PrioritizedEvents<Id> {
376    admin_actions: Vec<Id>,
377    priority_pos: HashMap<Id, usize>,

380fn prioritize_events<Id, C: crate::basespec::rezzy_types::EventContent + Clone, S1, K>(
381    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
382) -> PrioritizedEvents<Id>
383where
384    Id: crate::basespec::rezzy_types::EventId,
385    S1: core::hash::BuildHasher,
386    K: AsRef<str>,
387{
388    let admin_events_to_sort: Vec<&LeanEvent<Id, C, K>> = conflicted_events
389        .values()
390        .filter(|e| e.is_ban_or_kick() || e.is_demotion() || e.is_lockdown())
391        .collect();
392
393    let sorted_admin_events = sort_cdo_events(&admin_events_to_sort);
394    let mut admin_actions = Vec::with_capacity(sorted_admin_events.len());
395    for ev in sorted_admin_events {
396        admin_actions.push(ev.event_id.clone());
397    }
398
399    let all_sorted = sort_cdo_events(&conflicted_events.values().collect::<Vec<_>>());
400    let mut priority_pos = HashMap::with_capacity(all_sorted.len());
401    for (pos, ev) in all_sorted.into_iter().enumerate() {
402        priority_pos.insert(ev.event_id.clone(), pos);
403    }
404
405    PrioritizedEvents {
406        admin_actions,
407        priority_pos,
408    }
409}

411/// Returns `true` if `join_ev` (an `m.room.member` join) already carries its
412/// own authorization, independent of whatever `join_rules` state exists on
413/// an unrelated causal branch. Either of:
414/// - its `auth_events` cite a prior membership event for the same sender
415///   with membership `invite` or `join` (a returning/already-invited
416///   member), or
417/// - its `auth_events` cite a `m.room.join_rules` event that was *not*
418///   itself a lockdown (e.g. `public`) — the `join_rules` state the join
419///   was actually authorized against.
420///
421/// A CDO join-rules lockdown must not dominate a join that already has
422/// either of these: per the Matrix auth rules, a join authorized against a
423/// non-invite-only state (or a prior invite) remains valid even if a
424/// join-rules lockdown was concurrently, and causally independently,
425/// applied on another branch. Without this check, `restricts_event`'s
426/// lockdown case would drop *any* structurally-matching join regardless of
427/// whether the joiner actually lacked authorization — see
428/// `test_anomaly_17_sliced_dag_membership_desync` (an invited join dropped
429/// by an unrelated lockdown) and `test_anomaly_06b_mod_membership_evaporation`
430/// (a join into a still-public room dropped by a later, independent-branch
431/// lockdown, cascading to drop everything auth'd through that join).
432fn join_has_prior_authorization<Id, C, K, S1, S2>(
433    join_ev: &LeanEvent<Id, C, K>,
434    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
435    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
436) -> bool
437where
438    Id: crate::basespec::rezzy_types::EventId,
439    C: crate::basespec::rezzy_types::EventContent,
440    K: AsRef<str>,
441    S1: core::hash::BuildHasher,
442    S2: core::hash::BuildHasher,
443{
444    join_ev.auth_events.iter().any(|aid| {
445        conflicted_events
446            .get(aid)
447            .or_else(|| auth_context.get(aid))
448            .is_some_and(|ev| {
449                let cites_prior_membership = ev.event_type == M_ROOM_MEMBER
450                    && ev.state_key.as_ref().map(K::as_ref) == Some(join_ev.sender.as_str())
451                    && matches!(ev.get_membership(), Some(MEM_INVITE | MEM_JOIN));
452                let cites_non_lockdown_join_rules =
453                    ev.event_type == M_ROOM_JOIN_RULES && !ev.is_lockdown();
454                cites_prior_membership || cites_non_lockdown_join_rules
455            })
456    })
457}

459/// Returns `true` if `target_ev` cites, in its own `auth_events`, a
460/// `power_levels` event under which its sender was empowered *sufficiently*
461/// for the target event's type (meeting the `events.{type}` /
462/// `state_default` / `events_default` threshold -- not merely positive).
463///
464/// `is_demotion()` treats *any* `m.room.power_levels` event as a demotion
465/// (it does not check whether anyone was actually demoted), so
466/// `restricts_sender`'s domination check fires for every PL event on an
467/// independent branch, regardless of whether the target's own authorization
468/// predates it. This mirrors `join_has_prior_authorization`: an independent
469/// branch's demotion must not retroactively invalidate an action the sender
470/// took while validly empowered, if that empowerment is what the action's
471/// own `auth_events` actually cite. See
472/// `test_cdo_demotion_does_not_dominate_pre_demotion_authorized_action`.
473fn sender_has_pre_demotion_pl<Id, C, K, S1, S2>(
474    target_ev: &LeanEvent<Id, C, K>,
475    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
476    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
477) -> bool
478where
479    Id: crate::basespec::rezzy_types::EventId,
480    C: crate::basespec::rezzy_types::EventContent,
481    K: AsRef<str>,
482    S1: core::hash::BuildHasher,
483    S2: core::hash::BuildHasher,
484{
485    target_ev.auth_events.iter().any(|aid| {
486        conflicted_events
487            .get(aid)
488            .or_else(|| auth_context.get(aid))
489            .is_some_and(|ev| {
490                if ev.event_type != M_ROOM_POWER_LEVELS {
491                    return false;
492                }
493                // Effective PL: explicit `users[sender]` wins; else
494                // `users_default`; else 0.
495                let explicit = ev.get_user_power_level(target_ev.sender.as_str());
496                let effective = explicit.or_else(|| ev.get_users_default()).unwrap_or(0);
497
498                // Required PL for the target event type under *this* PL event.
499                // Uses the shared helper to stay in sync with auth checks.
500                let required = crate::auth::pl_threshold_for_event(
501                    ev,
502                    &target_ev.event_type,
503                    target_ev
504                        .state_key
505                        .as_ref()
506                        .map(core::convert::AsRef::as_ref),
507                );
508
509                effective >= required
510            })
511    })
512}

514fn process_direct_domination_chunks<
515    Id,
516    C: crate::basespec::rezzy_types::EventContent + Clone,
517    S1: core::hash::BuildHasher,
518    S2: core::hash::BuildHasher,
519    K,
520>(
521    adj: &AdjacencyStructures<'_, Id, C, K>,
522    prioritized: &PrioritizedEvents<Id>,
523    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
524    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
525) -> BTreeSet<Id>
526where
527    Id: crate::basespec::rezzy_types::EventId,
528    K: AsRef<str>,
529{
530    let n = adj.sorted_events.len();
531    let mut dropped_ids = BTreeSet::new();
532
533    // Allocate a strict O(N * WORDS_PER_CHUNK) matrix once, reused forever across passes
534    let mut and_masks = alloc::vec![0u64; n.saturating_mul(WORDS_PER_CHUNK)];
535    let mut desc_masks = alloc::vec![0u64; n.saturating_mul(WORDS_PER_CHUNK)];
536
537    let chunk_size = WORDS_PER_CHUNK.saturating_mul(64);
538
539    // Collect priority-sorted event IDs for iteration (positions only, no cloning events)
540    let mut priority_ordered_ids: Vec<&Id> = prioritized.priority_pos.keys().collect();
541    priority_ordered_ids.sort_by_key(|id| prioritized.priority_pos.get(*id));
542
543    for chunk in prioritized.admin_actions.chunks(chunk_size) {
544        compute_cdo_bit_masks_chunk(
545            chunk,
546            &adj.id_to_idx,
547            &adj.sorted_events,
548            &adj.parents,
549            &adj.children,
550            &mut and_masks,
551            &mut desc_masks,
552        );
553
554        // Build a map of active admin actions in this chunk to their relative index within the chunk
555        let mut chunk_admin_to_pos = HashMap::new();
556        for (i, admin_id) in chunk.iter().enumerate() {
557            if !dropped_ids.contains(admin_id) && !adj.unordered_ids.contains(admin_id) {
558                chunk_admin_to_pos.insert(admin_id, i);
559            }
560        }
561
562        // Check for direct domination against all non-dropped events
563        for event_id in &priority_ordered_ids {
564            if dropped_ids.contains(*event_id) || adj.unordered_ids.contains(*event_id) {
565                continue;
566            }
567
568            if let Some(&ev_idx) = adj.id_to_idx.get(*event_id) {
569                for (&admin_id, &orig_idx) in &chunk_admin_to_pos {
570                    if dropped_ids.contains(admin_id) {
571                        continue;
572                    }
573                    // Only higher-priority admin actions (occurring earlier in the sorted list) can dominate
574                    if let Some(&admin_pos) = prioritized.priority_pos.get(admin_id) {
575                        if let Some(&event_pos) = prioritized.priority_pos.get(*event_id) {
576                            if admin_pos >= event_pos {
577                                continue;
578                            }
579                        }
580                    }
581
582                    let word = orig_idx >> 6;
583                    let bit = 1u64 << (orig_idx & 63);
584
585                    let target_idx = ev_idx.saturating_mul(WORDS_PER_CHUNK).saturating_add(word);
586                    let is_ancestor_admin = (and_masks[target_idx] & bit) != 0;
587                    let is_descendant_admin = (desc_masks[target_idx] & bit) != 0;
588
589                    // Fast-path bitwise check first!
590                    if !is_ancestor_admin && !is_descendant_admin {
591                        if let Some(admin_ev) = conflicted_events.get(admin_id) {
592                            if let Some(target_ev) = conflicted_events.get(*event_id) {
593                                let dominates = admin_ev.restricts_event(target_ev)
594                                    && !(admin_ev.is_lockdown()
595                                        && join_has_prior_authorization(
596                                            target_ev,
597                                            conflicted_events,
598                                            auth_context,
599                                        ))
600                                    && !(admin_ev.is_demotion()
601                                        && sender_has_pre_demotion_pl(
602                                            target_ev,
603                                            conflicted_events,
604                                            auth_context,
605                                        ));
606                                if dominates {
607                                    dropped_ids.insert((*event_id).clone());
608                                    break;
609                                }
610                            }
611                        }
612                    }
613                }
614            }
615        }
616    }
617
618    dropped_ids
619}

621/// Cycle-0 Topological Filter: Vectorized Causal Domination Operator (CDO).
622///
623/// Executes strictly on the conflicted state subgraph. Returns the **safe set**
624/// of events that survived CDO filtering — i.e. events that are *not* causally
625/// dominated by any higher-priority administrative action.
626///
627/// The pipeline is:
628/// 1. **Build adjacency** — merge conflicted + auth context into a single DAG.
629/// 2. **Prioritize** — identify admin actions (bans, kicks, demotions, lockdowns)
630///    and sort all events by priority.
631/// 3. **Chunk-process** — compute ancestor/descendant bitmasks in SWAR chunks
632///    and mark dominated events.
633// jscpd:ignore-start
634#[must_use]
635pub fn apply_cdo_filter<
636    Id,
637    C: crate::basespec::rezzy_types::EventContent + Clone,
638    S1: core::hash::BuildHasher,
639    S2: core::hash::BuildHasher,
640    K,
641>(
642    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
643    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
644) -> HashMap<Id, LeanEvent<Id, C, K>>
645where
646    Id: crate::basespec::rezzy_types::EventId,
647    K: AsRef<str> + Clone,
648{
649    // jscpd:ignore-end
650    let adj = build_adjacency_structures(conflicted_events, auth_context);
651    let prioritized = prioritize_events(conflicted_events);
652    let dropped_ids =
653        process_direct_domination_chunks(&adj, &prioritized, conflicted_events, auth_context);
654
655    // Return strictly the safe set
656    let mut safe_set = HashMap::new();
657    for (id, event) in conflicted_events {
658        if !dropped_ids.contains(id) {
659            safe_set.insert(id.clone(), event.clone());
660        }
661    }
662
663    safe_set
664}
