15//! Multi-state resolution — resolve N parent state maps into one.
16//!
17//! This module provides the high-level entry point for resolving state across
18//! multiple DAG forks (e.g., multiple forward extremities). Given N state maps,
19//! it partitions entries into unconflicted (agreed by all forks) and conflicted
20//! (differing across forks), then delegates to [`resolve_iterative_sort`](crate::resolve::iterative::resolve_iterative_sort)
21//! for the conflicted subset.
22//!
23//! # Example
24//!
25//! ```rust,no_run
26//! use rezzy::{LeanEvent, SharedState, StateResVersion, HashMap};
27//! use rezzy::resolve::multi::resolve_state_maps;
28//!
29//! // Two forks with different member events
30//! let mut fork_a = SharedState::new();
31//! fork_a.insert(("m.room.create".into(), "".into()), "$create".into());
32//! fork_a.insert(("m.room.member".into(), "@alice:x".into()), "$join_a".into());
33//!
34//! let mut fork_b = SharedState::new();
35//! fork_b.insert(("m.room.create".into(), "".into()), "$create".into());
36//! fork_b.insert(("m.room.member".into(), "@alice:x".into()), "$join_b".into());
37//!
38//! // Build event context (auth chain + conflicted events)
39//! let mut ctx: HashMap<String, LeanEvent> = HashMap::new();
40//! // ... populate with conflicted events and their auth chains ...
41//!
42//! let resolved = resolve_state_maps(
43//!     &[fork_a, fork_b],
44//!     &ctx,
45//!     StateResVersion::V2,
46//! );
47//! ```

58/// Partitions N state maps into unconflicted state (agreed by all) and a set
59/// of conflicted event IDs (present in some forks with different values).
60///
61/// An entry is **unconflicted** if all N maps contain the same event ID for
62/// that `(event_type, state_key)` slot. Otherwise, all event IDs for that
63/// slot are added to the conflicted set.
64///
65/// # Parameters
66///
67/// - `state_maps`: Iterator of N state maps, each yielding
68///   `(&(event_type, state_key), &event_id)` pairs.
69/// - `num_maps`: The number of state maps (needed to determine unanimity).
70///
71/// # Returns
72///
73/// A tuple of:
74/// - `SharedState<Id>`: the unconflicted entries (agreed by all N maps).
75/// - `Vec<Id>`: the conflicted event IDs (present in at least one map
76///   but not unanimously agreed upon).
77///
78/// # Panics
79///
80/// This function does not panic.
81#[must_use]
82pub fn partition_state_maps<'a, Id, K, I, Iter>(
83    state_maps: I,
84    num_maps: usize,
85) -> (SharedState<Id, K>, Vec<Id>)
86where
87    Id: EventId,
88    I: IntoIterator<Item = Iter>,
89    K: StateKey + 'a,
90    Iter: IntoIterator<Item = (&'a (EventType, K), &'a Id)>,
91    Id: 'a,
92{
93    // Flattened borrowed-key scan: a single `FastMap` allocation instead of a
94    // nested `HashMap` per `(event_type, state_key)` slot, zero `String`/`Id`
95    // clones during the pass, and one hash lookup per `(key, id)` pair (no
96    // double hashing). Conflict vectors are allocated only on actual
97    // disagreement. `foldhash` (hashbrown's default hasher) uses a random seed
98    // per instance, preventing precomputed collision sets for these
99    // internal-only maps.
100    let mut occurrences: FastMap<&'a (EventType, K), Occurrence<'a, Id>> = FastMap::default();
101
102    for map in state_maps {
103        for (key, id) in map {
104            match occurrences.entry(key) {
105                hashbrown::hash_map::Entry::Vacant(e) => {
106                    e.insert(Occurrence {
107                        first_id: id,
108                        count: 1,
109                        conflicts: None,
110                    });
111                }
112                hashbrown::hash_map::Entry::Occupied(mut e) => {
113                    let occ = e.get_mut();
114                    occ.count = occ.count.saturating_add(1);
115                    if occ.first_id != id {
116                        // Seed the conflict vector with the first id so the
117                        // tail pass emits every distinct id as conflicted. The
118                        // membership set keeps dedup amortized O(1).
119                        match &mut occ.conflicts {
120                            None => {
121                                let mut set = FastSet::default();
122                                set.insert(occ.first_id);
123                                set.insert(id);
124                                occ.conflicts = Some((alloc::vec![occ.first_id, id], set));
125                            }
126                            Some((ids, seen)) => {
127                                if seen.insert(id) {
128                                    ids.push(id);
129                                }
130                            }
131                        }
132                    }
133                }
134            }
135        }
136    }
137
138    let mut unconflicted_state = SharedState::new();
139    let mut conflicted_ids = Vec::new();
140
141    for (key, occ) in occurrences {
142        if occ.count == num_maps && occ.conflicts.is_none() {
143            // Unanimous: seen in every map with a single, consistent id.
144            unconflicted_state.insert(key.clone(), occ.first_id.clone());
145        } else if let Some((conflicts, _)) = occ.conflicts {
146            // Disagreement: emit every distinct id as conflicted.
147            for id in conflicts {
148                conflicted_ids.push(id.clone());
149            }
150        } else {
151            // Seen in fewer than `num_maps` maps (absent from some forks) with
152            // no disagreement among the maps that do contain it.
153            conflicted_ids.push(occ.first_id.clone());
154        }
155    }
156
157    (unconflicted_state, conflicted_ids)
158}

160/// Per-`(event_type, state_key)` occurrence tally accumulated during the
161/// partition scan. Borrows from the input state maps so the scan performs zero
162/// clones; owned values are produced only when a `(key, id)` is written to the
163/// output `SharedState`/`Vec`.
164struct Occurrence<'a, Id> {
165    first_id: &'a Id,
166    count: usize,
167    /// Distinct conflicting ids in first-seen order, plus an O(1) membership
168    /// set so dedup stays amortized O(1) instead of a linear scan per id
169    /// (which would be quadratic for a slot with many distinct ids).
170    conflicts: Option<(Vec<&'a Id>, FastSet<&'a Id>)>,
171}

173/// Resolves N parent state maps into a single deterministic state map.
174///
175/// This is the high-level entry point for multi-fork state resolution.
176/// It handles the full pipeline:
177///
178/// 1. **Short-circuit**: if all maps are identical, returns the first one.
179/// 2. **Partition**: splits entries into unconflicted (unanimous) and
180///    conflicted (differing across forks).
181/// 3. **Subgraph** (V2.1+ only): computes the MSC4297 conflicted subgraph
182///    from the auth DAG and adds subgraph events to the conflicted set.
183/// 4. **Resolve**: delegates to [`resolve_iterative_sort`] with the
184///    partitioned state and conflicted events.
185///
186/// # Parameters
187///
188/// - `state_maps`: Slice of N state maps (one per fork/extremity).
189/// - `event_context`: The events needed for resolution. At minimum this
190///   must contain every conflicted state event (referenced by the state
191///   maps) **and** the transitive closure of their auth chains. Passing
192///   a full event map also works — extra events are harmless but waste
193///   memory. Homeservers with compressed auth-chain bitmaps can pass
194///   just the auth chain for optimal performance.
195/// - `version`: Which resolution algorithm to use.
196///
197/// # Returns
198///
199/// The resolved `SharedState<Id>` — the single deterministic state.
200///
201/// # Panics
202///
203/// Panics if `state_maps` is empty, or if a conflicted event ID from
204/// the state maps is not found in `event_context`.
205///
206/// [`resolve_iterative_sort`]: crate::resolve::iterative::resolve_iterative_sort
207#[must_use]
208pub fn resolve_state_maps<Id, C, S>(
209    state_maps: &[SharedState<Id>],
210    event_context: &HashMap<Id, LeanEvent<Id, C>, S>,
211    version: StateResVersion,
212) -> SharedState<Id>
213where
214    Id: EventId,
215    C: EventContent + Clone,
216    S: core::hash::BuildHasher,
217{
218    resolve_state_maps_generic(
219        state_maps,
220        event_context,
221        version,
222        &alloc::string::String::new(),
223    )
224    .0
225}

227fn resolve_state_maps_generic<Id, C, S, K>(
228    state_maps: &[SharedState<Id, K>],
229    event_context: &HashMap<Id, LeanEvent<Id, C, K>, S>,
230    version: StateResVersion,
231    empty_key: &K,
232) -> (SharedState<Id, K>, crate::FastSet<(EventType, K)>)
233where
234    Id: EventId,
235    C: EventContent + Clone,
236    S: core::hash::BuildHasher,
237    K: StateKey,
238    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
239{
240    assert!(
241        !state_maps.is_empty(),
242        "resolve_state_maps requires at least one state map"
243    );
244
245    // Fast path: all maps identical. Check pointer identity first (O(1) per
246    // comparison, O(N) total): sibling forks derived from a common ancestor
247    // often share the same `imbl::OrdMap` root, so a full structural `==`
248    // comparison is wasted. Generalized to N maps -- not just the 2-map
249    // case -- since `ptr_eq` is cheap enough that checking every map against
250    // the first costs nothing extra when it hits, and only degrades to the
251    // structural comparison below when maps have diverged.
252    let first = &state_maps[0];
253    if state_maps[1..].iter().all(|m| first.ptr_eq(m)) {
254        return (first.clone(), crate::FastSet::default());
255    }
256    let all_identical = state_maps[1..].iter().all(|m| m == first);
257    if all_identical {
258        return (first.clone(), crate::FastSet::default());
259    }
260
261    // Partition into unconflicted / conflicted
262    let (unconflicted_state, conflicted_ids) =
263        partition_state_maps(state_maps.iter().map(AsRef::as_ref), state_maps.len());
264
265    // Build the conflicted events map from the event context.
266    // Panic if a conflicted event is missing — event_context must contain all
267    // events referenced by the state maps.
268    let mut conflicted_events: HashMap<Id, LeanEvent<Id, C, K>> = HashMap::new();
269    for id in &conflicted_ids {
270        let ev = event_context
271            .get(id)
272            .unwrap_or_else(|| panic!("event_context missing conflicted event {id}"));
273        conflicted_events.insert(id.clone(), ev.clone());
274    }
275
276    // Genuinely conflicted keys: captured *before* the MSC4297 subgraph
277    // supplement below adds more events to `conflicted_events` purely as
278    // auth-chain context. Threading this narrower set into resolution
279    // prevents a subgraph-context event from clobbering a key every input
280    // state map actually agreed on (see resolve_iterative_sort_with_all_caches).
281    // TODO(perf): duplicates an EventType::from() conversion the power/
282    // non-power phases redo per event — see the identical TODO on
283    // resolve::iterative::derive_all_conflicted_keys.
284    let conflicted_keys =
285        crate::resolve::iterative::derive_all_conflicted_keys(&conflicted_events, empty_key);
286
287    // For V2.1+ rooms, compute the conflicted subgraph (MSC4297).
288    if matches!(version, StateResVersion::V2_1 | StateResVersion::V2_1_1) {
289        let subgraph = compute_v2_1_subgraph(event_context.iter(), &conflicted_ids);
290        for (id, _) in subgraph {
291            conflicted_events.entry(id.clone()).or_insert_with(|| {
292                event_context
293                    .get(&id)
294                    .expect("subgraph event must be in event_context")
295                    .clone()
296            });
297        }
298    }
299
300    let mut pl_cache: HashMap<Id, i64, hashbrown::DefaultHashBuilder> = HashMap::default();
301    let resolved = crate::resolve::iterative::resolve_iterative_sort_with_all_caches(
302        &unconflicted_state,
303        &conflicted_events,
304        event_context,
305        None,
306        version,
307        &mut pl_cache,
308        &mut crate::FastMap::default(),
309        &conflicted_keys,
310        empty_key,
311    );
312    (resolved, conflicted_keys)
313}

315/// Computes the exact state mutations needed to update the selected input
316/// state map to the result of resolving `state_maps`.
317///
318/// This is the incremental handoff API for callers that already persist the
319/// predecessor state elsewhere (for example, in a HAMT). The baseline is
320/// explicit by index and is never inferred from map allocation identity or
321/// iteration order.
322///
323/// The returned [`StateDiff`] contains final state changes only:
324/// additions, removals, and replacements. It does not contain the
325/// per-event auth-processing trace exposed by
326/// [`crate::resolve_iterative_sort_with_deltas`].
327///
328/// This function does not change the existing [`resolve_state_maps`] return
329/// type. Rezzy still materializes the resolved `SharedState` internally in
330/// order to perform resolution; this API avoids making callers retain or
331/// rebuild that full map at the persistence boundary. Because the baseline is
332/// an input fork, the final diff only examines the genuinely conflicted
333/// state-key slots.
334///
335/// `empty_key` is passed explicitly because [`StateKey`] does not require a
336/// default value. It is used for state events whose `state_key` is absent.
337///
338/// # Panics
339///
340/// Panics if `base_state_index` is outside `state_maps` or if a conflicted
341/// event is missing from `event_context`.
342///
343#[must_use]
344pub fn resolve_state_maps_diff<Id, C, S, K>(
345    base_state_index: usize,
346    state_maps: &[SharedState<Id, K>],
347    event_context: &HashMap<Id, LeanEvent<Id, C, K>, S>,
348    version: StateResVersion,
349    empty_key: &K,
350) -> StateDiff<Id, K>
351where
352    Id: EventId,
353    C: EventContent + Clone,
354    S: core::hash::BuildHasher,
355    K: StateKey,
356    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
357{
358    let base_state = state_maps
359        .get(base_state_index)
360        .unwrap_or_else(|| panic!("base_state_index out of range: {base_state_index}"));
361    let (resolved, conflicted_keys) =
362        resolve_state_maps_generic(state_maps, event_context, version, empty_key);
363
364    let mut entries = Vec::new();
365    // Sort keys to emit entries in a stable, deterministic order (matching
366    // compute_state_diff's sorted key order). Without this, the diff order
367    // depends on the hash-bucket iteration order of the HashSet, which is
368    // fragile and non-reproducible.
369    let mut sorted_keys: Vec<_> = conflicted_keys.into_iter().collect();
370    sorted_keys.sort();
371    for key in sorted_keys {
372        match (base_state.get(&key), resolved.get(&key)) {
373            (None, Some(new_id)) => entries.push(StateDiffEntry::Added {
374                key,
375                event_id: new_id.clone(),
376            }),
377            (Some(old_id), None) => entries.push(StateDiffEntry::Removed {
378                key,
379                event_id: old_id.clone(),
380            }),
381            (Some(old_id), Some(new_id)) if old_id != new_id => {
382                entries.push(StateDiffEntry::Changed {
383                    key,
384                    old_event_id: old_id.clone(),
385                    new_event_id: new_id.clone(),
386                });
387            }
388            _ => {}
389        }
390    }
391    StateDiff { entries }
392}

394/// Builds a stripped auth-only event map and computes the V2.1+ conflicted
395/// subgraph (MSC4297).
396///
397/// Events in the auth DAG that lie at the intersection of backwards-reachable
398/// (ancestors) and forwards-reachable (descendants) from the conflicted set
399/// must be added to the conflicted set so the mainline sort considers them.
400///
401/// The `events` iterator provides all events to consider (e.g., `event_context`
402/// for the eager path, or `auth_context.chain(conflicted_events)` for the lazy
403/// path). The returned subgraph events must be merged into `conflicted_events`
404/// by the caller.
405#[inline(never)]
406fn compute_v2_1_subgraph<'a, Id, C, I, K>(
407    events: I,
408    conflicted_ids: &[Id],
409) -> HashMap<Id, LeanEvent<Id>>
410where
411    Id: EventId + 'a,
412    C: 'a,
413    K: StateKey + 'a,
414    I: IntoIterator<Item = (&'a Id, &'a LeanEvent<Id, C, K>)>,
415{
416    let auth_only: HashMap<Id, LeanEvent<Id>> = events
417        .into_iter()
418        .map(|(id, ev)| {
419            (
420                id.clone(),
421                LeanEvent {
422                    rejected: false,
423                    soft_fail: false,
424                    event_id: ev.event_id.clone(),
425                    event_type: ev.event_type.clone(),
426                    state_key: ev
427                        .state_key
428                        .as_ref()
429                        .map(|key| alloc::string::String::from(key.as_ref())),
430                    sender: ev.sender.clone(),
431                    auth_events: ev.auth_events.clone(),
432                    prev_events: Vec::new(),
433                    content: serde_json::Value::Null,
434                    power_level: 0,
435                    origin_server_ts: 0,
436                    depth: 0,
437                    room_id: ev.room_id.clone(),
438                },
439            )
440        })
441        .collect();
442    crate::resolve::subgraph::compute_v2_1_conflicted_subgraph(&auth_only, conflicted_ids)
443}

445/// Populate `auth_context` from a precomputed auth diff, skipping events
446/// already in `conflicted_events`.
447///
448/// Extracted as a separate `#[inline(never)]` function to ensure LLVM
449/// coverage instruments it independently of the generic caller.
450#[inline(never)]
451fn populate_auth_from_diff<Id, C>(
452    auth_diff: impl IntoIterator<Item = Id>,
453    conflicted_events: &HashMap<Id, LeanEvent<Id, C>>,
454    provider: &impl EventProvider<Id, C>,
455    auth_context: &mut HashMap<Id, LeanEvent<Id, C>>,
456) where
457    Id: EventId,
458    C: Clone,
459{
460    for aid in auth_diff {
461        if conflicted_events.contains_key(&aid) {
462            continue;
463        }
464        if let Some(ev) = provider.get_event(&aid) {
465            auth_context.insert(aid, ev.clone());
466        }
467    }
468}

470/// Insert subgraph events into `conflicted_events`, sourcing them from
471/// `auth_context`.
472///
473/// # Invariant
474///
475/// `subgraph ⊆ auth_context ∪ conflicted_events`. If `or_insert_with`
476/// fires (event not in `conflicted_events`), it **must** be in
477/// `auth_context`.
478///
479/// # Panics
480///
481/// Panics if a subgraph event is found in neither `conflicted_events`
482/// nor `auth_context` (invariant violation).
483#[inline(never)]
484fn insert_subgraph_events<Id: EventId, C: Clone>(
485    subgraph: HashMap<Id, LeanEvent<Id>>,
486    auth_context: &HashMap<Id, LeanEvent<Id, C>>,
487    conflicted_events: &mut HashMap<Id, LeanEvent<Id, C>>,
488) {
489    for (id, _) in subgraph {
490        conflicted_events.entry(id.clone()).or_insert_with(|| {
491            auth_context
492                .get(&id)
493                .unwrap_or_else(|| panic!("subgraph event {id} must be in auth_context"))
494                .clone()
495        });
496    }
497}

499/// Like [`resolve_state_maps`], but accepts an [`EventProvider`] instead of a
500/// concrete `HashMap`, enabling lazy/on-demand event loading from a database or
501/// LRU cache.
502///
503/// Instead of requiring the caller to pre-materialize the entire auth context
504/// into a `HashMap`, this function:
505///
506/// 1. Partitions state maps into unconflicted/conflicted (no events needed).
507/// 2. Fetches **only** the conflicted events via `provider.get_event(id)`.
508/// 3. BFS-walks auth chains from conflicted events to build the minimal auth
509///    context (lazy — only touches events reachable from the conflicted set).
510/// 4. For V2.1+, computes the conflicted subgraph from the lazily-built context.
511/// 5. Delegates to [`resolve_iterative_sort`](crate::resolve::iterative::resolve_iterative_sort).
512///
513/// # Performance
514///
515/// For rooms with large auth chains, this can be significantly faster than
516/// [`resolve_state_maps`] because it never loads events outside the
517/// backwards-reachable set of the conflicted events.
518///
519/// # Panics
520///
521/// Panics if `state_maps` is empty, or if a conflicted event ID from
522/// the state maps is not found via the provider.
523///
524/// [`EventProvider`]: crate::basespec::rezzy_types::EventProvider
525#[must_use]
526pub fn resolve_state_maps_lazy_with_diff<Id, C>(
527    state_maps: &[SharedState<Id>],
528    provider: &impl crate::basespec::rezzy_types::EventProvider<Id, C>,
529    precomputed_auth_diff: Option<impl IntoIterator<Item = Id>>,
530    version: StateResVersion,
531) -> SharedState<Id>
532where
533    Id: EventId,
534    C: EventContent + Clone,
535{
536    assert!(
537        !state_maps.is_empty(),
538        "resolve_state_maps_lazy requires at least one state map"
539    );
540
541    // Fast path: all maps identical
542    let first = &state_maps[0];
543    if state_maps[1..].iter().all(|m| m == first) {
544        return first.clone();
545    }
546
547    // Partition into unconflicted / conflicted (no events needed)
548    let (unconflicted_state, conflicted_ids) =
549        partition_state_maps(state_maps.iter().map(AsRef::as_ref), state_maps.len());
550
551    // Lazily fetch conflicted events
552    let mut conflicted_events: HashMap<Id, LeanEvent<Id, C>> = HashMap::new();
553    for id in &conflicted_ids {
554        let ev = provider
555            .get_event(id)
556            .unwrap_or_else(|| panic!("provider missing conflicted event {id}"));
557        conflicted_events.insert(id.clone(), ev.clone());
558    }
559
560    // Genuinely conflicted keys, captured before the MSC4297 subgraph
561    // supplement below adds more (auth-chain-context-only) events — see
562    // resolve_state_maps's identical comment for why this matters.
563    let empty_key = alloc::string::String::new();
564    let conflicted_keys =
565        crate::resolve::iterative::derive_all_conflicted_keys(&conflicted_events, &empty_key);
566
567    // Lazily BFS auth chains from conflicted events to build minimal auth context
568    let mut auth_context: HashMap<Id, LeanEvent<Id, C>> = HashMap::new();
569
570    if let Some(auth_diff) = precomputed_auth_diff {
571        // Fast path: we already know exactly which events are in the auth diff.
572        populate_auth_from_diff(auth_diff, &conflicted_events, provider, &mut auth_context);
573    } else {
574        // Slow path: dynamically discover the auth diff via BFS
575        let mut auth_queue: alloc::collections::VecDeque<Id> = alloc::collections::VecDeque::new();
576        for ev in conflicted_events.values() {
577            for aid in &ev.auth_events {
578                if !conflicted_events.contains_key(aid) {
579                    auth_queue.push_back(aid.clone());
580                }
581            }
582        }
583        while let Some(aid) = auth_queue.pop_front() {
584            if auth_context.contains_key(&aid) || conflicted_events.contains_key(&aid) {
585                continue;
586            }
587            if let Some(ev) = provider.get_event(&aid) {
588                auth_context.insert(aid, ev.clone());
589                for parent_id in &ev.auth_events {
590                    if !auth_context.contains_key(parent_id)
591                        && !conflicted_events.contains_key(parent_id)
592                    {
593                        auth_queue.push_back(parent_id.clone());
594                    }
595                }
596            }
597        }
598    }
599
600    // V2.1+ subgraph computation from the lazily-built context
601    if matches!(version, StateResVersion::V2_1 | StateResVersion::V2_1_1) {
602        let subgraph = compute_v2_1_subgraph(
603            auth_context.iter().chain(conflicted_events.iter()),
604            &conflicted_ids,
605        );
606        insert_subgraph_events(subgraph, &auth_context, &mut conflicted_events);
607    }

610    // `route_msc4297_ancestral_power_events` (and `compute_local_auth`) can
611    // BFS through them — matching the non-lazy `resolve_state_maps` where
612    // `event_context` includes all events.
613    for (id, ev) in &conflicted_events {
614        auth_context.entry(id.clone()).or_insert_with(|| ev.clone());
615    }
616
617    let mut pl_cache: HashMap<Id, i64, hashbrown::DefaultHashBuilder> = HashMap::default();
618    crate::resolve::iterative::resolve_iterative_sort_with_all_caches(
619        &unconflicted_state,
620        &conflicted_events,
621        &auth_context,
622        None,
623        version,
624        &mut pl_cache,
625        &mut crate::FastMap::default(),
626        &conflicted_keys,
627        &empty_key,
628    )
629}

669    /// Parse a JSONL string into a `HashMap<String, LeanEvent>` keyed by `event_id`.
670    fn parse_jsonl_map(input: &str) -> HashMap<alloc::string::String, LeanEvent> {
671        let mut map = HashMap::new();
672        for line in input.lines() {
673            let line = line.trim();
674            if line.is_empty() || line.starts_with("//") {
675                continue;
676            }
677            let ev: LeanEvent = serde_json::from_str(line)
678                .unwrap_or_else(|e| panic!("bad JSONL: {e}\n  line: {line}"));
679            map.insert(ev.event_id.clone(), ev);
680        }
681        map
