24/// Dynamically fetches the sender's power level by inspecting the event's immediate `auth_events`.
25/// Recursive traversal of the auth chain is avoided to prevent bypassing immediate restrictions.
26pub(crate) fn get_power_level_from_auth_chain<Id, C, E>(
27    event: &E,
28    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
29    create_ev: Option<&E>,
30    version: StateResVersion,
31) -> i64
32where
33    Id: crate::basespec::rezzy_types::EventId,
34    C: crate::basespec::rezzy_types::EventContent,
35    E: EventLike<Id = Id, Content = C>,
36{
37    let mut pl_event = None;
38
39    // Spec compliance: only check immediate auth_events (or prev_state_events for V2.2).
40    for aid in event.dag_edges(version) {
41        if let Some(aev) = auth_context.get_event(aid) {
42            if aev.event_type().as_ref() == M_ROOM_POWER_LEVELS && aev.state_key() == Some("") {
43                pl_event = Some(aev);
44                break;
45            }
46        }
47    }
48
49    // V12+ (MSC4289): creators have spec-mandated infinite power level.
50    if matches!(
51        version,
52        StateResVersion::V2_1 | StateResVersion::V2_1_1 | StateResVersion::V2_2
53    ) {
54        let is_creator = create_ev.is_some_and(|ev| {
55            ev.sender() == event.sender() || ev.content().has_additional_creator(event.sender())
56        });
57        if is_creator {
58            return MAX_POWER_LEVEL_RUST;
59        }
60    }
61
62    if let Some(pl_ev) = pl_event {
63        if let Some(pl) = pl_ev.get_user_power_level(event.sender()) {
64            return pl;
65        }
66
67        if let Some(default_pl) = pl_ev.get_users_default() {
68            return default_pl;
69        }
70        return 0; // Default if PL event exists but no users_default
71    }
72
73    // No PL event in the auth chain — fall back to the pre-computed
74    // power_level field. This handles the bootstrap PL event (which only
75    // auths against $create) and simple unit-test events.
76    event.power_level()
77}

79/// Detailed Kahn's Topological Sort algorithm for event power resolution.
80///
81/// This function performs a reverse topological sort on a set of events, placing
82/// descendants before their ancestors. It returns diagnostic details about any cycles
83/// if they are detected.
84///
85/// # Panics
86///
87/// Will panic if graph invariants are violated during topological sorting (specifically, if
88/// the in-degree map lacks an entry for a child event during the queue processing phase).
89#[allow(clippy::implicit_hasher)]
90pub fn lean_kahn_sort_with_cycle_diagnostics<Id, C, E, S1, Spl>(
91    events: &HashMap<Id, E, S1>,
92    sort_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
93    create_ev: Option<&E>,
94    version: StateResVersion,
95    pl_cache: &mut HashMap<Id, i64, Spl>,
96) -> KahnSortResult<Id>
97where
98    Id: crate::basespec::rezzy_types::EventId,
99    S1: core::hash::BuildHasher,
100    Spl: core::hash::BuildHasher,
101    C: Clone + crate::basespec::rezzy_types::EventContent,
102    E: EventLike<Id = Id, Content = C>,
103{
104    pl_cache.clear();
105
106    let mut in_degree: FastMap<&Id, usize> = FastMap::default();
107    let mut adjacency: FastMap<&Id, Vec<&Id>> = FastMap::default();
108
109    for (id, event) in events {
110        in_degree.entry(id).or_insert(0);
111        for auth in event.dag_edges(version) {
112            if let Some((auth_key, _)) = events.get_key_value(auth) {
113                // Topological sort: ancestors come BEFORE descendants.
114                // But we want a REVERSE topological sort: descendants BEFORE ancestors.
115                // So we add edges from ancestors to descendants.
116                adjacency.entry(auth_key).or_default().push(id);
117                let val = in_degree.entry(id).or_insert(0);
118                *val = val.saturating_add(1);
119            }
120        }
121    }
122
123    // Pre-compute power levels once per event to avoid redundant auth chain walks
124    // inside the hot BinaryHeap push path.
125    for (id, ev) in events {
126        if !pl_cache.contains_key(id) {
127            pl_cache.insert(
128                id.clone(),
129                get_power_level_from_auth_chain(ev, sort_context, create_ev, version),
130            );
131        }
132    }
133
134    let mut queue: BinaryHeap<SortPriority<'_, E>> = BinaryHeap::new();
135    for (id, &degree) in &in_degree {
136        if degree == 0 {
137            if let Some(event) = events.get(*id) {
138                queue.push(SortPriority {
139                    event,
140                    power_level: pl_cache.get(*id).copied().unwrap_or(0),
141                    version,
142                });
143            }
144        }
145    }
146
147    let mut result = Vec::with_capacity(events.len());
148    while let Some(priority) = queue.pop() {
149        let event = priority.event;
150
151        result.push(event.event_id().clone());
152        if let Some(neighbors) = adjacency.get(event.event_id()) {
153            for &next_id in neighbors {
154                let degree = in_degree.get_mut(next_id).unwrap();
155                *degree = degree.saturating_sub(1);
156                if *degree == 0 {
157                    let next_ev = events.get(next_id).unwrap();
158                    queue.push(SortPriority {
159                        event: next_ev,
160                        power_level: pl_cache.get(next_id).copied().unwrap_or(0),
161                        version,
162                    });
163                }
164            }
165        }
166    }
167
168    // Detect cycles: events that never reached in-degree 0.
169    if result.len() != events.len() {
170        let sorted_set: crate::FastSet<&Id> = result.iter().collect();
171        let stuck: Vec<Id> = events
172            .keys()
173            .filter(|id| !sorted_set.contains(id))
174            .cloned()
175            .collect();
176        drop(sorted_set);
177        return KahnSortResult::CycleDetected {
178            sorted: result,
179            stuck,
180        };
181    }
182
183    KahnSortResult::Ok(result)
184}

186/// A simplified implementation of Kahn's Topological Sort.
187/// Backward-compatible wrapper that falls back to standard tie-breaking on cycles.
188///
189/// # Panics
190///
191/// Will panic if graph invariants are violated (specifically, if an event returned
192/// in the cycle-breaking list of stuck nodes is missing from the input `events` map).
193// jscpd:ignore-start
194#[must_use]
195#[allow(clippy::implicit_hasher)]
196pub fn lean_kahn_sort<Id, C, E, S1, Spl>(
197    events: &HashMap<Id, E, S1>,
198    sort_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
199    create_ev: Option<&E>,
200    version: StateResVersion,
201    pl_cache: &mut HashMap<Id, i64, Spl>,
202) -> Vec<Id>
203where
204    Id: crate::basespec::rezzy_types::EventId,
205    S1: core::hash::BuildHasher,
206    Spl: core::hash::BuildHasher,
207    C: Clone + crate::basespec::rezzy_types::EventContent,
208    E: EventLike<Id = Id, Content = C>,
209{
210    // jscpd:ignore-end
211    match lean_kahn_sort_with_cycle_diagnostics(events, sort_context, create_ev, version, pl_cache)
212    {
213        KahnSortResult::Ok(sorted) => sorted,
214        KahnSortResult::CycleDetected {
215            mut sorted,
216            mut stuck,
217        } => {
218            #[cfg(feature = "std")]
219            std::eprintln!("KAHN CYCLE DETECTED! Stuck: {stuck:?}");
220            stuck.sort_by(|a, b| {
221                let ev_a = events.get(a).unwrap();
222                let ev_b = events.get(b).unwrap();
223                // Standard tie-breaking fallback (origin_server_ts ascending, then event_id ascending)
224                ev_a.origin_server_ts()
225                    .cmp(&ev_b.origin_server_ts())
226                    .then_with(|| a.cmp(b))
227            });
228            sorted.append(&mut stuck);
229            sorted
230        }
231    }
232}

234pub(crate) fn build_mainline<Id, C, E, K>(
235    resolved: &crate::state::at::SharedState<Id, K>,
236    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
237    empty_key: &K,
238    version: StateResVersion,
239) -> Vec<Id>
240where
241    Id: crate::basespec::rezzy_types::EventId,
242    C: Clone + crate::basespec::rezzy_types::EventContent,
243    E: EventLike<Id = Id, Content = C>,
244    K: Ord + Clone,
245{
246    // The hot path (compute_state_at's fork-merge loop) now threads a persistent
247    // cache through `resolve_iterative_sort_with_all_caches`, so this fresh-cache
248    // fallback only matters for one-shot callers (e.g. resolve_semilattice_fold's V2
249    // path, which calls build_mainline exactly once per resolution).
250    build_mainline_with_cache(
251        resolved,
252        auth_context,
253        &mut FastMap::default(),
254        empty_key,
255        version,
256    )
257}

259/// Like [`build_mainline`], but populates a `pl_parent_cache` mapping each
260/// event ID to its nearest `m.room.power_levels` ancestor in the auth chain.
261///
262/// Subsequent calls sharing the same cache skip BFS entirely for events
263/// already resolved, turning the mainline walk from `O(M × B)` (M = mainline
264/// length, B = auth chain breadth) to `O(M)` on cache hits.
265pub(crate) fn build_mainline_with_cache<Id, C, E, K>(
266    resolved: &crate::state::at::SharedState<Id, K>,
267    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
268    pl_parent_cache: &mut FastMap<Id, Option<Id>>,
269    empty_key: &K,
270    version: StateResVersion,
271) -> Vec<Id>
272where
273    Id: crate::basespec::rezzy_types::EventId,
274    C: Clone + crate::basespec::rezzy_types::EventContent,
275    E: EventLike<Id = Id, Content = C>,
276    K: Ord + Clone,
277{
278    let mut mainline = Vec::new();
279    let mut seen_in_mainline = crate::FastSet::default();
280    let pl_key = (
281        crate::basespec::event_types::EventType::from(M_ROOM_POWER_LEVELS),
282        empty_key.clone(),
283    );
284    let mut current = resolved.get(&pl_key).cloned();
285
286    while let Some(eid) = current {
287        if !seen_in_mainline.insert(eid.clone()) {
288            #[cfg(feature = "std")]
289            std::eprintln!("REZZY_WARN: MAINLINE CYCLE DETECTED!");
290            break; // Cycle detected in the power-levels mainline!
291        }
292        mainline.push(eid.clone());
293
294        // Check cache first
295        if let Some(cached) = pl_parent_cache.get(&eid) {
296            current = None;
297            current.clone_from(cached);
298            continue;
299        }
300
301        // BFS to find the nearest PL ancestor
302        let mut found = None;
303        if let Some(ev) = auth_context.get_event(&eid) {
304            let mut queue: VecDeque<&Id> = ev.dag_edges(version).iter().collect();
305            let mut visited = crate::FastSet::default();
306            while let Some(q_id) = queue.pop_front() {
307                if !visited.insert(q_id) {
308                    continue;
309                }
310                if let Some(auth_ev) = auth_context.get_event(q_id) {
311                    if auth_ev.event_type().as_ref() == M_ROOM_POWER_LEVELS {
312                        found = Some(q_id.clone());
313                        break;
314                    }
315                    for aid in auth_ev.dag_edges(version) {
316                        queue.push_back(aid);
317                    }
318                }
319            }
320        }
321
322        pl_parent_cache.insert(eid, found.clone());
323        current = found;
324    }
325
326    mainline
327}

329/// Precompute the closest mainline position for every target event reachable via
330/// `auth_events` using a stack-safe O(V+E) iterative DFS upward search.
331///
332/// This entirely avoids `O(N)` cloning of the DAG, and prevents stack overflow
333/// by using an explicit stack to simulate recursion while memoizing distances.
334pub(crate) fn compute_closest_mainline_positions<Id, C, E>(
335    events: &mut [&E],
336    mainline: &[Id],
337    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
338    version: StateResVersion,
339) -> HashMap<Id, usize>
340where
341    Id: crate::basespec::rezzy_types::EventId,
342    C: Clone + crate::basespec::rezzy_types::EventContent,
343    E: EventLike<Id = Id, Content = C>,
344{
345    let mut memo = HashMap::new();
346
347    // Pre-populate mainline events with their exact index position
348    for (pos, id) in mainline.iter().enumerate() {
349        memo.insert(id.clone(), pos);
350    }
351
352    let mut stack = Vec::new();
353
354    for ev in events.iter() {
355        stack.push(ev.event_id());
356
357        while let Some(&top) = stack.last() {
358            if let Some(&val) = memo.get(top) {
359                if val != usize::MAX - 1 {
360                    stack.pop();
361                    continue;
362                }
363            } else {
364                memo.insert(top.clone(), usize::MAX - 1);
365            }
366
367            let mut all_children_done = true;
368            let mut min_pos = usize::MAX;
369
370            if let Some(node) = auth_context.get_event(top) {
371                for aid in node.dag_edges(version) {
372                    if let Some(&child_pos) = memo.get(aid) {
373                        if child_pos != usize::MAX - 1 {
374                            min_pos = min_pos.min(child_pos);
375                        }
376                    } else {
377                        all_children_done = false;
378                        stack.push(aid);
379                    }
380                }
381            }
382
383            if all_children_done {
384                // Clamp "no path found" to mainline.len() so callers
385                // don't see a raw usize::MAX leaking into comparisons.
386                let resolved_pos = if min_pos == usize::MAX {
387                    mainline.len()
388                } else {
389                    min_pos
390                };
391                memo.insert(top.clone(), resolved_pos);
392                stack.pop();
393            }
394        }
395    }
396
397    memo
398}

400/// Sorts non-power events by mainline ordering per the Matrix spec.
401///
402/// The mainline is the chain of `m.room.power_levels` events reachable from
403/// the currently resolved PL state. Each event's "mainline position" is the
404/// closest PL event in its auth chain. The sort order is:
405///
406/// 1. **Mainline position** descending (farther = worse = applied first).
407/// 2. **`origin_server_ts`** ascending (earlier = applied first, later wins).
408/// 3. **`event_id`** ascending (lexicographic tie-break).
409///
410/// Events closer to the current power-levels event are applied **last**
411/// and therefore win for same-key conflicts (last-write-wins).
412pub fn mainline_sort<Id, C, E>(
413    events: &mut [&E],
414    mainline: &[Id],
415    auth_context: &impl crate::basespec::rezzy_types::EventProvider<Id, C, E>,
416    version: StateResVersion,
417) where
418    Id: crate::basespec::rezzy_types::EventId,
419    C: Clone + crate::basespec::rezzy_types::EventContent,
420    E: EventLike<Id = Id, Content = C>,
421{
422    // O(V+E) iterative DFS to find the closest mainline index for all non-power events
423    let dist = compute_closest_mainline_positions(events, mainline, auth_context, version);
424
425    // Schwartzian transform: decorate each event with its precomputed mainline
426    // position so the comparator performs zero hash lookups. This cuts the
427    // `dist` hash lookups from O(N log N) (two per comparison) down to exactly
428    // N. The comparator order is identical to the pre-decoration version.
429    let mut decorated: Vec<(&E, usize)> = events
430        .iter()
431        .map(|&ev| (ev, *dist.get(ev.event_id()).unwrap_or(&mainline.len())))
432        .collect();
433
434    // Larger mainline position = farther from current PL = worse = comes first
435    // (so it gets overwritten by closer events via last-write-wins)
436    decorated.sort_by(|(a, pos_a), (b, pos_b)| match pos_b.cmp(pos_a) {
437        Ordering::Equal => {
438            // Earlier timestamp comes first (later wins via last-write)
439            match a.origin_server_ts().cmp(&b.origin_server_ts()) {
440                Ordering::Equal => a.event_id().cmp(b.event_id()),
441                ord => ord,
442            }
443        }
444        ord => ord,
445    });
446
447    for (i, (ev, _)) in decorated.into_iter().enumerate() {
448        events[i] = ev;
449    }
450}

710    /// call populates the cache for each PL event, and the second call hits the
711    /// cache early, skipping the BFS entirely.
712    #[test]
713    fn test_build_mainline_cache_hit() {
714        // Chain: PL2 → (auth) → PL1 → (auth) → PL0
715        let pl0 = LeanEvent::<String> {
716            event_id: "PL0".into(),
717            event_type: "m.room.power_levels".into(),
718            auth_events: alloc::vec![],
719            ..Default::default()
720        };
721        let pl1 = LeanEvent::<String> {
722            event_id: "PL1".into(),
723            event_type: "m.room.power_levels".into(),
724            auth_events: alloc::vec!["PL0".into()],
725            ..Default::default()
726        };
727        let pl2 = LeanEvent::<String> {
728            event_id: "PL2".into(),
729            event_type: "m.room.power_levels".into(),
730            auth_events: alloc::vec!["PL1".into()],
731            ..Default::default()
732        };
733
734        let mut ctx = HashMap::new();
735        ctx.insert("PL0".into(), pl0);
736        ctx.insert("PL1".into(), pl1);
737        ctx.insert("PL2".into(), pl2);
738
739        let mut resolved = imbl::OrdMap::new();
740        resolved.insert(("m.room.power_levels".into(), String::new()), "PL2".into());
741
742        // First call: populates cache for PL2 → Some(PL1), PL1 → Some(PL0), PL0 → None
743        let mut cache = FastMap::default();
744        let ml1 = build_mainline_with_cache(
745            &resolved,
746            &ctx,
747            &mut cache,
748            &String::new(),
749            StateResVersion::V2,
750        );
751        assert_eq!(ml1, alloc::vec!["PL2", "PL1", "PL0"]);
752        assert_eq!(cache.len(), 3, "all 3 PL events must be cached");
753
754        // Second call: hits cache immediately for PL2 → skips BFS
755        let ml2 = build_mainline_with_cache(
756            &resolved,
757            &ctx,
758            &mut cache,
759            &String::new(),
760            StateResVersion::V2,
761        );
762        assert_eq!(ml2, ml1, "cached mainline must match original");
763    }
