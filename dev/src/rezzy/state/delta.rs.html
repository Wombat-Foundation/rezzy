15//! State delta compression — efficient incremental state storage/checkpoint chains.
16//!
17//! Instead of storing the full `state_map` at *every* event, homeservers can
18//! store a base snapshot and a chain of deltas.
19//! This module provides the primitives for computing and applying those deltas.

26/// Which phase of state resolution produced a delta.
27#[derive(Debug, Clone, Copy, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
28pub enum ResolvePhase {
29    /// Power events: `m.room.create`, `m.room.power_levels`, `m.room.join_rules`,
30    /// bans, and kicks. Sorted by reverse topological order (Kahn's algorithm).
31    Power,
32    /// Non-power events: everything else. Sorted by mainline proximity.
33    NonPower,
34}

36/// A per-event record of what changed in the resolved state during resolution.
37///
38/// [`resolve_iterative_sort_with_deltas`](crate::resolve_iterative_sort_with_deltas) emits one of
39/// these for every conflicted event that is auth-checked, regardless of whether
40/// it was accepted or rejected.
41#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
42pub struct ResolutionDelta<Id: crate::basespec::rezzy_types::EventId = String, K = String> {
43    /// The event that was auth-checked.
44    pub event_id: Id,
45    /// Whether the event passed the iterative auth check.
46    pub accepted: bool,
47    /// The `(event_type, state_key)` slot this event targets.
48    pub key: (crate::basespec::event_types::EventType, K),
49    /// The event ID that was previously in this slot (if any).
50    /// `None` if the slot was empty before this event.
51    pub replaced: Option<Id>,
52    /// Whether this was processed in the power phase or non-power phase.
53    pub phase: ResolvePhase,
54}

56/// A single state delta entry — an addition, modification, or deletion.
57#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
58pub struct StateDelta<Id: crate::basespec::rezzy_types::EventId = String> {
59    /// The event type (e.g. `"m.room.member"`).
60    pub event_type: String,
61    /// The state key (e.g. `"@alice:example.com"` or `""`).
62    pub state_key: String,
63    /// The new event ID, or `None` if this key was deleted.
64    pub event_id: Option<Id>,
65}

67/// Computes the delta between a parent state and the current state.
68///
69/// Returns a list of [`StateDelta`] entries representing:
70/// - **Additions**: keys present in `current` but not in `parent`.
71/// - **Modifications**: keys present in both but with different event IDs.
72/// - **Deletions**: keys present in `parent` but not in `current`.
73///
74/// If the two states are identical, returns an empty `Vec`.
75#[must_use]
76pub fn compute_state_delta<Id: crate::basespec::rezzy_types::EventId>(
77    parent: &crate::state::at::SharedState<Id, String>,
78    current: &crate::state::at::SharedState<Id, String>,
79) -> Vec<StateDelta<Id>> {
80    let mut deltas = Vec::new();
81
82    // Additions and modifications
83    for ((event_type, state_key), event_id) in current {
84        match parent.get(&(event_type.clone(), state_key.clone())) {
85            Some(parent_event_id) if parent_event_id == event_id => {}
86            _ => {
87                deltas.push(StateDelta {
88                    event_type: event_type.to_string(),
89                    state_key: state_key.clone(),
90                    event_id: Some(event_id.clone()),
91                });
92            }
93        }
94    }
95
96    // Deletions
97    for key in parent.keys() {
98        if !current.contains_key(key) {
99            deltas.push(StateDelta {
100                event_type: key.0.to_string(),
101                state_key: key.1.clone(),
102                event_id: None,
103            });
104        }
105    }
106
107    deltas
108}

110/// Applies a list of deltas to a base state, producing the reconstructed state.
111///
112/// - Entries with `event_id = Some(id)` are inserted/overwritten.
113/// - Entries with `event_id = None` are removed from the base.
114#[must_use]
115pub fn apply_state_delta<Id: crate::basespec::rezzy_types::EventId>(
116    base: &crate::state::at::SharedState<Id>,
117    deltas: &[StateDelta<Id>],
118) -> crate::state::at::SharedState<Id> {
119    let mut result = base.clone();
120    for delta in deltas {
121        let key = (
122            crate::basespec::event_types::EventType::from(delta.event_type.as_str()),
123            delta.state_key.clone(),
124        );
125        if let Some(ref event_id) = delta.event_id {
126            result.insert(key, event_id.clone());
127        } else {
128            result.remove(&key);
129        }
130    }
131    result
132}

134/// Decodes a 64-character ASCII hex string into a 32-byte array.
135///
136/// Returns `Err` with a static message if the input is not exactly 64 ASCII
137/// hex characters. This rejects non-ASCII UTF-8 up front, preventing panics
138/// from slicing inside multibyte character boundaries.
139fn decode_hex_32(s: &str) -> Result<[u8; 32], &'static str> {
140    if s.len() != 64 {
141        return Err("expected 64-character hex string");
142    }
143    if !s.is_ascii() {
144        return Err("hex string contains non-ASCII characters");
145    }
146    let mut bytes = [0u8; 32];
147    for (i, byte) in bytes.iter_mut().enumerate() {
148        let start = i.saturating_mul(2);
149        let end = start.saturating_add(2);
150        // SAFETY: s.is_ascii() guarantees every byte is a single-byte character,
151        // so slicing at any byte offset is always on a character boundary.
152        let byte_str = &s[start..end];
153        *byte = u8::from_str_radix(byte_str, 16).map_err(|_| "invalid hex character")?;
154    }
155    Ok(bytes)
156}

158pub mod hex_serde {
159    use serde::{Deserialize, Deserializer, Serializer};
160
161    /// # Errors
162    /// Returns an error if the hex string fails to serialize.
163    pub fn serialize<S: Serializer>(bytes: &[u8; 32], serializer: S) -> Result<S::Ok, S::Error> {
164        let mut hex = alloc::string::String::with_capacity(64);
165        for b in bytes {
166            use core::fmt::Write;
167            write!(hex, "{b:02x}").expect("String formatting is infallible");
168        }
169        serializer.serialize_str(&hex)
170    }
171
172    /// # Errors
173    /// Returns an error if the string is not exactly 64 ASCII hex characters.
174    pub fn deserialize<'de, D: Deserializer<'de>>(deserializer: D) -> Result<[u8; 32], D::Error> {
175        let s = alloc::string::String::deserialize(deserializer)?;
176        super::decode_hex_32(&s).map_err(serde::de::Error::custom)
177    }
178}

204                .map_err(serde::de::Error::custom),
205            None => Ok(None),
206        }
207    }

210/// Maximum number of delta hops before a full snapshot is inserted (default: 100,
211/// configurable: true).
212///
213/// Matches Synapse's `MAX_STATE_DELTA_HOPS`. When a delta chain would exceed
214/// this length, [`compute_compacted_delta_chain_from_resolved`] inserts a full
215/// base snapshot instead of another delta, bounding reconstruction cost to at
216/// most this many hops.
217pub const MAX_DELTA_CHAIN_HOPS: usize = 100;
218
219/// A checkpoint that may be either a delta from a parent or a full snapshot.
220///
221/// When chain compaction triggers (every [`MAX_DELTA_CHAIN_HOPS`] events),
222/// the checkpoint stores the full state map as `snapshot` instead of a delta.
223/// Readers walk backwards from any checkpoint, applying deltas, until they
224/// hit a snapshot.
225#[derive(Debug, Clone, PartialEq, Eq, serde::Serialize, serde::Deserialize)]
226pub struct CompactedCheckpoint<Id: crate::basespec::rezzy_types::EventId = String> {
227    /// 256-bit hash of the state map at this point.
228    #[serde(with = "hex_serde")]
229    pub state_hash: [u8; 32],
230    /// Hash of the parent checkpoint (if any).
231    #[serde(with = "hex_serde_opt")]
232    pub parent_hash: Option<[u8; 32]>,
233    /// The event ID that produced this checkpoint.
234    pub event_id: Id,
235    /// Deltas from the parent state. Empty when `snapshot` is `Some`.
236    pub deltas: Vec<StateDelta<Id>>,
237    /// Full state snapshot, present every [`MAX_DELTA_CHAIN_HOPS`] checkpoints.
238    /// When this is `Some`, `deltas` is empty and reconstruction starts here.
239    pub snapshot: Option<crate::state::at::SharedState<Id>>,
240}

242/// Builds a compacted delta chain from pre-resolved `(event_id, state_map)` pairs,
243/// inserting full snapshots every `max_hops` events to bound reconstruction cost.
244///
245/// This is the recommended API for homeserver state storage. The caller supplies
246/// pre-resolved states (from [`compute_state_at_batch`](crate::compute_state_at_batch)
247/// / [`resolve_iterative_sort`](crate::resolve_iterative_sort)), and this function handles only the
248/// storage compression concern.
249///
250/// A custom `max_hops` can be provided; pass `None` to use the default
251/// [`MAX_DELTA_CHAIN_HOPS`] (100).
252#[must_use]
253pub fn compute_compacted_delta_chain_from_resolved<Id: crate::basespec::rezzy_types::EventId>(
254    resolved_states: impl IntoIterator<Item = (Id, crate::state::at::SharedState<Id>)>,
255    max_hops: Option<usize>,
256) -> Vec<CompactedCheckpoint<Id>> {
257    let max_hops = max_hops.unwrap_or(MAX_DELTA_CHAIN_HOPS);
258    let mut checkpoints = Vec::new();
259    let mut prev_state: Option<crate::state::at::SharedState<Id>> = None;
260    let mut prev_hash: Option<[u8; 32]> = None;
261    let mut hops_since_snapshot: usize = 0;
262
263    for (event_id, state) in resolved_states {
264        let state_hash = compute_state_hash(&state);
265        let current_hops = hops_since_snapshot.saturating_add(1);
266
267        let (deltas, snapshot, recorded_hops) = match prev_state {
268            Some(ref base) if current_hops < max_hops => {
269                let deltas = compute_state_delta(base, &state);
270                (deltas, None, current_hops)
271            }
272            _ => {
273                // First event or chain exceeded max_hops — insert a
274                // full snapshot to reset the chain.
275                (Vec::new(), Some(state.clone()), 0)
276            }
277        };
278
279        checkpoints.push(CompactedCheckpoint {
280            state_hash,
281            parent_hash: prev_hash,
282            event_id,
283            deltas,
284            snapshot,
285        });
286
287        prev_hash = Some(state_hash);
288        prev_state = Some(state);
289        hops_since_snapshot = recorded_hops;
290    }
291
292    checkpoints
293}

295/// Reconstruct the full state map by looking up a checkpoint by event ID.
296///
297/// This is a convenience wrapper around [`reconstruct_state_at`] that finds
298/// the checkpoint index for the given `event_id` and reconstructs the state.
299///
300/// Returns `None` if the event ID is not found in the checkpoint chain or
301/// the chain is broken.
302#[must_use]
303pub fn reconstruct_state_at_by_event_id<Id: crate::basespec::rezzy_types::EventId>(
304    checkpoints: &[CompactedCheckpoint<Id>],
305    event_id: &Id,
306) -> Option<crate::state::at::SharedState<Id>> {
307    let idx = checkpoints.iter().position(|cp| &cp.event_id == event_id)?;
308    reconstruct_state_at(checkpoints, idx)
309}

311/// Reconstructs the full state map from a stored delta chain.
312///
313/// Walks `checkpoints` **backwards** from `target_index`, applying deltas
314/// until a snapshot is found. Returns the reconstructed state at that index.
315///
316/// # Arguments
317///
318/// * `checkpoints` — a slice of [`CompactedCheckpoint`]s in chronological order.
319/// * `target_index` — the index of the checkpoint to reconstruct.
320///
321/// # Returns
322///
323/// `Some(state_map)` if a snapshot base was found, `None` if the chain is
324/// broken (no snapshot ancestor exists).
325#[must_use]
326pub fn reconstruct_state_at<Id: crate::basespec::rezzy_types::EventId>(
327    checkpoints: &[CompactedCheckpoint<Id>],
328    target_index: usize,
329) -> Option<crate::state::at::SharedState<Id>> {
330    if target_index >= checkpoints.len() {
331        return None;
332    }
333
334    // Collect deltas by walking backwards until we hit a snapshot
335    let mut delta_stack: Vec<&[StateDelta<Id>]> = Vec::new();
336    let mut current_idx = target_index;
337
338    loop {
339        let cp = &checkpoints[current_idx];
340
341        if let Some(ref snapshot) = cp.snapshot {
342            // Found a base snapshot — apply accumulated deltas forward
343            let mut state = snapshot.clone();
344            while let Some(deltas) = delta_stack.pop() {
345                for delta in deltas {
346                    let key = (
347                        crate::basespec::event_types::EventType::from(delta.event_type.as_str()),
348                        delta.state_key.clone(),
349                    );
350                    if let Some(ref event_id) = delta.event_id {
351                        state.insert(key, event_id.clone());
352                    } else {
353                        state.remove(&key);
354                    }
355                }
356            }
357            return Some(state);
358        }
359
360        // No snapshot — push deltas and walk to parent
361        delta_stack.push(&cp.deltas);
362
363        let parent_hash = cp.parent_hash.as_ref()?;
364        // Prefer immediate predecessor; fall back to scanning earlier checkpoints.
365        if let Some(prev) = current_idx
366            .checked_sub(1)
367            .filter(|&pi| checkpoints[pi].state_hash == *parent_hash)
368        {
369            current_idx = prev;
370        } else {
371            current_idx = checkpoints[..current_idx]
372                .iter()
373                .rposition(|cp| cp.state_hash == *parent_hash)?;
374        }
375    }
376}

378/// Reconstructs the full state map at multiple checkpoint indices in one pass.
379///
380/// Instead of calling [`reconstruct_state_at`] N times (which re-walks shared
381/// ancestors), this function finds the nearest snapshot before the earliest
382/// target and walks forward once, capturing state at each requested index.
383///
384/// **Complexity**: `O(span)` where `span = max(targets) - snapshot_base`,
385/// versus `O(N * span)` for N individual calls.
386///
387/// # Arguments
388///
389/// * `checkpoints` — a slice of [`CompactedCheckpoint`]s in chronological order.
390/// * `target_indices` — the indices to reconstruct (duplicates and out-of-bounds
391///   are silently ignored).
392///
393/// # Returns
394///
395/// A `imbl::OrdMap<usize, state_map>` keyed by the requested indices. Missing
396/// entries indicate broken chains or out-of-bounds indices.
397///
398/// # Panics
399///
400/// Panics if an internal index increment overflows `usize` (should never
401/// happen with valid checkpoint data).
402#[must_use]
403pub fn reconstruct_state_batch<Id: crate::basespec::rezzy_types::EventId>(
404    checkpoints: &[CompactedCheckpoint<Id>],
405    target_indices: &[usize],
406) -> imbl::OrdMap<usize, crate::state::at::SharedState<Id>> {
407    use crate::HashMap;
408
409    let mut sorted_targets: Vec<usize> = target_indices
410        .iter()
411        .copied()
412        .filter(|&i| i < checkpoints.len())
413        .collect();
414    sorted_targets.sort_unstable();
415    sorted_targets.dedup();
416
417    if sorted_targets.is_empty() {
418        return imbl::OrdMap::new();
419    }
420
421    // Backward-only parent lookup: prefer immediate predecessor, fall back to
422    // scanning earlier checkpoints. This avoids the HashMap shadowing bug where
423    // duplicate hashes overwrite earlier entries.
424    let find_parent_idx = |idx: usize, parent_hash: &[u8; 32]| -> Option<usize> {
425        idx.checked_sub(1)
426            .filter(|&pi| checkpoints[pi].state_hash == *parent_hash)
427            .or_else(|| {
428                checkpoints[..idx]
429                    .iter()
430                    .rposition(|cp| cp.state_hash == *parent_hash)
431            })
432    };
433
434    // Walk backwards from all targets to find all required ancestors
435    let mut required_indices = alloc::collections::BTreeSet::new();
436    let mut queue: Vec<usize> = sorted_targets.clone();
437    while let Some(idx) = queue.pop() {
438        if required_indices.insert(idx) && checkpoints[idx].snapshot.is_none() {
439            if let Some(parent_hash) = &checkpoints[idx].parent_hash {
440                if let Some(p_idx) = find_parent_idx(idx, parent_hash) {
441                    queue.push(p_idx);
442                }
443            }
444        }
445    }
446
447    let mut known_states = HashMap::new();
448    let mut results = imbl::OrdMap::new();
449
450    // Iterate forward only through the required indices
451    for idx in required_indices {
452        let cp = &checkpoints[idx];
453        let state = if let Some(ref snapshot) = cp.snapshot {
454            snapshot.clone()
455        } else if let Some(parent_hash) = &cp.parent_hash {
456            if let Some(p_idx) = find_parent_idx(idx, parent_hash) {
457                if let Some(parent_state) = known_states.get(&p_idx) {
458                    apply_state_delta(parent_state, &cp.deltas)
459                } else {
460                    #[cfg(feature = "std")]
461                    std::eprintln!(
462                        "state_delta: checkpoint {idx} parent {p_idx} not reconstructed"
463                    );
464                    continue; // BROKEN CHAIN: Parent was required but not reconstructed (gap in chain)
465                }
466            } else {
467                #[cfg(feature = "std")]
468                std::eprintln!(
469                    "state_delta: checkpoint {idx} has no earlier match for parent_hash {parent_hash:?}"
470                );
471                continue; // BROKEN CHAIN: No earlier checkpoint matches parent_hash
472            }
473        } else {
474            #[cfg(feature = "std")]
475            std::eprintln!(
476                "state_delta: checkpoint {idx} is an orphan (no snapshot, no parent_hash)"
477            );
478            continue; // BROKEN CHAIN: No snapshot and no parent_hash — orphan checkpoint
479        };
480
481        if sorted_targets.binary_search(&idx).is_ok() {
482            results.insert(idx, state.clone());
483        }
484        known_states.insert(idx, state);
485    }
486
487    results
488}

564    #[test]
565    fn test_compaction_inserts_snapshots() {
566        // NOTE: This test creates `O(N^2)` state items as it builds the DAG.
567        // `LtHash` computes a SHAKE256 XOF expansion per item, making large `N` values
568        // extremely slow in debug builds. We keep `N=35` to ensure fast tests.
569        // Build 35 pre-resolved states — should trigger snapshots at 0, 10, 20, 30
570        let states: ResolvedStates = (1..=35)
571            .map(|i| {
572                let mut state = StateMap::new();
573                for j in 1..=i {
574                    state.insert(
575                        (
576                            "m.room.member".into(),
577                            alloc::format!("@user_{j}:example.com"),
578                        ),
579                        alloc::format!("${j}"),
580                    );
581                }
582                (alloc::format!("${i}"), state)
583            })
584            .collect();
585
586        let checkpoints = compute_compacted_delta_chain_from_resolved(states, Some(10));
587        assert_eq!(checkpoints.len(), 35);
588
589        // First event always gets a snapshot (no parent)
590        assert!(
591            checkpoints[0].snapshot.is_some(),
592            "first event must have snapshot"
593        );
594
595        // Count snapshots
596        let snapshot_count = checkpoints
597            .iter()
598            .filter(|cp| cp.snapshot.is_some())
599            .count();
600        assert!(
601            snapshot_count >= 3,
602            "35 events with max_hops=10 should produce at least 3 snapshots, got {snapshot_count}"
603        );
604
605        // No delta chain should exceed max_hops
606        for (i, cp) in checkpoints.iter().enumerate() {
607            if cp.snapshot.is_some() {
608                continue;
609            }
610            // Walk back to find the nearest snapshot
611            let mut hops = 0;
612            let mut idx = i;
613            while checkpoints[idx].snapshot.is_none() {
614                hops += 1;
615                // Find parent by hash
616                let parent_hash = checkpoints[idx].parent_hash.as_ref().unwrap();
617                idx = checkpoints
618                    .iter()
619                    .position(|c| c.state_hash == *parent_hash)
620                    .unwrap();
621            }
622            assert!(
623                hops < 10,
624                "delta chain at index {i} has {hops} hops, exceeds max 10"

979    /// verifies that index 3 (whose parent is `HASH_A`) correctly resolves to
980    /// index 0, not silently fails because index 4 shadowed the lookup.
981    #[test]
982    fn test_batch_hash_shadowing_regression() {
983        let hash_a = [0xAA; 32];
984        let hash_b = [0xBB; 32];

1017                snapshot: None,

1071    /// predecessor — e.g. after a fork/merge or gap in the chain.
