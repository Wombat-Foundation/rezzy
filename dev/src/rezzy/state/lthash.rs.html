15//! Homomorphic state hashing via `LtHash16` (MSC4500).
16//!
17//! `LtHash` (Lattice Hash) is based on the homomorphic hashing paradigm first
18//! introduced by Bellare and Micciancio in their 1997 paper *"A New Paradigm
19//! for Collision-Free Hashing: Incrementality at Reduced Cost"*. This specific
20//! 2048-byte instantiation (using 1024 16-bit integers and wrapping addition)
21//! is modeled after the industry-standard implementation in Meta's Folly library
22//! (`folly::crypto::LtHash`).
23//!
24//! Each `(event_type, state_key, event_id)` entry is expanded to 1024 16-bit
25//! integers via SHAKE256 XOF (NIST FIPS 202). The state hash is the wrapping
26//! addition of all these vectors. This provides:
27//!
28//! - **O(1) incremental updates**: insert = `hash + expanded`,
29//!   remove = `hash - expanded`.
30//! - **Order independence**: addition is commutative + associative.
31//! - **Cryptographic security**: hard to find set collisions (SVP).
32
33/// A 2048-byte homomorphic state hash using `LtHash`.
34///
35/// `LtHash` (Lattice Hash) is based on the homomorphic hashing paradigm first introduced
36/// by Bellare and Micciancio in their 1997 paper *"A New Paradigm for Collision-Free
37/// Hashing: Incrementality at Reduced Cost"*. This specific 2048-byte instantiation
38/// (using 1024 16-bit integers and wrapping addition) is modeled after the industry-standard
39/// implementation in Meta's Folly library (`folly::crypto::LtHash`).
40///
41/// Each `(event_type, state_key, event_id)` entry is expanded to 1024 16-bit
42/// integers via SHAKE256 XOF (NIST FIPS 202). The state hash is the wrapping
43/// addition of all these vectors. This provides:
44///
45/// - **O(1) incremental updates**: insert = `hash + expanded`,
46///   remove = `hash - expanded`.
47/// - **Order independence**: addition is commutative + associative.
48/// - **Cryptographic security**: hard to find set collisions (SVP).
49///
50/// `StateUpdate::New/Unchanged` now carry `&LtHash` (borrowed, zero-copy); callers
51/// that need to retain the hash (e.g. across a thread channel) copy it explicitly.
52#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
53pub struct LtHash(pub [u16; 1024]);
54
55impl Default for LtHash {
56    fn default() -> Self {
57        Self::ZERO
58    }
59}

61struct HashWriter<'a, H> {
62    hasher: &'a mut H,
63}
64
65impl<H> core::fmt::Write for HashWriter<'_, H>
66where
67    H: sha3::digest::Update,
68{
69    #[inline]
70    fn write_str(&mut self, s: &str) -> core::fmt::Result {
71        self.hasher.update(s.as_bytes());
72        Ok(())
73    }
74}

76/// Truncate a string to fit within a `u16` length prefix (65535 bytes).
77///
78/// Valid Matrix events are capped at 64KiB total, so real event types and state keys
79/// can never reach this limit. Truncation only applies to malformed/adversarial input.
80#[inline]
81fn truncate_to_u16_limit(s: &str) -> (&str, u16) {
82    let limit = usize::from(u16::MAX);
83    let s_len = s.len();
84    if s_len > limit {
85        let mut end = limit;
86        while !s.is_char_boundary(end) {
87            end = end.saturating_sub(1);
88        }
89        (&s[..end], u16::try_from(end).unwrap())
90    } else {
91        (s, u16::try_from(s_len).unwrap())
92    }
93}

95impl LtHash {
96    /// The identity element (empty state).
97    pub const ZERO: Self = Self([0u16; 1024]);
98
99    /// MSC4500 v1 domain separation tag for the primary state accumulator.
100    const DST: &'static [u8] = b"msc4500:lthash16:v1";
101
102    /// Compute the 2048-byte SHAKE256 expansion for a single state entry.
103    ///
104    /// Input encoding (MSC4500 §1): `len(type) || type || len(state_key) || state_key || event_id`
105    /// where each `len()` is an unsigned 16-bit little-endian byte count.
106    ///
107    /// Expansion (MSC4500 §2): `SHAKE256(tag || element, 2048)`
108    ///
109    /// # Performance & Validation
110    ///
111    /// Full cryptographic and syntactic validation of the Matrix Event ID (e.g., verifying
112    /// length, prefix, character sets, or room-version-specific syntax) is intentionally
113    /// **not** performed within this function for performance reasons and to allow flexible
114    /// event ID formats across legacy/modern room versions. Any syntactic validation of
115    /// event IDs must be enforced by the caller at the application ingestion boundary if desired.
116    #[must_use]
117    fn seed(event_type: &str, state_key: &str, event_id: &dyn core::fmt::Display) -> Self {
118        use core::fmt::Write;
119        use sha3::digest::{ExtendableOutput, Update};
120
121        let (event_type, type_len) = truncate_to_u16_limit(event_type);
122        let (state_key, sk_len) = truncate_to_u16_limit(state_key);
123
124        let mut xof = sha3::Shake256::default();
125        xof.update(Self::DST);
126        xof.update(&type_len.to_le_bytes());
127        xof.update(event_type.as_bytes());
128        xof.update(&sk_len.to_le_bytes());
129        xof.update(state_key.as_bytes());
130
131        let mut writer = HashWriter { hasher: &mut xof };
132        write!(writer, "{event_id}").expect("failed to write event_id to hasher");
133
134        let mut buf = [0u8; 2048];
135        xof.finalize_xof_into(&mut buf);
136
137        let mut out = [0u16; 1024];
138        for (i, chunk) in buf.chunks_exact(2).enumerate() {
139            out[i] = u16::from_le_bytes([chunk[0], chunk[1]]);
140        }
141        Self(out)
142    }
143
144    /// Add a seed into the hash (insert).
145    #[inline]
146    fn add_seed(&mut self, seed: &Self) {
147        // Process in 8-lane chunks to assist SIMD auto-vectorization
148        for (a, b) in self.0.chunks_exact_mut(8).zip(seed.0.chunks_exact(8)) {
149            a[0] = a[0].wrapping_add(b[0]);
150            a[1] = a[1].wrapping_add(b[1]);
151            a[2] = a[2].wrapping_add(b[2]);
152            a[3] = a[3].wrapping_add(b[3]);
153            a[4] = a[4].wrapping_add(b[4]);
154            a[5] = a[5].wrapping_add(b[5]);
155            a[6] = a[6].wrapping_add(b[6]);
156            a[7] = a[7].wrapping_add(b[7]);
157        }
158    }
159
160    /// Subtract a seed from the hash (remove).
161    #[inline]
162    fn sub_seed(&mut self, seed: &Self) {
163        // Process in 8-lane chunks to assist SIMD auto-vectorization
164        for (a, b) in self.0.chunks_exact_mut(8).zip(seed.0.chunks_exact(8)) {
165            a[0] = a[0].wrapping_sub(b[0]);
166            a[1] = a[1].wrapping_sub(b[1]);
167            a[2] = a[2].wrapping_sub(b[2]);
168            a[3] = a[3].wrapping_sub(b[3]);
169            a[4] = a[4].wrapping_sub(b[4]);
170            a[5] = a[5].wrapping_sub(b[5]);
171            a[6] = a[6].wrapping_sub(b[6]);
172            a[7] = a[7].wrapping_sub(b[7]);
173        }
174    }

176    /// Record a state entry being inserted.
177    pub fn insert(
178        &mut self,
179        event_type: &str,
180        state_key: &str,
181        event_id: &(impl core::fmt::Display + ?Sized),
182    ) {
183        let s = Self::seed(event_type, state_key, &event_id);
184        self.add_seed(&s);
185    }

187    /// Record a state entry being removed.
188    pub fn remove(
189        &mut self,
190        event_type: &str,
191        state_key: &str,
192        event_id: &(impl core::fmt::Display + ?Sized),
193    ) {
194        let s = Self::seed(event_type, state_key, &event_id);
195        self.sub_seed(&s);
196    }

198    /// Record a state entry being replaced (old → new).
199    pub fn replace(
200        &mut self,
201        event_type: &str,
202        state_key: &str,
203        old_event_id: &(impl core::fmt::Display + ?Sized),
204        new_event_id: &(impl core::fmt::Display + ?Sized),
205    ) {
206        let old = Self::seed(event_type, state_key, &old_event_id);
207        let new = Self::seed(event_type, state_key, &new_event_id);
208        self.sub_seed(&old);
209        self.add_seed(&new);
210    }
211
212    /// Record a replacement that is required to stay on the same `(event_type, state_key)`.
213    ///
214    /// This is a defensive wrapper for callers that want an explicit invariant check before
215    /// performing the remove/add pair.
216    ///
217    /// # Panics
218    ///
219    /// Panics if `old_event_type != new_event_type` or `old_state_key != new_state_key`.
220    pub fn replace_checked(
221        &mut self,
222        old_event_type: &str,
223        old_state_key: &str,
224        old_event_id: &(impl core::fmt::Display + ?Sized),
225        new_event_type: &str,
226        new_state_key: &str,
227        new_event_id: &(impl core::fmt::Display + ?Sized),
228    ) {
229        assert!(
230            old_event_type == new_event_type && old_state_key == new_state_key,
231            "mismatched replacement key: ({old_event_type}, {old_state_key}) -> ({new_event_type}, {new_state_key})",
232        );
233        self.sub_seed(&Self::seed(old_event_type, old_state_key, &old_event_id));
234        self.add_seed(&Self::seed(new_event_type, new_state_key, &new_event_id));
235    }

237    /// Compute the full hash from a state map (non-incremental).
238    #[must_use]
239    pub fn from_state<Id, K>(state: &crate::state::at::SharedState<Id, K>) -> Self
240    where
241        Id: crate::basespec::rezzy_types::EventId,
242        K: Ord + AsRef<str>,
243    {
244        let mut hash = Self::ZERO;
245        for ((event_type, state_key), event_id) in state {
246            let s = Self::seed(event_type.as_str(), state_key.as_ref(), event_id);
247            hash.add_seed(&s);
248        }
249        hash
250    }

252    /// Finalize into the 32-byte wire digest per MSC4500 §6:
253    /// `BLAKE2b-256(S)`, where `S` is the 2048-byte lattice.
254    #[must_use]
255    pub fn digest(&self) -> [u8; 32] {
256        use blake2::digest::consts::U32;
257        use blake2::{Blake2b, Digest};
258        let mut hasher = Blake2b::<U32>::new();
259        let mut bytes = [0u8; 2048];
260        for (i, val) in self.0.iter().enumerate() {
261            let le = val.to_le_bytes();
262            let idx = i.wrapping_mul(2);
263            bytes[idx] = le[0];
264            bytes[idx.wrapping_add(1)] = le[1];
265        }
266        hasher.update(&bytes[..]);
267        hasher.finalize().into()
268    }

271/// A homomorphic digest of the redaction overlay associated with a resolved
272/// state.  The overlay is deliberately a separate accumulator from
273/// [`LtHash`]: it does not describe another state snapshot.  Each entry names
274/// one selected state event that is effectively redacted at the DAG point.
275///
276/// Callers should insert only selected state events that are effectively
277/// redacted by authorized causal redactions at the state point being
278/// described.  An empty overlay is a known empty overlay; `None` in
279/// [`StateDigest`] means that the sender did not compute one.
280#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
281pub struct RedactionOverlay(pub [u16; 1024]);
282
283impl Default for RedactionOverlay {
284    fn default() -> Self {
285        Self::ZERO
286    }
287}
288
289impl RedactionOverlay {
290    /// The identity element (no effectively redacted selected events).
291    pub const ZERO: Self = Self([0u16; 1024]);
292
293    const DST: &'static [u8] = b"msc4500:redactions:v1";
294
295    // TODO: The overlay duplicates `LtHash`'s tuple encoding, lattice updates,
296    // and digest serialization instead of sharing them. Future changes to
297    // framing, truncation, or byte order can update one path and silently make
298    // the two digest implementations inconsistent; factor the common
299    // accumulator/encoding logic and inject the domain-separation tag.
300    fn seed(
301        event_type: &str,
302        state_key: &str,
303        event_id: &(impl core::fmt::Display + ?Sized),
304    ) -> Self {
305        use core::fmt::Write;
306        use sha3::digest::{ExtendableOutput, Update};
307
308        let mut xof = sha3::Shake256::default();
309        xof.update(Self::DST);
310        for value in [event_type, state_key] {
311            let (value, len) = truncate_to_u16_limit(value);
312            xof.update(&len.to_le_bytes());
313            xof.update(value.as_bytes());
314        }
315        // The event ID is appended raw, matching the primary MSC4500 element
316        // encoding. It is already self-delimiting under Matrix event-ID
317        // syntax and must not acquire a second length prefix here.
318        let mut writer = HashWriter { hasher: &mut xof };
319        write!(writer, "{event_id}").expect("failed to write event_id to hasher");
320
321        let mut bytes = [0u8; 2048];
322        xof.finalize_xof_into(&mut bytes);
323        let mut out = [0u16; 1024];
324        for (i, chunk) in bytes.chunks_exact(2).enumerate() {
325            out[i] = u16::from_le_bytes([chunk[0], chunk[1]]);
326        }
327        Self(out)
328    }

330    /// Adds one effectively redacted selected state event to the overlay.
331    ///
332    /// The caller must maintain set semantics: inserting the same tuple more
333    /// than once intentionally changes the lattice, just as it does for the
334    /// primary accumulator.
335    pub fn insert(
336        &mut self,
337        event_type: &str,
338        state_key: &str,
339        event_id: &(impl core::fmt::Display + ?Sized),
340    ) {
341        let seed = Self::seed(event_type, state_key, event_id);
342        for (left, right) in self.0.chunks_exact_mut(8).zip(seed.0.chunks_exact(8)) {
343            for i in 0..8 {
344                left[i] = left[i].wrapping_add(right[i]);
345            }
346        }
347    }
348
349    /// Removes one overlay entry previously inserted with [`Self::insert`].
350    /// Callers must not remove an entry that is absent from the authoritative
351    /// overlay set.
352    pub fn remove(
353        &mut self,
354        event_type: &str,
355        state_key: &str,
356        event_id: &(impl core::fmt::Display + ?Sized),
357    ) {
358        let seed = Self::seed(event_type, state_key, event_id);
359        for (left, right) in self.0.chunks_exact_mut(8).zip(seed.0.chunks_exact(8)) {
360            for i in 0..8 {
361                left[i] = left[i].wrapping_sub(right[i]);
362            }
363        }
364    }

366    /// Collapses the overlay lattice to its 32-byte MSC4500 wire digest.
367    #[must_use]
368    pub fn digest(&self) -> [u8; 32] {
369        use blake2::digest::consts::U32;
370        use blake2::{Blake2b, Digest};
371        let mut hasher = Blake2b::<U32>::new();
372        let mut bytes = [0u8; 2048];
373        for (i, value) in self.0.iter().enumerate() {
374            let pair = value.to_le_bytes();
375            let index = i.wrapping_mul(2);
376            bytes[index] = pair[0];
377            bytes[index.wrapping_add(1)] = pair[1];
378        }
379        hasher.update(bytes);
380        hasher.finalize().into()
381    }
382}

384/// The MSC4500 state digest for one DAG point: the primary resolved-state
385/// digest and, when supported, its causal redaction overlay digest.
386///
387/// MSC4500 carries these values as `before` and `after` fields around a state
388/// transition (alongside `redactions_before` and `redactions_after`). This
389/// type represents one such point; [`StateDigestTransition`] represents the
390/// pair. `overlay` is optional for wire compatibility and must never be
391/// interpreted as agreement when absent.
392#[derive(Debug, Clone, Copy, PartialEq, Eq)]
393pub struct StateDigest {
394    pub primary: [u8; 32],
395    pub overlay: Option<[u8; 32]>,
396}
397
398/// The before/after digest pair carried for one MSC4500 state transition.
399#[derive(Debug, Clone, Copy, PartialEq, Eq)]
400pub struct StateDigestTransition {
401    pub before: StateDigest,
402    pub after: StateDigest,
403}
404
405/// Result of comparing two state digest advertisements.
406#[derive(Debug, Clone, Copy, PartialEq, Eq)]
407pub enum DigestAgreement {
408    /// The selected `(type, state_key, event_id)` maps differ.
409    PrimaryMismatch,
410    /// Primary maps agree and both overlay digests agree.
411    FullySynchronized,
412    /// Primary maps agree but causal redaction overlays differ.
413    OverlayMismatch,
414    /// Primary maps agree, but at least one side omitted its overlay.
415    OverlayUnknown,
416}
417
418impl StateDigest {
419    /// Compares primary state first, then treats the overlay as a diagnostic.
420    #[must_use]
421    pub fn compare(self, remote: Self) -> DigestAgreement {
422        if self.primary == remote.primary {
423            match (self.overlay, remote.overlay) {
424                (Some(left), Some(right)) if left == right => DigestAgreement::FullySynchronized,
425                (Some(_), Some(_)) => DigestAgreement::OverlayMismatch,
426                _ => DigestAgreement::OverlayUnknown,
427            }
428        } else {
429            DigestAgreement::PrimaryMismatch
430        }
431    }
432}

434/// Computes a deterministic 256-bit `LtHash` fingerprint of a
435/// state map, returned as a 32-byte array.
436///
437/// This is a convenience wrapper around
438/// [`LtHash::from_state`]. Each
439/// `(event_type, state_key, event_id)` entry is expanded via
440/// SHAKE256 to a 2048-byte seed, and the state hash is the
441/// wrapping addition of all seeds — making it order-independent and
442/// incrementally updatable.
443#[must_use]
444pub fn compute_state_hash<Id: crate::basespec::rezzy_types::EventId, K: Ord + AsRef<str>>(
445    state: &crate::state::at::SharedState<Id, K>,
446) -> [u8; 32] {
447    LtHash::from_state(state).digest()
448}

699    /// Validate against the official MSC4500 test vectors.
700    ///
701    /// These vectors use SHAKE256 expansion (with domain separation tag
702    /// `msc4500:lthash16:v1`) and BLAKE2b-256 collapse.
703    #[test]
704    fn test_msc4500_vectors() {
705        fn hex(bytes: &[u8]) -> alloc::string::String {
706            use core::fmt::Write;
707            bytes.iter().fold(
708                alloc::string::String::with_capacity(bytes.len() * 2),
709                |mut s, b| {
710                    write!(s, "{b:02x}").unwrap();
711                    s
712                },
713            )
714        }
715
716        // Digest vectors are published by MSC4500 in unpadded base64url (the
717        // wire form); lattice/expansion prefixes remain hex in the spec.
718        fn b64u(bytes: &[u8]) -> alloc::string::String {
719            use base64::{engine::general_purpose::URL_SAFE_NO_PAD, Engine as _};
720            URL_SAFE_NO_PAD.encode(bytes)
721        }
722
723        // --- Empty state (n=0) ---
724        let s0 = LtHash::ZERO;
725        assert_eq!(
726            b64u(&s0.digest()),
727            "IAgj5RWLN3TBG1xhhQradi-CZBRKm-vsPrrFoq3eZ7g"
728        );
729
730        // --- Scenario 1: Add element 1 ---
731        let seed1 = LtHash::seed("m.room.member", "@alice:example.com", &"$event_1");
732        let exp1_bytes: Vec<u8> = seed1.0[..8].iter().flat_map(|v| v.to_le_bytes()).collect();
733        assert_eq!(hex(&exp1_bytes), "dbcadc58c85d7be0efca00e478a66697");
734
735        let mut s1 = s0;
736        s1.add_seed(&seed1);
737        let s1_bytes: Vec<u8> = s1.0[..8].iter().flat_map(|v| v.to_le_bytes()).collect();
738        assert_eq!(hex(&s1_bytes), "dbcadc58c85d7be0efca00e478a66697");
739        assert_eq!(
740            b64u(&s1.digest()),
741            "bX7ccIPg0lyRZyBYO_UZs5nC4iVitD62L6cJfL2iAiU"
742        );
743
744        // --- Scenario 2: Remove element 1 ---
745        let mut s_back = s1;
746        s_back.sub_seed(&seed1);
747        assert_eq!(s_back, LtHash::ZERO);
748        assert_eq!(
749            b64u(&s_back.digest()),
750            "IAgj5RWLN3TBG1xhhQradi-CZBRKm-vsPrrFoq3eZ7g"
751        );
752
753        // --- Scenario 3: Add element 2 ---
754        let seed2 = LtHash::seed("m.room.name", "", &"$event_2");
755        let exp2_bytes: Vec<u8> = seed2.0[..8].iter().flat_map(|v| v.to_le_bytes()).collect();
756        assert_eq!(hex(&exp2_bytes), "118e0b32fac730c01f1351378389793a");
757
758        let mut s2 = s1;
759        s2.add_seed(&seed2);
760        let s2_bytes: Vec<u8> = s2.0[..8].iter().flat_map(|v| v.to_le_bytes()).collect();
761        assert_eq!(hex(&s2_bytes), "ec58e78ac225aba00ede511bfb2fdfd1");
762        assert_eq!(
763            b64u(&s2.digest()),
764            "uPdh4wkYWs0awGqFQmf3ieHSoFoMXFPwZmdqrwSPhkM"
765        );
766
767        // --- Scenario 4: Replace element 1 with element 3 ---
768        let seed3 = LtHash::seed("m.room.member", "@alice:example.com", &"$event_3");
769        let exp3_bytes: Vec<u8> = seed3.0[..8].iter().flat_map(|v| v.to_le_bytes()).collect();
770        assert_eq!(hex(&exp3_bytes), "4f026432409d32757f83fd088659c6c6");
771
772        let mut s3 = s2;
773        s3.sub_seed(&seed1);
774        s3.add_seed(&seed3);
775        let s3_bytes: Vec<u8> = s3.0[..8].iter().flat_map(|v| v.to_le_bytes()).collect();
776        assert_eq!(hex(&s3_bytes), "60906f643a6562359e964e4009e33f01");
777        assert_eq!(
778            b64u(&s3.digest()),
779            "eqev6DfKxlhX6RocDu97tQghpBYRRQ9TfbGXiiQiSZA"
780        );
781    }

785        let seed = LtHash::seed("m.room.message", "", &"$1");

811    #[test]
812    fn test_lthash_differential_random_mutations() {
813        struct Lcg(u32);
814        impl Lcg {
815            fn next(&mut self) -> u32 {
816                self.0 = self.0.wrapping_mul(1_664_525).wrapping_add(1_013_904_223);
817                self.0
818            }
819            fn next_range(&mut self, min: u32, max: u32) -> u32 {
820                min + (self.next() % (max - min + 1))
821            }
822        }
823
824        let mut rng = Lcg(12345);
825        let mut state = StateMap::new();
826        let mut running_hash = LtHash::ZERO;
827
828        // Populate state with some initial keys to work with
829        let mut keys = Vec::new();
830        for i in 0..15 {
831            let key = (
832                crate::basespec::event_types::EventType::from(alloc::format!("type_{i}")),
833                alloc::format!("state_key_{i}"),
834            );
835            let val = alloc::format!("$initial_event_{i}");
836            state.insert(key.clone(), val.clone());
837            running_hash.insert(key.0.as_str(), &key.1, &val);
838            keys.push(key);
839        }
840
841        assert_eq!(running_hash, LtHash::from_state(&state));
842
843        // Let's do 200 mutations
844        for step in 0..200 {
845            let op = rng.next_range(0, 2); // 0 = insert/overwrite, 1 = remove, 2 = replace
846            if op == 0 || keys.is_empty() {
847                // Insert a new key or overwrite an existing one
848                let key = if !keys.is_empty() && rng.next_range(0, 1) == 1 {
849                    // Overwrite an existing key
850                    let keys_len = u32::try_from(keys.len()).unwrap();
851                    let idx = rng.next_range(0, keys_len - 1) as usize;
852                    keys[idx].clone()
853                } else {
854                    // Create a new key
855                    let id = rng.next();
856                    let key = (
857                        crate::basespec::event_types::EventType::from(alloc::format!("type_{id}")),
858                        alloc::format!("state_key_{id}"),
859                    );
860                    keys.push(key.clone());
861                    key
862                };
863
864                let new_val = alloc::format!("$event_{}", rng.next());
865
866                // If it existed, we do a replace under the hood, or insert/remove.
867                if let Some(old_val) = state.get(&key) {
868                    running_hash.replace(key.0.as_str(), &key.1, old_val, &new_val);
869                } else {
870                    running_hash.insert(key.0.as_str(), &key.1, &new_val);
871                }
872                state.insert(key, new_val);
873            } else if op == 1 && !keys.is_empty() {
874                // Remove an existing key
875                let keys_len = u32::try_from(keys.len()).unwrap();
876                let idx = rng.next_range(0, keys_len - 1) as usize;
877                let key = keys.swap_remove(idx);
878                if let Some(val) = state.remove(&key) {
879                    running_hash.remove(key.0.as_str(), &key.1, &val);
880                }
881            } else {
882                // Replace via explicit .replace API
883                let keys_len = u32::try_from(keys.len()).unwrap();
884                let idx = rng.next_range(0, keys_len - 1) as usize;
885                let key = &keys[idx];
886                if let Some(old_val) = state.get(key).cloned() {
887                    let new_val = alloc::format!("$replaced_{}", rng.next());
888                    running_hash.replace(key.0.as_str(), &key.1, &old_val, &new_val);
889                    state.insert(key.clone(), new_val);
890                }
891            }
892
893            // Verify parity at every single step!
894            assert_eq!(
895                running_hash,
896                LtHash::from_state(&state),
897                "Hash mismatch at step {step}"
898            );
899        }
900    }

951
952        // Avalanche Effect: seed1 and seed2 event_id differ by only 1 character ('1' vs '2').
953        let mut different_elements = 0;

957            }
958        }

962            "Avalanche effect failed: only {different_elements} / 1024 elements differed"
