15//! Incremental state computation — room state at arbitrary DAG positions.
16//!
17//! This module computes the resolved room state *after* any given event in the
18//! DAG, without requiring external state snapshots. It walks the `prev_events`
19//! graph backwards, builds the state at each ancestor, and merges fork points.
20//!
21//! Key optimizations:
22//!
23//! - `O(1)` structural sharing: persistent state is represented via
24//!   [`imbl::OrdMap`](`SharedState`). Fork branches are created and merged
25//!   incrementally with zero allocations for identical shared subtrees.
26//! - **Batch mode:** computes state at multiple targets in a single topological
27//!   pass, amortizing the graph traversal cost.

37/// An entry in the local auth cache, pairing an event with its discovery depth.
38///
39/// The `depth` field tracks how many hops through `auth_events` it took to
40/// reach this event. When the same `(type, state_key)` is found at multiple
41/// depths, the shallowest (closest) entry wins.
42#[derive(Debug, Clone)]
43pub struct LocalAuthEntry<Id, C = serde_json::Value, K = String> {
44    /// The auth event itself.
45    pub event: LeanEvent<Id, C, K>,
46    /// Number of auth-chain hops from the original event to this one.
47    pub auth_depth: usize,
48}
49
50/// Inner type for the local auth cache to satisfy clippy's `type_complexity` lint.
51pub type LocalAuthCacheMap<Id, C, K> = BTreeMap<(EventType, K), LocalAuthEntry<Id, C, K>>;
52
53/// Memoization cache for local auth context computation.
54///
55/// Maps `event_id -> BTreeMap<(type, state_key) -> LocalAuthEntry>`, allowing
56/// the local auth context to be computed once and reused for all events that
57/// share auth chain prefixes.
58///
59/// This cache tracks which `StateResVersion` its entries were computed for.
60/// Callers must clear the cache when reusing it with a different `StateResVersion`
61/// (higher-level helpers like `resolve_iterative_sort_with_cache*` do this automatically).
62pub struct LocalAuthCache<Id = String, C = serde_json::Value, K = String> {
63    pub version: StateResVersion,
64    pub map: crate::HashMap<Id, LocalAuthCacheMap<Id, C, K>>,
65}

71        Self {
72            version,
73            map: crate::HashMap::default(),
74        }

80    pub(crate) auth_context: &'a HashMap<Id, LeanEvent<Id, C, K>, S1>,
81    pub(crate) sort_set: &'a HashMap<Id, LeanEvent<Id, C, K>, S2>,
82    pub(crate) local_auth: BTreeMap<(EventType, K), LeanEvent<Id, C, K>>,
83    pub(crate) create_ev: Option<&'a LeanEvent<Id, C, K>>,

85    pub(crate) is_power_phase: bool,
86    pub(crate) candidate_event_type: &'a str,

89impl<
90        Id: crate::basespec::rezzy_types::EventId,
91        C: crate::basespec::rezzy_types::EventContent,
92        S1: core::hash::BuildHasher,
93        S2: core::hash::BuildHasher,
94        K,
95    > crate::auth::StateProvider<Id, C, LeanEvent<Id, C, K>> for OverlayState<'_, Id, C, S1, S2, K>
96where
97    K: Ord + Clone + AsRef<str>,
98    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
99{
100    /// Returns the resolved event or a limited local-auth fallback for the query.
101    fn get_event(&self, event_type: &str, state_key: &str) -> Option<&LeanEvent<Id, C, K>> {
102        use crate::basespec::event_types::{M_ROOM_JOIN_RULES, M_ROOM_MEMBER, M_ROOM_POWER_LEVELS};
103
104        let query: &dyn crate::auth::StateKeyDyn = &(event_type, state_key);
105
106        // V2.1+ (MSC4297): required auth keys come from the (resolved) state in
107        // ALL phases — power phase included. The prior PL-only power-phase
108        // restriction and the V2.1.1 ban/kick-only member supplement were both
109        // non-spec "causal-domination" scaffolding; a banned sender's later
110        // power event must be rejected against the resolved ban.
111
112        // Check consensus resolved state
113        let resolved_ev = self.resolved.get(query).and_then(|eid| {
114            self.auth_context
115                .get(eid)
116                .or_else(|| self.sort_set.get(eid))
117        });
118
119        if let Some(ev) = resolved_ev {
120            return Some(ev);
121        }
122
123        if let Some(ev) = self.local_auth.get(query) {
124            // Under Matrix State Resolution, during the power phase, a required auth event in the conflicted set
125            // can ONLY be used if it has been successfully authorized and resolved
126            // (i.e. is present in the resolved state).
127            let is_required_type =
128                event_type == M_ROOM_POWER_LEVELS || event_type == M_ROOM_JOIN_RULES;
129
130            // Gate the power-phase fallback behind V2.1.1+ only: per
131            // `StateResVersion::has_ban_evasion_hardening`'s documentation,
132            // this is rezzy-internal hardening V2.1.1 adds over stock V2.1,
133            // not something V2.1 also does. Stock V2.1 falls through to the
134            // unconditional `Some(ev)` below. Go through the shared method
135            // (not a local `matches!` copy) so this can't silently drift
136            // from `resolve::iterative`'s `is_sender_banned` gate again.
137            if self.is_power_phase && self.version.has_ban_evasion_hardening() && is_required_type {
138                if self.resolved.contains_key(query) {
139                    // A resolved ID that cannot be found in either backing
140                    // collection is not safe to replace with local auth.
141                    return None;
142                }
143                // No resolved event was found above. Under V2.1+, allow the
144                // local auth event only while resolving a power/required event.
145                let candidate_is_power = self.candidate_event_type == M_ROOM_POWER_LEVELS
146                    || self.candidate_event_type == M_ROOM_JOIN_RULES
147                    || self.candidate_event_type == M_ROOM_MEMBER;
148                if candidate_is_power {
149                    Some(ev)
150                } else {
151                    None
152                }
153            } else {
154                Some(ev)
155            }
156        } else {
157            // Fallback for create
158            if event_type == crate::basespec::event_types::M_ROOM_CREATE
159                && state_key == crate::basespec::event_types::M_EMPTY_STATE_KEY
160            {
161                return self.create_ev;
162            }
163            None
164        }
165    }
166}

168/// Evaluates whether an event passes authentication checks given a resolved state map,
169/// delegating to the core `crate::auth::check_auth` logic via a temporary `OverlayState` view.
170///
171/// Authenticates an event against the current resolved state and an optional local auth context.
172/// Ensures the event complies with the Matrix spec rules for its given type.
173///
174/// NOTE: In V2.1/MSC4297, progressive state starts empty. The first event's sender membership
175/// check must use its own `auth_events` (via `local_auth` / `OverlayState` fallback), not the
176/// empty state. This is critical for competing bans where both senders need membership validation.
177#[allow(clippy::too_many_arguments)]
178pub(crate) fn iterative_auth_ok<Id, C, S1, S2, K>(
179    ev: &LeanEvent<Id, C, K>,
180    resolved: &crate::state::at::SharedState<Id, K>,
181    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
182    sort_set: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
183    local_auth: BTreeMap<(EventType, K), LeanEvent<Id, C, K>>,
184    cached_create: Option<&LeanEvent<Id, C, K>>,
185    version: StateResVersion,
186    is_power_phase: bool,
187) -> bool
188where
189    Id: crate::basespec::rezzy_types::EventId,
190    S1: core::hash::BuildHasher,
191    S2: core::hash::BuildHasher,
192    C: crate::basespec::rezzy_types::EventContent,
193    K: crate::basespec::rezzy_types::StateKey,
194    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
195{
196    // Rejected events must never be admitted into state (spec rooms/v9). Soft-failed
197    // events, however, participate in state resolution as normal (spec server-server-api
198    // "Soft failure"): they are auth-checked like any other event and admitted if they
199    // pass. Blanket-rejecting them here would diverge from Synapse/spec, which re-auth-checks
200    // them during `_iterative_auth_checks`.
201    if ev.rejected {
202        return false;
203    }
204
205    let overlay = OverlayState {
206        resolved,
207        auth_context,
208        sort_set,
209        local_auth,
210        create_ev: cached_create,
211        version,
212        is_power_phase,
213        candidate_event_type: &ev.event_type,
214    };
215
216    crate::auth::check_auth(ev, &overlay, version, None).is_ok()
217}

219/// Merges an event into a local auth map if it is an auth event (e.g. power levels, join rules).
220/// Ensures that newer auth events replace older ones during chain traversal.
221pub(crate) fn update_local_auth<Id: Clone + Ord, C: Clone, K: Clone + Ord>(
222    local_auth: &mut BTreeMap<(EventType, K), LocalAuthEntry<Id, C, K>>,
223    aev: &LeanEvent<Id, C, K>,
224    depth: usize,
225) {
226    let Some(sk) = &aev.state_key else {
227        return;
228    };
229    let key = (EventType::from(aev.event_type.as_str()), sk.clone());
230    match local_auth.entry(key) {
231        alloc::collections::btree_map::Entry::Vacant(e) => {
232            e.insert(LocalAuthEntry {
233                event: aev.clone(),
234                auth_depth: depth,
235            });
236        }
237        alloc::collections::btree_map::Entry::Occupied(mut e) => {
238            if depth < e.get().auth_depth {
239                e.insert(LocalAuthEntry {
240                    event: aev.clone(),
241                    auth_depth: depth,
242                });
243            }
244        }
245    }
246}

248/// Resolves the auth chain context incrementally and stores it in the shared cache.
249pub(crate) fn compute_local_auth<Id, C, S1, S2, K>(
250    event: &LeanEvent<Id, C, K>,
251    auth_context: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
252    conflicted_events: &HashMap<Id, LeanEvent<Id, C, K>, S2>,
253    cache: &mut LocalAuthCache<Id, C, K>,
254    version: StateResVersion,
255) -> BTreeMap<(EventType, K), LeanEvent<Id, C, K>>
256where
257    Id: crate::basespec::rezzy_types::EventId,
258    C: Clone,
259    S1: core::hash::BuildHasher,
260    S2: core::hash::BuildHasher,
261    K: Clone + Ord,
262{
263    if let Some(cached) = cache.map.get(&event.event_id) {
264        return cached
265            .clone()
266            .into_iter()
267            .map(|(k, entry)| (k, entry.event))
268            .collect();
269    }
270
271    let mut local_auth: BTreeMap<(EventType, K), LocalAuthEntry<Id, C, K>> = BTreeMap::new();
272    let mut queue = alloc::collections::VecDeque::new();
273    for aid in &event.auth_events {
274        queue.push_back((aid, 1));
275    }
276    // Membership-only dedup; do NOT iterate (foldhash seed is per-process,
277    // so iteration order is non-deterministic). Traversal order is strictly
278    // driven by the FIFO VecDeque queue.
279    let mut visited = crate::FastSet::default();
280
281    while let Some((aid, current_depth)) = queue.pop_front() {
282        if !visited.insert(aid) {
283            continue;
284        }
285
286        if let Some(cached_ancestor) = cache.map.get(aid) {
287            // The cache only contains the parents of `aid`. We must also insert `aid` itself!
288            if let Some(aev) = auth_context.get(aid).or_else(|| conflicted_events.get(aid)) {
289                update_local_auth(&mut local_auth, aev, current_depth);
290            }
291
292            // NOTE: V2.1.1 (Proposed) replaces unbounded DFS with a pure memoized BFS traversal.
293            // Therefore, both V2.1.1 and V2.2 natively gather transitive auth context!
294            if matches!(version, StateResVersion::V2_1_1 | StateResVersion::V2_2) {
295                for (key, entry) in cached_ancestor {
296                    let total_depth = current_depth.saturating_add(entry.auth_depth);
297                    match local_auth.entry(key.clone()) {
298                        alloc::collections::btree_map::Entry::Vacant(e) => {
299                            e.insert(LocalAuthEntry {
300                                event: entry.event.clone(),
301                                auth_depth: total_depth,
302                            });
303                        }
304                        alloc::collections::btree_map::Entry::Occupied(mut e) => {
305                            if total_depth < e.get().auth_depth {
306                                e.insert(LocalAuthEntry {
307                                    event: entry.event.clone(),
308                                    auth_depth: total_depth,
309                                });
310                            }
311                        }
312                    }
313                }
314            }
315            continue;
316        }
317
318        if let Some(aev) = auth_context.get(aid).or_else(|| conflicted_events.get(aid)) {
319            update_local_auth(&mut local_auth, aev, current_depth);
320
321            // TODO: confirm the V2.2 auth traversal rule remains aligned with V2.1.1.
322            // For V2.1 and below, we only check the immediate auth_events.
323            if matches!(version, StateResVersion::V2_1_1 | StateResVersion::V2_2) {
324                for parent_id in &aev.auth_events {
325                    queue.push_back((parent_id, current_depth.saturating_add(1)));
326                }
327            }
328        }
329    }
330
331    cache.map.insert(event.event_id.clone(), local_auth.clone());
332    local_auth
333        .into_iter()
334        .map(|(k, entry)| (k, entry.event))
335        .collect()
336}

338/// An O(1) cloneable, persistent state map. Note that `state_key: ""`
339/// is _never_ `null` or `None`.
340///
341/// The `event_type` half of the key is interned via [`EventType`] so that
342/// well-known types use compact enum representations and avoid per-entry
343/// type-string allocations in state keys. Equality, ordering, and hashing
344/// still follow the canonical string form.
345///
346/// Generic over the state-key type `K` (defaults to `String`); see
347/// [`crate::basespec::rezzy_types::StateKey`].
348///
349/// A HAMT-backed state map was benchmarked as a replacement
350/// (`benches/state_backend.rs`) and lost on the access pattern that
351/// matters most here: forking a state map into several branches and
352/// diverging each (what conflict resolution does), where it was 6-22x
353/// slower than `OrdMap`'s clone. `imbl::OrdMap`'s RRB-tree is tuned
354/// specifically for cheap-clone/structural-sharing workloads, so it stays.
355pub type SharedState<Id = String, K = String> = imbl::OrdMap<(EventType, K), Id>;

357/// Computes the resolved room state *after* a given event.
358///
359/// This walks the `prev_events` graph backwards from `target_event_id`,
360/// topologically sorts all reachable ancestors, and incrementally builds
361/// the state by applying each state event in order. Fork points are resolved
362/// via [`crate::resolve::iterative::resolve_iterative_sort`] with the given `version` semantics.
363///
364/// Returns `None` if `target_event_id` is not found in `events_map`.
365///
366/// # Panics
367///
368/// Will panic if graph invariants are violated (specifically, if an ancestor event
369/// present in the reachable subgraph is missing from `events_map` during topological processing).
370#[must_use]
371pub fn compute_state_at<Id, C, Q, S, K>(
372    target_event_id: &Q,
373    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
374    version: StateResVersion,
375    empty_key: &K,
376) -> Option<BTreeMap<(EventType, K), Id>>
377where
378    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
379    Q: ?Sized + Eq + Ord + core::hash::Hash,
380    S: core::hash::BuildHasher,
381    C: crate::basespec::rezzy_types::EventContent,
382    K: crate::basespec::rezzy_types::StateKey,
383    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
384{
385    if !events_map.contains_key(target_event_id) {
386        return None;
387    }
388
389    let mut result = None;
390    compute_state_at_streaming(
391        &[target_event_id],
392        events_map,
393        version,
394        |_, state| {
395            result = Some(state.into_iter().collect());
396        },
397        empty_key,
398    );
399    result
400}

402/// Computes the resolved room state at multiple target events in a single pass.
403///
404/// This is the batch variant of [`compute_state_at`]. It shares the topological
405/// sort and ancestor traversal across all targets, which is significantly faster
406/// than calling `compute_state_at` in a loop when the targets share ancestors.
407/// TODO: once the incremental state pipeline uses `crate::hamt`, this should
408/// be able to reuse more structure between forked states.
409///
410/// Returns a map from each found target event ID to its resolved state.
411/// Target IDs not found in `events_map` are silently skipped.
412///
413/// # Memory and Performance
414///
415/// This function materializes and returns a complete `BTreeMap` for every
416/// target event. For large rooms with many target events, this will cause
417/// massive memory spikes and allocation overhead.
418///
419/// For processing multiple events in production (e.g., full room rebuilds),
420/// use [`compute_state_at_streaming`] instead to stream states via a callback
421/// and keep memory bounded to the DAG's width.
422///
423/// # Panics
424///
425/// Will panic if graph invariants are violated (specifically, if an ancestor event
426/// present in the reachable subgraph is missing from `events_map` during topological processing).
427#[must_use]
428pub fn compute_state_at_batch<Id, C, Q, S, K>(
429    target_event_ids: &[&Q],
430    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
431    version: StateResVersion,
432    empty_key: &K,
433) -> HashMap<Id, BTreeMap<(EventType, K), Id>>
434where
435    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
436    Q: ?Sized + Eq + core::hash::Hash + Ord,
437    S: core::hash::BuildHasher,
438    C: crate::basespec::rezzy_types::EventContent,
439    K: crate::basespec::rezzy_types::StateKey,
440    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
441{
442    let mut results = HashMap::with_capacity(target_event_ids.len());
443
444    compute_state_at_streaming(
445        target_event_ids,
446        events_map,
447        version,
448        |id, state| {
449            results.insert(id, state.into_iter().collect());
450        },
451        empty_key,
452    );
453
454    results
455}

457/// Errors that can occur during streaming state computation.
458#[derive(Debug, Clone, PartialEq, Eq)]
459pub enum StateComputationError<E> {
460    /// The timeline DAG contains a cycle, making topological sorting impossible.
461    CycleDetected,
462    /// The caller-provided callback returned an error.
463    Callback(E),
464}
465
466impl<E: core::fmt::Display> core::fmt::Display for StateComputationError<E> {
467    fn fmt(&self, f: &mut core::fmt::Formatter<'_>) -> core::fmt::Result {
468        match self {
469            Self::CycleDetected => {
470                write!(f, "Cycle detected in DAG. Reachable subgraph is malformed.")
471            }
472            Self::Callback(e) => write!(f, "Callback error: {e}"),
473        }
474    }
475}

479/// Same as [`compute_state_at_batch`] but yields each resolved room state
480/// to a callback (as soon as it is ready).
481///
482/// This function is **strictly superior** to [`compute_state_at_batch`] for
483/// large-scale state reconstruction (e.g. homeserver full state rebuilds).
484/// By passing ownership of the computed state to the callback, callers can
485/// immediately compress and store the state (e.g. directly into a `RocksDB`
486/// buffer), bounding the peak memory for materialized state maps to the live
487/// frontier/DAG width under strict `O(n_reachable_ancestors)` indexing metadata.
488/// TODO: pair this with the HAMT-backed state map to reduce clone pressure on
489/// large fork-heavy DAGs.
490///
491/// **NOTE:** Target IDs not found in `events_map` are silently skipped!
492///
493/// # Panics
494///
495/// Will panic if graph invariants are violated (specifically, if an ancestor event
496/// present in the reachable subgraph is missing from `events_map` during topological processing).
497pub fn compute_state_at_streaming<Id, C, Q, S, F, K>(
498    target_event_ids: &[&Q],
499    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
500    version: StateResVersion,
501    mut on_target_resolved: F,
502    empty_key: &K,
503) where
504    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
505    Q: ?Sized + Eq + core::hash::Hash + Ord,
506    S: core::hash::BuildHasher,
507    C: crate::basespec::rezzy_types::EventContent,
508    F: FnMut(Id, SharedState<Id, K>),
509    K: crate::basespec::rezzy_types::StateKey,
510    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
511{
512    let result = try_compute_state_at_streaming(
513        target_event_ids,
514        events_map,
515        version,
516        |id, state| -> Result<(), core::convert::Infallible> {
517            on_target_resolved(id, state);
518            Ok(())
519        },
520        empty_key,
521    );
522
523    match result {
524        Ok(()) => {}
525        Err(StateComputationError::CycleDetected) => {
526            #[cfg(feature = "std")]
527            std::eprintln!(
528                "rezzy::compute_state_at: Cycle detected! Reachable subgraph is malformed."
529            );
530        }
531        Err(StateComputationError::Callback(infallible)) => match infallible {},
532    }
533}

535/// A fallible variant of [`compute_state_at_streaming`].
536///
537/// Functions identically to `compute_state_at_streaming`, but threads a `Result` through
538/// the callback so that callers can abort early (e.g. on I/O errors during storage).
539///
540/// # Errors
541/// Returns `StateComputationError::CycleDetected` if a cycle is found in the reachable graph.
542/// Returns `StateComputationError::Callback(e)` if the callback yields an error.
543pub fn try_compute_state_at_streaming<Id, C, Q, S, F, E, K>(
544    target_event_ids: &[&Q],
545    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
546    version: StateResVersion,
547    mut on_target_resolved: F,
548    empty_key: &K,
549) -> Result<(), StateComputationError<E>>
550where
551    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
552    Q: ?Sized + Eq + core::hash::Hash + Ord,
553    S: core::hash::BuildHasher,
554    C: crate::basespec::rezzy_types::EventContent,
555    F: FnMut(Id, SharedState<Id, K>) -> Result<(), E>,
556    K: crate::basespec::rezzy_types::StateKey,
557    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
558{
559    let mut actual_target_ids = Vec::new();
560    let mut seen = alloc::collections::BTreeSet::new();
561    for &tid in target_event_ids {
562        if let Some((k, _)) = events_map.get_key_value(tid) {
563            if seen.insert(k) {
564                actual_target_ids.push(k.clone());
565            }
566        }
567    }
568
569    if actual_target_ids.is_empty() {
570        return Ok(());
571    }
572
573    let target_refs: Vec<&Id> = actual_target_ids.iter().collect();
574    let index = collect_ancestor_short_ids_batch(&target_refs, events_map);
575
576    let mut is_target = alloc::vec![false; index.len()];
577    for tid in &actual_target_ids {
578        if let Some(idx) = index.index_of(&tid) {
579            is_target[idx] = true;
580        }
581    }
582
583    run_state_pipeline_streaming(
584        &index,
585        &is_target,
586        events_map,
587        version,
588        |idx, shared_state| {
589            let id = index.items()[idx].clone();
590            on_target_resolved(id, shared_state)
591        },
592        empty_key,
593    )
594}

596/// Core topological graph traversal loop for batch state reconstruction.
597///
598/// Topologically sorts all reachable ancestors, incrementally merges state at forks,
599/// and yields the target states as they are completed.
600fn run_state_pipeline_streaming<Id, C, S, F, E, K>(
601    index: &DenseIndex<&Id, usize>,
602    is_target: &[bool],
603    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
604    version: StateResVersion,
605    mut on_target: F,
606    empty_key: &K,
607) -> Result<(), StateComputationError<E>>
608where
609    Id: crate::basespec::rezzy_types::EventId,
610    S: core::hash::BuildHasher,
611    C: crate::basespec::rezzy_types::EventContent,
612    F: FnMut(usize, SharedState<Id, K>) -> Result<(), E>,
613    K: crate::basespec::rezzy_types::StateKey,
614    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
615{
616    let (sorted_ancestors, mut out_degree) = topological_sort_short_ids(index, events_map);
617
618    if sorted_ancestors.len() != index.len() {
619        return Err(StateComputationError::CycleDetected);
620    }
621
622    let mut global_auth_cache = LocalAuthCache::new(version);
623    let mut mainline_cache: FastMap<Id, Option<Id>> = FastMap::default();
624
625    let mut state_after_map: Vec<Option<SharedState<Id, K>>> =
626        core::iter::repeat_with(|| None).take(index.len()).collect();
627
628    for idx in sorted_ancestors {
629        let id_val = index.items()[idx];
630        let ev = events_map.get(id_val).unwrap();
631
632        let mut prev_states = Vec::with_capacity(ev.prev_events.len());
633        let mut seen_parents = if ev.prev_events.len() > 1 {
634            Some(crate::FastSet::default())
635        } else {
636            None
637        };
638        for pe in &ev.prev_events {
639            let Some(pe_idx) = index.index_of(&pe) else {
640                continue;
641            };
642            if let Some(seen) = &mut seen_parents {
643                if !seen.insert(pe_idx) {
644                    continue;
645                }
646            }
647            if out_degree[pe_idx] == 0 {
648                continue;
649            }
650            out_degree[pe_idx] = out_degree[pe_idx].saturating_sub(1);
651            if out_degree[pe_idx] == 0 {
652                if let Some(pe_state) = state_after_map[pe_idx].take() {
653                    prev_states.push(pe_state);
654                }
655            } else if let Some(ref pe_state) = state_after_map[pe_idx] {
656                prev_states.push(pe_state.clone());
657            }
658        }
659
660        let mut state_before: SharedState<Id, K> = if prev_states.is_empty() {
661            SharedState::new()
662        } else if prev_states.len() == 1 {
663            prev_states.into_iter().next().unwrap()
664        } else {
665            resolve_merge_fast_path(
666                &prev_states,
667                events_map,
668                &mut global_auth_cache,
669                &mut mainline_cache,
670                version,
671                empty_key,
672            )
673        };
674
675        if let Some(state_key) = ev.state_key.as_ref().filter(|_| !ev.rejected) {
676            state_before.insert(
677                (EventType::from(ev.event_type.as_str()), state_key.clone()),
678                ev.event_id.clone(),
679            );
680        }
681
682        if is_target[idx] {
683            on_target(idx, state_before.clone()).map_err(StateComputationError::Callback)?;
684        }
685
686        if out_degree[idx] > 0 {
687            state_after_map[idx] = Some(state_before);
688        }
689    }
690
691    Ok(())
692}

694/// A point in the DAG where a subset of forward extremities converge.
695///
696/// Returned by [`compute_merge_bases`]. Each junction records which extremities
697/// are reachable (via `mask`), the event at the convergence point, and its depth.
698#[derive(Debug, Clone, PartialEq, Eq)]
699pub struct MergeBase<Id> {
700    /// The event ID at the junction point.
701    pub event_id: Id,
702    /// Bitmask of which extremities can reach this node.
703    /// Bit `i` is set iff extremity `i` is an ancestor-or-self.
704    pub mask: u8,
705    /// DAG depth of the junction event.
706    pub depth: u64,
707}
708
709/// Default hard cap on backward walk steps for [`compute_merge_bases`].
710pub const MERGE_BASE_MAX_STEPS: usize = 5_000;
711
712/// Finds **all** primitive merge bases for up to 8 forward extremities in a
713/// single backward pass.
714///
715/// Unlike [`compute_merge_base`] (which returns only the global common
716/// ancestor), this function discovers every subset junction — the points where
717/// different subsets of extremities first converge. The result is a minimal set
718/// of [`MergeBase`] entries after superseding pruning.
719///
720/// # Algorithm (bitmask-propagating backward walk)
721///
722/// 1. Each extremity gets a unique bit in a `u8` mask.
723/// 2. A max-heap ordered by depth walks backward through `prev_events`,
724///    propagating masks via bitwise OR.
725/// 3. When a node's mask gains `popcount ≥ 2`, it is recorded as a candidate
726///    junction for that mask value.
727/// 4. Walk terminates when the global merge base is found (all bits set) and
728///    no unexplored paths remain, or when `max_steps` is exceeded.
729/// 5. Superseded junctions are pruned: if mask M₂ ⊃ M₁ and J₂.depth ≥ J₁.depth,
730///    then J₁ is redundant.
731///
732/// # Complexity
733///
734/// - **Time**: `O((V + E) · k)` where V/E are visited nodes/edges, k ≤ 8.
735///   Bitmask ops are single CPU instructions.
736/// - **Space**: `O(V)` for the mask map (one `u8` per visited event).
737///
738/// # Panics
739///
740/// Panics if `extremities.len() > 8` (bitmask overflow).
741///
742/// # Example
743///
744/// ```rust
745/// use rezzy::{compute_merge_bases, MERGE_BASE_MAX_STEPS, DagNode};
746/// use rezzy::{LeanEvent, HashMap};
747///
748/// let events: HashMap<String, LeanEvent<String>> = HashMap::new();
749/// let tips = vec!["$tip_a", "$tip_b", "$tip_c"];
750/// let junctions = compute_merge_bases(&tips, &events, MERGE_BASE_MAX_STEPS);
751/// for j in &junctions {
752///     println!("junction {:?} mask={:08b} depth={}", j.event_id, j.mask, j.depth);
753/// }
754/// ```
755#[must_use]
756pub fn compute_merge_bases<'a, Id, Q, S, Node>(
757    extremities: &[&Q],
758    events_map: &'a HashMap<Id, Node, S>,
759    max_steps: usize,
760) -> Vec<MergeBase<&'a Id>>
761where
762    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
763    Q: ?Sized + Eq + core::hash::Hash + Ord,
764    S: core::hash::BuildHasher,
765    Node: crate::basespec::rezzy_types::DagNode<Id = Id>,
766{
767    use alloc::collections::BinaryHeap;
768
769    assert!(
770        extremities.len() <= 8,
771        "compute_merge_bases supports at most 8 extremities"
772    );
773
774    if extremities.len() < 2 {
775        return Vec::new();
776    }
777
778    let k = extremities.len();
779    let full_mask: u8 = 1u8
780        .checked_shl(u32::try_from(k).expect("extremity count overflow"))
781        .and_then(|v| v.checked_sub(1))
782        .expect("bitmask overflow: k must be <= 8");
783
784    // Max-heap: (depth, &Id) — highest depth pops first.
785    let mut queue: BinaryHeap<(u64, &Id)> = BinaryHeap::new();
786    let mut masks: FastMap<&Id, u8> = FastMap::default();
787
788    // Track the highest-depth (closest to tips) junction found per mask.
789    let mut best_junction: FastMap<u8, (&Id, u64)> = FastMap::default();
790
791    for (i, &head) in extremities.iter().enumerate() {
792        if let Some((k, ev)) = events_map.get_key_value(head) {
793            let bit = 1u8 << i;
794            let entry = masks.entry(k).or_insert(0);
795            *entry |= bit;
796            queue.push((ev.depth(), k));
797        }
798    }
799
800    let mut steps: usize = 0;
801
802    while let Some((depth, current_id)) = queue.pop() {
803        if steps >= max_steps {
804            break;
805        }
806        steps = steps.saturating_add(1);
807
808        // Every id ever pushed onto `queue` was inserted into `masks` first
809        // (initial extremity seeding above, or `masks.entry(pk)` before the
810        // parent push below); `masks` entries are never removed. So this
811        // lookup cannot miss.
812        let &current_mask = masks
813            .get(current_id)
814            .expect("queue entries are only pushed for ids already present in `masks`");
815
816        let popcount = current_mask.count_ones();
817
818        // Record junction if this is a convergence point (≥ 2 extremities).
819        if popcount >= 2 {
820            best_junction
821                .entry(current_mask)
822                .or_insert((current_id, depth));
823            // We use or_insert because the first time we see a mask, it's at
824            // the highest depth (closest to tips) due to max-heap ordering.
825        }
826
827        // Global merge base found — ancestors are redundant.
828        if current_mask == full_mask {
829            break;
830        }
831
832        // Propagate mask to parents.
833        if let Some(ev) = events_map.get(current_id.borrow()) {
834            for parent_id in ev.prev_events() {
835                let parent_q: &Q = parent_id.borrow();
836                if let Some((pk, parent_ev)) = events_map.get_key_value(parent_q) {
837                    let parent_mask = masks.entry(pk).or_insert(0);
838                    let old = *parent_mask;
839                    *parent_mask |= current_mask;
840
841                    if *parent_mask != old {
842                        queue.push((parent_ev.depth(), pk));
843                    }
844                }
845            }
846        }
847    }
848
849    // If we didn't find even a 2-bit convergence, return empty.
850    if best_junction.is_empty() {
851        return Vec::new();
852    }
853
854    // Superseding pruning: remove junction J₁ (mask M₁) if there exists
855    // J₂ (mask M₂ ⊃ M₁) where J₂.depth ≥ J₁.depth (the larger subset
856    // converged at least as close to the tips).
857    let mut junctions: Vec<MergeBase<&'a Id>> = Vec::new();
858    let masks_vec: Vec<(u8, &Id, u64)> = best_junction
859        .into_iter()
860        .map(|(mask, (id, depth))| (mask, id, depth))
861        .collect();
862
863    for &(mask, id, depth) in &masks_vec {
864        let superseded = masks_vec
865            .iter()
866            .any(|&(m2, _, d2)| m2 != mask && (m2 & mask) == mask && d2 >= depth);
867        if !superseded {
868            junctions.push(MergeBase {
869                event_id: id,
870                mask,
871                depth,
872            });
873        }
874    }
875
876    // Sort by descending depth (closest to tips first).
877    junctions.sort_by_key(|j| core::cmp::Reverse(j.depth));
878    junctions
879}

881/// Computes the most recent common ancestor (merge base) of multiple DAG tips.
882///
883/// Uses a max-heap ordered by event `depth` with roaring bitmap reachability
884/// masks. Each extremity gets a unique bit index; as the heap walks backward
885/// through `prev_events`, bitmasks propagate via bitwise OR. The first event
886/// whose bitmask contains all extremity bits is the merge base.
887///
888/// Returns `None` if the extremities have no common ancestor (disjoint DAGs)
889/// or if `extremities` is empty.
890///
891/// # Complexity
892///
893/// - **Time**: `O(V + E)` bounded to the subgraph between the extremities and
894///   their merge base. Events below the merge base are never visited.
895/// - **Space**: `O(V)` for the bitmask map, where each bitmask is a compressed
896///   roaring bitmap.
897///
898/// ## **TODO:** Future optimization
899///
900/// With offline preprocessing (binary lifting or Euler tour + sparse table),
901/// repeated LCA queries against the same DAG could be answered in `O(log V)`
902/// per query after `O(V log V)` pre-processing.
903///
904/// # Panics
905///
906/// Panics if there are more than `2^32` extremities (practically unreachable).
907///
908/// # Example
909///
910/// ```rust
911/// use rezzy::{compute_merge_base, DagNode};
912/// use rezzy::{LeanEvent, HashMap};
913///
914/// let mut events: HashMap<String, LeanEvent<String>> = HashMap::new();
915/// // ... populate events ...
916/// let tips = vec!["$tip_a", "$tip_b"];
917/// let merge_base = compute_merge_base(&tips, &events);
918/// ```
919#[must_use]
920pub fn compute_merge_base<'a, Id, Q, S, Node>(
921    extremities: &[&Q],
922    events_map: &'a HashMap<Id, Node, S>,
923) -> Option<&'a Id>
924where
925    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
926    Q: ?Sized + Eq + core::hash::Hash + Ord,
927    S: core::hash::BuildHasher,
928    Node: crate::basespec::rezzy_types::DagNode<Id = Id>,
929{
930    use alloc::collections::BinaryHeap;
931
932    use roaring::RoaringBitmap;
933
934    if extremities.is_empty() {
935        return None;
936    }
937
938    // Single extremity: it is its own merge base.
939    if extremities.len() == 1 {
940        return events_map.get_key_value(extremities[0]).map(|(k, _)| k);
941    }
942
943    let target_count = extremities.len() as u64;
944
945    // Max-heap: (depth, &Id) — highest depth pops first, ensuring a parent
946    // is never processed until all of its descendants have propagated bits.
947    let mut queue: BinaryHeap<(u64, &Id)> = BinaryHeap::new();
948    let mut masks: FastMap<&Id, RoaringBitmap> = FastMap::default();
949
950    for (i, &head) in extremities.iter().enumerate() {
951        if let Some((k, ev)) = events_map.get_key_value(head) {
952            let idx = u32::try_from(i).expect("more than 2^32 extremities");
953            let entry = masks.entry(k).or_default();
954            entry.insert(idx);
955            queue.push((ev.depth(), k));
956        }
957    }
958
959    while let Some((_, current_id)) = queue.pop() {
960        // Same invariant as `compute_merge_bases`: every id pushed onto
961        // `queue` was inserted into `masks` first, and entries are never
962        // removed, so this lookup cannot miss.
963        let current_mask = masks
964            .get(current_id)
965            .cloned()
966            .expect("queue entries are only pushed for ids already present in `masks`");
967
968        // If reachable by ALL extremities, this is the merge base.
969        if current_mask.len() == target_count {
970            return Some(current_id);
971        }
972
973        if let Some(ev) = events_map.get(current_id.borrow()) {
974            for parent_id in ev.prev_events() {
975                let parent_q: &Q = parent_id.borrow();
976                if let Some((pk, parent_ev)) = events_map.get_key_value(parent_q) {
977                    let is_new = !masks.contains_key(pk);
978                    let parent_mask = masks.entry(pk).or_default();
979                    let old_len = parent_mask.len();
980                    *parent_mask |= &current_mask;
981                    let new_len = parent_mask.len();
982
983                    if is_new || new_len > old_len {
984                        queue.push((parent_ev.depth(), pk));
985                    }
986                }
987            }
988        }
989    }
990
991    None // Disjoint DAGs (no common ancestor)
992}

994/// Collects all reachable ancestor events across a batch of target events and assigns them
995/// contiguous integer IDs (short IDs) for fast topological processing and array lookups.
996fn collect_ancestor_short_ids_batch<'a, Id, C, S, K>(
997    target_event_ids: &[&'a Id],
998    events_map: &'a HashMap<Id, LeanEvent<Id, C, K>, S>,
999) -> DenseIndex<&'a Id, usize>
1000where
1001    Id: crate::basespec::rezzy_types::EventId,
1002    S: core::hash::BuildHasher,
1003    C: Clone,
1004{
1005    let mut index_to_id: Vec<&'a Id> = Vec::new();
1006    let mut seen: crate::FastSet<&'a Id> = crate::FastSet::default();
1007    let mut queue = Vec::new();
1008
1009    for &tid in target_event_ids {
1010        if seen.insert(tid) {
1011            index_to_id.push(tid);
1012            queue.push(tid);
1013        }
1014    }
1015
1016    let mut head = 0;
1017    while head < queue.len() {
1018        let current_id = queue[head];
1019        head = head.saturating_add(1);
1020
1021        let Some(ev) = events_map.get(current_id) else {
1022            continue;
1023        };
1024        for pe in &ev.prev_events {
1025            if events_map.contains_key(pe) && seen.insert(pe) {
1026                index_to_id.push(pe);
1027                queue.push(pe);
1028            }
1029        }
1030    }
1031
1032    DenseIndex::try_build(index_to_id)
1033        .expect("ancestor short-id index exceeds the usize address space")
1034}

1036/// Performs a topological sort of the graph represented by short `usize` indexes.
1037/// Performs Kahn's topological sort on the collected ancestor graph.
1038/// Returns the events sorted such that parents always appear before their children.
1039fn topological_sort_short_ids<Id, C, S, K>(
1040    index: &DenseIndex<&Id, usize>,
1041    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
1042) -> (Vec<usize>, Vec<usize>)
1043where
1044    Id: crate::basespec::rezzy_types::EventId,
1045    S: core::hash::BuildHasher,
1046    C: Clone,
1047{
1048    let num_reachable = index.len();
1049    let mut in_degree = alloc::vec![0usize; num_reachable];
1050    let mut adjacency = alloc::vec![Vec::new(); num_reachable];
1051    let mut out_degree = alloc::vec![0usize; num_reachable];
1052
1053    for (i, id) in index.items().iter().enumerate() {
1054        let Some(ev) = events_map.get(*id) else {
1055            continue;
1056        };
1057        let mut seen = if ev.prev_events.len() > 1 {
1058            Some(crate::FastSet::default())
1059        } else {
1060            None
1061        };
1062        for parent in &ev.prev_events {
1063            if let Some(parent_idx) = index.index_of(&parent) {
1064                // Dedup: only count each parent edge once, even if prev_events has duplicates.
1065                if let Some(seen_set) = &mut seen {
1066                    if !seen_set.insert(parent_idx) {
1067                        continue;
1068                    }
1069                }
1070                in_degree[i] = in_degree[i].saturating_add(1);
1071                adjacency[parent_idx].push(i);
1072                out_degree[parent_idx] = out_degree[parent_idx].saturating_add(1);
1073            }
1074        }
1075    }
1076
1077    let mut topo_queue = alloc::collections::VecDeque::new();
1078    for (i, &deg) in in_degree.iter().enumerate() {
1079        if deg == 0 {
1080            topo_queue.push_back(i);
1081        }
1082    }
1083
1084    let mut sorted_ancestors = Vec::with_capacity(num_reachable);
1085    while let Some(idx) = topo_queue.pop_front() {
1086        sorted_ancestors.push(idx);
1087        for &child_idx in &adjacency[idx] {
1088            in_degree[child_idx] = in_degree[child_idx].saturating_sub(1);
1089            if in_degree[child_idx] == 0 {
1090                topo_queue.push_back(child_idx);
1091            }
1092        }
1093    }
1094
1095    (sorted_ancestors, out_degree)
1096}

1134/// Slow path for merging multiple parent states via the state resolution algorithm.
1135/// Full state resolution path for DAG nodes with multiple parents (forks).
1136/// Groups the unconflicted state and runs `resolve_iterative_sort` on the conflicted subset.
1137fn resolve_multiple_prev_states<Id, C, S, K>(
1138    prev_states: &[SharedState<Id, K>],
1139    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
1140    global_auth_cache: &mut LocalAuthCache<Id, C, K>,
1141    mainline_cache: &mut FastMap<Id, Option<Id>>,
1142    version: StateResVersion,
1143    empty_key: &K,
1144) -> SharedState<Id, K>
1145where
1146    Id: crate::basespec::rezzy_types::EventId,
1147    S: core::hash::BuildHasher,
1148    C: crate::basespec::rezzy_types::EventContent,
1149    K: crate::basespec::rezzy_types::StateKey,
1150    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
1151{
1152    let mut conflicted_keys = crate::FastSet::default();
1153    let mut conflicted_state_set = crate::HashSet::new();
1154    let base = &prev_states[0];
1155
1156    for other in &prev_states[1..] {
1157        for diff_item in base.diff(other) {
1158            match diff_item {
1159                imbl::ordmap::DiffItem::Add(k, v) | imbl::ordmap::DiffItem::Remove(k, v) => {
1160                    conflicted_keys.insert(k.clone());
1161                    conflicted_state_set.insert(v.clone());
1162                }
1163                imbl::ordmap::DiffItem::Update {
1164                    old: (k, old_v),
1165                    new: (_, new_v),
1166                } => {
1167                    conflicted_keys.insert(k.clone());
1168                    conflicted_state_set.insert(old_v.clone());
1169                    conflicted_state_set.insert(new_v.clone());
1170                }
1171            }
1172        }
1173    }
1174
1175    let mut unconflicted_state = base.clone();
1176    for k in &conflicted_keys {
1177        unconflicted_state.remove(k);
1178    }
1179
1180    let mut conflicted_events = HashMap::with_capacity(conflicted_state_set.len());
1181    for id_val in &conflicted_state_set {
1182        if let Some(event) = events_map.get(id_val) {
1183            conflicted_events.insert(id_val.clone(), event.clone());
1184        }
1185    }
1186
1187    // Supplement conflicted_events with the auth difference auth(C) \ auth(U)
1188    let auth_diff = compute_auth_chain_diff(&unconflicted_state, &conflicted_state_set, events_map);
1189    for id_val in auth_diff {
1190        if let Some(event) = events_map.get(&id_val) {
1191            conflicted_events.insert(id_val, event.clone());
1192        }
1193    }
1194
1195    let mut pl_cache: HashMap<Id, i64, hashbrown::DefaultHashBuilder> = HashMap::default();
1196    crate::resolve::iterative::resolve_iterative_sort_with_all_caches(
1197        &unconflicted_state,
1198        &conflicted_events,
1199        events_map,
1200        Some(global_auth_cache),
1201        version,
1202        &mut pl_cache,
1203        mainline_cache,
1204        &conflicted_keys,
1205        empty_key,
1206    )
1207}

1209/// Computes the **auth chain difference**: `auth(C) \ auth(U)`.
1210///
1211/// Walks the unconflicted (U) and conflicted (C) auth chains in
1212/// parallel by depth, pruning C-side events already reachable
1213/// from U. Returns the set of event IDs in the conflicted auth
1214/// chains that are NOT reachable from unconflicted state.
1215///
1216/// This is the core input to state resolution — the set of
1217/// events that must be considered during iterative auth. By
1218/// exposing this as a public API, homeservers can compute the
1219/// auth difference without reimplementing the bounded dual-heap
1220/// traversal.
1221///
1222/// # Parameters
1223///
1224/// - `unconflicted_state`: The agreed-upon state (values are
1225///   event IDs whose auth chains define the "known" baseline).
1226/// - `conflicted_state_set`: Event IDs that differ across forks.
1227/// - `events_map`: Full event context containing all referenced
1228///   events and their auth chains.
1229///
1230/// # Returns
1231///
1232/// The set of event IDs reachable from `conflicted_state_set`'s
1233/// auth chains but NOT reachable from `unconflicted_state`'s
1234/// auth chains.
1235///
1236/// # Complexity
1237///
1238/// - **Time**: `O((|U| + |C|) · log(|U| + |C|))` — bounded by
1239///   the total auth chain size, with early pruning.
1240/// - **Space**: `O(|U| + |C|)` for visited sets.
1241///
1242/// # Panics
1243///
1244/// Internal `unwrap()` calls are guarded by `peek()`
1245/// checks and cannot panic under normal operation.
1246pub fn compute_auth_chain_diff<Id, C, S1, S2, K>(
1247    unconflicted_state: &SharedState<Id, K>,
1248    conflicted_state_set: &crate::HashSet<Id, S2>,
1249    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S1>,
1250) -> crate::HashSet<Id>
1251where
1252    Id: crate::basespec::rezzy_types::EventId,
1253    S1: core::hash::BuildHasher,
1254    S2: core::hash::BuildHasher,
1255    C: crate::basespec::rezzy_types::EventContent,
1256    K: Ord + Clone,
1257{
1258    let mut u_visited = crate::FastSet::default();
1259    let mut u_heap_elements = Vec::with_capacity(unconflicted_state.len());
1260    for id in unconflicted_state.values() {
1261        if u_visited.insert(id.clone()) {
1262            if let Some(ev) = events_map.get(id) {
1263                u_heap_elements.push((ev.depth, id.clone()));
1264            }
1265        }
1266    }
1267    let mut u_heap = alloc::collections::BinaryHeap::from(u_heap_elements);
1268
1269    let mut c_visited = crate::FastSet::default();
1270    let mut c_heap = alloc::collections::BinaryHeap::new();
1271    for id in conflicted_state_set {
1272        if u_visited.contains(id) {
1273            continue; // PRUNE EARLY
1274        }
1275        if c_visited.insert(id.clone()) {
1276            if let Some(ev) = events_map.get(id) {
1277                c_heap.push((ev.depth, id.clone()));
1278            }
1279        }
1280    }
1281
1282    let mut auth_diff = crate::HashSet::new();
1283
1284    while let Some(&(c_depth, _)) = c_heap.peek() {
1285        // Catch up U's traversal to C's current depth
1286        while let Some(&(u_depth, _)) = u_heap.peek() {
1287            if u_depth < c_depth {
1288                break;
1289            }
1290            let (_, u_id) = u_heap
1291                .pop()
1292                .expect("invariant: heap peek implies non-empty pop");
1293            let ev = events_map.get(&u_id).expect(
1294                "invariant: every heap entry corresponds to an event present in events_map",
1295            );
1296            for auth_id in &ev.auth_events {
1297                if u_visited.insert(auth_id.clone()) {
1298                    if let Some(a_ev) = events_map.get(auth_id) {
1299                        u_heap.push((a_ev.depth, auth_id.clone()));
1300                    }
1301                }
1302            }
1303        }
1304
1305        let (_, c_id) = c_heap
1306            .pop()
1307            .expect("invariant: heap peek implies non-empty pop");
1308        if !u_visited.contains(&c_id) {
1309            auth_diff.insert(c_id.clone());
1310            let ev = events_map.get(&c_id).expect(
1311                "invariant: every heap entry corresponds to an event present in events_map",
1312            );
1313            for auth_id in &ev.auth_events {
1314                if u_visited.contains(auth_id) {
1315                    continue; // PRUNE EARLY
1316                }
1317                if c_visited.insert(auth_id.clone()) {
1318                    if let Some(a_ev) = events_map.get(auth_id) {
1319                        c_heap.push((a_ev.depth, auth_id.clone()));
1320                    }
1321                }
1322            }
1323        }
1324    }
1325
1326    auth_diff
1327}

1349/// Scans a set of DAG events and identifies **backward extremities** —
1350/// events whose `prev_events` reference parent IDs that are missing from
1351/// both the provided `events` map and the caller's `exists` oracle.
1352///
1353/// This is the pure graph-analysis core of a homeserver's backfill loop.
1354/// By extracting it into rezzy, it becomes testable and reusable without
1355/// async database I/O or federation networking.
1356///
1357/// # Arguments
1358///
1359/// - `events`: The local event map to scan.
1360/// - `exists`: A predicate that returns `true` if an event ID is known to
1361///   exist outside `events` (e.g. in a database). This prevents reporting
1362///   false gaps for events that are stored but not loaded into memory.
1363///
1364/// # Returns
1365///
1366/// A `Vec<BackwardExtremity<Id>>` for every event that has at least one
1367/// missing parent. Events whose parents are all accounted for (either in
1368/// `events` or via `exists`) are not included.
1369///
1370/// # Example
1371///
1372/// ```rust
1373/// use rezzy::{find_backward_extremities, LeanEvent, HashMap};
1374///
1375/// let mut events: HashMap<String, LeanEvent> = HashMap::new();
1376/// // ... populate events ...
1377/// let gaps = find_backward_extremities(&events, |_id| false);
1378/// for gap in &gaps {
1379///     println!("Event {} missing parents: {:?}", gap.event_id, gap.missing_prev_events);
1380/// }
1381/// ```
1382///
1383/// # Complexity
1384///
1385/// - **Time**: `O(Σ |prev_events|)` — linear in the total number of parent
1386///   references across all events.
1387/// - **Space**: `O(G)` where `G` is the total number of missing parent IDs
1388///   across all backward extremities.
1389#[must_use]
1390pub fn find_backward_extremities<Id, Node, S, F>(
1391    events: &crate::HashMap<Id, Node, S>,
1392    exists: F,
1393) -> Vec<BackwardExtremity<Id>>
1394where
1395    Id: crate::basespec::rezzy_types::EventId,
1396    Node: crate::basespec::rezzy_types::DagNode<Id = Id>,
1397    S: core::hash::BuildHasher,
1398    F: Fn(&Id) -> bool,
1399{
1400    let mut result = Vec::new();
1401
1402    for node in events.values() {
1403        let mut missing = Vec::new();
1404        for prev_id in node.prev_events() {
1405            if !events.contains_key(prev_id) && !exists(prev_id) {
1406                missing.push(prev_id.clone());
1407            }
1408        }
1409        if !missing.is_empty() {
1410            result.push(BackwardExtremity {
1411                event_id: node.event_id().clone(),
1412                missing_prev_events: missing,
1413            });
1414        }
1415    }
1416
1417    result
1418}

1488/// Computes position-based topological depths for all events in the map.
1489///
1490/// Unlike [`compute_depths`] (which returns `1 + max(parent_depths)` —
1491/// the spec-correct DAG depth), this function returns the **1-indexed
1492/// position** of each event in Kahn's topological sort. This produces a
1493/// total ordering suitable for building a database index where every
1494/// event gets a unique depth value.
1495///
1496/// The `tiebreak` closure determines the ordering of events at the same
1497/// topological level (zero in-degree simultaneously). Typical choices:
1498/// - `|a, b| ts(a).cmp(&ts(b)).then(a.cmp(b))` — chronological with
1499///   lexicographic event ID fallback (deterministic).
1500/// - `|_, _| Ordering::Equal` — arbitrary (fastest, non-deterministic).
1501///
1502/// # Difference from `compute_depths`
1503///
1504/// ```text
1505///          A
1506///         / \
1507///        B   C
1508///         \ /
1509///          D
1510///
1511///  compute_depths:         A=1, B=2, C=2, D=3
1512///  compute_topo_positions: A=1, B=2, C=3, D=4  (total order)
1513/// ```
1514///
1515/// `compute_depths` preserves DAG structure (siblings share a depth).
1516/// `compute_topo_positions` produces a strict total order (every event
1517/// gets a unique position). The latter is what a homeserver needs for
1518/// its `roomid_topologicalorder_pducount` index.
1519///
1520/// # Complexity
1521///
1522/// - **Time**: `O(V log V + E)` — Kahn sort plus a comparison sort for
1523///   deterministic tiebreaking within topological levels.
1524/// - **Space**: `O(V)` for the position map.
1525#[must_use]
1526pub fn compute_topo_positions<Id, C, S, F, K>(
1527    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
1528    tiebreak: F,
1529) -> Vec<Id>
1530where
1531    Id: crate::basespec::rezzy_types::EventId,
1532    S: core::hash::BuildHasher,
1533    C: Clone,
1534    F: Fn(&Id, &Id) -> core::cmp::Ordering,
1535{
1536    if events_map.is_empty() {
1537        return Vec::new();
1538    }
1539
1540    let all_ids: Vec<&Id> = events_map.keys().collect();
1541    let index = collect_ancestor_short_ids_batch(&all_ids, events_map);
1542    let (sorted, _) = topological_sort_short_ids(&index, events_map);
1543
1544    debug_assert_eq!(
1545        sorted.len(),
1546        index.len(),
1547        "compute_topo_positions: Kahn sort returned fewer nodes than expected — \
1548         the input graph contains a cycle"
1549    );
1550
1551    // Kahn sort gives a valid topological order; apply tiebreak within
1552    // each topological level for deterministic output.
1553    // First compute parent-max depths to identify levels.
1554    let mut depth_by_idx = alloc::vec![0u64; index.len()];
1555    for &idx in &sorted {
1556        let id = index.items()[idx];
1557        if let Some(ev) = events_map.get(id) {
1558            let max_parent = ev
1559                .prev_events
1560                .iter()
1561                .filter_map(|pe| index.index_of(&pe))
1562                .map(|pi| depth_by_idx[pi])
1563                .max()
1564                .unwrap_or(0);
1565            depth_by_idx[idx] = max_parent.saturating_add(1);
1566        }
1567    }
1568
1569    // Sort by depth ascending (parents first), tiebreak within level.
1570    let mut result: Vec<Id> = sorted
1571        .iter()
1572        .map(|&idx| index.items()[idx].clone())
1573        .collect();
1574
1575    result.sort_by(|a, b| {
1576        let da = index.index_of(&a).map_or(0, |i| depth_by_idx[i]);
1577        let db = index.index_of(&b).map_or(0, |i| depth_by_idx[i]);
1578        da.cmp(&db).then_with(|| tiebreak(a, b))
1579    });
1580
1581    result
1582}

1586/// Computes the topological depth of every event reachable from the given
1587/// targets in the DAG. Depth is defined as `1 + max(parent depths)`, with
1588/// root events (no parents in the map) having depth 1.
1589///
1590/// This is the reference depth computation. A homeserver should use these
1591/// values when building its topological index — any mismatch is a bug in
1592/// the storage layer.
1593///
1594/// # Complexity
1595///
1596/// - **Time**: `O(V + E)` — one Kahn sort pass over the reachable subgraph.
1597/// - **Space**: `O(V)` for the depth map.
1598///
1599/// # Panics
1600///
1601/// Panics if a sorted event ID is not found in `events_map` (indicates a
1602/// bug in the topological sort).
1603#[must_use]
1604pub fn compute_depths<Id, C, S, K>(
1605    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
1606) -> HashMap<Id, u64>
1607where
1608    Id: crate::basespec::rezzy_types::EventId,
1609    S: core::hash::BuildHasher,
1610    C: Clone,
1611{
1612    if events_map.is_empty() {
1613        return HashMap::new();
1614    }
1615
1616    let all_ids: Vec<&Id> = events_map.keys().collect();
1617    let index = collect_ancestor_short_ids_batch(&all_ids, events_map);
1618    let (sorted, _) = topological_sort_short_ids(&index, events_map);
1619
1620    let mut depths = alloc::vec![0u64; index.len()];
1621
1622    for idx in &sorted {
1623        let id = index.items()[*idx];
1624        let ev = events_map.get(id).unwrap();
1625        let max_parent_depth = ev
1626            .prev_events
1627            .iter()
1628            .filter_map(|pe| index.index_of(&pe))
1629            .map(|pi| depths[pi])
1630            .max()
1631            .unwrap_or(0);
1632        depths[*idx] = max_parent_depth.saturating_add(1);
1633    }
1634
1635    let mut result = HashMap::with_capacity(index.len());
1636    for (i, &id) in index.items().iter().enumerate() {
1637        result.insert(id.clone(), depths[i]);
1638    }
1639    result
1640}

1644/// A `prev_events` edge whose two endpoints' wire-supplied `depth` values
1645/// are not consistent with the DAG structure: the child's claimed `depth`
1646/// does not exceed its parent's.
1647///
1648/// Both `depth` and `prev_events` are covered by an event's content hash
1649/// (and therefore its signature), so a signature proves the sender *said*
1650/// `depth: N` — it doesn't prove `N` is a truthful function of the parents
1651/// it also signed. `prev_events` is checkable against the DAG the local
1652/// server already holds; `depth` is not independently checkable at all
1653/// except by comparing it back against that same DAG. That comparison is
1654/// what this type reports: a byzantine or buggy peer can attach any
1655/// `depth` it likes, and this flags claims that are structurally
1656/// impossible given the parent it also claims.
1657///
1658/// This only checks *relative* monotonicity across a single edge, not
1659/// absolute depth-from-room-creation — an edge check works on a partial
1660/// view (e.g. a backfill gap-fill batch, where ancestors above the gap
1661/// aren't in `events_map`), whereas an absolute check would require the
1662/// full DAG back to `m.room.create` and would flag every legitimate
1663/// partial batch as a false positive.
1664#[derive(Debug, Clone, PartialEq, Eq)]
1665pub struct DepthDivergence<Id> {
1666    /// The parent event (referenced via `prev_events`).
1667    pub parent: Id,
1668    /// The child event (the one whose `prev_events` includes `parent`).
1669    pub child: Id,
1670    /// The parent's claimed wire `depth`.
1671    pub parent_depth: u64,
1672    /// The child's claimed wire `depth`.
1673    pub child_depth: u64,
1674}
1675
1676/// Scans a set of events and reports every `prev_events` edge whose
1677/// endpoints' wire `depth` values violate DAG monotonicity — i.e. a child
1678/// whose claimed `depth` does not exceed its parent's claimed `depth`.
1679/// Only edges where **both** endpoints are present in `events_map` are
1680/// checked; an edge to a parent outside the map (a genuine gap) is
1681/// silently skipped rather than treated as a violation.
1682///
1683/// This is a pure detector — it doesn't decide what to do about a
1684/// divergence, only surfaces it. A homeserver can use this to log or alert
1685/// on peers sending events with implausible depths, independent of whether
1686/// it also chooses to use [`resolve_gap_fill_order`] to sidestep the
1687/// untrusted value entirely when merging a gap-fill batch.
1688///
1689/// # Complexity
1690///
1691/// `O(Σ |prev_events|)` — linear in the total number of parent references.
1692#[must_use]
1693pub fn find_depth_divergences<Id, C, S, K>(
1694    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
1695) -> Vec<DepthDivergence<Id>>
1696where
1697    Id: crate::basespec::rezzy_types::EventId,
1698    S: core::hash::BuildHasher,
1699    C: Clone,
1700{
1701    let mut divergences = Vec::new();
1702    let mut seen_edges = BTreeSet::new();
1703
1704    for child in events_map.values() {
1705        for parent_id in &child.prev_events {
1706            let Some(parent) = events_map.get(parent_id) else {
1707                continue; // parent outside the map — a gap, not a violation
1708            };
1709            if !seen_edges.insert((parent_id.clone(), child.event_id.clone())) {
1710                continue;
1711            }
1712            if parent.depth == u64::MAX && child.depth == u64::MAX {
1713                continue; // saturated clamp: both endpoints reached the cap
1714            }
1715            if child.depth <= parent.depth {
1716                divergences.push(DepthDivergence {
1717                    parent: parent_id.clone(),
1718                    child: child.event_id.clone(),
1719                    parent_depth: parent.depth,
1720                    child_depth: child.depth,
1721                });
1722            }
1723        }
1724    }
1725
1726    // `events_map` iteration order is hash-map-arbitrary; sort for a
1727    // deterministic, reproducible report.
1728    divergences.sort_by(|a, b| (&a.parent, &a.child).cmp(&(&b.parent, &b.child)));
1729    divergences
1730}

1732/// Re-derives the ordering of a backfill gap-fill batch directly from
1733/// `prev_events` DAG edges, ignoring every event's wire-supplied `depth`
1734/// field entirely.
1735///
1736/// This is not new logic — it's [`compute_topo_positions`] under a name
1737/// that documents the intended call site: the point where a homeserver
1738/// merges a `/backfill` response into its timeline index, where trusting
1739/// wire `depth` for ordering would otherwise be tempting because the value
1740/// is "right there" on each PDU. `compute_topo_positions` already ignores
1741/// `depth` and derives order purely from `prev_events`, which is why it's
1742/// safe to reuse as-is rather than needing new derivation logic; only the
1743/// entry point is new.
1744///
1745/// Works correctly on a partial batch: the derived order is a valid
1746/// topological order of the subgraph you pass in, which is what a gap-fill

1771/// Returns events reachable from `tip` in **reverse topological order**
1772/// (newest first). This is the spec-correct ordering for
1773/// `/messages?dir=b` backward pagination.
1774///
1775/// Tie-breaking within the same topological level is determined by
1776/// `tiebreak`. Typical choices:
1777/// - Homeserver: `|a, b| pdu_count(a).cmp(&pdu_count(b)).reverse()` (insertion order)
1778/// - Tests: `|a, b| a.cmp(&b).reverse()` (lexicographic event ID)
1779///
1780/// # Complexity
1781///
1782/// - **Time**: `O(V + E)` for ancestor collection + Kahn sort.
1783/// - **Space**: `O(V)`.
1784#[must_use]
1785pub fn reverse_topological_order<Id, C, Q, S, F, K>(
1786    tip: &Q,
1787    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
1788    tiebreak: F,
1789) -> Vec<Id>
1790where
1791    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
1792    Q: ?Sized + Eq + core::hash::Hash + Ord,
1793    S: core::hash::BuildHasher,
1794    C: Clone,
1795    F: Fn(&Id, &Id) -> core::cmp::Ordering,
1796{
1797    let Some((tip_key, _)) = events_map.get_key_value(tip) else {
1798        return Vec::new();
1799    };
1800
1801    let targets = [tip_key];
1802    let index = collect_ancestor_short_ids_batch(&targets, events_map);
1803    let (sorted, _) = topological_sort_short_ids(&index, events_map);
1804
1805    // Compute depths inline using the index arrays
1806    let mut depth_by_idx = alloc::vec![0u64; index.len()];
1807    for &idx in &sorted {
1808        let id = index.items()[idx];
1809        if let Some(ev) = events_map.get(id.borrow()) {
1810            let max_parent = ev
1811                .prev_events
1812                .iter()
1813                .filter_map(|pe| index.index_of(&pe))
1814                .map(|pi| depth_by_idx[pi])
1815                .max()
1816                .unwrap_or(0);
1817
1818            depth_by_idx[idx] = max_parent.saturating_add(1);
1819        }
1820    }
1821
1822    // Kahn sort gives parents-first; reverse for newest-first,
1823    // then stable-sort by depth descending with tiebreak.
1824    let mut result: Vec<Id> = sorted
1825        .iter()
1826        .rev()
1827        .map(|&idx| index.items()[idx].clone())
1828        .collect();
1829
1830    result.sort_by(|a, b| {
1831        let da = index.index_of(&a).map_or(0, |i| depth_by_idx[i]);
1832        let db = index.index_of(&b).map_or(0, |i| depth_by_idx[i]);
1833        db.cmp(&da).then_with(|| tiebreak(a, b))
1834    });
1835
1836    result
1837}

1839/// The kind of violation detected by [`verify_pagination`].
1840#[derive(Debug, Clone, PartialEq, Eq)]
1841pub enum PaginationViolation<Id> {
1842    /// An event appeared on more than one page.
1843    Duplicate {
1844        event_id: Id,
1845        first_page: usize,
1846        second_page: usize,
1847    },
1848    /// An ancestor appeared *after* its descendant in the page sequence
1849    /// (violates the reverse-topological ordering invariant).
1850    AncestorAfterDescendant {
1851        ancestor: Id,
1852        descendant: Id,
1853        ancestor_page: usize,
1854        descendant_page: usize,
1855    },
1856}

1858/// Verifies that a sequence of pagination pages respects DAG ordering
1859/// invariants:
1860///
1861/// 1. **No duplicates** — each event ID appears on at most one page.
1862/// 2. **Topological monotonicity** — if event A is an ancestor of B,
1863///    then A must not appear on an *earlier* page than B (in backward
1864///    pagination, descendants come first).
1865///
1866/// This is a pure property checker. Feed it the actual pages from your
1867/// paginator; any violation is a bug in the storage/pagination layer.
1868///
1869/// Completeness (every reachable event present) is intentionally NOT
1870/// checked — pagination may stop at room creation, budget limits, or
1871/// ACL boundaries.
1872///
1873/// # Returns
1874///
1875/// A `Vec` of violations. Empty means the pages are well-formed.
1876#[must_use]
1877pub fn verify_pagination<Id, C, S, K>(
1878    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
1879    pages: &[Vec<Id>],
1880) -> Vec<PaginationViolation<Id>>
1881where
1882    Id: crate::basespec::rezzy_types::EventId,
1883    S: core::hash::BuildHasher,
1884    C: Clone,
1885{
1886    let mut violations = Vec::new();
1887
1888    // 1. Check for duplicates
1889    let mut seen: FastMap<&Id, usize> = FastMap::default();
1890    for (page_idx, page) in pages.iter().enumerate() {
1891        for id in page {
1892            if let Some(&first_page) = seen.get(id) {
1893                violations.push(PaginationViolation::Duplicate {
1894                    event_id: id.clone(),
1895                    first_page,
1896                    second_page: page_idx,
1897                });
1898            } else {
1899                seen.insert(id, page_idx);
1900            }
1901        }
1902    }
1903
1904    // 2. Check topological monotonicity (an ancestor must not appear before a
1905    // descendant). Walk transitively: a paginator may omit an intermediate
1906    // event while still returning both endpoints of an ancestor relationship.
1907    // In backward pagination, page 0 has the newest events. If event B is on
1908    // page 1 and B's ancestor A is on page 0 (earlier), that's wrong — A should
1909    // be on a later page (higher index).
1910    for (page_idx, page) in pages.iter().enumerate() {
1911        for id in page {
1912            let mut ancestors = Vec::new();
1913            let mut visited: FastSet<&Id> = FastSet::default();
1914            if let Some(event) = events_map.get(id) {
1915                ancestors.extend(event.prev_events.iter());
1916            }
1917
1918            let mut ancestor_index = 0;
1919            while ancestor_index < ancestors.len() {
1920                let parent_id = ancestors[ancestor_index];
1921                ancestor_index = ancestor_index.saturating_add(1);
1922                if !visited.insert(parent_id) {
1923                    continue;
1924                }
1925
1926                if let Some(&parent_page) = seen.get(parent_id) {
1927                    if parent_page < page_idx {
1928                        violations.push(PaginationViolation::AncestorAfterDescendant {
1929                            ancestor: parent_id.clone(),
1930                            descendant: id.clone(),
1931                            ancestor_page: parent_page,
1932                            descendant_page: page_idx,
1933                        });
1934                    }
1935                }
1936
1937                if let Some(parent) = events_map.get(parent_id) {
1938                    ancestors.extend(parent.prev_events.iter());
1939                }
1940            }
1941        }
1942    }
1943
1944    violations
1945}

2229            empty_key,

2231

2235            match diff_item {
2236                imbl::ordmap::DiffItem::Add(key, new_id) => {
2237                    hash.insert(key.0.as_str(), key.1.as_ref(), new_id);
2238                }
2239                imbl::ordmap::DiffItem::Remove(key, old_id) => {
2240                    hash.remove(key.0.as_str(), key.1.as_ref(), old_id);
2241                }

2243                    old: (key, old_id),
2244                    new: (_, new_id),
2245                } => {
2246                    hash.remove(key.0.as_str(), key.1.as_ref(), old_id);
2247                    hash.insert(key.0.as_str(), key.1.as_ref(), new_id);
2248                }
2249            }

2251

2253            state: resolved,

2260///
2261/// This variant reuses the persistent `mainline_cache` across the whole
2262/// traversal and yields `StateUpdate` values for target events, avoiding the
2263/// extra cloning performed by the plain streaming path.
2264fn run_state_pipeline_streaming_optimized<'a, Id, C, S, F, E, K>(
2265    index: &DenseIndex<&'a Id, usize>,
2266    is_target: &[bool],
2267    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
2268    version: StateResVersion,
2269    mut on_target: F,
2270    empty_key: &K,
2271) -> Result<(), StateComputationError<E>>
2272where
2273    Id: crate::basespec::rezzy_types::EventId,

2275    C: crate::basespec::rezzy_types::EventContent,
2276    F: for<'b> FnMut(usize, StateUpdate<'b, Id, K>) -> Result<(), E>,
2277    K: crate::basespec::rezzy_types::StateKey,

2283        return Err(StateComputationError::CycleDetected);
2284    }

2286    let mut global_auth_cache = LocalAuthCache::new(version);
2287    let mut mainline_cache: FastMap<Id, Option<Id>> = FastMap::default();
2288
2289    let mut state_after_map: Vec<Option<HashedState<Id, K>>> =
2290        core::iter::repeat_with(|| None).take(index.len()).collect();
2291

2293        let id_val = index.items()[idx];
2294        let ev = events_map.get(id_val).unwrap();
2295
2296        let mut prev_states = Vec::with_capacity(ev.prev_events.len());
2297        let mut seen_parents = if ev.prev_events.len() > 1 {
2298            Some(crate::FastSet::default())
2299        } else {
2300            None
2301        };
2302        for pe in &ev.prev_events {
2303            let Some(pe_idx) = index.index_of(&pe) else {
2304                continue;
2305            };
2306            // Dedup: adversarial events may carry duplicate prev_events.
2307            // Without this, out_degree is decremented twice for one child,
2308            // causing premature take() and wrong merge results.
2309            if let Some(seen) = &mut seen_parents {
2310                if !seen.insert(pe_idx) {
2311                    continue;
2312                }
2313            }
2314            out_degree[pe_idx] = out_degree[pe_idx].saturating_sub(1);
2315            if out_degree[pe_idx] == 0 {
2316                if let Some(pe_state) = state_after_map[pe_idx].take() {
2317                    prev_states.push(pe_state);
2318                }
2319            } else if let Some(ref pe_state) = state_after_map[pe_idx] {
2320                prev_states.push(pe_state.clone());
2321            }
2322        }
2323
2324        let is_state = ev.state_key.is_some();
2325        let has_single_parent = prev_states.len() == 1;
2326
2327        let mut state_before: HashedState<Id, K> = if prev_states.is_empty() {
2328            HashedState::new()
2329        } else if has_single_parent && !is_state {
2330            let parent_state = prev_states.into_iter().next().unwrap();
2331            if is_target[idx] {
2332                on_target(
2333                    idx,
2334                    StateUpdate::Unchanged {
2335                        parent_event_id: ev
2336                            .prev_events
2337                            .iter()
2338                            .find(|pe| index.index_of(pe).is_some())
2339                            .expect(
2340                                "has_single_parent implies at least one prev_event is in the index",
2341                            ),
2342                        hash: &parent_state.hash,
2343                    },
2344                )
2345                .map_err(StateComputationError::Callback)?;
2346            }
2347            if out_degree[idx] > 0 {
2348                state_after_map[idx] = Some(parent_state);
2349            }
2350            continue;
2351        } else if has_single_parent {
2352            prev_states.into_iter().next().unwrap()
2353        } else {
2354            resolve_merge_fast_path_hashed_with_cache(
2355                &prev_states,
2356                events_map,
2357                &mut global_auth_cache,
2358                &mut mainline_cache,
2359                version,
2360                empty_key,
2361            )
2362        };
2363
2364        if let Some(state_key) = ev.state_key.as_ref().filter(|_| !ev.rejected) {
2365            let key = (EventType::from(ev.event_type.as_str()), state_key.clone());
2366            state_before.insert(key, ev.event_id.clone());
2367        }
2368
2369        if is_target[idx] {
2370            on_target(
2371                idx,
2372                StateUpdate::New {
2373                    state: state_before.state.clone(),
2374                    hash: &state_before.hash,
2375                },
2376            )
2377            .map_err(StateComputationError::Callback)?;
2378        }
2379
2380        if out_degree[idx] > 0 {
2381            state_after_map[idx] = Some(state_before);
2382        }
2383    }
2384
2385    Ok(())
2386}

2388/// A high-performance, fallible variant of [`compute_state_at_streaming`] designed for
2389/// massive rebuild pipelines.
2390///
2391/// Rather than cloning full `SharedState` maps for every target, this function yields
2392/// `StateUpdate` events that support zero-clone streaming when states are unchanged
2393/// from their parent, and $O(1)$ LtHash-based matching.
2394///
2395/// # Errors
2396/// Returns `StateComputationError::CycleDetected` if a cycle is found in the reachable graph.
2397/// Returns `StateComputationError::Callback(e)` if the callback yields an error.
2398///
2399/// # Behavior
2400/// Duplicate target IDs are silently deduplicated, and targets absent from `events_map`
2401/// are dropped. The callback count may therefore be less than the input count.
2402pub fn try_compute_state_at_streaming_optimized<Id, C, Q, S, F, E, K>(
2403    target_event_ids: &[&Q],
2404    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
2405    version: StateResVersion,
2406    mut on_target_resolved: F,
2407    empty_key: &K,
2408) -> Result<(), StateComputationError<E>>
2409where
2410    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
2411    Q: ?Sized + Eq + core::hash::Hash + Ord,
2412    S: core::hash::BuildHasher,
2413    C: crate::basespec::rezzy_types::EventContent,
2414    F: for<'b> FnMut(Id, StateUpdate<'b, Id, K>) -> Result<(), E>,
2415    K: crate::basespec::rezzy_types::StateKey,
2416    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
2417{
2418    let mut actual_target_ids = Vec::new();
2419    let mut seen = alloc::collections::BTreeSet::new();
2420    for &tid in target_event_ids {
2421        if let Some((k, _)) = events_map.get_key_value(tid) {
2422            if seen.insert(k) {
2423                actual_target_ids.push(k.clone());
2424            }
2425        }
2426    }
2427
2428    if actual_target_ids.is_empty() {
2429        return Ok(());
2430    }
2431
2432    let target_refs: Vec<&Id> = actual_target_ids.iter().collect();
2433    let index = collect_ancestor_short_ids_batch(&target_refs, events_map);
2434
2435    let mut is_target = alloc::vec![false; index.len()];
2436    for tid in &actual_target_ids {
2437        if let Some(idx) = index.index_of(&tid) {
2438            is_target[idx] = true;
2439        }
2440    }
2441
2442    run_state_pipeline_streaming_optimized(
2443        &index,
2444        &is_target,
2445        events_map,
2446        version,
2447        |idx, update| {
2448            let id = index.items()[idx].clone();
2449            on_target_resolved(id, update)
2450        },
2451        empty_key,
2452    )
2453}

2455/// A high-performance, non-fallible variant of [`compute_state_at_streaming`] designed for
2456/// massive rebuild pipelines.
2457///
2458/// Returns `true` if the graph traversal completed successfully, or `false` if a cycle
2459/// was detected in the reachable subgraph.
2460#[must_use = "a `false` return means a cycle was detected and results are incomplete; silently discarding it defeats the purpose of cycle detection"]
2461pub fn compute_state_at_streaming_optimized<Id, C, Q, S, F, K>(
2462    target_event_ids: &[&Q],
2463    events_map: &HashMap<Id, LeanEvent<Id, C, K>, S>,
2464    version: StateResVersion,
2465    mut on_target_resolved: F,
2466    empty_key: &K,
2467) -> bool
2468where
2469    Id: crate::basespec::rezzy_types::EventId + core::borrow::Borrow<Q>,
2470    Q: ?Sized + Eq + core::hash::Hash + Ord,
2471    S: core::hash::BuildHasher,
2472    C: crate::basespec::rezzy_types::EventContent,
2473    F: for<'b> FnMut(Id, StateUpdate<'b, Id, K>),
2474    K: crate::basespec::rezzy_types::StateKey,
2475    for<'q> (EventType, K): core::borrow::Borrow<dyn crate::auth::StateKeyDyn + 'q>,
2476{
2477    let result = try_compute_state_at_streaming_optimized(
2478        target_event_ids,
2479        events_map,
2480        version,
2481        |id, update| -> Result<(), core::convert::Infallible> {
2482            on_target_resolved(id, update);
2483            Ok(())
2484        },
2485        empty_key,
2486    );
2487
2488    match result {
2489        Ok(()) => true,
2490        Err(StateComputationError::CycleDetected) => false,
2491        Err(StateComputationError::Callback(infallible)) => match infallible {},
2492    }
2493}

2495/// Computes the true forward extremities (DAG leaves) from a batched set of events.
2496/// This uses `RoaringBitmap` set differences (`all_events - all_parents`) to
2497/// instantly find the leaves of a DAG, no matter how deep.
2498///
2499/// # Arguments
2500/// - `events`: An iterator yielding tuples of `(event_id, prev_event_ids)`.
2501///
2502/// # Returns
2503/// A `Vec<Id>` of all events that are not referenced as a `prev_event` by any other event in the set.
2504///
2505/// # Panics
2506/// Panics if the number of distinct event IDs exceeds `u32::MAX`.
2507pub fn find_forward_extremities_roaring<Id, I, P>(events: I) -> alloc::vec::Vec<Id>
2508where
2509    Id: core::hash::Hash + Eq + Clone,
2510    I: IntoIterator<Item = (Id, P)>,
2511    P: IntoIterator<Item = Id>,
2512{
2513    use roaring::RoaringBitmap;
2514    let mut id_map = crate::HashMap::default();
2515    let mut reverse_map = alloc::vec::Vec::new();
2516
2517    let get_or_insert = |id: Id,
2518                         id_map: &mut crate::HashMap<Id, u32>,
2519                         reverse_map: &mut alloc::vec::Vec<Id>|
2520     -> u32 {
2521        *id_map.entry(id).or_insert_with_key(|id| {
2522            let idx = u32::try_from(reverse_map.len()).expect("event count exceeds u32");
2523            reverse_map.push(id.clone());
2524            idx
2525        })
2526    };
2527
2528    let mut all_events = RoaringBitmap::new();
2529    let mut has_children = RoaringBitmap::new();
2530
2531    for (id, prevs) in events {
2532        let idx = get_or_insert(id, &mut id_map, &mut reverse_map);
2533        all_events.insert(idx);
2534
2535        for prev_id in prevs {
2536            let prev_idx = get_or_insert(prev_id, &mut id_map, &mut reverse_map);
2537            has_children.insert(prev_idx);
2538        }
2539    }
2540
2541    let extremities_bitmap = core::ops::Sub::sub(all_events, has_children);
2542
2543    let mut extremities =
2544        alloc::vec::Vec::with_capacity(usize::try_from(extremities_bitmap.len()).unwrap());
2545    for idx in extremities_bitmap {
2546        extremities.push(reverse_map[idx as usize].clone());
2547    }
2548
2549    extremities
2550}
