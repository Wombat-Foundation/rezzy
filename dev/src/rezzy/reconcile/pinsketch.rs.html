16const FIELD_BITS: usize = 64;
17const MIXED_FACTOR_TRIALS: usize = 8;
18const FACTOR_TRIALS: usize = MIXED_FACTOR_TRIALS + FIELD_BITS;
19const FACTOR_PARAMETER_SEED: u64 = 0x9e37_79b9_7f4a_7c15;
20// Only the test-only reference `trace_mod` still uses this directly; the
21// production path (`build_frobenius_basis`) derives the same round count
22// from `FIELD_BITS - 1`.
23#[cfg(test)]
24const TRACE_SQUARES: usize = 63;
25// KNOWN GAP, still open: this covers observed *balanced* recursive
26// splitting, not a proven worst case.
27//
28// Measured directly (`find_roots_with_budget`'s own consumption, not a
29// hand derivation) against both random and adversarial-consecutive-value
30// degree-256 inputs: a full successful decode costs ~9.0-9.3M, and a
31// single node's full non-splitting 72-trial ladder (the shape a genuinely
32// undecodable degree-256 sketch produces) costs ~10.0M -- both comfortably
33// inside this budget with real, if modest, headroom.
34//
35// What's still unbounded is a *chain* of nodes each needing a full
36// non-splitting ladder before finally splitting -- e.g. a degenerate
37// sequence of (1, d-1) splits. Computed from this file's own cost
38// functions (`frobenius_basis_cost` + `FACTOR_TRIALS` *
39// `factor_trial_cost_with_basis` + `split_cost`, summed over degrees
40// d, d-1, ..., 2), that chain totals ~917M at d=256 -- ~57x this budget.
41// Random locators split near-binomially in practice (matching the
42// measurements above), so this degenerate shape is not something a
43// benchmark surfaces, and it fails safe into `BudgetExhausted` (routing
44// to the same fallback ladder callers already handle) rather than
45// corrupting anything -- but "degree 256 is decodable within
46// MAX_FACTOR_WORK" remains a statement about typical inputs, not a proven
47// bound. A real worst-case bound on the recursion, or a balance guarantee
48// on the splitting itself, would be needed to close this properly.
49const MAX_FACTOR_WORK: usize = 16_000_000;

51pub(crate) fn decode(
52    odd_syndromes: &[u64],
53    max_elements: usize,
54) -> Result<Vec<u64>, AlgebraicError> {
55    let all = reconstruct_syndromes(odd_syndromes);
56    let mut locator = berlekamp_massey(&all, max_elements).ok_or(AlgebraicError::DecodeFailure)?;
57    if locator.len() == 1 {
58        return Ok(Vec::new());
59    }
60    locator.reverse();
61    let expected = locator
62        .len()
63        .checked_sub(1)
64        .ok_or(AlgebraicError::DecodeFailure)?;
65    let mut roots = Vec::with_capacity(expected);
66    find_roots(locator, &mut roots)?;
67    if roots.len() != expected || roots.contains(&0) {
68        return Err(AlgebraicError::DecodeFailure);
69    }
70    roots.sort_unstable();
71    Ok(roots)
72}

74/// Like [`decode`], but draws factoring work from `budget` and leaves the
75/// unspent remainder in it, so a caller decoding multiple sketches can share
76/// one budget across all of them instead of each call getting its own
77/// implicit ceiling.
78pub(crate) fn decode_with_budget(
79    odd_syndromes: &[u64],
80    max_elements: usize,
81    budget: &mut usize,
82) -> Result<Vec<u64>, AlgebraicError> {
83    let all = reconstruct_syndromes(odd_syndromes);
84    let mut locator = berlekamp_massey(&all, max_elements).ok_or(AlgebraicError::DecodeFailure)?;
85    if locator.len() == 1 {
86        return Ok(Vec::new());
87    }
88    locator.reverse();
89    let expected = locator
90        .len()
91        .checked_sub(1)
92        .ok_or(AlgebraicError::DecodeFailure)?;
93    let mut roots = Vec::with_capacity(expected);
94    find_roots_with_budget(locator, &mut roots, budget)?;
95    if roots.len() != expected || roots.contains(&0) {
96        return Err(AlgebraicError::DecodeFailure);
97    }
98    roots.sort_unstable();
99    Ok(roots)
100}

102fn reconstruct_syndromes(odd: &[u64]) -> Vec<u64> {
103    let mut all = vec![0; odd.len().saturating_mul(2)];
104    for (index, value) in odd.iter().copied().enumerate() {
105        all[index.saturating_mul(2)] = value;
106        // Earlier iterations have already reconstructed the source at `index`.
107        all[index.saturating_mul(2).saturating_add(1)] = gf64_mul(all[index], all[index]);
108    }
109    all
110}

112fn berlekamp_massey(syndromes: &[u64], max_degree: usize) -> Option<Polynomial> {
113    let mut current = vec![1];
114    let mut previous = vec![1];
115    let mut previous_discrepancy = 1;
116
117    for (n, syndrome) in syndromes.iter().copied().enumerate() {
118        let mut discrepancy = syndrome;
119        for (i, coefficient) in current.iter().copied().enumerate().skip(1) {
120            let Some(syndrome_index) = n.checked_sub(i) else {
121                continue;
122            };
123            discrepancy ^= gf64_mul(syndromes[syndrome_index], coefficient);
124        }
125        if discrepancy == 0 {
126            continue;
127        }
128
129        let current_degree = current.len().checked_sub(1)?;
130        let previous_degree = previous.len().checked_sub(1)?;
131        let shift = n
132            .checked_add(1)?
133            .checked_sub(current_degree)?
134            .checked_sub(previous_degree)?;
135        let swap = current_degree.checked_mul(2)? <= n;
136        let old_current = current.clone();
137        if swap {
138            let new_len = previous.len().checked_add(shift)?;
139            if new_len.checked_sub(1)? > max_degree {
140                return None;
141            }
142            current.resize(new_len, 0);
143        }
144        let scale = gf64_mul(discrepancy, gf64_inv(previous_discrepancy)?);
145        for (i, coefficient) in previous.iter().copied().enumerate() {
146            let target = i.checked_add(shift)?;
147            current[target] ^= gf64_mul(scale, coefficient);
148        }
149        if swap {
150            previous = old_current;
151            previous_discrepancy = discrepancy;
152        }
153    }
154    (current.last().copied().unwrap_or(0) != 0).then_some(current)
155}

157fn gf64_inv(value: u64) -> Option<u64> {
158    if value == 0 {
159        return None;
160    }
161    // a^(2^64-2), using a fixed square-and-multiply schedule.
162    let mut result = 1;
163    let exponent = u64::MAX - 1;
164    for bit in (0..64).rev() {
165        result = gf64_mul(result, result);
166        if exponent & (1_u64 << bit) != 0 {
167            result = gf64_mul(result, value);
168        }
169    }
170    Some(result)
171}

173fn trim(poly: &mut Polynomial) {
174    while poly.last() == Some(&0) {
175        poly.pop();
176    }
177}

179fn make_monic(poly: &mut Polynomial) -> Option<()> {
180    let leading = *poly.last()?;
181    if leading == 1 {
182        return Some(());
183    }
184    let inverse = gf64_inv(leading)?;
185    for coefficient in poly {
186        *coefficient = gf64_mul(*coefficient, inverse);
187    }
188    Some(())
189}

191fn poly_mod(modulus: &[u64], value: &mut Polynomial) -> Option<()> {
192    let modulus_degree = modulus.len().checked_sub(1)?;
193    if modulus.last() != Some(&1) {
194        return None;
195    }
196    // Dispatch on the evaluator backend once, outside the reduction loop,
197    // and let each arm monomorphize its own copy of the loop via the
198    // generic `poly_mod_reduce`, instead of matching on the backend once
199    // per row inside a single shared loop. The match cost was already
200    // trivial (branch-predicted, hoisted out of the loop in the prior
201    // pass), but this removes it from the loop body entirely so the
202    // compiler can optimize each backend's reduction independently.
203    match crate::reconcile::gf64_simd::get_evaluator() {
204        #[cfg(all(target_arch = "x86_64", has_avx512_support))]
205        crate::reconcile::gf64_simd::EvaluatorBackend::Avx512 => {
206            poly_mod_reduce::<crate::reconcile::gf64_simd::Avx512Evaluator>(
207                modulus_degree,
208                modulus,
209                value,
210            )?;
211        }
212        #[cfg(target_arch = "x86_64")]
213        crate::reconcile::gf64_simd::EvaluatorBackend::Sse => {
214            poly_mod_reduce::<crate::reconcile::gf64_simd::SseEvaluator>(
215                modulus_degree,
216                modulus,
217                value,
218            )?;
219        }
220        crate::reconcile::gf64_simd::EvaluatorBackend::Scalar => {
221            poly_mod_reduce::<crate::reconcile::gf64_simd::ScalarEvaluator>(
222                modulus_degree,
223                modulus,
224                value,
225            )?;
226        }
227    }
228    trim(value);
229    Some(())
230}

232fn poly_mod_reduce<E: Gf64Evaluator>(
233    modulus_degree: usize,
234    modulus: &[u64],
235    value: &mut Polynomial,
236) -> Option<()> {
237    while value.len() >= modulus.len() {
238        let term = value.pop()?;
239        if term != 0 {
240            let offset = value.len().checked_sub(modulus_degree)?;
241            let target = &mut value[offset..offset.checked_add(modulus_degree)?];
242            let source = &modulus[..modulus_degree];
243            E::poly_mac(term, source, target);
244        }
245    }
246    Some(())
247}

249fn poly_div(mut dividend: Polynomial, divisor: &[u64]) -> Option<Polynomial> {
250    if divisor.last() != Some(&1) || dividend.len() < divisor.len() {
251        return None;
252    }
253    let mut quotient = vec![0; dividend.len().checked_sub(divisor.len())?.checked_add(1)?];
254    let divisor_degree = divisor.len().checked_sub(1)?;
255    // See `poly_mod` for why the backend dispatch happens once here rather
256    // than once per reduction row inside a shared loop.
257    match crate::reconcile::gf64_simd::get_evaluator() {
258        #[cfg(all(target_arch = "x86_64", has_avx512_support))]
259        crate::reconcile::gf64_simd::EvaluatorBackend::Avx512 => {
260            poly_div_reduce::<crate::reconcile::gf64_simd::Avx512Evaluator>(
261                divisor_degree,
262                divisor,
263                &mut dividend,
264                &mut quotient,
265            )?;
266        }
267        #[cfg(target_arch = "x86_64")]
268        crate::reconcile::gf64_simd::EvaluatorBackend::Sse => {
269            poly_div_reduce::<crate::reconcile::gf64_simd::SseEvaluator>(
270                divisor_degree,
271                divisor,
272                &mut dividend,
273                &mut quotient,
274            )?;
275        }
276        crate::reconcile::gf64_simd::EvaluatorBackend::Scalar => {
277            poly_div_reduce::<crate::reconcile::gf64_simd::ScalarEvaluator>(
278                divisor_degree,
279                divisor,
280                &mut dividend,
281                &mut quotient,
282            )?;
283        }
284    }
285    trim(&mut quotient);
286    Some(quotient)
287}

289fn poly_div_reduce<E: Gf64Evaluator>(
290    divisor_degree: usize,
291    divisor: &[u64],
292    dividend: &mut Polynomial,
293    quotient: &mut Polynomial,
294) -> Option<()> {
295    while dividend.len() >= divisor.len() {
296        let term = dividend.pop()?;
297        let position = dividend.len().checked_sub(divisor_degree)?;
298        quotient[position] = term;
299        if term != 0 {
300            let target = &mut dividend[position..position.checked_add(divisor_degree)?];
301            let source = &divisor[..divisor_degree];
302            E::poly_mac(term, source, target);
303        }
304    }
305    Some(())
306}

308fn poly_gcd(mut left: Polynomial, mut right: Polynomial) -> Option<Polynomial> {
309    if left.len() < right.len() {
310        core::mem::swap(&mut left, &mut right);
311    }
312    while !right.is_empty() {
313        make_monic(&mut right)?;
314        poly_mod(&right, &mut left)?;
315        core::mem::swap(&mut left, &mut right);
316    }
317    make_monic(&mut left)?;
318    Some(left)
319}

321fn poly_square(poly: &mut Polynomial) -> Option<()> {
322    if poly.is_empty() {
323        return Some(());
324    }
325    let new_len = poly.len().checked_mul(2)?.checked_sub(1)?;
326    let old = core::mem::take(poly);
327    poly.resize(new_len, 0);
328    for (target, coefficient) in (0..new_len).step_by(2).zip(old) {
329        poly[target] = gf64_mul(coefficient, coefficient);
330    }
331    Some(())
332}

334/// Naive reference implementation of `Tr(parameter * X) mod modulus`,
335/// re-deriving the full squaring recurrence from scratch for every
336/// `parameter`. Cost is `O(TRACE_SQUARES * degree^2)` *per call*. Superseded
337/// in the production path by `build_frobenius_basis` +
338/// `trace_from_basis`, which amortize the squaring recurrence across every
339/// trial for a given `modulus` instead of repeating it per trial (see that
340/// pair's doc comments). Kept as a test-only cross-check that the
341/// precomputed path computes the same trace.
342#[cfg(test)]
343fn trace_mod(modulus: &[u64], parameter: u64) -> Option<Polynomial> {
344    let mut trace = vec![0, parameter];
345    for _ in 0..TRACE_SQUARES {
346        poly_square(&mut trace)?;
347        if trace.len() < 2 {
348            trace.resize(2, 0);
349        }
350        trace[1] = parameter;
351        poly_mod(modulus, &mut trace)?;
352    }
353    Some(trace)
354}

356/// One-time-per-polynomial precomputation consumed by `trace_from_basis`:
357/// the images `X^(2^0), X^(2^1), ..., X^(2^63) mod modulus`.
358///
359/// Squaring is GF(2)-linear in characteristic 2 (`(a+b)^2 = a^2+b^2`), so
360/// `(parameter*X)^(2^i) = parameter^(2^i) * X^(2^i)` -- the trace
361/// `sum_i (parameter*X)^(2^i) mod modulus` decomposes into
362/// `sum_i parameter^(2^i) * B_i`, where `B_i = X^(2^i) mod modulus` does not
363/// depend on `parameter` at all. Building this basis once per polynomial and
364/// reusing it across every trial in `find_roots_with_budget`'s ladder turns
365/// each trial's dominant cost from `O(TRACE_SQUARES * degree^2)` (the naive
366/// `trace_mod`, repeating the squaring-and-reduce recurrence per trial) into
367/// `O(FIELD_BITS * degree)` for the linear combination, leaving only the
368/// unavoidable `O(degree^2)` GCD/division per trial. See
369/// `frobenius_basis_cost` and `factor_trial_cost_with_basis`.
370fn build_frobenius_basis(modulus: &[u64]) -> Option<Vec<Polynomial>> {
371    let mut basis = Vec::with_capacity(FIELD_BITS);
372    let mut current = vec![0, 1]; // X
373    poly_mod(modulus, &mut current)?;
374    basis.push(current.clone());
375    for _ in 1..FIELD_BITS {
376        poly_square(&mut current)?;
377        poly_mod(modulus, &mut current)?;
378        basis.push(current.clone());
379    }
380    Some(basis)
381}

383/// Computes `Tr(parameter * X) mod modulus` from a basis built by
384/// `build_frobenius_basis` for the same `modulus`.
385fn trace_from_basis(basis: &[Polynomial], parameter: u64) -> Polynomial {
386    let width = basis.iter().map(Vec::len).max().unwrap_or(0);
387    let mut trace = vec![0_u64; width];
388    let mut power = parameter;
389    for basis_term in basis {
390        if power != 0 {
391            for (coefficient, term) in trace.iter_mut().zip(basis_term.iter()) {
392                *coefficient ^= gf64_mul(power, *term);
393            }
394        }
395        power = gf64_mul(power, power);
396    }
397    trim(&mut trace);
398    trace
399}

404    state ^ (state << 17)

407fn solve_quadratic_form(target: u64) -> Option<u64> {
408    // Solve the GF(2)-linear map z -> z^2 + z using a reduced 64x65 matrix.
409    #[derive(Clone, Copy)]
410    struct Row {
411        coefficients: u64,
412        rhs: bool,
413    }
414
415    let mut rows = [Row {
416        coefficients: 0,
417        rhs: false,
418    }; 64];
419    for column in 0..64 {
420        let basis = 1_u64 << column;
421        let image = gf64_mul(basis, basis) ^ basis;
422        for (row, equation) in rows.iter_mut().enumerate() {
423            if image & (1_u64 << row) != 0 {
424                equation.coefficients |= 1_u64 << column;
425            }
426        }
427    }
428    for (row, equation) in rows.iter_mut().enumerate() {
429        if target & (1_u64 << row) != 0 {
430            equation.rhs = true;
431        }
432    }
433
434    let mut rank = 0;
435    for column in 0..64 {
436        let pivot = (rank..64).find(|row| rows[*row].coefficients & (1_u64 << column) != 0);
437        let Some(pivot) = pivot else { continue };
438        rows.swap(rank, pivot);
439        let pivot_row = rows[rank];
440        for (row, equation) in rows.iter_mut().enumerate() {
441            if row != rank && equation.coefficients & (1_u64 << column) != 0 {
442                equation.coefficients ^= pivot_row.coefficients;
443                equation.rhs ^= pivot_row.rhs;
444            }
445        }
446        rank = rank.checked_add(1)?;
447    }
448    if rows.iter().any(|row| row.coefficients == 0 && row.rhs) {
449        return None;
450    }
451    let mut solution = 0_u64;
452    for row in rows.iter().take(rank) {
453        if row.coefficients == 0 {
454            return None;
455        }
456        let pivot = row.coefficients.trailing_zeros();
457        if row.rhs {
458            solution |= 1_u64 << pivot;
459        }
460    }
461    (gf64_mul(solution, solution) ^ solution == target).then_some(solution)
462}

464fn find_roots(poly: Polynomial, roots: &mut Vec<u64>) -> Result<(), AlgebraicError> {
465    let mut work = MAX_FACTOR_WORK;
466    find_roots_with_budget(poly, roots, &mut work)
467}

469/// One-time cost of `build_frobenius_basis` for a degree-`degree`
470/// polynomial: `FIELD_BITS - 1` squaring+reduce rounds, each `O(degree^2)`
471/// -- the same per-round cost the naive `trace_mod` paid per trial, now
472/// paid once per recursion node instead.
473const fn frobenius_basis_cost(degree: usize) -> Option<usize> {
474    let Some(squared) = degree.checked_mul(degree) else {
475        return None;
476    };
477    let Some(rounds) = FIELD_BITS.checked_sub(1) else {
478        return None;
479    };
480    squared.checked_mul(rounds)
481}

483/// Per-trial cost once the node's Frobenius basis is already built:
484/// `O(FIELD_BITS * degree)` for `trace_from_basis`'s linear combination,
485/// plus `O(degree^2)` for `poly_gcd` (which runs a `poly_mod` internally).
486/// This is charged on every trial, whether or not it finds a split --
487/// `poly_div` is charged separately, only on the trial that actually
488/// splits, via `split_cost`. (An earlier version folded `poly_div`'s cost
489/// in here unconditionally on the theory that it kept this a safe upper
490/// bound; it didn't just pad the bound, it overcharged every non-splitting
491/// trial for work that never happened -- an average split takes ~3.7
492/// trials, so that wasted roughly 2.7x this term's cost of budget per
493/// node.)
494const fn factor_trial_cost_with_basis(degree: usize) -> Option<usize> {
495    let Some(trace_cost) = FIELD_BITS.checked_mul(degree) else {
496        return None;
497    };
498    let Some(gcd_cost) = degree.checked_mul(degree) else {
499        return None;
500    };
501    trace_cost.checked_add(gcd_cost)
502}
503
504/// Cost of `poly_div` on a successful split: `O(degree^2)`. Charged once,
505/// only on the trial that actually splits -- see `factor_trial_cost_with_basis`.
506const fn split_cost(degree: usize) -> Option<usize> {
507    degree.checked_mul(degree)
508}

510/// Upper bound on the work one `find_roots_with_budget` call can need for a
511/// locator of degree `degree`, covering the basis build plus a full
512/// `FACTOR_TRIALS`-trial ladder at that degree alone.
513///
514/// This deliberately does **not** try to bound the cost of the recursive
515/// splits below the root -- that recursion has no proven worst-case bound
516/// (see `MAX_FACTOR_WORK`'s comment). It exists so a caller decoding many
517/// sketches under one shared budget (`triage::decode_bucket_sketches`) can
518/// clamp how much of that shared budget any single sketch's decode is
519/// allowed to draw, so one pathological sketch can't starve the rest of a
520/// batch by exhausting the shared pool before later, cheaper sketches are
521/// even attempted. A sketch whose own recursion exceeds this ceiling will
522/// simply hit `BudgetExhausted` for its own allotment -- which is the
523/// intended outcome here, not a bug in this bound.
524pub(crate) const fn single_call_work_ceiling(degree: usize) -> Option<usize> {
525    let Some(basis) = frobenius_basis_cost(degree) else {
526        return None;
527    };
528    let Some(per_trial) = factor_trial_cost_with_basis(degree) else {
529        return None;
530    };
531    let Some(ladder) = per_trial.checked_mul(FACTOR_TRIALS) else {
532        return None;
533    };
534    // `split_cost` is charged at most once per node -- on the one trial
535    // (if any) that actually splits -- not per trial in the ladder.
536    let Some(split) = split_cost(degree) else {
537        return None;
538    };
539    let Some(subtotal) = basis.checked_add(ladder) else {
540        return None;
541    };
542    subtotal.checked_add(split)
543}

545fn find_roots_with_budget(
546    poly: Polynomial,
547    roots: &mut Vec<u64>,
548    work: &mut usize,
549) -> Result<(), AlgebraicError> {
550    let mut pending = vec![poly];
551    while let Some(poly) = pending.pop() {
552        let degree = poly
553            .len()
554            .checked_sub(1)
555            .ok_or(AlgebraicError::DecodeFailure)?;
556        if degree == 0 {
557            continue;
558        }
559        if degree == 1 {
560            roots.push(poly[0]);
561            continue;
562        }
563        if degree == 2 {
564            let linear = poly[1];
565            if linear == 0 {
566                return Err(AlgebraicError::DecodeFailure);
567            }
568            let inverse = gf64_inv(linear).ok_or(AlgebraicError::DecodeFailure)?;
569            let normalized = gf64_mul(poly[0], gf64_mul(inverse, inverse));
570            let root = gf64_mul(
571                solve_quadratic_form(normalized).ok_or(AlgebraicError::DecodeFailure)?,
572                linear,
573            );
574            roots.push(root);
575            roots.push(root ^ linear);
576            continue;
577        }
578
579        let basis_cost = frobenius_basis_cost(degree).ok_or(AlgebraicError::DecodeFailure)?;
580        *work = work
581            .checked_sub(basis_cost)
582            .ok_or(AlgebraicError::BudgetExhausted)?;
583        let basis = build_frobenius_basis(&poly).ok_or(AlgebraicError::DecodeFailure)?;
584
585        let mut split = None;
586        let mut parameter = FACTOR_PARAMETER_SEED;
587        for trial in 0..FACTOR_TRIALS {
588            if trial >= MIXED_FACTOR_TRIALS {
589                let basis_bit = trial
590                    .checked_sub(MIXED_FACTOR_TRIALS)
591                    .ok_or(AlgebraicError::DecodeFailure)?;
592                let basis_bit =
593                    u32::try_from(basis_bit).map_err(|_| AlgebraicError::DecodeFailure)?;
594                parameter = 1_u64
595                    .checked_shl(basis_bit)
596                    .ok_or(AlgebraicError::DecodeFailure)?;
597            }
598            let cost = factor_trial_cost_with_basis(degree).ok_or(AlgebraicError::DecodeFailure)?;
599            *work = work
600                .checked_sub(cost)
601                .ok_or(AlgebraicError::BudgetExhausted)?;
602            let trace = trace_from_basis(&basis, parameter);
603            let factor = poly_gcd(poly.clone(), trace).ok_or(AlgebraicError::DecodeFailure)?;
604            if factor.len() > 1 && factor.len() < poly.len() {
605                // poly_div only runs on this, the one trial that actually
606                // splits -- charged separately from the per-trial cost
607                // above so trials that don't split aren't overcharged for
608                // division work they never performed.
609                let div_cost = split_cost(degree).ok_or(AlgebraicError::DecodeFailure)?;
610                *work = work
611                    .checked_sub(div_cost)
612                    .ok_or(AlgebraicError::BudgetExhausted)?;
613                let quotient = poly_div(poly, &factor).ok_or(AlgebraicError::DecodeFailure)?;
614                split = Some((factor, quotient));
615                break;
616            }
617            if trial < MIXED_FACTOR_TRIALS {
618                parameter = next_factor_parameter(parameter);
619            }
620        }
621        let (factor, quotient) = split.ok_or(AlgebraicError::DecodeFailure)?;
622        pending.push(quotient);
623        pending.push(factor);
624    }
625    Ok(())
626}

692    /// bounds the recursive-splitting cost, only measurements against
693    /// random and adversarial-consecutive-value degree-256 inputs, plus a
694    /// from-first-principles computation of one specific pathological shape
695    /// (a chain of (1, d-1) splits) that would exceed the budget ~57x over
696    /// if it occurred. This test does not close that gap -- constructing an
697    /// input that actually forces that exact recursive shape is a separate,
698    /// nontrivial exercise in choosing a locator polynomial with a
699    /// specific factorization structure over GF(2^64), not attempted here.
700    ///
701    /// What this *does* verify, on a real degree-256 decode: the
702    /// documented "fails safe" claim. `decode_with_budget` with a budget
703    /// far too small to complete any real decode must return
704    /// `BudgetExhausted` cleanly -- not panic, not loop unboundedly, not
