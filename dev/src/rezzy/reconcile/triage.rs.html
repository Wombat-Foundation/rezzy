13/// Maximum sum of capacities in one bucketed sketch request.
14pub const MAX_BUCKETED_SKETCH_CAPACITY: usize = 4_096;
15/// Maximum extraction capacity assigned to one bucket.
16pub const MAX_BUCKET_SKETCH_CAPACITY: usize = 32;
17/// Recommended default work budget for [`decode_bucket_sketches`], sized to
18/// cover a full normal-path batch: [`MAX_BUCKETS_PER_ROUND`] buckets each at
19/// the worst-case per-bucket work ceiling for [`MAX_BUCKET_SKETCH_CAPACITY`].
20/// A caller passing less than this to `decode_bucket_sketches` risks a
21/// bucket near the end of a full-size batch reporting `BudgetExhausted` for
22/// no reason but this constant not being wired through -- see that
23/// function's doc comment.
24pub const MAX_BATCH_FACTOR_WORK: usize = MAX_BUCKETS_PER_ROUND
25    * match pinsketch::single_call_work_ceiling(MAX_BUCKET_SKETCH_CAPACITY) {
26        Some(ceiling) => ceiling,
27        None => panic!("single_call_work_ceiling overflowed for MAX_BUCKET_SKETCH_CAPACITY"),
28    };
29/// Maximum capacity permitted only through the local overflow request path.
30///
31/// This is not a negotiated protocol capability. Normal bucket requests remain
32/// limited to [`MAX_BUCKET_SKETCH_CAPACITY`].
33// Kept equal to `algebraic::MAX_OVERFLOW_SKETCH_CAPACITY` — see its comment.
34pub const MAX_OVERFLOW_BUCKET_CAPACITY: usize = 256;
35/// Client-side sketch-mode cutoff for estimates in the saturated regime.
36pub const SATURATED_DELTA_ESTIMATE: u64 = 8 * (1_u64 << 31);
37/// Minimum cardinality implied by an over-capacity stratum-0 decode failure.
38const OVER_CAPACITY_DELTA_FLOOR: u64 = (STRATUM_CAPACITY as u64) + 1;

40/// One localized sketch request.
41#[derive(Debug, Clone, Copy, PartialEq, Eq)]
42pub struct BucketRequest {
43    pub depth: u8,
44    pub prefix: u64,
45    pub capacity: usize,
46    /// Explicit overflow marker. Must be set to `true` only after the overflow
47    /// request validation path passes. Never infer from `capacity` alone.
48    pub overflow: bool,
49}

51impl BucketRequest {
52    /// Standard bucket request (overflow = false).
53    #[must_use]
54    pub fn new(depth: u8, prefix: u64, capacity: usize) -> Self {

56            depth,
57            prefix,
58            capacity,
59            overflow: false,
60        }
61    }

63    /// Overflow bucket request (overflow = true).
64    #[must_use]
65    pub fn with_overflow(depth: u8, prefix: u64, capacity: usize) -> Self {
66        Self {
67            depth,
68            prefix,
69            capacity,
70            overflow: true,
71        }
72    }
73}

91/// Returns the canonical start of a bucket's key-space range.
92///
93/// This is shared between bucket ordering and request validation so the two
94/// paths stay aligned if the bucket geometry changes.
95#[must_use]
96pub(crate) fn bucket_range_start(request: &BucketRequest) -> u128 {
97    let shift = u32::from(MAX_DEPTH.saturating_sub(request.depth));
98    u128::from(request.prefix) << shift
99}

111/// Estimates the symmetric difference from corresponding strata sketches.
112///
113/// This helper stays test-only so production callers use the structured
114/// [`StrataEstimate`] API rather than a bare scalar estimate.
115///
116/// # Errors
117/// Returns an error when root finding exceeds its work budget.
118#[cfg(test)]
119#[cfg_attr(coverage_nightly, coverage(off))]
120fn estimate_delta(
121    local: &[[u64; STRATUM_CAPACITY]; STRATA_COUNT],
122    remote: &[[u64; STRATUM_CAPACITY]; STRATA_COUNT],
123) -> Result<u64, AlgebraicError> {
124    Ok(estimate_delta_internal(local, remote)?.0)
125}

127/// Estimates the symmetric difference and whether that estimate is provisional.
128///
129/// Starting at the sparsest stratum, this decodes the longest consecutive tail.
130/// If `r` is the lowest decoded stratum and `T` is the decoded tail cardinality,
131/// `T * 2^r` estimates the complete difference. Decoding every stratum yields
132/// the exact cardinality.
133///
134/// If even the sparsest residual stratum overflows, this returns a saturated
135/// estimate in [`StrataEstimate::delta`] and marks it
136/// [`StrataEstimate::low_confidence`] so the caller can route away from sketch
137/// mode.
138///
139/// # Errors
140/// Returns an error when root finding exceeds its work budget.
141pub fn estimate_strata(
142    local: &[[u64; STRATUM_CAPACITY]; STRATA_COUNT],
143    remote: &[[u64; STRATUM_CAPACITY]; STRATA_COUNT],
144) -> Result<StrataEstimate, AlgebraicError> {
145    let (delta, low_confidence) = estimate_delta_internal(local, remote)?;
146    Ok(StrataEstimate {
147        delta,
148        low_confidence,
149    })
150}
151
152fn estimate_delta_internal(
153    local: &[[u64; STRATUM_CAPACITY]; STRATA_COUNT],
154    remote: &[[u64; STRATUM_CAPACITY]; STRATA_COUNT],
155) -> Result<(u64, bool), AlgebraicError> {
156    let mut decoded_tail = 0_u64;
157    let mut lowest_decoded = None;
158
159    for stratum in (0..STRATA_COUNT).rev() {
160        let residual: [u64; STRATUM_CAPACITY] =
161            core::array::from_fn(|index| local[stratum][index] ^ remote[stratum][index]);
162
163        match pinsketch::decode(&residual, STRATUM_CAPACITY) {
164            Ok(roots) => {
165                let cardinality =
166                    u64::try_from(roots.len()).map_err(|_| AlgebraicError::CountOverflow)?;
167                decoded_tail = decoded_tail
168                    .checked_add(cardinality)
169                    .ok_or(AlgebraicError::CountOverflow)?;
170                lowest_decoded = Some(stratum);
171            }
172
173            Err(AlgebraicError::DecodeFailure) => {
174                if lowest_decoded.is_none() && stratum == STRATA_COUNT - 1 {
175                    return Ok((SATURATED_DELTA_ESTIMATE, true));
176                }
177
178                let scaled_stratum = lowest_decoded.unwrap_or(stratum);
179                let shift =
180                    u32::try_from(scaled_stratum).map_err(|_| AlgebraicError::CountOverflow)?;
181                let estimate = decoded_tail
182                    .max(OVER_CAPACITY_DELTA_FLOOR)
183                    .saturating_mul(1_u64 << shift);
184
185                return Ok((estimate, true));
186            }
187            Err(error) => return Err(error),
188        }
189    }
190
191    let stratum = lowest_decoded.expect("all strata decoded implies stratum 0 decoded");
192    let shift = u32::try_from(stratum).map_err(|_| AlgebraicError::CountOverflow)?;
193    // saturating_mul overflows to u64::MAX rather than silently collapsing to 0
194    // (which the old checked_shl(shift).unwrap_or(0) scale factor could do).
195    Ok((decoded_tail.saturating_mul(1_u64 << shift), false))
196}

198/// Parses and independently decodes concatenated residual bucket sketches.
199///
200/// Requests are validated in canonical key-space range order. Each requested
201/// sketch is serialized as little-endian syndrome coordinates with no length
202/// prefix; the request capacities define the boundaries.
203///
204/// Structural and budget errors abort the batch. A normal decode failure is
205/// isolated to its bucket so successfully decoded roots can be retained.
206///
207/// Decodes a batch of bucket sketches under a single, shared work budget.
208///
209/// `budget` bounds the *total* factoring work across every bucket in the
210/// batch, not just each bucket individually. Without this, each bucket gets
211/// its own implicit `MAX_FACTOR_WORK` from the unbudgeted decode path, so a
212/// batch of many buckets (up to `MAX_BUCKETED_SKETCH_CAPACITY` worth of
213/// aggregate capacity) could cost the caller an unbounded multiple of a
214/// single bucket's work -- the batch size itself was not a cost input.
215///
216/// Each bucket's draw against `budget` is further clamped to a per-bucket
217/// work ceiling for that bucket's declared capacity (an internal
218/// `pinsketch` cost-model helper, not part of this crate's public API), so
219/// one pathological sketch (corrupt, over capacity, or deliberately ground
220/// to resist factoring) cannot exhaust the shared pool before later,
221/// cheaper buckets in `requests` are even attempted -- otherwise batch
222/// outcomes would depend on request order, since a successful decode and a
223/// proven-undecodable one differ in cost by roughly two orders of
224/// magnitude (see `MAX_FACTOR_WORK`'s comment). Callers sizing `budget`
225/// should account for this clamp: budgeting less than that per-bucket
226/// ceiling for even one request in the batch means that request may report
227/// `BudgetExhausted` for reasons unrelated to how much of `budget` the
228/// rest of the batch has used. [`MAX_BATCH_FACTOR_WORK`] is sized to cover
229/// a full normal-path batch and is the right default absent a
230/// caller-specific budget.
231///
232/// # Errors
233///
234/// Returns an error for invalid ordering, capacity, aggregate or byte
235/// length, or [`AlgebraicError::BudgetExhausted`] once the shared budget
236/// runs out, even if some buckets in the batch would otherwise have
237/// decoded.
238pub fn decode_bucket_sketches(
239    encoded: &[u8],
240    requests: &[BucketRequest],
241    mut budget: usize,
242) -> Result<BucketDecodeBatch, AlgebraicError> {
243    validate_bucket_requests(requests)?;
244
245    let mut offset = 0_usize;
246    let mut successful_buckets = Vec::new();
247    let mut failed_buckets = Vec::new();
248    for request in requests {
249        let byte_len = request
250            .capacity
251            .checked_mul(8)
252            .ok_or(AlgebraicError::InvalidSketchLength)?;
253        let end = offset
254            .checked_add(byte_len)
255            .ok_or(AlgebraicError::InvalidSketchLength)?;
256        let bytes = encoded
257            .get(offset..end)
258            .ok_or(AlgebraicError::InvalidSketchLength)?;
259        offset = end;
260
261        let coordinates = bytes
262            .chunks_exact(8)
263            .map(|coordinate| {
264                let mut value = [0; 8];
265                value.copy_from_slice(coordinate);
266                u64::from_le_bytes(value)
267            })
268            .collect();
269
270        let sketch = SyndromeSketch::from_coordinates(coordinates)?;
271        // Clamp this bucket's draw so it can't outspend its own fair
272        // ceiling and starve buckets later in `requests` of the shared
273        // pool -- see this function's doc comment.
274        let ceiling = pinsketch::single_call_work_ceiling(request.capacity)
275            .ok_or(AlgebraicError::InvalidSketchCapacity)?;
276        let allowance = budget.min(ceiling);
277        let mut remaining = allowance;
278        let result = sketch.decode_elements_with_shared_budget(request.capacity, &mut remaining);
279        // `remaining` only ever decreases from `allowance` (budgeted
280        // decoders only subtract) and `allowance <= budget` by
281        // construction above, so both `checked_sub`s below are provably
282        // non-underflowing. `debug_assert!` turns a violation of that
283        // invariant into a loud dev-time failure instead of a silent
284        // `BudgetExhausted` that would read as a routine protocol
285        // fallback -- but the production path still degrades to an
286        // error rather than panicking: this function processes
287        // untrusted wire input, and a release-mode `assert!` reachable
288        // from that input would trade a decode-accounting bug for a
289        // remotely triggerable crash loop, which is a worse failure
290        // mode than the one it replaces.
291        debug_assert!(
292            remaining <= allowance,
293            "decoder violated budget bounds: remaining={remaining} > allowance={allowance}"
294        );
295        let spent = allowance
296            .checked_sub(remaining)
297            .ok_or(AlgebraicError::BudgetExhausted)?;
298        budget = budget
299            .checked_sub(spent)
300            .ok_or(AlgebraicError::BudgetExhausted)?;
301        match result {
302            Ok(roots) => successful_buckets.push(BucketDecodeSuccess {
303                depth: request.depth,
304                prefix: request.prefix,
305                roots,
306            }),
307            Err(AlgebraicError::DecodeFailure) => {
308                failed_buckets.push((request.depth, request.prefix));
309            }
310            Err(error) => return Err(error),
311        }
312    }
313    if offset != encoded.len() {
314        return Err(AlgebraicError::InvalidSketchLength);
315    }
316    Ok(BucketDecodeBatch {
317        successful_buckets,
318        failed_buckets,
319    })
320}

322/// Validates a list of bucket requests to ensure they adhere to limits and form an antichain.
323///
324/// Normatively, the request set MUST be an antichain under the prefix-containment relation.
325/// For two requests `R_i = (d_i, p_i)` and `R_j = (d_j, p_j)`, `R_i` is an ancestor of `R_j`
326/// if and only if `d_i <= d_j` and the `d_i` most-significant bits of `p_j` equal `p_i`.
327/// A receiver MUST reject any request list containing an ancestor/descendant pair before
328/// performing sketch subtraction or field operations.
329///
330/// This function also ensures no single request exceeds the per-node extraction limit
331/// (`MAX_BUCKET_SKETCH_CAPACITY`), that the overall extraction respects
332/// `MAX_BUCKETED_SKETCH_CAPACITY`, and that bucket indices are well-formed.
333///
334/// Implementation note (non-normative): a canonical `O(N log N)` verifier can sort requests
335/// by ascending `depth`, then compare each candidate only against previously validated shallower
336/// requests using the same prefix test. A binary prefix trie can reduce this to `O(N)`.
337///
338/// # Errors
339/// Returns an error if any capacity or bound constraint is violated, or if the requests
340/// overlap.
341pub fn validate_bucket_requests(requests: &[BucketRequest]) -> Result<(), AlgebraicError> {
342    for request in requests {
343        if request.overflow {
344            return Err(AlgebraicError::InvalidSketchCapacity);
345        }
346    }
347    validate_bucket_requests_with_limit(requests, MAX_BUCKET_SKETCH_CAPACITY)
348}
349
350/// Validates requests issued through the explicit local overflow path.
351///
352/// Normal request handling must continue to call [`validate_bucket_requests`].
353/// This helper does not negotiate or advertise overflow support to peers.
354///
355/// # Errors
356/// Returns an error when a request exceeds the overflow or aggregate capacity
357/// limit, or when requests are malformed or overlap.
358pub fn validate_overflow_bucket_requests(requests: &[BucketRequest]) -> Result<(), AlgebraicError> {
359    for request in requests {
360        if !request.overflow {
361            return Err(AlgebraicError::InvalidSketchCapacity);
362        }
363    }
364    validate_bucket_requests_with_limit(requests, MAX_OVERFLOW_BUCKET_CAPACITY)
365}
366
367fn validate_bucket_requests_with_limit(
368    requests: &[BucketRequest],
369    max_bucket_capacity: usize,
370) -> Result<(), AlgebraicError> {
371    let mut total_capacity = 0_usize;
372    let mut previous_end = 0_u128;
373    for request in requests {
374        if request.capacity == 0 || request.capacity > max_bucket_capacity {
375            return Err(AlgebraicError::InvalidSketchCapacity);
376        }
377        if request.depth > MAX_DEPTH {
378            return Err(AlgebraicError::InvalidBucketIndex);
379        }
380
381        if request.depth < MAX_DEPTH && request.prefix >= (1_u64 << request.depth) {
382            return Err(AlgebraicError::InvalidBucketIndex);
383        }
384
385        total_capacity = total_capacity
386            .checked_add(request.capacity)
387            .ok_or(AlgebraicError::InvalidSketchCapacity)?;
388        if total_capacity > MAX_BUCKETED_SKETCH_CAPACITY {
389            return Err(AlgebraicError::InvalidSketchCapacity);
390        }
391
392        let start = bucket_range_start(request);
393        let shift = u32::from(MAX_DEPTH.saturating_sub(request.depth));
394        let end = start
395            .checked_add(1_u128 << shift)
396            .ok_or(AlgebraicError::InvalidBucketIndex)?;
397
398        if start < previous_end {
399            return Err(AlgebraicError::InvalidBucketIndex);
400        }
401        previous_end = end;
402    }
403    Ok(())
404}

490    fn toggle_stratum(strata: &mut [[u64; STRATUM_CAPACITY]; STRATA_COUNT], value: u64) {
491        let event = ElementHash {
492            h128: u128::from(value),
493            h64: value,
494        };
495        let mut resident = ResidentKernel::new();
496        resident.insert(event).unwrap();
497        for (target, source) in strata.iter_mut().zip(resident.strata()) {
498            for (coordinate, value) in target.iter_mut().zip(source) {
499                *coordinate ^= value;
500            }
501        }
502    }

638    #[test]
639    fn one_undecodable_bucket_does_not_starve_a_later_decodable_bucket() {
640        // Bucket at prefix 1 is over capacity and forces the full
641        // factoring ladder to run and fail. Bucket at prefix 9 is
642        // trivially decodable. The shared budget is exactly one
643        // capacity-2 ceiling plus a small margin.
644        //
645        // For an unclamped root-only failure (no further splitting -- the
646        // only case a small deterministic test can construct without
647        // engineering a specific locator factorization by hand) the spend
648        // is bounded by `single_call_work_ceiling` regardless of whether
649        // this clamp exists, so this test cannot by itself distinguish
650        // clamped from unclamped code; it pins the intended budget-sharing
651        // behavior (the easy bucket must not starve) as a regression guard
652        // going forward. The clamp's real justification is a bucket whose
653        // recursion visits several nodes before failing, where the total
654        // cost can exceed one node's ceiling -- see `MAX_FACTOR_WORK`'s
655        // comment on unbalanced split trees; that scenario needs a
656        // deliberately engineered locator to reproduce deterministically
657        // and isn't covered by this test.
658        let requests = [BucketRequest::new(8, 1, 2), BucketRequest::new(8, 9, 2)];
659        let mut over_capacity = SyndromeSketch::new(2).unwrap();
660        for value in [1, 2, 3] {
661            over_capacity.toggle(value).unwrap();
662        }
663        let mut encoded = over_capacity
664            .coordinates()
665            .iter()
666            .flat_map(|value| value.to_le_bytes())
667            .collect::<Vec<_>>();
668        let mut easy = SyndromeSketch::new(2).unwrap();
669        easy.toggle(7).unwrap();
670        encoded.extend(
671            easy.coordinates()
672                .iter()
673                .flat_map(|value| value.to_le_bytes()),
674        );
675
676        let ceiling = pinsketch::single_call_work_ceiling(2).unwrap();
677        let budget = ceiling.saturating_add(1_000);
678
679        let batch = decode_bucket_sketches(&encoded, &requests, budget).unwrap();
680        assert!(
681            batch
682                .successful_buckets
683                .iter()
684                .any(|success| success.prefix == 9 && success.roots == vec![7]),
685            "the decodable bucket at prefix 9 must not be starved by the \
686             undecodable bucket at prefix 1 running first: {batch:?}"
687        );
688    }
