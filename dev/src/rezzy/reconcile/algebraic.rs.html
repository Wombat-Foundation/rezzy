26/// Maximum extraction capacity for an unbucketed `algebraic_v1` sketch.
27pub const MAX_SKETCH_CAPACITY: usize = 32;
28/// Default local extraction limit for CPU-bounded sketch decoding.
29pub const MAX_LOCAL_SKETCH_DECODE_CAPACITY: usize = MAX_SKETCH_CAPACITY;
30/// Hard-capped overflow capacity for adaptive reconciliation.
31/// Separate from `MAX_SKETCH_CAPACITY`; only reachable after overflow request
32/// validation passes.
33// A full degree-256 decode is only affordable within `MAX_FACTOR_WORK`
34// because `pinsketch::build_frobenius_basis` amortizes the root-finding
35// ladder's dominant cost across all trials for a node instead of repeating
36// it per trial -- see `MAX_FACTOR_WORK`'s comment for the measurement this
37// capacity relies on, including the caveat that it covers observed
38// balanced-split behavior, not a proven worst case.
39pub const MAX_OVERFLOW_SKETCH_CAPACITY: usize = 256;
40const EVENT_HASH_ENCODED_LEN: usize = 43;

69/// The two truncations of a reconciled element's canonical 32-byte digest.
70#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
71pub struct ElementHash {
72    /// First 128 bits, interpreted in network byte order.
73    pub h128: u128,
74    /// First non-zero 64-bit chunk of the element digest (network byte order),
75    /// falling back to 1 if all four 64-bit chunks are zero.
76    pub h64: u64,
77}
78
79impl ElementHash {
80    // TODO(prefix-grinding): `h64` is derived unkeyed from the element
81    // digest below. For V3/V4+ Matrix event IDs the digest *is* a content
82    // hash the event's author controls (message body, custom content keys,
83    // timestamp within clock-skew tolerance all give grinding room), so an
84    // attacker with grinding room can search offline for events whose h64
85    // shares a long common prefix -- roughly 2^k SHA-256/BLAKE-family
86    // hashes for k bits, i.e. seconds of commodity CPU for k in the
87    // mid-20s. Note this needs more than "can post events": an event that
88    // replicated normally is present in both sides' sets and cancels out
89    // of the symmetric difference, so it never reaches the bucket splitter
90    // at all. The attacker needs the ground events to actually be *in* the
91    // difference at reconciliation time -- e.g. selective federation, or
92    // an attacker-operated homeserver that delivers them to some peers and
93    // withholds them from others. A materially lower bar than compromising
94    // a server, but a different (and narrower) threat model than "any room
95    // member," and worth stating precisely before this goes into 4511-C.
96    //
97    // Given that precondition, `client.rs`'s bucket splitter descends one
98    // h64 bit per round (`bucket_range_start`/the depth-increment in
99    // `select_action`) over `MAX_RECONCILIATION_ROUNDS` (20) rounds before
100    // giving up to `ClientAction::ExtremityDiff`, so a ~20+ bit shared
101    // prefix reliably exhausts every round on that bucket before any real
102    // split happens. This isn't a one-off cost either: h64 is fixed
103    // forever once an event exists, so a single grinding pass taxes
104    // *every future* pairwise reconciliation of the room that touches
105    // those events (subject to the withholding precondition above), on
106    // any pair of servers, indefinitely -- not just the session the
107    // attacker ran it against.
108    //
109    // The fix from the literature (Yang et al., "Practical Rateless Set
110    // Reconciliation" §4.3) is a keyed hash (e.g. SipHash) negotiated
111    // per-session so placement isn't predictable offline. That is NOT a
112    // drop-in here: `ResidentKernel` (resident.rs) is deliberately a
113    // server-local structure built once and incrementally maintained
114    // across the room's lifetime, then reused to serve *any* peer that
115    // reconciles against it -- the whole point is amortizing the
116    // build cost across many peers/sessions rather than rebuilding
117    // per-session. Keying `h64` per-session breaks that: the bucket
118    // geometry (and therefore the resident trie's shape) would become
119    // session-specific, so the server would need either a separate
120    // resident structure per active peer (defeats the amortization this
121    // module exists for) or a coarser shared secret.
122    //
123    // A *static* room-scoped key doesn't solve it: the realistic
124    // adversary is a room member grinding their own event content, and a
125    // static room-scoped key is known to every room member by
126    // construction. But what defeats grinding isn't secrecy of the key --
127    // it's unpredictability at authoring time. Event IDs (and therefore
128    // h64) are fixed when the event is created; if the placement key
129    // didn't exist yet, no amount of offline grinding could have targeted
130    // it. A room-scoped key that *rotates on an epoch* is still public to
131    // every member and still defeats precomputation, because a
132    // pre-ground event lands in an unpredictable bucket after the next
133    // rotation. That preserves `ResidentKernel`'s amortization within an
134    // epoch (one rebuild per rotation, not per session) -- a materially
135    // different, more promising tradeoff than per-session keying. Needs
136    // an MSC-level design decision (epoch-rotated placement key vs.
137    // accepting the bounded liveness cost and documenting it, informed by
138    // the client-side mitigation noted on `MAX_RECONCILIATION_ROUNDS` in
139    // client.rs), not a code-level patch -- see 4511-C.
140    /// Derives the MSC4521 profile truncations from a canonical 32-byte element digest.
141    #[must_use]
142    pub fn from_digest32(digest: [u8; 32]) -> Self {
143        let mut wide = [0; 16];
144        wide.copy_from_slice(&digest[..16]);
145        let mut short = [0; 8];
146        let h64 = digest
147            .chunks_exact(8)
148            .take(4)
149            .map(|chunk| {
150                short.copy_from_slice(chunk);
151                u64::from_be_bytes(short)
152            })
153            .find(|value| *value != 0)
154            .unwrap_or(1);
155        Self {
156            h128: u128::from_be_bytes(wide),
157            h64,
158        }
159    }
160
161    /// Derives an element hash from a Matrix event ID.
162    ///
163    /// This is the MSC4521 Matrix event-ID binding. The algebraic kernel itself
164    /// is generic over canonical 32-byte element digests.
165    ///
166    /// # Errors
167    /// Returns an error when the ID has no `$` sigil, contains invalid base64,
168    /// or its decoded hash is not exactly 32 bytes (256 bits).
169    pub fn from_matrix_event_id(
170        event_id: &str,
171        format: EventIdFormat,
172    ) -> Result<Self, AlgebraicError> {
173        Self::matrix_event_digest32(event_id, format).map(Self::from_digest32)
174    }
175
176    /// Derives the MSC4521 Matrix event-ID binding digest `D(e)`.
177    ///
178    /// # Errors
179    /// Returns an error when the ID has no `$` sigil, contains invalid base64,
180    /// or its decoded hash is not exactly 32 bytes.
181    pub fn matrix_event_digest32(
182        event_id: &str,
183        format: EventIdFormat,
184    ) -> Result<[u8; 32], AlgebraicError> {
185        let encoded = event_id
186            .strip_prefix('$')
187            .ok_or(AlgebraicError::InvalidEventId)?;
188        if format != EventIdFormat::Legacy && encoded.len() > EVENT_HASH_ENCODED_LEN {
189            return Err(AlgebraicError::InvalidBase64);
190        }
191        let digest = match format {
192            EventIdFormat::Legacy => Sha256::digest(event_id.as_bytes()).to_vec(),
193            EventIdFormat::V3 => STANDARD_NO_PAD
194                .decode(encoded)
195                .map_err(|_| AlgebraicError::InvalidBase64)?,
196            EventIdFormat::V4Plus => URL_SAFE_NO_PAD
197                .decode(encoded)
198                .map_err(|_| AlgebraicError::InvalidBase64)?,
199        };
200        if digest.len() != 32 {
201            return Err(AlgebraicError::InvalidEventId);
202        }
203        digest
204            .try_into()
205            .map_err(|_| AlgebraicError::InvalidEventId)
206    }
207}

209/// Incrementally maintained level-0 set digest and exact element count.
210#[derive(Debug, Clone, Copy, PartialEq, Eq, Default)]
211pub struct RoomAccumulator {
212    digest: u128,
213    count: u64,
214}
215
216impl RoomAccumulator {
217    #[must_use]
218    pub const fn new() -> Self {
219        Self {
220            digest: 0,
221            count: 0,
222        }
223    }
224    #[must_use]
225    pub const fn digest(self) -> u128 {
226        self.digest
227    }
228    #[must_use]
229    pub const fn known_event_count(self) -> u64 {
230        self.count
231    }
232
233    /// Adds a known element.
234    ///
235    /// # Errors
236    /// Returns [`AlgebraicError::CountOverflow`] at `u64::MAX` events.
237    pub fn insert(&mut self, hash: ElementHash) -> Result<(), AlgebraicError> {
238        self.count = self
239            .count
240            .checked_add(1)
241            .ok_or(AlgebraicError::CountOverflow)?;
242        self.digest ^= hash.h128;
243        Ok(())
244    }
245
246    /// Removes a known element.
247    ///
248    /// # Errors
249    /// Returns [`AlgebraicError::CountUnderflow`] when the accumulator is empty.
250    pub fn remove(&mut self, hash: ElementHash) -> Result<(), AlgebraicError> {
251        self.count = self
252            .count
253            .checked_sub(1)
254            .ok_or(AlgebraicError::CountUnderflow)?;
255        self.digest ^= hash.h128;
256        Ok(())
257    }
258
259    #[must_use]
260    pub fn encode_digest(self) -> String {
261        URL_SAFE_NO_PAD.encode(self.digest.to_be_bytes())
262    }
263
264    /// Decodes a level-0 digest.
265    ///
266    /// # Errors
267    /// Returns an error for invalid base64 or any decoded length other than 16 bytes.
268    ///
269    /// # Panics
270    /// Panics only if the prior length check is violated internally.
271    pub fn decode_digest(encoded: &str) -> Result<u128, AlgebraicError> {
272        if encoded.len() != 22 {
273            return Err(AlgebraicError::InvalidDigestLength);
274        }
275        let bytes = URL_SAFE_NO_PAD
276            .decode(encoded)
277            .map_err(|_| AlgebraicError::InvalidBase64)?;
278        let bytes: [u8; 16] = bytes
279            .as_slice()
280            .try_into()
281            .expect("digest length is validated before decode");
282        Ok(u128::from_be_bytes(bytes))
283    }
284
285    #[must_use]
286    pub const fn residual(self, other: Self) -> u128 {
287        self.digest ^ other.digest
288    }
289
290    /// Computes the unquoted opaque value for the MSC0501 HTTP `ETag`.
291    ///
292    /// # Panics
293    /// Panics only if `serde_json` fails to serialize a sequence of strings.
294    #[must_use]
295    pub fn etag<'a>(self, extremity_event_ids: impl IntoIterator<Item = &'a str>) -> String {
296        let mut extremities: Vec<&str> = extremity_event_ids.into_iter().collect();
297        extremities.sort_unstable();
298        let canonical = serde_json::to_vec(&extremities).expect("string arrays are serializable");
299        let frontier_hash = Sha256::digest(canonical);
300        let mut etag = Vec::with_capacity(24);
301        etag.extend_from_slice(&self.digest.to_be_bytes());
302        etag.extend_from_slice(&frontier_hash[..8]);
303        URL_SAFE_NO_PAD.encode(etag)
304    }
305}

307/// Checks decoded difference identifiers against the 128-bit integrity residual.
308#[must_use]
309pub fn verify_residual(
310    expected_residual: u128,
311    hashes: impl IntoIterator<Item = ElementHash>,
312) -> bool {
313    hashes
314        .into_iter()
315        .fold(0, |residual, hash| residual ^ hash.h128)
316        == expected_residual
317}

319/// Odd syndrome coordinates `s1, s3, ... s(2k-1)` over GF(2^64).
320#[derive(Debug, Clone, PartialEq, Eq)]
321pub struct SyndromeSketch {
322    coordinates: Vec<u64>,
323}
324
325impl SyndromeSketch {
326    pub(crate) fn from_coordinates(coordinates: Vec<u64>) -> Result<Self, AlgebraicError> {
327        Self::from_coordinates_checked(coordinates)
328    }
329
330    fn from_coordinates_checked(coordinates: Vec<u64>) -> Result<Self, AlgebraicError> {
331        if coordinates.is_empty() || coordinates.len() > MAX_SKETCH_CAPACITY {
332            return Err(AlgebraicError::InvalidSketchCapacity);
333        }
334        Ok(Self { coordinates })
335    }

337    /// Allocates an empty sketch with the requested extraction capacity.
338    ///
339    /// # Errors
340    /// Returns an error for zero capacity or capacity above the profile maximum.
341    pub fn new(capacity: usize) -> Result<Self, AlgebraicError> {
342        if capacity == 0 || capacity > MAX_SKETCH_CAPACITY {
343            return Err(AlgebraicError::InvalidSketchCapacity);
344        }
345        Ok(Self {
346            coordinates: vec![0; capacity],
347        })
348    }

350    #[must_use]
351    pub fn capacity(&self) -> usize {
352        self.coordinates.len()
353    }
354    #[must_use]
355    pub fn coordinates(&self) -> &[u64] {
356        &self.coordinates
357    }

359    /// Subtracts (XORs) another sketch's coordinates from this one.
360    ///
361    /// # Errors
362    /// Returns an error when sketch capacities differ.
363    pub fn xor(&mut self, other: &Self) -> Result<(), AlgebraicError> {
364        if self.capacity() != other.capacity() {
365            return Err(AlgebraicError::InvalidSketchLength);
366        }
367        for (a, b) in self.coordinates.iter_mut().zip(other.coordinates.iter()) {
368            *a ^= b;
369        }
370        Ok(())
371    }
372
373    /// Inserts or removes a short identifier. Both operations are XOR in characteristic two.
374    /// # Errors
375    /// Returns an error because zero is not representable by a `PinSketch`.
376    pub fn toggle(&mut self, value: u64) -> Result<(), AlgebraicError> {
377        if value == 0 {
378            return Err(AlgebraicError::ZeroShortIdentifier);
379        }
380        let squared = gf64_mul(value, value);
381        let mut odd_power = value;
382        for coordinate in &mut self.coordinates {
383            *coordinate ^= odd_power;
384            odd_power = gf64_mul(odd_power, squared);
385        }
386        Ok(())
387    }

389    /// Decodes up to `max_elements` from this residual sketch.
390    ///
391    /// # Errors
392    /// Returns [`AlgebraicError::DecodeFailure`] when the residual exceeds the
393    /// bound, is malformed, or does not factor into distinct field elements.
394    /// Returns [`AlgebraicError::InvalidSketchCapacity`] when `max_elements`
395    /// exceeds the sketch capacity or the local decode policy, and
396    /// [`AlgebraicError::BudgetExhausted`] when root finding reaches its work limit.
397    pub fn decode_elements(&self, max_elements: usize) -> Result<Vec<u64>, AlgebraicError> {
398        if max_elements == 0
399            || max_elements > self.capacity()
400            || max_elements > MAX_LOCAL_SKETCH_DECODE_CAPACITY
401        {
402            return Err(AlgebraicError::InvalidSketchCapacity);
403        }
404        let decoded = super::pinsketch::decode(&self.coordinates[..max_elements], max_elements)?;
405        self.validate_decoded_elements(decoded)
406    }
407
408    /// # Errors
409    ///
410    /// Returns [`AlgebraicError::DecodeFailure`] when the residual exceeds the
411    /// bound, is malformed, or does not factor into distinct field elements.
412    /// Returns [`AlgebraicError::InvalidSketchCapacity`] when `max_elements`
413    /// exceeds the sketch capacity or the local decode policy, and
414    /// [`AlgebraicError::BudgetExhausted`] when root finding reaches its work limit.
415    pub fn decode_elements_with_budget(
416        &self,
417        max_elements: usize,
418        budget: usize,
419    ) -> Result<Vec<u64>, AlgebraicError> {
420        let mut remaining = budget;
421        self.decode_elements_with_shared_budget(max_elements, &mut remaining)
422    }
423
424    /// Like [`decode_elements_with_budget`](Self::decode_elements_with_budget),
425    /// but draws from and updates a caller-owned `budget` in place, so a
426    /// caller decoding many sketches in a batch (e.g.
427    /// [`super::triage::decode_bucket_sketches`]) can enforce one shared
428    /// work ceiling across the whole batch instead of each sketch getting
429    /// its own independent allowance.
430    ///
431    /// # Errors
432    ///
433    /// Same as [`decode_elements_with_budget`](Self::decode_elements_with_budget).
434    pub(crate) fn decode_elements_with_shared_budget(
435        &self,
436        max_elements: usize,
437        budget: &mut usize,
438    ) -> Result<Vec<u64>, AlgebraicError> {
439        if max_elements == 0
440            || max_elements > self.capacity()
441            || max_elements > MAX_LOCAL_SKETCH_DECODE_CAPACITY
442        {
443            return Err(AlgebraicError::InvalidSketchCapacity);
444        }
445        let decoded = super::pinsketch::decode_with_budget(
446            &self.coordinates[..max_elements],
447            max_elements,
448            budget,
449        )?;
450        self.validate_decoded_elements(decoded)
451    }
452
453    fn validate_decoded_elements(&self, decoded: Vec<u64>) -> Result<Vec<u64>, AlgebraicError> {
454        if decoded.contains(&0) {
455            return Err(AlgebraicError::DecodeFailure);
456        }
457        let mut check = Self::new(self.capacity())?;
458        for element in &decoded {
459            check
460                .toggle(*element)
461                .expect("decoded elements are validated to be nonzero");
462        }
463        (check == *self)
464            .then_some(decoded)
465            .ok_or(AlgebraicError::DecodeFailure)
466    }

468    /// XOR-subtracts another sketch.
469    ///
470    /// # Errors
471    /// Returns an error when sketch capacities differ.
472    pub fn subtract(&self, other: &Self) -> Result<Self, AlgebraicError> {
473        if self.capacity() != other.capacity() {
474            return Err(AlgebraicError::InvalidSketchLength);
475        }
476        Ok(Self {
477            coordinates: self
478                .coordinates
479                .iter()
480                .zip(&other.coordinates)
481                .map(|(a, b)| a ^ b)
482                .collect(),
483        })
484    }

486    #[must_use]
487    pub fn encode(&self) -> String {
488        let byte_len = self.coordinates.len().checked_mul(8).unwrap_or(0);
489        let mut bytes = Vec::with_capacity(byte_len);
490        for coordinate in &self.coordinates {
491            bytes.extend_from_slice(&coordinate.to_le_bytes());
492        }
493        URL_SAFE_NO_PAD.encode(bytes)
494    }

496    /// Decodes a sketch with an externally negotiated capacity.
497    ///
498    /// # Errors
499    /// Returns an error for invalid capacity, base64, or encoded byte length.
500    pub fn decode(capacity: usize, encoded: &str) -> Result<Self, AlgebraicError> {
501        if capacity == 0 || capacity > MAX_SKETCH_CAPACITY {
502            return Err(AlgebraicError::InvalidSketchCapacity);
503        }
504        let expected_len = capacity
505            .checked_mul(8)
506            .ok_or(AlgebraicError::InvalidSketchLength)?;
507        let expected_encoded_len =
508            base64::encoded_len(expected_len, false).ok_or(AlgebraicError::InvalidSketchLength)?;
509        if encoded.len() != expected_encoded_len {
510            return Err(AlgebraicError::InvalidSketchLength);
511        }
512        let bytes = URL_SAFE_NO_PAD
513            .decode(encoded)
514            .map_err(|_| AlgebraicError::InvalidBase64)?;
515        Self::from_encoded_bytes(capacity, &bytes)
516    }
517
518    fn from_encoded_bytes(capacity: usize, bytes: &[u8]) -> Result<Self, AlgebraicError> {
519        let expected_len = capacity
520            .checked_mul(8)
521            .ok_or(AlgebraicError::InvalidSketchLength)?;
522        if bytes.len() != expected_len {
523            return Err(AlgebraicError::InvalidSketchLength);
524        }
525        let coordinates = bytes
526            .chunks_exact(8)
527            .map(|chunk| {
528                let mut value = [0; 8];
529                value.copy_from_slice(chunk);
530                u64::from_le_bytes(value)
531            })
532            .collect();
533        Ok(Self { coordinates })
534    }

536    /// Allocates an empty sketch with overflow extraction capacity.
537    ///
538    /// Overflow sketches are used when a standard 32-element sketch fails to
539    /// decode and a larger sketch is requested under the local overflow policy.
540    /// The capacity is validated against [`MAX_OVERFLOW_SKETCH_CAPACITY`].
541    ///
542    /// # Errors
543    /// Returns an error for zero capacity or capacity above the overflow maximum.
544    pub fn new_overflow(capacity: usize) -> Result<Self, AlgebraicError> {
545        if capacity == 0 || capacity > MAX_OVERFLOW_SKETCH_CAPACITY {
546            return Err(AlgebraicError::InvalidSketchCapacity);
547        }
548        Ok(Self {
549            coordinates: vec![0; capacity],
550        })
551    }

553    /// Decodes a sketch with overflow capacity from wire encoding.
554    ///
555    /// Like [`decode`](Self::decode) but validates against
556    /// [`MAX_OVERFLOW_SKETCH_CAPACITY`].
557    ///
558    /// # Errors
559    /// Returns an error for invalid capacity, base64, or encoded byte length.
560    pub fn decode_overflow(capacity: usize, encoded: &str) -> Result<Self, AlgebraicError> {
561        if capacity == 0 || capacity > MAX_OVERFLOW_SKETCH_CAPACITY {
562            return Err(AlgebraicError::InvalidSketchCapacity);
563        }
564        let expected_len = capacity
565            .checked_mul(8)
566            .ok_or(AlgebraicError::InvalidSketchLength)?;
567        let expected_encoded_len =
568            base64::encoded_len(expected_len, false).ok_or(AlgebraicError::InvalidSketchLength)?;
569        if encoded.len() != expected_encoded_len {
570            return Err(AlgebraicError::InvalidSketchLength);
571        }
572        let bytes = URL_SAFE_NO_PAD
573            .decode(encoded)
574            .map_err(|_| AlgebraicError::InvalidBase64)?;
575        Self::from_encoded_bytes(capacity, &bytes)
576    }

578    /// Decodes up to `max_elements` with overflow capacity support.
579    ///
580    /// Like [`decode_elements_with_budget`](Self::decode_elements_with_budget)
581    /// but validates against [`MAX_OVERFLOW_SKETCH_CAPACITY`] instead of
582    /// [`MAX_LOCAL_SKETCH_DECODE_CAPACITY`].
583    ///
584    /// # Errors
585    /// Returns [`AlgebraicError::InvalidSketchCapacity`] when `max_elements`
586    /// exceeds the sketch capacity or the overflow capacity limit, and
587    /// [`AlgebraicError::BudgetExhausted`] when root finding reaches its work
588    /// limit.
589    pub fn decode_elements_overflow_budget(
590        &self,
591        max_elements: usize,
592        budget: usize,
593    ) -> Result<Vec<u64>, AlgebraicError> {
594        if max_elements == 0
595            || max_elements > self.capacity()
596            || max_elements > MAX_OVERFLOW_SKETCH_CAPACITY
597        {
598            return Err(AlgebraicError::InvalidSketchCapacity);
599        }
600        let mut remaining = budget;
601        let decoded = super::pinsketch::decode_with_budget(
602            &self.coordinates[..max_elements],
603            max_elements,
604            &mut remaining,
605        )?;
606        self.validate_decoded_overflow(decoded)
607    }
608
609    fn validate_decoded_overflow(&self, decoded: Vec<u64>) -> Result<Vec<u64>, AlgebraicError> {
610        if decoded.contains(&0) {
611            return Err(AlgebraicError::DecodeFailure);
612        }
613        let mut check = Self::new_overflow(self.capacity())?;
614        for element in &decoded {
615            check
616                .toggle(*element)
617                .expect("decoded elements are validated to be nonzero");
618        }
619        (check == *self)
620            .then_some(decoded)
621            .ok_or(AlgebraicError::DecodeFailure)
622    }
623}
